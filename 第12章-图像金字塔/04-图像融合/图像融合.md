# 图像融合（Image Blending）

## 1. 图像融合概述

图像融合是将两张或多张图像合成为一张图像的技术。使用拉普拉斯金字塔进行图像融合可以实现无缝、自然的混合效果，避免简单拼接产生的明显边界。

### 1.1 为什么需要金字塔融合

**直接融合的问题**：
- 边界明显，过渡不自然
- 颜色突变
- 缺乏层次感

**金字塔融合的优势**：
- 多尺度混合，边界平滑
- 保留两张图像的细节
- 过渡自然无缝

### 1.2 金字塔融合原理

```
图像A                        图像B
  ↓                            ↓
拉普拉斯金字塔A              拉普拉斯金字塔B
  ↓                            ↓
  L_A[0], L_A[1], ...         L_B[0], L_B[1], ...
        ↘                    ↙
         融合（使用掩码金字塔）
               ↓
        融合后的拉普拉斯金字塔
               ↓
            重建
               ↓
          融合后的图像
```

### 1.3 融合公式

对于每一层的融合：

$$L_{blend}[i] = L_A[i] \cdot M[i] + L_B[i] \cdot (1 - M[i])$$

其中 $M[i]$ 是掩码在第 $i$ 层的高斯金字塔版本。

## 2. 基础图像融合

### 示例1：简单的左右融合

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def build_gaussian_pyramid(image, levels):
    """Build Gaussian pyramid"""
    pyramid = [image.astype(np.float64)]
    current = image.astype(np.float64)
    for i in range(levels - 1):
        current = cv2.pyrDown(current)
        pyramid.append(current)
    return pyramid

def build_laplacian_pyramid(image, levels):
    """Build Laplacian pyramid"""
    G = build_gaussian_pyramid(image, levels)
    L = []
    for i in range(levels - 1):
        expanded = cv2.pyrUp(G[i + 1])
        if expanded.shape != G[i].shape:
            expanded = cv2.resize(expanded, (G[i].shape[1], G[i].shape[0]))
        L.append(G[i] - expanded)
    L.append(G[-1])
    return L

def reconstruct_from_laplacian(pyramid):
    """Reconstruct image from Laplacian pyramid"""
    result = pyramid[-1].copy()
    for i in range(len(pyramid) - 2, -1, -1):
        expanded = cv2.pyrUp(result)
        if expanded.shape != pyramid[i].shape:
            expanded = cv2.resize(expanded, (pyramid[i].shape[1], pyramid[i].shape[0]))
        result = expanded + pyramid[i]
    return result

def pyramid_blend(img1, img2, mask, levels=6):
    """
    Blend two images using Laplacian pyramid

    Parameters:
        img1, img2: Input images (same size)
        mask: Blending mask (0-1, same size as images)
        levels: Number of pyramid levels

    Returns:
        Blended image
    """
    # Build Laplacian pyramids for both images
    L1 = build_laplacian_pyramid(img1, levels)
    L2 = build_laplacian_pyramid(img2, levels)

    # Build Gaussian pyramid for mask
    GM = build_gaussian_pyramid(mask, levels)

    # Blend at each level
    L_blend = []
    for l1, l2, gm in zip(L1, L2, GM):
        # Expand mask to match image channels if needed
        if len(l1.shape) == 3 and len(gm.shape) == 2:
            gm = np.stack([gm] * 3, axis=2)

        # Blend
        blended = l1 * gm + l2 * (1 - gm)
        L_blend.append(blended)

    # Reconstruct
    result = reconstruct_from_laplacian(L_blend)
    result = np.clip(result, 0, 255).astype(np.uint8)

    return result

# Create test images
size = (256, 256)

# Image 1: Orange
img1 = np.zeros((*size, 3), dtype=np.uint8)
cv2.rectangle(img1, (0, 0), size, (0, 165, 255), -1)  # Orange in BGR
cv2.circle(img1, (128, 128), 60, (0, 100, 200), -1)

# Image 2: Apple (red/green)
img2 = np.zeros((*size, 3), dtype=np.uint8)
cv2.rectangle(img2, (0, 0), size, (0, 255, 0), -1)  # Green
cv2.circle(img2, (128, 80), 80, (0, 0, 255), -1)  # Red

# Create mask (left half = 1, right half = 0)
mask = np.zeros(size, dtype=np.float64)
mask[:, :size[1]//2] = 1.0

# Direct blending (for comparison)
direct_blend = (img1.astype(np.float64) * mask[:,:,np.newaxis] +
                img2.astype(np.float64) * (1 - mask[:,:,np.newaxis]))
direct_blend = direct_blend.astype(np.uint8)

# Pyramid blending
pyramid_blended = pyramid_blend(img1, img2, mask, levels=6)

# Display
plt.figure(figsize=(16, 10))

plt.subplot(2, 3, 1)
plt.imshow(cv2.cvtColor(img1, cv2.COLOR_BGR2RGB))
plt.title('Image 1')
plt.axis('off')

plt.subplot(2, 3, 2)
plt.imshow(cv2.cvtColor(img2, cv2.COLOR_BGR2RGB))
plt.title('Image 2')
plt.axis('off')

plt.subplot(2, 3, 3)
plt.imshow(mask, cmap='gray')
plt.title('Mask')
plt.axis('off')

plt.subplot(2, 3, 4)
plt.imshow(cv2.cvtColor(direct_blend, cv2.COLOR_BGR2RGB))
plt.title('Direct Blend\n(visible seam)')
plt.axis('off')

plt.subplot(2, 3, 5)
plt.imshow(cv2.cvtColor(pyramid_blended, cv2.COLOR_BGR2RGB))
plt.title('Pyramid Blend\n(seamless)')
plt.axis('off')

# Show difference
plt.subplot(2, 3, 6)
diff = cv2.absdiff(direct_blend, pyramid_blended)
diff_enhanced = cv2.convertScaleAbs(diff, alpha=5)
plt.imshow(cv2.cvtColor(diff_enhanced, cv2.COLOR_BGR2RGB))
plt.title('Difference (5x)')
plt.axis('off')

plt.tight_layout()
plt.savefig('basic_blend.png', dpi=150, bbox_inches='tight')
plt.show()

print("Basic pyramid blending completed!")
```

### 示例2：渐变掩码融合

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def create_gradient_mask(size, direction='horizontal', center=0.5, width=0.3):
    """
    Create a gradient blending mask

    Parameters:
        size: (height, width)
        direction: 'horizontal' or 'vertical'
        center: Position of center (0-1)
        width: Width of transition zone (0-1)

    Returns:
        Gradient mask
    """
    h, w = size
    mask = np.zeros((h, w), dtype=np.float64)

    if direction == 'horizontal':
        x = np.linspace(0, 1, w)
        # Smooth transition using sigmoid-like function
        mask[:, :] = 1 - 1 / (1 + np.exp(-10 * (x - center) / width))
    else:
        y = np.linspace(0, 1, h)
        mask = 1 - 1 / (1 + np.exp(-10 * (y - center) / width))
        mask = mask.reshape(-1, 1)
        mask = np.tile(mask, (1, w))

    return mask

# Create test images
size = (256, 256)

# Image 1: Sky scene
img1 = np.zeros((*size, 3), dtype=np.uint8)
# Sky gradient
for i in range(size[0]):
    img1[i, :] = [255 - i//2, 200 - i//3, 100]
cv2.circle(img1, (200, 50), 30, (0, 255, 255), -1)  # Sun

# Image 2: Ground scene
img2 = np.zeros((*size, 3), dtype=np.uint8)
# Ground gradient
for i in range(size[0]):
    img2[i, :] = [50 + i//4, 100 + i//3, 50 + i//2]
cv2.rectangle(img2, (50, 100), (100, 200), (80, 80, 80), -1)  # Building

# Different mask types
masks = {
    'Sharp': create_gradient_mask(size, 'vertical', 0.5, 0.01),

}

# Blend with different masks

plt.title('Image 2 (Ground)')

# Show masks and results
```

```python
import cv2

def blend_with_arbitrary_mask(img1, img2, mask, levels=6, feather=True):
    """
    Blend images with arbitrary shaped mask

    Parameters:
        img1, img2: Input images
        mask: Binary or grayscale mask
        levels: Pyramid levels
        feather: Whether to feather the mask edges

    Returns:
        Blended image
    """
    # Ensure mask is float
    mask = mask.astype(np.float64)
    if mask.max() > 1:
        mask = mask / 255.0

    # Feather mask edges if requested
    if feather:
        mask = cv2.GaussianBlur(mask, (21, 21), 0)

    return pyramid_blend(img1, img2, mask, levels)
```

### 示例4：多图像融合

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def multi_image_blend(images, masks, levels=6):
    """
    Blend multiple images using Laplacian pyramids

    Parameters:
        images: List of images
        masks: List of corresponding masks (should sum to 1)
        levels: Pyramid levels

    Returns:
        Blended image
    """
    assert len(images) == len(masks), "Number of images must match masks"

    # Build Laplacian pyramids for all images
    L_pyramids = [build_laplacian_pyramid(img, levels) for img in images]

    # Build Gaussian pyramids for all masks
    G_masks = [build_gaussian_pyramid(mask.astype(np.float64), levels) for mask in masks]

    # Blend at each level
    L_blend = []
    for level in range(levels):
        blended_level = np.zeros_like(L_pyramids[0][level])

        for L_pyr, G_mask in zip(L_pyramids, G_masks):
            lap_level = L_pyr[level]
            mask_level = G_mask[level]

            # Expand mask to match channels
            if len(lap_level.shape) == 3 and len(mask_level.shape) == 2:
                mask_level = np.stack([mask_level] * 3, axis=2)

            blended_level += lap_level * mask_level

        L_blend.append(blended_level)

    # Reconstruct
    result = reconstruct_from_laplacian(L_blend)
    result = np.clip(result, 0, 255).astype(np.uint8)

    return result
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def panorama_blend_demo():
    """
    Demonstrate panorama-style image blending
    """
    # Create simulated panorama images with overlap
    h, w = 200, 300
    overlap = 100

    # Left image
    left = np.zeros((h, w, 3), dtype=np.uint8)
    # Gradient background
    for i in range(w):
        left[:, i] = [50 + i//3, 100, 200 - i//3]
    cv2.rectangle(left, (50, 50), (150, 150), (0, 200, 255), -1)
    cv2.putText(left, 'LEFT', (80, 180), cv2.FONT_HERSHEY_SIMPLEX, 0.8, (255, 255, 255), 2)

    # Right image (with overlap region)
    right = np.zeros((h, w, 3), dtype=np.uint8)
    for i in range(w):
        right[:, i] = [150 + (i-100)//3, 100, 100 - (i-100)//5]
    cv2.circle(right, (200, 100), 50, (255, 100, 0), -1)
    cv2.putText(right, 'RIGHT', (150, 180), cv2.FONT_HERSHEY_SIMPLEX, 0.8, (255, 255, 255), 2)

    # Create panorama canvas
    panorama_width = w * 2 - overlap
    canvas = np.zeros((h, panorama_width, 3), dtype=np.uint8)

    # Place images
    left_region = np.zeros((h, panorama_width, 3), dtype=np.uint8)
    left_region[:, :w] = left

    right_region = np.zeros((h, panorama_width, 3), dtype=np.uint8)
    right_region[:, w-overlap:] = right

    # Create blending mask
    mask = np.zeros((h, panorama_width), dtype=np.float64)
    mask[:, :w-overlap] = 1.0  # Left region

    # Gradient in overlap region
    for i in range(overlap):
        mask[:, w-overlap+i] = 1.0 - i/overlap

    # Apply pyramid blending
    blended = pyramid_blend(left_region, right_region, mask, levels=5)

    # Simple overlay for comparison
    simple = left_region.copy()
    alpha = 0.5
    simple[:, w-overlap:w] = cv2.addWeighted(
        left_region[:, w-overlap:w], alpha,
        right_region[:, w-overlap:w], 1-alpha, 0
    )
    simple[:, w:] = right_region[:, w:]

    # Display
    plt.figure(figsize=(18, 12))

    plt.subplot(3, 3, 1)
    plt.imshow(cv2.cvtColor(left, cv2.COLOR_BGR2RGB))
    plt.title('Left Image')
    plt.axis('off')

    plt.subplot(3, 3, 2)
    plt.imshow(cv2.cvtColor(right, cv2.COLOR_BGR2RGB))
    plt.title('Right Image')
    plt.axis('off')

    plt.subplot(3, 3, 3)
    plt.imshow(mask, cmap='gray')
    plt.title('Blending Mask')
    plt.axis('off')

    plt.subplot(3, 3, 4)
    plt.imshow(cv2.cvtColor(left_region, cv2.COLOR_BGR2RGB))
    plt.title('Left on Canvas')
    plt.axis('off')

    plt.subplot(3, 3, 5)
    plt.imshow(cv2.cvtColor(right_region, cv2.COLOR_BGR2RGB))
    plt.title('Right on Canvas')
    plt.axis('off')

    plt.subplot(3, 3, 6)
    plt.imshow(cv2.cvtColor(simple, cv2.COLOR_BGR2RGB))
    plt.title('Simple Alpha Blend')
    plt.axis('off')

    plt.subplot(3, 1, 3)
    plt.imshow(cv2.cvtColor(blended, cv2.COLOR_BGR2RGB))
    plt.title('Pyramid Blend (Seamless)')
    plt.axis('off')

    plt.tight_layout()
    plt.savefig('panorama_blend.png', dpi=150, bbox_inches='tight')
    plt.show()
```

```python
def face_swap_demo():
    """
    Demonstrate face swapping effect using pyramid blending
    """
    # Create simulated face images
    size = (256, 256)

    # Face 1: Round face
    face1 = np.ones((*size, 3), dtype=np.uint8) * 200  # Light skin
    cv2.ellipse(face1, (128, 140), (80, 100), 0, 0, 360, (180, 160, 140), -1)
    # Eyes
    cv2.ellipse(face1, (95, 110), (15, 10), 0, 0, 360, (255, 255, 255), -1)
    cv2.ellipse(face1, (161, 110), (15, 10), 0, 0, 360, (255, 255, 255), -1)
    cv2.circle(face1, (95, 110), 5, (50, 30, 20), -1)
    cv2.circle(face1, (161, 110), 5, (50, 30, 20), -1)
    # Nose
    cv2.line(face1, (128, 120), (128, 150), (150, 130, 110), 2)
    # Mouth
    cv2.ellipse(face1, (128, 180), (25, 10), 0, 0, 180, (100, 80, 150), -1)
    # Hair
    cv2.ellipse(face1, (128, 60), (90, 50), 0, 180, 360, (50, 30, 20), -1)
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

class PyramidBlender:
    """
    Complete image blending class using Laplacian pyramids
    """

    def __init__(self, levels=6):
        self.levels = levels

    def _build_gaussian_pyramid(self, image):
        """Build Gaussian pyramid"""
        pyramid = [image.astype(np.float64)]
        current = image.astype(np.float64)
        for i in range(self.levels - 1):
            current = cv2.pyrDown(current)
            pyramid.append(current)
        return pyramid

    def _build_laplacian_pyramid(self, image):
        """Build Laplacian pyramid"""
        G = self._build_gaussian_pyramid(image)
        L = []
        for i in range(self.levels - 1):
            expanded = cv2.pyrUp(G[i + 1])
            if expanded.shape != G[i].shape:
                expanded = cv2.resize(expanded, (G[i].shape[1], G[i].shape[0]))
            L.append(G[i] - expanded)
        L.append(G[-1])
        return L

    def _reconstruct(self, pyramid):
        """Reconstruct from Laplacian pyramid"""
        result = pyramid[-1].copy()
        for i in range(len(pyramid) - 2, -1, -1):
            expanded = cv2.pyrUp(result)
            if expanded.shape != pyramid[i].shape:
                expanded = cv2.resize(expanded, (pyramid[i].shape[1], pyramid[i].shape[0]))
            result = expanded + pyramid[i]
        return result

    def blend(self, img1, img2, mask):
        """
        Blend two images using mask

        Parameters:
            img1, img2: Input images (same size)
            mask: Blending mask (0-1)

        Returns:
            Blended image
        """
        # Ensure same size
        assert img1.shape == img2.shape, "Images must have same size"

        # Ensure mask is float
        mask = mask.astype(np.float64)
        if mask.max() > 1:
            mask = mask / 255.0

        # Build pyramids
        L1 = self._build_laplacian_pyramid(img1)
        L2 = self._build_laplacian_pyramid(img2)
        GM = self._build_gaussian_pyramid(mask)

        # Blend
        L_blend = []
        for l1, l2, gm in zip(L1, L2, GM):
            if len(l1.shape) == 3 and len(gm.shape) == 2:
                gm = np.stack([gm] * 3, axis=2)
            blended = l1 * gm + l2 * (1 - gm)
            L_blend.append(blended)

        # Reconstruct
        result = self._reconstruct(L_blend)
        return np.clip(result, 0, 255).astype(np.uint8)

    def blend_multiple(self, images, masks):
        """
        Blend multiple images

        Parameters:
            images: List of images
            masks: List of masks (should sum to 1)

        Returns:
            Blended image
        """
        assert len(images) == len(masks)

        # Build pyramids
        L_pyramids = [self._build_laplacian_pyramid(img) for img in images]
        G_masks = [self._build_gaussian_pyramid(m.astype(np.float64)) for m in masks]

        # Blend
        L_blend = []
        for level in range(self.levels):
            blended_level = np.zeros_like(L_pyramids[0][level])
            for L_pyr, G_mask in zip(L_pyramids, G_masks):
                lap = L_pyr[level]
                weight = G_mask[level]
                if len(lap.shape) == 3 and len(weight.shape) == 2:
                    weight = np.stack([weight] * 3, axis=2)
                blended_level += lap * weight
            L_blend.append(blended_level)

        result = self._reconstruct(L_blend)
        return np.clip(result, 0, 255).astype(np.uint8)

    def create_gradient_mask(self, size, direction='horizontal', position=0.5, width=0.2):
        """Create gradient blending mask"""
        h, w = size
        mask = np.zeros((h, w), dtype=np.float64)

        if direction == 'horizontal':
            x = np.linspace(0, 1, w)
            mask[:, :] = 1 - 1 / (1 + np.exp(-10 * (x - position) / width))
        else:
            y = np.linspace(0, 1, h)
            mask = 1 - 1 / (1 + np.exp(-10 * (y - position) / width))
            mask = mask.reshape(-1, 1)
            mask = np.tile(mask, (1, w))

        return mask
```

## 6. 总结

### 6.1 图像融合核心概念

| 概念 | 说明 |
|------|------|
| **金字塔融合** | 在多尺度上分别融合再重建 |
| **掩码金字塔** | 掩码也需要构建高斯金字塔 |
| **无缝融合** | 消除可见边界 |
| **细节保留** | 保留两张图像的细节信息 |

### 6.2 融合步骤

```python
# 1. 构建图像的拉普拉斯金字塔
L1 = build_laplacian_pyramid(img1)
L2 = build_laplacian_pyramid(img2)

# 2. 构建掩码的高斯金字塔
GM = build_gaussian_pyramid(mask)

# 3. 逐层融合
L_blend = [L1[i] * GM[i] + L2[i] * (1 - GM[i]) for i in range(levels)]

# 4. 重建
result = reconstruct(L_blend)
```
