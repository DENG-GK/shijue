# 拉普拉斯金字塔（Laplacian Pyramid）

## 1. 拉普拉斯金字塔概述

拉普拉斯金字塔是一种用于存储图像细节信息的多分辨率表示方法。与高斯金字塔不同，拉普拉斯金字塔保存的是每一层的细节信息（边缘、纹理等），这使得它在图像融合、压缩和增强等应用中非常有用。

### 1.1 什么是拉普拉斯金字塔

拉普拉斯金字塔的每一层是高斯金字塔**相邻两层之间的差值**：

$$L_i = G_i - \text{expand}(G_{i+1})$$

其中：
- $G_i$：高斯金字塔第 $i$ 层
- $\text{expand}(G_{i+1})$：将第 $i+1$ 层上采样到第 $i$ 层的大小
- $L_i$：拉普拉斯金字塔第 $i$ 层

### 1.2 拉普拉斯金字塔的结构

```
高斯金字塔                    拉普拉斯金字塔
    G₀ ─────────────────────→ L₀ = G₀ - expand(G₁)
    │                             (高频细节)
    ↓ pyrDown
    G₁ ─────────────────────→ L₁ = G₁ - expand(G₂)
    │                             (中频细节)
    ↓ pyrDown
    G₂ ─────────────────────→ L₂ = G₂ - expand(G₃)
    │                             (低频细节)
    ↓ pyrDown
    G₃ ─────────────────────→ L₃ = G₃ (顶层，低频信息)
```

### 1.3 拉普拉斯金字塔的特点

| 特点 | 说明 |
|------|------|
| **带通特性** | 每层包含特定频带的信息 |
| **完美重建** | 可以无损恢复原始图像 |
| **细节分离** | 分离不同尺度的细节 |
| **零均值** | 除顶层外，其他层均值接近零 |

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def build_laplacian_pyramid(image, levels=5):
    """
    Build a Laplacian pyramid

    Parameters:
        image: Input image
        levels: Number of pyramid levels

    Returns:
        Laplacian pyramid (list of images)
    """
    # First build Gaussian pyramid
    gaussian_pyramid = [image.astype(np.float64)]
    current = image.astype(np.float64)

    for i in range(levels - 1):
        current = cv2.pyrDown(current)
        gaussian_pyramid.append(current)

    # Build Laplacian pyramid
    laplacian_pyramid = []

    for i in range(levels - 1):
        # Expand smaller image
        expanded = cv2.pyrUp(gaussian_pyramid[i + 1])

        # Match sizes if needed
        if expanded.shape != gaussian_pyramid[i].shape:
            expanded = cv2.resize(expanded,
                                   (gaussian_pyramid[i].shape[1],
                                    gaussian_pyramid[i].shape[0]))

        # Laplacian = current level - expanded next level
        laplacian = gaussian_pyramid[i] - expanded
        laplacian_pyramid.append(laplacian)

    # Add the top of Gaussian pyramid as the last level
    laplacian_pyramid.append(gaussian_pyramid[-1])

    return laplacian_pyramid
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def visualize_laplacian_pyramid(image, levels=4):
    """
    Enhanced visualization of Laplacian pyramid
    """
    # Build Gaussian pyramid
    G = [image.astype(np.float64)]
    for i in range(levels):
        G.append(cv2.pyrDown(G[-1]))

    # Build Laplacian pyramid
    L = []
    for i in range(levels):
        expanded = cv2.pyrUp(G[i + 1])
        if expanded.shape != G[i].shape:
            expanded = cv2.resize(expanded, (G[i].shape[1], G[i].shape[0]))
        L.append(G[i] - expanded)
    L.append(G[-1])

    # Create visualization
    plt.figure(figsize=(20, 12))

    # Row 1: Gaussian pyramid
    for i, g in enumerate(G):
        plt.subplot(3, levels+1, i+1)
        display = np.clip(g, 0, 255).astype(np.uint8)
        plt.imshow(cv2.cvtColor(display, cv2.COLOR_BGR2RGB))
        plt.title(f'G{i}: {g.shape[1]}x{g.shape[0]}')
        plt.axis('off')

    # Row 2: Laplacian pyramid (normalized)
    for i, lap in enumerate(L):
        plt.subplot(3, levels+1, levels+2+i)

        if i < levels:
            # Normalize to [0, 255] for display
            lap_norm = lap - lap.min()
            if lap_norm.max() > 0:
                lap_norm = lap_norm / lap_norm.max() * 255
            display = lap_norm.astype(np.uint8)
        else:
            display = np.clip(lap, 0, 255).astype(np.uint8)

    - Progressive blur & downsample

    Row 2: Laplacian Levels
    - Difference between adjacent
      Gaussian levels

    Row 3: Absolute Values
    - Highlights edges and textures
    - Higher values = more detail
    """

             verticalalignment='center', transform=plt.gca().transAxes)
    plt.axis('off')

    plt.tight_layout()
    plt.savefig('laplacian_visualization.png', dpi=150, bbox_inches='tight')
```

## 3. 从拉普拉斯金字塔重建图像

### 3.1 重建原理

从拉普拉斯金字塔重建原始图像的过程：

$$G_i = L_i + \text{expand}(G_{i+1})$$

从最顶层开始，逐层向下重建：

```
L₃ (顶层) ─────────────→ G₃
                          ↓ expand + L₂
L₂ ───────────────────→ G₂
                          ↓ expand + L₁
L₁ ───────────────────→ G₁
                          ↓ expand + L₀
L₀ ───────────────────→ G₀ (重建图像)
```

### 示例3：完美重建

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def reconstruct_from_laplacian(laplacian_pyramid):
    """
    Reconstruct image from Laplacian pyramid

    Parameters:
        laplacian_pyramid: List of Laplacian levels

    Returns:
        Reconstructed image
    """
    # Start from the top (smallest) level
    reconstructed = laplacian_pyramid[-1].copy()

    # Work down the pyramid
    for i in range(len(laplacian_pyramid) - 2, -1, -1):
        # Expand current reconstruction
        expanded = cv2.pyrUp(reconstructed)

        # Match size with current Laplacian level
        if expanded.shape != laplacian_pyramid[i].shape:
            expanded = cv2.resize(expanded,
                                   (laplacian_pyramid[i].shape[1],
                                    laplacian_pyramid[i].shape[0]))

        # Add Laplacian details
        reconstructed = expanded + laplacian_pyramid[i]

    return reconstructed
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def verify_perfect_reconstruction():
    """
    Verify that Laplacian pyramid provides perfect reconstruction
    """
    # Test with different image types
    test_cases = []

    # Case 1: Random noise image
    random_img = np.random.randint(0, 256, (128, 128, 3), dtype=np.uint8)
    test_cases.append(('Random Noise', random_img))

    # Case 2: Gradient image
    gradient_img = np.zeros((128, 128, 3), dtype=np.uint8)
    for i in range(128):
        gradient_img[:, i] = [i*2, 128, 255-i*2]
    test_cases.append(('Gradient', gradient_img))

    # Case 3: Sharp edges
    edge_img = np.zeros((128, 128, 3), dtype=np.uint8)
    edge_img[:, :64] = [255, 255, 255]
    edge_img[:, 64:] = [0, 0, 0]
    test_cases.append(('Sharp Edge', edge_img))

    # Case 4: Complex image
    complex_img = np.random.randint(50, 200, (128, 128, 3), dtype=np.uint8)
    cv2.circle(complex_img, (64, 64), 30, (255, 0, 0), -1)
    cv2.rectangle(complex_img, (80, 80), (120, 120), (0, 255, 0), -1)
    test_cases.append(('Complex', complex_img))

    results = []

    for name, img in test_cases:
        # Build and reconstruct
        lap_pyr = build_laplacian_pyramid(img, levels=4)
        reconstructed = reconstruct_from_laplacian(lap_pyr)
        reconstructed = np.clip(reconstructed, 0, 255).astype(np.uint8)

        # Calculate metrics
        error = cv2.absdiff(img, reconstructed)
        max_err = np.max(error)
        mean_err = np.mean(error)
        psnr = cv2.PSNR(img, reconstructed) if max_err > 0 else float('inf')

        results.append({
            'name': name,
            'original': img,
            'reconstructed': reconstructed,
            'error': error,
            'max_error': max_err,
            'mean_error': mean_err,
            'psnr': psnr

    plt.figure(figsize=(20, 12))

        plt.title(f"Mean: {r['mean_error']:.4f}")
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def laplacian_sharpen(image, levels=4, sharpen_factor=1.5):
    """
    Sharpen image by enhancing Laplacian pyramid details

    Parameters:
        image: Input image
        levels: Number of pyramid levels
        sharpen_factor: Factor to enhance details (>1 sharpens)

    Returns:
        Sharpened image
    """
    # Build Laplacian pyramid
    lap_pyr = build_laplacian_pyramid(image, levels)

    # Enhance detail layers (not the top level)
    enhanced_pyr = []
    for i, level in enumerate(lap_pyr):
        if i < len(lap_pyr) - 1:
            # Enhance details
            enhanced = level * sharpen_factor
        else:
            # Keep top level unchanged
            enhanced = level
        enhanced_pyr.append(enhanced)

    # Reconstruct
    result = reconstruct_from_laplacian(enhanced_pyr)
    result = np.clip(result, 0, 255).astype(np.uint8)

    return result

# Load image
image = cv2.imread('image.jpg')
if image is None:
    # Create slightly blurred test image
    image = np.zeros((256, 256, 3), dtype=np.uint8)

    image = cv2.GaussianBlur(image, (5, 5), 0)

# Test different sharpening factors
factors = [0.5, 1.0, 1.5, 2.0, 3.0]

# Display
```

### 示例6：图像压缩概念

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def laplacian_compression_demo(image, levels=4, threshold=10):
    """
    Demonstrate image compression using Laplacian pyramid

    Parameters:
        image: Input image
        levels: Number of pyramid levels
        threshold: Values below this are set to zero

    Returns:
        Compressed pyramid, reconstructed image, compression stats
    """
    # Build Laplacian pyramid
    lap_pyr = build_laplacian_pyramid(image, levels)

    # Threshold small values (lossy compression concept)
    compressed_pyr = []
    stats = []

    for i, level in enumerate(lap_pyr):
        # Count non-zero before
        total_values = level.size
        original_nonzero = np.sum(np.abs(level) > 0)

        if i < len(lap_pyr) - 1:
            # Threshold detail levels
            compressed = level.copy()
            compressed[np.abs(compressed) < threshold] = 0
        else:
            # Keep top level
            compressed = level

        # Count non-zero after
        compressed_nonzero = np.sum(np.abs(compressed) > 0)
        compression_ratio = 1 - (compressed_nonzero / total_values)

        stats.append({
            'level': i,
            'total': total_values,
            'original_nonzero': original_nonzero,
            'compressed_nonzero': compressed_nonzero,
            'ratio': compression_ratio
        })

        compressed_pyr.append(compressed)

    # Reconstruct from compressed pyramid
    reconstructed = reconstruct_from_laplacian(compressed_pyr)
    reconstructed = np.clip(reconstructed, 0, 255).astype(np.uint8)

    return compressed_pyr, reconstructed, stats

# Load image
image = cv2.imread('image.jpg')
if image is None:
    image = np.zeros((256, 256, 3), dtype=np.uint8)
    cv2.rectangle(image, (30, 30), (226, 226), (100, 150, 200), -1)
    cv2.circle(image, (128, 128), 60, (200, 100, 50), -1)
    noise = np.random.randint(0, 20, image.shape, dtype=np.uint8)
    image = cv2.add(image, noise)

thresholds = [0, 5, 10, 20, 40]

    comp_pyr, recon, stats = laplacian_compression_demo(image, threshold=thresh)
    error = cv2.absdiff(image, recon)
    psnr = cv2.PSNR(image, recon) if np.max(error) > 0 else float('inf')

    results.append({
        'threshold': thresh,
        'reconstructed': recon,
        'error': error,
        'psnr': psnr,
        'stats': stats
    })

# Display
plt.figure(figsize=(20, 12))

# Reconstructed images
for i, r in enumerate(results):
    plt.subplot(3, 5, i+1)
    plt.imshow(cv2.cvtColor(r['reconstructed'], cv2.COLOR_BGR2RGB))
    plt.title(f'Threshold = {r["threshold"]}\nPSNR: {r["psnr"]:.1f} dB')
    plt.axis('off')

# Error images
for i, r in enumerate(results):
    plt.subplot(3, 5, 6+i)
    error_display = cv2.convertScaleAbs(r['error'], alpha=5)
    plt.imshow(cv2.cvtColor(error_display, cv2.COLOR_BGR2RGB))
```

```python
def multiscale_edge_extraction(image, levels=4):
    """
    Extract edges at multiple scales using Laplacian pyramid

    Parameters:
        image: Input image
        levels: Number of pyramid levels

    Returns:
        Edge maps at each scale
    """
    # Convert to grayscale
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image

    # Build Laplacian pyramid
    lap_pyr = build_laplacian_pyramid(gray, levels)

    # Extract edges from each level
    edge_maps = []
    for i, level in enumerate(lap_pyr[:-1]):  # Exclude top level
        # Laplacian already contains edge information
        # Take absolute value for edge magnitude
        edges = np.abs(level)

        # Normalize to 0-255
        edges = (edges / edges.max() * 255).astype(np.uint8)
        edge_maps.append(edges)

    # Combined edge map (weighted sum)
    combined = np.zeros_like(gray, dtype=np.float64)
    for i, edges in enumerate(edge_maps):
        # Resize to original size
        resized = cv2.resize(edges, (gray.shape[1], gray.shape[0]))
        # Weight by level (fine details have higher weight)
        weight = 1.0 / (i + 1)
        combined += resized * weight

    combined = (combined / combined.max() * 255).astype(np.uint8)

    return edge_maps, combined
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

class LaplacianPyramidAnalyzer:
    """
    Comprehensive Laplacian pyramid analysis tool
    """

    def __init__(self, image, levels=5):
        self.original = image
        self.levels = levels
        self.gaussian_pyr = None
        self.laplacian_pyr = None
        self._build_pyramids()

    def _build_pyramids(self):
        """Build both Gaussian and Laplacian pyramids"""
        # Gaussian pyramid
        self.gaussian_pyr = [self.original.astype(np.float64)]
        current = self.original.astype(np.float64)

        for i in range(self.levels - 1):
            current = cv2.pyrDown(current)
            self.gaussian_pyr.append(current)

        # Laplacian pyramid
        self.laplacian_pyr = []
        for i in range(self.levels - 1):
            expanded = cv2.pyrUp(self.gaussian_pyr[i + 1])
            if expanded.shape != self.gaussian_pyr[i].shape:
                expanded = cv2.resize(expanded,
                                       (self.gaussian_pyr[i].shape[1],
                                        self.gaussian_pyr[i].shape[0]))
            self.laplacian_pyr.append(self.gaussian_pyr[i] - expanded)
        self.laplacian_pyr.append(self.gaussian_pyr[-1])

    def get_statistics(self):
        """Calculate statistics for each level"""
        stats = []
        for i, level in enumerate(self.laplacian_pyr):
            stats.append({
                'level': i,
                'shape': level.shape,
                'mean': np.mean(level),
                'std': np.std(level),
                'min': np.min(level),
                'max': np.max(level),
                'energy': np.sum(level ** 2)
            })
        return stats

    def reconstruct(self, level_factors=None):
        """Reconstruct with optional level modifications"""
        if level_factors is None:
            level_factors = [1.0] * self.levels

        # Modify pyramid
        modified = []
        for i, level in enumerate(self.laplacian_pyr):
            if i < len(level_factors):
                modified.append(level * level_factors[i])
            else:
                modified.append(level)

        # Reconstruct
        result = modified[-1].copy()
        for i in range(len(modified) - 2, -1, -1):
            expanded = cv2.pyrUp(result)
            if expanded.shape != modified[i].shape:
                expanded = cv2.resize(expanded,
                                       (modified[i].shape[1], modified[i].shape[0]))
            result = expanded + modified[i]

        return np.clip(result, 0, 255).astype(np.uint8)

    def visualize(self):
        """Create comprehensive visualization"""
        fig, axes = plt.subplots(3, self.levels, figsize=(18, 12))

        # Gaussian pyramid
        for i, g in enumerate(self.gaussian_pyr):
            ax = axes[0, i]
            display = np.clip(g, 0, 255).astype(np.uint8)
            if len(display.shape) == 3:
                ax.imshow(cv2.cvtColor(display, cv2.COLOR_BGR2RGB))
            else:
                ax.imshow(display, cmap='gray')
            ax.set_title(f'G{i}: {g.shape[1]}x{g.shape[0]}')
            ax.axis('off')

        # Laplacian pyramid
        for i, lap in enumerate(self.laplacian_pyr):
            ax = axes[1, i]
            if i < len(self.laplacian_pyr) - 1:
                display = lap - lap.min()
                if display.max() > 0:
                    display = display / display.max() * 255
                display = display.astype(np.uint8)
            else:
                display = np.clip(lap, 0, 255).astype(np.uint8)

            if len(display.shape) == 3:
                ax.imshow(cv2.cvtColor(display, cv2.COLOR_BGR2RGB))
            else:
                ax.imshow(display, cmap='gray')
            ax.set_title(f'L{i}: {lap.shape[1]}x{lap.shape[0]}')
            ax.axis('off')

        # Statistics
        stats = self.get_statistics()
        for i, s in enumerate(stats):
            ax = axes[2, i]
            text = f"Mean: {s['mean']:.2f}\nStd: {s['std']:.2f}\n"
            text += f"Range: [{s['min']:.1f}, {s['max']:.1f}]"
            ax.text(0.5, 0.5, text, ha='center', va='center',
                    fontsize=10, family='monospace',
                    transform=ax.transAxes)
            ax.set_title(f'Stats L{i}')
            ax.axis('off')

        plt.tight_layout()
        return fig
```

```python
# 构建拉普拉斯金字塔
def build_laplacian(image, levels):
    G = [image.astype(np.float64)]
    for i in range(levels - 1):
        G.append(cv2.pyrDown(G[-1]))

    L = []
    for i in range(levels - 1):
        exp = cv2.pyrUp(G[i+1])
        if exp.shape != G[i].shape:
            exp = cv2.resize(exp, (G[i].shape[1], G[i].shape[0]))
        L.append(G[i] - exp)
    L.append(G[-1])
    return L

# 重建
def reconstruct(L):
    result = L[-1].copy()
    for i in range(len(L) - 2, -1, -1):
        exp = cv2.pyrUp(result)
        if exp.shape != L[i].shape:
            exp = cv2.resize(exp, (L[i].shape[1], L[i].shape[0]))
        result = exp + L[i]
    return np.clip(result, 0, 255).astype(np.uint8)
```
