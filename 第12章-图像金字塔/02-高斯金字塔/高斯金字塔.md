# 高斯金字塔（Gaussian Pyramid）

## 1. 高斯金字塔概述

高斯金字塔是最常用的图像金字塔类型，它通过对图像进行高斯模糊和下采样来构建多分辨率表示。这种方法能够有效避免下采样时产生的混叠效应。

### 1.1 高斯金字塔的定义

高斯金字塔是一系列分辨率逐渐降低的图像集合：

$$G_0 \rightarrow G_1 \rightarrow G_2 \rightarrow \cdots \rightarrow G_n$$

其中：
- $G_0$：原始图像（最高分辨率）
- $G_n$：最顶层图像（最低分辨率）
- 每一层的尺寸是上一层的一半

### 1.2 构建过程

```
G₀ (原始图像)
 │
 ├─→ 高斯滤波 ─→ 下采样 ─→ G₁
 │                           │
 │                           ├─→ 高斯滤波 ─→ 下采样 ─→ G₂
 │                           │                           │
 │                           │                           └─→ ...
 │
 └─ 保留原始分辨率
```

### 1.3 高斯金字塔的特点

| 特点 | 说明 |
|------|------|
| **低通特性** | 每层保留低频信息，丢弃高频细节 |
| **尺寸递减** | 每层尺寸减半 |
| **存储高效** | 总存储约为原图的4/3 |
| **无混叠** | 高斯滤波避免下采样混叠 |

### 示例1：基本高斯金字塔构建

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def build_gaussian_pyramid(image, levels=5):
    """
    Build a Gaussian pyramid

    Parameters:
        image: Input image
        levels: Number of pyramid levels

    Returns:
        List of images forming the pyramid
    """
    pyramid = [image]
    current = image

    for i in range(levels - 1):
        # Apply pyrDown to get next level
        current = cv2.pyrDown(current)
        pyramid.append(current)
        print(f"Level {i+1}: {current.shape}")

    return pyramid

# Load or create test image
image = cv2.imread('image.jpg')
if image is None:
    # Create colorful test image
    image = np.zeros((512, 512, 3), dtype=np.uint8)
    # Add some patterns
    cv2.rectangle(image, (50, 50), (200, 200), (0, 0, 255), -1)
    cv2.circle(image, (350, 150), 80, (0, 255, 0), -1)
    cv2.ellipse(image, (256, 350), (100, 50), 45, 0, 360, (255, 0, 0), -1)
    # Add text
    cv2.putText(image, 'Gaussian Pyramid', (100, 450),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (255, 255, 255), 2)

print(f"Original image: {image.shape}")

# Build pyramid
pyramid = build_gaussian_pyramid(image, levels=6)

# Visualize pyramid
plt.figure(figsize=(18, 8))

for i, level in enumerate(pyramid):
    plt.subplot(2, 6, i+1)
    plt.imshow(cv2.cvtColor(level, cv2.COLOR_BGR2RGB))
    plt.title(f'Level {i}\n{level.shape[1]}x{level.shape[0]}')
    plt.axis('off')

# Create stacked visualization
# All levels resized to fit in one row
max_height = pyramid[0].shape[0]
stacked_width = sum(p.shape[1] for p in pyramid) + 10 * (len(pyramid) - 1)
stacked = np.ones((max_height, stacked_width, 3), dtype=np.uint8) * 255

x_offset = 0
for level in pyramid:
    h, w = level.shape[:2]
    y_offset = max_height - h
    stacked[y_offset:y_offset+h, x_offset:x_offset+w] = level
    x_offset += w + 10

plt.subplot(2, 1, 2)
plt.imshow(cv2.cvtColor(stacked, cv2.COLOR_BGR2RGB))
plt.title('Gaussian Pyramid (Stacked View)')
plt.axis('off')

plt.tight_layout()
plt.savefig('gaussian_pyramid_basic.png', dpi=150, bbox_inches='tight')
plt.show()

print("\nGaussian pyramid construction completed!")
print(f"Total levels: {len(pyramid)}")
```

### 示例2：金字塔的内存和计算分析

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def analyze_pyramid_efficiency():
    """
    Analyze memory and computational aspects of Gaussian pyramid
    """
    # Test with different image sizes
    test_sizes = [(256, 256), (512, 512), (1024, 1024), (2048, 2048)]

    results = []

    for width, height in test_sizes:
        # Create test image
        image = np.random.randint(0, 256, (height, width, 3), dtype=np.uint8)

        # Build pyramid
        pyramid = [image]
        current = image
        levels = int(np.log2(min(width, height)))

        for i in range(levels - 1):
            current = cv2.pyrDown(current)
            pyramid.append(current)

        # Calculate statistics
        original_pixels = width * height
        total_pixels = sum(p.shape[0] * p.shape[1] for p in pyramid)
        memory_ratio = total_pixels / original_pixels

        results.append({
            'size': f'{width}x{height}',
            'levels': len(pyramid),
            'original_pixels': original_pixels,
            'total_pixels': total_pixels,
            'memory_ratio': memory_ratio,
            'overhead_percent': (memory_ratio - 1) * 100
        })

    # Display results
    print("="*70)
    print("Gaussian Pyramid Efficiency Analysis")
    print("="*70)
    print(f"{'Size':<12} {'Levels':<8} {'Original':<12} {'Total':<12} {'Ratio':<8} {'Overhead':<10}")
    print("-"*70)

    for r in results:
        print(f"{r['size']:<12} {r['levels']:<8} {r['original_pixels']:<12,} "
              f"{r['total_pixels']:<12,} {r['memory_ratio']:.3f}    {r['overhead_percent']:.1f}%")

    # Theoretical analysis
    print("\n" + "="*70)
    print("Theoretical Analysis:")
    print("For infinite levels, total storage = 1 + 1/4 + 1/16 + ... = 4/3 ≈ 1.333")
    print("Overhead = 33.3%")
    print("="*70)

    # Visualize
    plt.figure(figsize=(14, 5))

    # Memory ratio
    plt.subplot(1, 3, 1)
    sizes = [r['size'] for r in results]
    ratios = [r['memory_ratio'] for r in results]
    bars = plt.bar(sizes, ratios, color='steelblue', edgecolor='black')
    plt.axhline(y=4/3, color='red', linestyle='--', label='Theoretical limit (4/3)')
    plt.xlabel('Image Size')
    plt.ylabel('Memory Ratio')
    plt.title('Pyramid Memory Ratio')
    plt.legend()
    plt.ylim(0, 1.5)

    # Add value labels
    for bar, ratio in zip(bars, ratios):
        plt.text(bar.get_x() + bar.get_width()/2, bar.get_height() + 0.02,
                 f'{ratio:.3f}', ha='center', va='bottom', fontsize=9)

    # Pixels per level
    plt.subplot(1, 3, 2)
    # Use 512x512 as example
    image_512 = np.random.randint(0, 256, (512, 512, 3), dtype=np.uint8)
    pyramid_512 = [image_512]
    current = image_512
    for i in range(8):
        current = cv2.pyrDown(current)
        pyramid_512.append(current)

    level_pixels = [p.shape[0] * p.shape[1] for p in pyramid_512]
    level_nums = list(range(len(pyramid_512)))

    plt.bar(level_nums, level_pixels, color='coral', edgecolor='black')
    plt.xlabel('Pyramid Level')
    plt.ylabel('Number of Pixels')
    plt.title('Pixels per Level (512x512)')
    plt.yscale('log')

    # Cumulative storage
    plt.subplot(1, 3, 3)
    cumulative = np.cumsum(level_pixels) / level_pixels[0]
    plt.plot(level_nums, cumulative, 'go-', linewidth=2, markersize=8)
    plt.axhline(y=4/3, color='red', linestyle='--', label='4/3 limit')
    plt.xlabel('Pyramid Level')
    plt.ylabel('Cumulative Ratio')
    plt.title('Cumulative Storage Ratio')
    plt.legend()
    plt.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig('pyramid_efficiency.png', dpi=150, bbox_inches='tight')
    plt.show()

analyze_pyramid_efficiency()
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt
import time

def compare_scaling_methods():
    """
    Compare pyramid-based scaling with direct resize
    """
    # Create test image with fine details
    image = cv2.imread('image.jpg')
    if image is None:
        image = np.zeros((512, 512, 3), dtype=np.uint8)
        # Add checkerboard pattern (high frequency)
        for i in range(64):
            for j in range(64):
                if (i + j) % 2 == 0:
                    cv2.rectangle(image, (i*8, j*8), ((i+1)*8, (j+1)*8), (200, 200, 200), -1)
        # Add some shapes
        cv2.circle(image, (256, 256), 100, (0, 255, 0), -1)
        cv2.rectangle(image, (50, 50), (150, 150), (255, 0, 0), -1)

    target_size = (image.shape[1] // 8, image.shape[0] // 8)  # 1/8 size

    # Method 1: Direct resize (INTER_AREA - good for shrinking)
    start = time.time()
    direct_area = cv2.resize(image, target_size, interpolation=cv2.INTER_AREA)
    time_area = time.time() - start

    # Method 2: Direct resize (INTER_LINEAR)
    start = time.time()
    direct_linear = cv2.resize(image, target_size, interpolation=cv2.INTER_LINEAR)
    time_linear = time.time() - start

    # Method 3: Direct resize (INTER_NEAREST)
    start = time.time()
    direct_nearest = cv2.resize(image, target_size, interpolation=cv2.INTER_NEAREST)
    time_nearest = time.time() - start

    # Method 4: Pyramid-based (3 levels of pyrDown for ~1/8)
    start = time.time()
    pyramid = image
    for _ in range(3):
        pyramid = cv2.pyrDown(pyramid)
    # Adjust to exact target size if needed
    pyramid_result = cv2.resize(pyramid, target_size, interpolation=cv2.INTER_LINEAR)
    time_pyramid = time.time() - start

    # Display comparison
    plt.figure(figsize=(18, 12))

    # Original
    plt.subplot(2, 4, 1)
    plt.imshow(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
    plt.title(f'Original\n{image.shape[1]}x{image.shape[0]}')
    plt.axis('off')

    # Direct INTER_AREA
    plt.subplot(2, 4, 2)
    plt.imshow(cv2.cvtColor(direct_area, cv2.COLOR_BGR2RGB))
    plt.title(f'INTER_AREA\n{direct_area.shape[1]}x{direct_area.shape[0]}')
    plt.axis('off')

    # Direct INTER_LINEAR
    plt.subplot(2, 4, 3)
    plt.imshow(cv2.cvtColor(direct_linear, cv2.COLOR_BGR2RGB))
    plt.title(f'INTER_LINEAR\n{direct_linear.shape[1]}x{direct_linear.shape[0]}')
    plt.axis('off')

    # Direct INTER_NEAREST
    plt.subplot(2, 4, 4)
    plt.imshow(cv2.cvtColor(direct_nearest, cv2.COLOR_BGR2RGB))
    plt.title(f'INTER_NEAREST\n{direct_nearest.shape[1]}x{direct_nearest.shape[0]}')
    plt.axis('off')

    # Pyramid result
    plt.subplot(2, 4, 5)
    plt.imshow(cv2.cvtColor(pyramid_result, cv2.COLOR_BGR2RGB))
    plt.title(f'Pyramid\n{pyramid_result.shape[1]}x{pyramid_result.shape[0]}')
    plt.axis('off')

    # Upscaled comparisons (to see differences better)
    upscale_size = (256, 256)

    plt.subplot(2, 4, 6)
    up_area = cv2.resize(direct_area, upscale_size, interpolation=cv2.INTER_NEAREST)
    plt.imshow(cv2.cvtColor(up_area, cv2.COLOR_BGR2RGB))
    plt.title('INTER_AREA (upscaled)')
    plt.axis('off')

    plt.subplot(2, 4, 7)
    up_nearest = cv2.resize(direct_nearest, upscale_size, interpolation=cv2.INTER_NEAREST)
    plt.imshow(cv2.cvtColor(up_nearest, cv2.COLOR_BGR2RGB))
    plt.title('INTER_NEAREST (upscaled)')
    plt.axis('off')

    plt.subplot(2, 4, 8)
    up_pyramid = cv2.resize(pyramid_result, upscale_size, interpolation=cv2.INTER_NEAREST)
    plt.imshow(cv2.cvtColor(up_pyramid, cv2.COLOR_BGR2RGB))
    plt.title('Pyramid (upscaled)')
    plt.axis('off')

    plt.tight_layout()
    plt.savefig('scaling_comparison.png', dpi=150, bbox_inches='tight')
    plt.show()

    # Print timing
    print("\nTiming Comparison (for reference):")
    print(f"  INTER_AREA: {time_area*1000:.2f} ms")
    print(f"  INTER_LINEAR: {time_linear*1000:.2f} ms")
    print(f"  INTER_NEAREST: {time_nearest*1000:.2f} ms")
    print(f"  Pyramid: {time_pyramid*1000:.2f} ms")
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def multiscale_feature_visualization():
    """
    Visualize features at different pyramid levels
    """
    # Create or load image
    image = cv2.imread('image.jpg')
    if image is None:
        image = np.zeros((512, 512, 3), dtype=np.uint8)
        # Large features
        cv2.rectangle(image, (50, 50), (250, 250), (0, 100, 200), -1)
        cv2.rectangle(image, (260, 260), (460, 460), (200, 100, 0), -1)
        # Medium features
        for i in range(4):
            cv2.circle(image, (100 + i*100, 350), 30, (0, 255, 0), -1)
        # Small features
        for i in range(16):
            cv2.circle(image, (50 + i*30, 480), 8, (255, 255, 0), -1)
        # Very small features
        for i in range(32):
            cv2.circle(image, (16 + i*16, 500), 3, (255, 0, 255), -1)

    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    # Build pyramid
    pyramid = [gray]
    current = gray
    for i in range(4):
        current = cv2.pyrDown(current)
        pyramid.append(current)

    # Apply edge detection at each level
    edges = []
    for level in pyramid:
        edge = cv2.Canny(level, 50, 150)
        edges.append(edge)

    # Apply corner detection at each level
    corners = []
    for level in pyramid:
        corner = cv2.cornerHarris(level.astype(np.float32), 2, 3, 0.04)
        corner = cv2.dilate(corner, None)
        corner_img = np.zeros_like(level)
        corner_img[corner > 0.01 * corner.max()] = 255
        corners.append(corner_img)

    # Display
    plt.figure(figsize=(20, 12))

    # Original pyramid levels
    for i, level in enumerate(pyramid):
        plt.subplot(3, 5, i+1)
        plt.imshow(level, cmap='gray')
        plt.title(f'Level {i}\n{level.shape[1]}x{level.shape[0]}')
        plt.axis('off')

    # Edge detection at each level
    for i, edge in enumerate(edges):
        plt.subplot(3, 5, i+6)
        plt.imshow(edge, cmap='gray')
        plt.title(f'Edges L{i}')
        plt.axis('off')

    # Corner detection at each level

        plt.subplot(3, 5, i+11)

        plt.title(f'Corners L{i}')

    plt.tight_layout()
    plt.savefig('multiscale_features.png', dpi=150, bbox_inches='tight')

    # Feature count analysis

        corner_count = np.sum(corner > 0)
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def pyramid_template_matching():
    """
    Use pyramid for faster template matching
    """
    # Create scene with objects
    scene = np.ones((600, 800, 3), dtype=np.uint8) * 200

    # Add various objects
    cv2.rectangle(scene, (100, 100), (180, 180), (0, 0, 255), -1)  # Target
    cv2.rectangle(scene, (400, 200), (480, 280), (0, 0, 255), -1)  # Another match
    cv2.circle(scene, (300, 400), 60, (0, 255, 0), -1)  # Different object
    cv2.rectangle(scene, (600, 400), (680, 480), (0, 0, 255), -1)  # Another match

    # Add noise and texture
    noise = np.random.randint(0, 30, scene.shape, dtype=np.uint8)
    scene = cv2.add(scene, noise)

    # Create template (red square)
    template = np.ones((80, 80, 3), dtype=np.uint8) * 200
    cv2.rectangle(template, (0, 0), (80, 80), (0, 0, 255), -1)
    template = cv2.add(template, np.random.randint(0, 30, template.shape, dtype=np.uint8))

    # Standard template matching
    result_standard = cv2.matchTemplate(scene, template, cv2.TM_CCOEFF_NORMED)

    # Pyramid-based matching
    def pyramid_match(scene, template, levels=3):
        """Coarse-to-fine template matching"""
        # Build pyramids
        scene_pyr = [scene]
        template_pyr = [template]

        for i in range(levels - 1):
            scene_pyr.append(cv2.pyrDown(scene_pyr[-1]))
            template_pyr.append(cv2.pyrDown(template_pyr[-1]))

        # Start from coarsest level
        candidates = []
        for level in range(levels - 1, -1, -1):
            s = scene_pyr[level]
            t = template_pyr[level]

            if t.shape[0] > s.shape[0] or t.shape[1] > s.shape[1]:
                continue

            result = cv2.matchTemplate(s, t, cv2.TM_CCOEFF_NORMED)

            # Find peaks
            threshold = 0.7
            locations = np.where(result >= threshold)

            for pt in zip(*locations[::-1]):
                # Scale back to original coordinates
                scale = 2 ** level
                original_pt = (pt[0] * scale, pt[1] * scale)
                score = result[pt[1], pt[0]]
                candidates.append((original_pt, score, level))

        return candidates

    candidates = pyramid_match(scene, template, levels=3)

    # Display results
    plt.figure(figsize=(18, 10))

    # Scene with matches
    plt.subplot(2, 3, 1)
    scene_marked = scene.copy()
    for (x, y), score, level in candidates:
        cv2.rectangle(scene_marked, (x, y),
                      (x + template.shape[1], y + template.shape[0]),
                      (0, 255, 0), 2)
        cv2.putText(scene_marked, f'{score:.2f}', (x, y-5),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
    plt.imshow(cv2.cvtColor(scene_marked, cv2.COLOR_BGR2RGB))
    plt.title('Scene with Matches')
    plt.axis('off')

    # Template
    plt.subplot(2, 3, 2)
    plt.imshow(cv2.cvtColor(template, cv2.COLOR_BGR2RGB))
    plt.title('Template')
    plt.axis('off')

    # Standard matching result
    plt.subplot(2, 3, 3)
    plt.imshow(result_standard, cmap='hot')
    plt.colorbar()
    plt.title('Standard Matching Result')

    # Pyramid levels
    scene_pyr = [scene]
    template_pyr = [template]
    for i in range(2):
        scene_pyr.append(cv2.pyrDown(scene_pyr[-1]))
        template_pyr.append(cv2.pyrDown(template_pyr[-1]))

    for i in range(3):
        plt.subplot(2, 3, 4+i)
        plt.imshow(cv2.cvtColor(scene_pyr[i], cv2.COLOR_BGR2RGB))
        plt.title(f'Pyramid Level {i}')
        plt.axis('off')

    plt.tight_layout()
    plt.savefig('pyramid_matching.png', dpi=150, bbox_inches='tight')
    plt.show()

    print("\nMatching Results:")
    print("-" * 50)
    for i, ((x, y), score, level) in enumerate(candidates):
        print(f"Match {i+1}: Position=({x}, {y}), Score={score:.3f}, Found at Level={level}")

pyramid_template_matching()
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.animation import FuncAnimation

def create_pyramid_animation():
    """
    Create animation showing pyramid construction
    """
    # Create test image
    image = cv2.imread('image.jpg')
    if image is None:
        image = np.zeros((256, 256, 3), dtype=np.uint8)
        cv2.circle(image, (128, 128), 80, (0, 255, 0), -1)
        cv2.rectangle(image, (30, 30), (100, 100), (255, 0, 0), -1)
        cv2.rectangle(image, (156, 156), (226, 226), (0, 0, 255), -1)

    # Build pyramid
    pyramid = [image]
    current = image
    for i in range(5):
        current = cv2.pyrDown(current)
        pyramid.append(current)

    # Create static visualization of pyramid building process
    fig, axes = plt.subplots(2, 6, figsize=(18, 8))

    # Show each level
    for i, level in enumerate(pyramid):
        if i < 6:
            axes[0, i].imshow(cv2.cvtColor(level, cv2.COLOR_BGR2RGB))
            axes[0, i].set_title(f'Level {i}: {level.shape[1]}x{level.shape[0]}')
            axes[0, i].axis('off')

    # Show the process: original → blur → downsample
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    # Step 1: Original
    axes[1, 0].imshow(gray, cmap='gray')
    axes[1, 0].set_title('Step 1: Original')
    axes[1, 0].axis('off')

    # Step 2: Apply Gaussian blur
    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    axes[1, 1].imshow(blurred, cmap='gray')
    axes[1, 1].set_title('Step 2: Gaussian Blur')
    axes[1, 1].axis('off')

    # Step 3: Downsample
    downsampled = blurred[::2, ::2]
    axes[1, 2].imshow(downsampled, cmap='gray')
    axes[1, 2].set_title('Step 3: Downsample')
    axes[1, 2].axis('off')

    # Repeat for next level
    blurred2 = cv2.GaussianBlur(downsampled, (5, 5), 0)
    axes[1, 3].imshow(blurred2, cmap='gray')
    axes[1, 3].set_title('Step 4: Blur Again')
    axes[1, 3].axis('off')

    downsampled2 = blurred2[::2, ::2]
    axes[1, 4].imshow(downsampled2, cmap='gray')
    axes[1, 4].set_title('Step 5: Downsample')
    axes[1, 4].axis('off')

    # Compare OpenCV result
    opencv_result = cv2.pyrDown(cv2.pyrDown(gray))
    axes[1, 5].imshow(opencv_result, cmap='gray')
    axes[1, 5].set_title('OpenCV pyrDown x2')
    axes[1, 5].axis('off')

    plt.tight_layout()
    plt.savefig('pyramid_animation.png', dpi=150, bbox_inches='tight')
    plt.show()

    print("Pyramid construction visualization completed!")
```

```python
def scale_space_analysis():

                cv2.circle(image, (30 + i*20, 160 + j*8), 2, 255, -1)

    laplacians = []
    for i, level in enumerate(pyramid):
        lap = cv2.Laplacian(level, cv2.CV_64F)
        laplacians.append(lap)

    # Calculate DoG (Difference of Gaussians)
    dogs = []
    for i in range(len(pyramid) - 1):
        # Upsample smaller image to match larger
        larger = pyramid[i]
        smaller = pyramid[i + 1]
        upsampled = cv2.pyrUp(smaller)

        # Resize to match exactly
        if upsampled.shape != larger.shape:
            upsampled = cv2.resize(upsampled, (larger.shape[1], larger.shape[0]))

        dog = larger - upsampled
        dogs.append(dog)

    # Display
    plt.figure(figsize=(20, 12))

    # Pyramid levels
    for i, level in enumerate(pyramid):
        plt.subplot(3, 5, i+1)

        plt.title(f'Scale {i}')

    for i, lap in enumerate(laplacians):

        plt.title(f'Laplacian {i}')

    for i, dog in enumerate(dogs):

    print("-" * 50)

        energy = np.sum(np.abs(lap))
        max_response = np.max(np.abs(lap))
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def antialiased_resize():
    """
    Demonstrate antialiased resizing using pyramid
    """
    # Create image with high-frequency content
    image = np.zeros((512, 512, 3), dtype=np.uint8)

    # Checkerboard pattern
    for i in range(64):
        for j in range(64):
            if (i + j) % 2 == 0:
                cv2.rectangle(image, (i*8, j*8), ((i+1)*8, (j+1)*8),
                              (255, 255, 255), -1)

    # Add some colored shapes
    cv2.circle(image, (256, 256), 100, (0, 255, 0), 3)
    cv2.rectangle(image, (100, 100), (200, 200), (255, 0, 0), 3)

    target_size = (64, 64)

    # Method 1: Direct resize (naive)
    naive_resize = cv2.resize(image, target_size, interpolation=cv2.INTER_LINEAR)

    # Method 2: Direct resize with INTER_AREA
    area_resize = cv2.resize(image, target_size, interpolation=cv2.INTER_AREA)

    # Method 3: Pyramid-based (progressive downsampling)
    def pyramid_resize(img, target):
        current = img.copy()
        current_size = (img.shape[1], img.shape[0])

        # Progressive downsampling
        while current_size[0] > target[0] * 2 and current_size[1] > target[1] * 2:
            current = cv2.pyrDown(current)
            current_size = (current.shape[1], current.shape[0])

        # Final resize
        return cv2.resize(current, target, interpolation=cv2.INTER_LINEAR)

    pyramid_resize_result = pyramid_resize(image, target_size)

    # Compare quality
    def calculate_aliasing_score(img):
        """Higher score = more aliasing artifacts"""
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) if len(img.shape) == 3 else img
        # Sobel gradients
        gx = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3)
        gy = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3)
        magnitude = np.sqrt(gx**2 + gy**2)
        return np.std(magnitude)

    # Display
    plt.figure(figsize=(16, 10))

    plt.subplot(2, 4, 1)
    plt.imshow(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
    plt.title(f'Original\n{image.shape[1]}x{image.shape[0]}')
    plt.axis('off')

    plt.subplot(2, 4, 2)
    plt.imshow(cv2.cvtColor(naive_resize, cv2.COLOR_BGR2RGB))
    plt.title(f'INTER_LINEAR\n{target_size}')
    plt.axis('off')

    plt.subplot(2, 4, 3)
    plt.imshow(cv2.cvtColor(area_resize, cv2.COLOR_BGR2RGB))
    plt.title(f'INTER_AREA\n{target_size}')
    plt.axis('off')

    plt.subplot(2, 4, 4)
    plt.imshow(cv2.cvtColor(pyramid_resize_result, cv2.COLOR_BGR2RGB))
    plt.title(f'Pyramid\n{target_size}')
    plt.axis('off')

    # Upscaled views
    upscale = (256, 256)

    plt.subplot(2, 4, 6)
    up_naive = cv2.resize(naive_resize, upscale, interpolation=cv2.INTER_NEAREST)
    plt.imshow(cv2.cvtColor(up_naive, cv2.COLOR_BGR2RGB))
    plt.title('INTER_LINEAR (upscaled)')
    plt.axis('off')

    plt.subplot(2, 4, 7)
    up_area = cv2.resize(area_resize, upscale, interpolation=cv2.INTER_NEAREST)
    plt.imshow(cv2.cvtColor(up_area, cv2.COLOR_BGR2RGB))
    plt.title('INTER_AREA (upscaled)')
    plt.axis('off')

    plt.subplot(2, 4, 8)
    up_pyramid = cv2.resize(pyramid_resize_result, upscale, interpolation=cv2.INTER_NEAREST)
    plt.imshow(cv2.cvtColor(up_pyramid, cv2.COLOR_BGR2RGB))
    plt.title('Pyramid (upscaled)')
    plt.axis('off')

    # Quality comparison
    plt.subplot(2, 4, 5)
    methods = ['INTER_LINEAR', 'INTER_AREA', 'Pyramid']
    scores = [
        calculate_aliasing_score(naive_resize),
        calculate_aliasing_score(area_resize),
        calculate_aliasing_score(pyramid_resize_result)
    ]
    bars = plt.bar(methods, scores, color=['red', 'orange', 'green'], edgecolor='black')
    plt.ylabel('Aliasing Score')
    plt.title('Quality Comparison\n(Lower is Better)')

    plt.tight_layout()
    plt.savefig('antialiased_resize.png', dpi=150, bbox_inches='tight')
    plt.show()

    print("\nAntialiasing Comparison:")
    print("-" * 40)
    for method, score in zip(methods, scores):
        print(f"{method}: Aliasing Score = {score:.2f}")
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def pyramid_reconstruction():
    """
    Demonstrate image reconstruction from pyramid
    """
    # Create test image
    image = cv2.imread('image.jpg')
    if image is None:
        image = np.zeros((256, 256, 3), dtype=np.uint8)
        cv2.rectangle(image, (50, 50), (200, 200), (0, 255, 0), -1)
        cv2.circle(image, (128, 128), 50, (255, 0, 0), -1)
        cv2.putText(image, 'TEST', (80, 140), cv2.FONT_HERSHEY_SIMPLEX, 1, (255, 255, 255), 2)

    # Build Gaussian pyramid
    n_levels = 5
    gaussian_pyramid = [image]
    current = image

    for i in range(n_levels - 1):
        current = cv2.pyrDown(current)
        gaussian_pyramid.append(current)

    # Reconstruct by upsampling from top
    reconstructed_levels = [gaussian_pyramid[-1]]

    for i in range(n_levels - 2, -1, -1):
        upsampled = cv2.pyrUp(reconstructed_levels[-1])

        # Match size with original level
        target_size = gaussian_pyramid[i].shape[:2][::-1]
        if upsampled.shape[:2][::-1] != target_size:
            upsampled = cv2.resize(upsampled, target_size)

        reconstructed_levels.append(upsampled)

    # Reverse to match pyramid order
    reconstructed_levels = reconstructed_levels[::-1]

    # Calculate errors
    errors = []
    for i, (orig, recon) in enumerate(zip(gaussian_pyramid, reconstructed_levels)):
        if orig.shape == recon.shape:
            error = cv2.absdiff(orig, recon)
            errors.append(error)
        else:
            errors.append(None)

    # Display
    plt.figure(figsize=(20, 12))

    # Original pyramid
    for i, level in enumerate(gaussian_pyramid):
        plt.subplot(3, 5, i+1)
        plt.imshow(cv2.cvtColor(level, cv2.COLOR_BGR2RGB))
        plt.title(f'Original L{i}')
        plt.axis('off')

    # Reconstructed
    for i, level in enumerate(reconstructed_levels):
        plt.subplot(3, 5, i+6)
        plt.imshow(cv2.cvtColor(level, cv2.COLOR_BGR2RGB))
        plt.title(f'Reconstructed L{i}')
        plt.axis('off')

    # Errors
    for i, error in enumerate(errors):
        if error is not None:
            plt.subplot(3, 5, i+11)
            error_amplified = cv2.convertScaleAbs(error, alpha=5)
            plt.imshow(cv2.cvtColor(error_amplified, cv2.COLOR_BGR2RGB))
            mean_error = np.mean(error)
            plt.title(f'Error L{i}\nMean: {mean_error:.1f}')
            plt.axis('off')

    plt.tight_layout()
    plt.savefig('pyramid_reconstruction.png', dpi=150, bbox_inches='tight')
    plt.show()

    # Print error statistics
    print("\nReconstruction Error Analysis:")
    print("-" * 50)
    for i, error in enumerate(errors):
        if error is not None:
            print(f"Level {i}: Mean Error = {np.mean(error):.2f}, "
                  f"Max Error = {np.max(error)}, PSNR = {cv2.PSNR(gaussian_pyramid[i], reconstructed_levels[i]):.2f} dB")
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def pyramid_decompose_recompose():
    """
    Full decomposition and recomposition of image using pyramid
    """
    # Load image
    image = cv2.imread('image.jpg')
    if image is None:
        image = np.zeros((256, 256, 3), dtype=np.uint8)
        # Create gradient background
        for i in range(256):
            image[:, i] = [i, 128, 255-i]
        # Add some features
        cv2.circle(image, (128, 128), 60, (255, 255, 255), -1)
        cv2.rectangle(image, (50, 50), (100, 100), (0, 0, 0), -1)

    n_levels = 4

    # Build Gaussian pyramid
    G = [image.astype(np.float64)]
    for i in range(n_levels):
        G.append(cv2.pyrDown(G[-1]))

    # Build Laplacian pyramid (for later use in reconstruction)
    L = []
    for i in range(n_levels):
        upsampled = cv2.pyrUp(G[i+1])
        if upsampled.shape != G[i].shape:
            upsampled = cv2.resize(upsampled, (G[i].shape[1], G[i].shape[0]))
        L.append(G[i] - upsampled)
    L.append(G[-1])  # Top of Gaussian pyramid

    # Perfect reconstruction from Laplacian pyramid
    reconstructed = L[-1].copy()
    for i in range(n_levels - 1, -1, -1):
        upsampled = cv2.pyrUp(reconstructed)
        if upsampled.shape != L[i].shape:
            upsampled = cv2.resize(upsampled, (L[i].shape[1], L[i].shape[0]))
        reconstructed = upsampled + L[i]

    reconstructed = np.clip(reconstructed, 0, 255).astype(np.uint8)

    # Display
    plt.figure(figsize=(20, 12))

    # Original
    plt.subplot(3, 6, 1)
    plt.imshow(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
    plt.title('Original')
    plt.axis('off')

    # Gaussian pyramid
    for i, g in enumerate(G):
        if i < 5:
            plt.subplot(3, 6, i+2)
            plt.imshow(cv2.cvtColor(np.clip(g, 0, 255).astype(np.uint8), cv2.COLOR_BGR2RGB))
            plt.title(f'G{i}')
            plt.axis('off')

    # Laplacian pyramid
    for i, l in enumerate(L):
        if i < 6:
            plt.subplot(3, 6, i+7)
            # Normalize for display
            l_display = l - l.min()
            l_display = (l_display / l_display.max() * 255).astype(np.uint8)
            plt.imshow(cv2.cvtColor(l_display, cv2.COLOR_BGR2RGB))
            plt.title(f'L{i}')
            plt.axis('off')

    # Reconstructed
    plt.subplot(3, 6, 13)
    plt.imshow(cv2.cvtColor(reconstructed, cv2.COLOR_BGR2RGB))
    plt.title('Reconstructed')
    plt.axis('off')

    # Difference
    plt.subplot(3, 6, 14)
    diff = cv2.absdiff(image, reconstructed)
    diff_amplified = cv2.convertScaleAbs(diff, alpha=10)
    plt.imshow(cv2.cvtColor(diff_amplified, cv2.COLOR_BGR2RGB))
    plt.title(f'Difference (10x)\nMax: {np.max(diff)}')
    plt.axis('off')

    # Error histogram
    plt.subplot(3, 6, 15)
    plt.hist(diff.flatten(), bins=50, color='steelblue', alpha=0.7)
    plt.xlabel('Error Value')
    plt.ylabel('Frequency')
    plt.title('Error Distribution')

    # PSNR
    plt.subplot(3, 6, 16)
    psnr = cv2.PSNR(image, reconstructed)
    mse = np.mean((image.astype(float) - reconstructed.astype(float))**2)
    info_text = f"PSNR: {psnr:.2f} dB\nMSE: {mse:.4f}\nMax Error: {np.max(diff)}"
    plt.text(0.5, 0.5, info_text, ha='center', va='center',
             fontsize=14, family='monospace',
             transform=plt.gca().transAxes)
    plt.axis('off')
    plt.title('Quality Metrics')

    plt.tight_layout()
    plt.savefig('pyramid_decompose.png', dpi=150, bbox_inches='tight')
    plt.show()

    print("\nPyramid Decomposition Analysis:")
    print("="*50)
    print(f"Original Size: {image.shape}")
    print(f"Number of Levels: {n_levels + 1}")
    print(f"PSNR: {psnr:.2f} dB")
    print(f"MSE: {mse:.6f}")
```

## 6. 总结

### 6.1 高斯金字塔核心要点

| 要点 | 说明 |
|------|------|
| **构建方法** | 高斯滤波 + 下采样 |
| **每层关系** | 尺寸减半，分辨率降低 |
| **主要函数** | `cv2.pyrDown()`, `cv2.pyrUp()` |
| **存储效率** | 约为原图的4/3 |
| **应用场景** | 多尺度处理、快速搜索、抗锯齿 |

### 6.2 最佳实践

1. **下采样前滤波**：始终使用高斯滤波避免混叠
2. **渐进式缩放**：大比例缩放时使用多级金字塔
3. **尺寸匹配**：重建时注意尺寸匹配
4. **精度保持**：使用浮点数进行中间计算
