# 畸变矫正

## 1. 畸变矫正概述

哼，笨蛋，有了相机标定结果，就可以矫正镜头畸变了！本小姐今天教你怎么把扭曲的图像变回正常！(￣▽￣)／

### 1.1 为什么需要畸变矫正

| 应用场景 | 需要矫正的原因 |
|----------|----------------|
| 测量应用 | 畸变影响距离和角度测量精度 |
| 3D重建 | 需要精确的2D-3D对应关系 |
| 拼接全景 | 消除边缘变形 |
| 机器视觉 | 提高检测和识别精度 |

### 1.2 OpenCV畸变矫正函数

- `cv2.undistort()`：直接矫正图像
- `cv2.initUndistortRectifyMap()` + `cv2.remap()`：创建映射表后矫正

## 2. 使用cv2.undistort()

### 2.1 代码示例1：基本畸变矫正

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def create_distorted_image():
    """创建带畸变的模拟图像"""

    # 创建网格图案
    img = np.ones((480, 640), dtype=np.uint8) * 200

    # 绘制网格
    for x in range(0, 640, 40):
        cv2.line(img, (x, 0), (x, 480), 100, 1)
    for y in range(0, 480, 40):
        cv2.line(img, (0, y), (640, y), 100, 1)

    # 添加一些特征
    cv2.rectangle(img, (200, 150), (440, 330), 50, 2)
    cv2.circle(img, (320, 240), 80, 50, 2)

    return img

def apply_distortion(img, K, dist_coeffs):
    """对图像应用畸变（模拟畸变图像）"""

    h, w = img.shape[:2]

    # 生成网格坐标
    map_x = np.zeros((h, w), dtype=np.float32)
    map_y = np.zeros((h, w), dtype=np.float32)

    fx, fy = K[0, 0], K[1, 1]
    cx, cy = K[0, 2], K[1, 2]
    k1, k2, p1, p2, k3 = dist_coeffs

    for v in range(h):
        for u in range(w):
            # 归一化坐标
            x = (u - cx) / fx
            y = (v - cy) / fy

            r2 = x*x + y*y
            r4 = r2 * r2
            r6 = r4 * r2

            # 径向畸变
            radial = 1 + k1*r2 + k2*r4 + k3*r6

            # 切向畸变
            x_dist = x * radial + 2*p1*x*y + p2*(r2 + 2*x*x)
            y_dist = y * radial + p1*(r2 + 2*y*y) + 2*p2*x*y

            # 转回像素坐标
            map_x[v, u] = fx * x_dist + cx
            map_y[v, u] = fy * y_dist + cy

    distorted = cv2.remap(img, map_x, map_y, cv2.INTER_LINEAR)
    return distorted

# 相机参数
K = np.array([
    [500, 0, 320],
    [0, 500, 240],
    [0, 0, 1]
], dtype=np.float64)

dist_coeffs = np.array([-0.3, 0.1, 0.001, -0.001, 0.0])

# 创建原始图像
original = create_distorted_image()

# 应用畸变
distorted = apply_distortion(original, K, dist_coeffs)

# 矫正畸变
undistorted = cv2.undistort(distorted, K, dist_coeffs)

# 可视化
fig, axes = plt.subplots(1, 3, figsize=(15, 5))

axes[0].imshow(original, cmap='gray')
axes[0].set_title('原始图像（无畸变）')
axes[0].axis('off')

axes[1].imshow(distorted, cmap='gray')
axes[1].set_title('畸变图像')
axes[1].axis('off')

axes[2].imshow(undistorted, cmap='gray')
axes[2].set_title('矫正后图像')
axes[2].axis('off')

plt.suptitle('畸变矫正效果', fontsize=14)
plt.tight_layout()
plt.savefig('undistort_basic.png', dpi=150, bbox_inches='tight')
plt.show()

print(f"原始图像尺寸: {original.shape}")
print(f"畸变图像尺寸: {distorted.shape}")
print(f"矫正后尺寸: {undistorted.shape}")
```

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def undistort_with_new_camera_matrix():
    """使用新相机矩阵进行畸变矫正"""

    # 创建测试图像
    img = create_distorted_image()

    # 相机参数
    K = np.array([[500, 0, 320], [0, 500, 240], [0, 0, 1]], dtype=np.float64)
    dist_coeffs = np.array([-0.35, 0.12, 0.001, -0.001, 0.0])

    # 应用畸变
    distorted = apply_distortion(img, K, dist_coeffs)

    h, w = distorted.shape[:2]

    # 方法1: 直接矫正（可能有黑边）
    undist1 = cv2.undistort(distorted, K, dist_coeffs)

    # 方法2: 使用getOptimalNewCameraMatrix获取新相机矩阵
    # alpha=0: 只保留有效像素（无黑边，但可能裁剪）
    # alpha=1: 保留所有像素（有黑边）
    new_K_0, roi_0 = cv2.getOptimalNewCameraMatrix(K, dist_coeffs, (w, h), 0)
    new_K_1, roi_1 = cv2.getOptimalNewCameraMatrix(K, dist_coeffs, (w, h), 1)

    undist_alpha0 = cv2.undistort(distorted, K, dist_coeffs, None, new_K_0)
    undist_alpha1 = cv2.undistort(distorted, K, dist_coeffs, None, new_K_1)

    # 可选：裁剪ROI
    x, y, rw, rh = roi_0
    undist_cropped = undist_alpha0[y:y+rh, x:x+rw]

    print("新相机矩阵 (alpha=0):")
    print(new_K_0)
    print(f"ROI: {roi_0}")

    print("\n新相机矩阵 (alpha=1):")
    print(new_K_1)
    print(f"ROI: {roi_1}")

    # 可视化
    fig, axes = plt.subplots(2, 3, figsize=(15, 10))

    axes[0, 0].imshow(distorted, cmap='gray')
    axes[0, 0].set_title('畸变图像')
    axes[0, 0].axis('off')

    axes[0, 1].imshow(undist1, cmap='gray')
    axes[0, 1].set_title('直接矫正')
    axes[0, 1].axis('off')

    axes[0, 2].imshow(undist_alpha0, cmap='gray')
    axes[0, 2].set_title('alpha=0（无黑边）')
    axes[0, 2].axis('off')

    axes[1, 0].imshow(undist_alpha1, cmap='gray')
    axes[1, 0].set_title('alpha=1（保留全部）')
    axes[1, 0].axis('off')

    axes[1, 1].imshow(undist_cropped, cmap='gray')
    axes[1, 1].set_title('裁剪后')
    axes[1, 1].axis('off')

    # 对比原始和矫正
    diff = cv2.absdiff(img, undist_alpha0)
    axes[1, 2].imshow(diff, cmap='hot')
    axes[1, 2].set_title('原始与矫正差异')
    axes[1, 2].axis('off')

    plt.tight_layout()
    plt.savefig('undistort_new_matrix.png', dpi=150, bbox_inches='tight')
    plt.show()
```

## 3. 使用映射表矫正

### 3.1 为什么使用映射表

- **效率高**：预计算映射表，多次使用
- **灵活性**：可以组合多种变换
- **批量处理**：视频处理时只需计算一次

### 3.2 代码示例3：使用映射表

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt
import time

def undistort_with_map():
    """使用映射表进行畸变矫正"""

    # 创建测试图像
    img = create_distorted_image()

    # 相机参数
    K = np.array([[500, 0, 320], [0, 500, 240], [0, 0, 1]], dtype=np.float64)
    dist_coeffs = np.array([-0.3, 0.1, 0.001, -0.001, 0.0])

    # 应用畸变
    distorted = apply_distortion(img, K, dist_coeffs)

    h, w = distorted.shape[:2]
    new_K, roi = cv2.getOptimalNewCameraMatrix(K, dist_coeffs, (w, h), 0)

    # 方法1: 每次使用cv2.undistort
    start1 = time.time()
    for _ in range(100):
        undist1 = cv2.undistort(distorted, K, dist_coeffs, None, new_K)
    time1 = (time.time() - start1) / 100 * 1000

    # 方法2: 预计算映射表
    map_x, map_y = cv2.initUndistortRectifyMap(
        K, dist_coeffs, None, new_K, (w, h), cv2.CV_32FC1
    )

    start2 = time.time()
    for _ in range(100):
        undist2 = cv2.remap(distorted, map_x, map_y, cv2.INTER_LINEAR)
    time2 = (time.time() - start2) / 100 * 1000

    print(f"cv2.undistort 平均时间: {time1:.3f} ms")
    print(f"cv2.remap 平均时间: {time2:.3f} ms")
    print(f"加速比: {time1/time2:.2f}x")

    # 可视化映射表
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))

    axes[0, 0].imshow(distorted, cmap='gray')
    axes[0, 0].set_title('畸变图像')
    axes[0, 0].axis('off')

    axes[0, 1].imshow(undist2, cmap='gray')
    axes[0, 1].set_title('矫正后图像')
    axes[0, 1].axis('off')

    im1 = axes[1, 0].imshow(map_x, cmap='jet')
    axes[1, 0].set_title('X映射表')
    plt.colorbar(im1, ax=axes[1, 0], fraction=0.046)

    im2 = axes[1, 1].imshow(map_y, cmap='jet')
    axes[1, 1].set_title('Y映射表')
    plt.colorbar(im2, ax=axes[1, 1], fraction=0.046)

    plt.tight_layout()
    plt.savefig('undistort_map.png', dpi=150, bbox_inches='tight')
    plt.show()

    return map_x, map_y

map_x, map_y = undistort_with_map()
```

### 3.3 代码示例4：视频畸变矫正

```python
import numpy as np
import cv2
import time
import matplotlib.pyplot as plt

class VideoUndistorter:
    """视频畸变矫正器"""

    def __init__(self, camera_matrix, dist_coeffs, image_size, alpha=0):
        """
        初始化矫正器

        参数:
            camera_matrix: 相机内参
            dist_coeffs: 畸变系数
            image_size: 图像尺寸 (width, height)
            alpha: 0-1, 控制视野范围
        """
        self.K = camera_matrix
        self.dist = dist_coeffs

        # 计算新相机矩阵和ROI
        self.new_K, self.roi = cv2.getOptimalNewCameraMatrix(
            camera_matrix, dist_coeffs, image_size, alpha
        )

        # 预计算映射表
        self.map_x, self.map_y = cv2.initUndistortRectifyMap(
            camera_matrix, dist_coeffs, None, self.new_K,
            image_size, cv2.CV_32FC1
        )

        print(f"映射表已初始化: {image_size}")

    def undistort(self, frame, crop=False):
        """矫正单帧"""
        undistorted = cv2.remap(frame, self.map_x, self.map_y, cv2.INTER_LINEAR)

        if crop and self.roi[2] > 0 and self.roi[3] > 0:
            x, y, w, h = self.roi
            undistorted = undistorted[y:y+h, x:x+w]

        return undistorted

def simulate_video_processing():
    """模拟视频处理"""

    # 相机参数
    K = np.array([[500, 0, 320], [0, 500, 240], [0, 0, 1]], dtype=np.float64)
    dist_coeffs = np.array([-0.25, 0.08, 0.001, -0.001, 0.0])

    # 创建矫正器
    undistorter = VideoUndistorter(K, dist_coeffs, (640, 480))

    # 模拟视频帧
    print("\n模拟处理100帧视频...")
    frames = []
    for i in range(100):
        # 创建变化的帧
        frame = np.ones((480, 640), dtype=np.uint8) * 180
        cv2.putText(frame, f'Frame {i}', (200, 250),
                   cv2.FONT_HERSHEY_SIMPLEX, 2, 50, 3)
        # 添加移动的圆
        cx = 320 + int(100 * np.sin(i * 0.1))
        cy = 240 + int(50 * np.cos(i * 0.1))
        cv2.circle(frame, (cx, cy), 30, 100, -1)
        frames.append(frame)

    # 处理所有帧
    start = time.time()
    processed = []
    for frame in frames:
        # 模拟畸变
        distorted = apply_distortion(frame, K, dist_coeffs)
        # 矫正
        undistorted = undistorter.undistort(distorted)
        processed.append(undistorted)

    elapsed = time.time() - start
    fps = len(frames) / elapsed

    print(f"处理时间: {elapsed:.3f}秒")
    print(f"帧率: {fps:.1f} FPS")

    # 可视化部分结果
    fig, axes = plt.subplots(2, 4, figsize=(16, 8))

    for i, idx in enumerate([0, 25, 50, 75]):
        # 原始帧
        axes[0, i].imshow(frames[idx], cmap='gray')
        axes[0, i].set_title(f'原始帧 {idx}')
        axes[0, i].axis('off')

        # 处理后
        axes[1, i].imshow(processed[idx], cmap='gray')
        axes[1, i].set_title(f'矫正后 {idx}')
        axes[1, i].axis('off')

    plt.suptitle(f'视频畸变矫正 ({fps:.1f} FPS)', fontsize=14)
    plt.tight_layout()
    plt.savefig('video_undistort.png', dpi=150, bbox_inches='tight')
    plt.show()

simulate_video_processing()
```

## 5. 点坐标矫正

### 5.1 代码示例7：矫正单个点坐标

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def undistort_points_demo():
    """演示点坐标的畸变矫正"""

    K = np.array([[500, 0, 320], [0, 500, 240], [0, 0, 1]], dtype=np.float64)
    dist_coeffs = np.array([-0.3, 0.1, 0.001, -0.001, 0.0])

    # 原始点（无畸变）
    original_points = np.array([
        [100, 100], [540, 100], [320, 240],
        [100, 380], [540, 380], [200, 200],
        [440, 200], [200, 280], [440, 280]
    ], dtype=np.float32)

    # 模拟畸变点（实际中是从畸变图像检测到的点）
    # 这里我们手动应用畸变模型
    distorted_points = []
    fx, fy = K[0, 0], K[1, 1]
    cx, cy = K[0, 2], K[1, 2]
    k1, k2, p1, p2, k3 = dist_coeffs

    for pt in original_points:
        x = (pt[0] - cx) / fx
        y = (pt[1] - cy) / fy
        r2 = x*x + y*y

        radial = 1 + k1*r2 + k2*r2*r2 + k3*r2*r2*r2
        x_dist = x * radial + 2*p1*x*y + p2*(r2 + 2*x*x)
        y_dist = y * radial + p1*(r2 + 2*y*y) + 2*p2*x*y

        u_dist = fx * x_dist + cx
        v_dist = fy * y_dist + cy
        distorted_points.append([u_dist, v_dist])

    distorted_points = np.array(distorted_points, dtype=np.float32)

    # 使用cv2.undistortPoints矫正
    undistorted_points = cv2.undistortPoints(
        distorted_points.reshape(-1, 1, 2),
        K, dist_coeffs, P=K
    ).reshape(-1, 2)

    # 计算误差
    errors = np.linalg.norm(original_points - undistorted_points, axis=1)
    print("点坐标矫正结果:")
    print(f"  平均误差: {np.mean(errors):.4f} 像素")
    print(f"  最大误差: {np.max(errors):.4f} 像素")

    # 可视化
    fig, axes = plt.subplots(1, 2, figsize=(14, 6))

    # 点位置对比
    axes[0].scatter(original_points[:, 0], original_points[:, 1],
                   c='green', s=100, marker='o', label='原始点', zorder=3)
    axes[0].scatter(distorted_points[:, 0], distorted_points[:, 1],
                   c='red', s=100, marker='x', label='畸变点', zorder=3)
    axes[0].scatter(undistorted_points[:, 0], undistorted_points[:, 1],
                   c='blue', s=100, marker='+', label='矫正点', zorder=3)

    # 绘制连线
    for orig, dist, undist in zip(original_points, distorted_points, undistorted_points):
        axes[0].plot([orig[0], dist[0]], [orig[1], dist[1]], 'r--', alpha=0.5)
```

## 6. 实际应用案例

### 6.1 代码示例8：测量应用中的畸变矫正

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def measurement_application():
    """测量应用中的畸变矫正示例"""

    # 创建带尺寸标注的测试图像
    img = np.ones((480, 640), dtype=np.uint8) * 220

    # 绘制一个已知尺寸的矩形（模拟测量目标）
    # 实际尺寸：100mm x 60mm
    rect_points = np.array([
        [220, 180], [420, 180], [420, 300], [220, 300]
    ], dtype=np.float32)

    cv2.polylines(img, [rect_points.astype(np.int32)], True, 50, 3)

    # 绘制对角线用于测量
    cv2.line(img, (220, 180), (420, 300), 80, 2)
    cv2.line(img, (420, 180), (220, 300), 80, 2)

    # 添加参考网格
    for x in range(0, 640, 50):
        cv2.line(img, (x, 0), (x, 480), 200, 1)
    for y in range(0, 480, 50):
        cv2.line(img, (0, y), (640, y), 200, 1)

    K = np.array([[500, 0, 320], [0, 500, 240], [0, 0, 1]], dtype=np.float64)
    dist_coeffs = np.array([-0.25, 0.08, 0.001, -0.001, 0.0])

    # 应用畸变
    distorted = apply_distortion(img, K, dist_coeffs)

    # 在畸变图像上检测矩形角点（模拟）
    distorted_rect = []
    fx, fy = K[0, 0], K[1, 1]
    cx, cy = K[0, 2], K[1, 2]
    k1, k2, p1, p2, k3 = dist_coeffs

    for pt in rect_points:
        x = (pt[0] - cx) / fx
        y = (pt[1] - cy) / fy
        r2 = x*x + y*y
        radial = 1 + k1*r2 + k2*r2*r2
        x_dist = x * radial + 2*p1*x*y + p2*(r2 + 2*x*x)
        y_dist = y * radial + p1*(r2 + 2*y*y) + 2*p2*x*y
        distorted_rect.append([fx*x_dist + cx, fy*y_dist + cy])

    distorted_rect = np.array(distorted_rect, dtype=np.float32)

    # 矫正点坐标
    undistorted_rect = cv2.undistortPoints(
        distorted_rect.reshape(-1, 1, 2), K, dist_coeffs, P=K
    ).reshape(-1, 2)

    # 计算尺寸
    def calculate_dimensions(pts):
        width = np.linalg.norm(pts[1] - pts[0])
        height = np.linalg.norm(pts[3] - pts[0])
        diagonal = np.linalg.norm(pts[2] - pts[0])
        return width, height, diagonal

    orig_w, orig_h, orig_d = calculate_dimensions(rect_points)
    dist_w, dist_h, dist_d = calculate_dimensions(distorted_rect)
    undist_w, undist_h, undist_d = calculate_dimensions(undistorted_rect)

    print("尺寸测量对比:")
    print(f"{'':15} {'宽度':>10} {'高度':>10} {'对角线':>10}")
    print("-" * 50)
    print(f"{'原始(真值)':15} {orig_w:>10.2f} {orig_h:>10.2f} {orig_d:>10.2f}")
    print(f"{'畸变后':15} {dist_w:>10.2f} {dist_h:>10.2f} {dist_d:>10.2f}")
    print(f"{'矫正后':15} {undist_w:>10.2f} {undist_h:>10.2f} {undist_d:>10.2f}")
    print("-" * 50)
    print(f"{'畸变误差(%)':15} {(dist_w/orig_w-1)*100:>10.2f} {(dist_h/orig_h-1)*100:>10.2f} {(dist_d/orig_d-1)*100:>10.2f}")
    print(f"{'矫正误差(%)':15} {(undist_w/orig_w-1)*100:>10.2f} {(undist_h/orig_h-1)*100:>10.2f} {(undist_d/orig_d-1)*100:>10.2f}")

    # 可视化
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))

    # 原始
    axes[0].imshow(img, cmap='gray')
    axes[0].scatter(rect_points[:, 0], rect_points[:, 1], c='green', s=100, zorder=5)
    axes[0].set_title(f'原始图像\n宽:{orig_w:.1f} 高:{orig_h:.1f}')
    axes[0].axis('off')

    # 畸变
    axes[1].imshow(distorted, cmap='gray')
    axes[1].scatter(distorted_rect[:, 0], distorted_rect[:, 1], c='red', s=100, zorder=5)
    axes[1].set_title(f'畸变图像\n宽:{dist_w:.1f} 高:{dist_h:.1f} (误差较大)')
    axes[1].axis('off')

    # 矫正后
    undistorted = cv2.undistort(distorted, K, dist_coeffs)
    axes[2].imshow(undistorted, cmap='gray')
    axes[2].scatter(undistorted_rect[:, 0], undistorted_rect[:, 1], c='blue', s=100, zorder=5)
    axes[2].set_title(f'矫正后\n宽:{undist_w:.1f} 高:{undist_h:.1f} (接近真值)')
    axes[2].axis('off')

    plt.suptitle('测量应用中畸变矫正的重要性', fontsize=14)
    plt.tight_layout()
    plt.savefig('measurement_application.png', dpi=150, bbox_inches='tight')
    plt.show()

measurement_application()
```

### 核心要点

1. **两种矫正方法**
   - `cv2.undistort()`：简单直接
   - 映射表法：适合批量处理

2. **新相机矩阵**
   - alpha=0：无黑边
   - alpha=1：保留全部

3. **点坐标矫正**
   - `cv2.undistortPoints()`
   - 用于精确测量

4. **应用场景**
   - 视频处理用映射表提速
   - 测量应用必须矫正
