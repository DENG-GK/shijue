# 立体视觉

## 1. 立体视觉概述

哼，笨蛋，立体视觉是计算机视觉中非常重要的技术！它模仿人眼的双目视觉原理，从两个视角恢复场景的3D结构！本小姐今天就来给你讲讲！(￣▽￣)／

### 1.1 什么是立体视觉

立体视觉（Stereo Vision）通过两个相机从不同角度观察同一场景，利用视差（Disparity）计算深度信息。

```
左相机                     右相机
  ↓                         ↓
  *-----------基线B-----------*
  \           |             /
   \          |            /
    \         |           /
     \        |          /
      \       |         /
       \      ↓        /
        \   场景点P   /
         \    *     /
          \       /
           \     /
            \   /
             \ /
```

### 1.2 立体视觉的应用

| 应用领域 | 具体场景 |
|----------|----------|
| 自动驾驶 | 障碍物检测、距离测量 |
| 机器人 | 导航、避障、抓取 |
| 3D重建 | 建模、测绘 |
| AR/VR | 深度感知、交互 |

## 2. 双目标定

### 2.1 代码示例1：双目相机标定

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def stereo_calibration_simulation():
    """模拟双目相机标定"""

    # 棋盘格参数
    pattern_size = (9, 6)
    square_size = 25.0  # mm

    # 生成3D点
    objp = np.zeros((pattern_size[0] * pattern_size[1], 3), np.float32)
    objp[:, :2] = np.mgrid[0:pattern_size[0], 0:pattern_size[1]].T.reshape(-1, 2)
    objp *= square_size

    # 模拟真实的双目相机参数
    # 左相机
    K_left = np.array([
        [800, 0, 320],
        [0, 800, 240],
        [0, 0, 1]
    ], dtype=np.float64)
    dist_left = np.array([-0.15, 0.05, 0, 0, 0])

    # 右相机
    K_right = np.array([
        [805, 0, 325],
        [0, 805, 238],
        [0, 0, 1]
    ], dtype=np.float64)
    dist_right = np.array([-0.16, 0.06, 0, 0, 0])

    # 相机间的相对位姿
    # 右相机相对于左相机：平移60mm（基线）
    R_stereo = np.eye(3)  # 相机平行
    T_stereo = np.array([[60], [0], [0]], dtype=np.float64)  # 60mm基线

    # 生成模拟标定数据
    num_images = 15
    obj_points = []
    img_points_left = []
    img_points_right = []

    for i in range(num_images):
        # 随机标定板位姿
        rvec = np.random.uniform(-0.3, 0.3, 3)
        tvec = np.array([
            np.random.uniform(-50, 50),
            np.random.uniform(-50, 50),
            np.random.uniform(250, 400)
        ])

        # 左相机投影
        pts_left, _ = cv2.projectPoints(objp, rvec, tvec, K_left, dist_left)
        pts_left = pts_left.reshape(-1, 2)

        # 右相机投影（考虑相对位姿）
        R_left, _ = cv2.Rodrigues(rvec)
        R_right = R_stereo @ R_left
        T_right = R_stereo @ tvec.reshape(3, 1) + T_stereo
        rvec_right, _ = cv2.Rodrigues(R_right)

        pts_right, _ = cv2.projectPoints(objp, rvec_right, T_right, K_right, dist_right)
        pts_right = pts_right.reshape(-1, 2)

        # 添加噪声
        pts_left += np.random.randn(*pts_left.shape) * 0.3
        pts_right += np.random.randn(*pts_right.shape) * 0.3

        # 检查点是否在图像范围内
        if (pts_left.min() > 10 and pts_left.max() < 630 and
            pts_right.min() > 10 and pts_right.max() < 630):
            obj_points.append(objp)
            img_points_left.append(pts_left.astype(np.float32))
            img_points_right.append(pts_right.astype(np.float32))

    print(f"有效标定图像对: {len(obj_points)}")

    # 执行立体标定
    flags = cv2.CALIB_FIX_INTRINSIC  # 使用已知内参

    # 先单独标定两个相机
    ret_l, K_l, dist_l, _, _ = cv2.calibrateCamera(
        obj_points, img_points_left, (640, 480), None, None
    )
    ret_r, K_r, dist_r, _, _ = cv2.calibrateCamera(
        obj_points, img_points_right, (640, 480), None, None
    )

    print(f"\n左相机标定误差: {ret_l:.4f}")
    print(f"右相机标定误差: {ret_r:.4f}")

    # 立体标定
    ret_stereo, K_l, dist_l, K_r, dist_r, R, T, E, F = cv2.stereoCalibrate(
        obj_points, img_points_left, img_points_right,
        K_l, dist_l, K_r, dist_r,
        (640, 480),
        criteria=(cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 1e-5),
        flags=cv2.CALIB_FIX_INTRINSIC
    )

    print(f"\n立体标定误差: {ret_stereo:.4f}")
    print(f"\n旋转矩阵 R:")
    print(R)
    print(f"\n平移向量 T:")
    print(T.flatten())
    print(f"\n基线长度: {np.linalg.norm(T):.2f} mm")

    return {
        'K_left': K_l, 'dist_left': dist_l,
        'K_right': K_r, 'dist_right': dist_r,
        'R': R, 'T': T, 'E': E, 'F': F
    }

stereo_params = stereo_calibration_simulation()
```

## 3. 立体校正

### 3.1 代码示例2：立体校正(Stereo Rectification)

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def stereo_rectification():
    """立体校正演示"""

    # 使用模拟参数
    K_left = np.array([[800, 0, 320], [0, 800, 240], [0, 0, 1]], dtype=np.float64)
    K_right = np.array([[800, 0, 320], [0, 800, 240], [0, 0, 1]], dtype=np.float64)
    dist_left = np.array([-0.1, 0.05, 0, 0, 0])
    dist_right = np.array([-0.1, 0.05, 0, 0, 0])

    # 相机间位姿（轻微不平行）
    angle = np.radians(2)  # 2度偏角
    R = np.array([
        [np.cos(angle), 0, np.sin(angle)],
        [0, 1, 0],
        [-np.sin(angle), 0, np.cos(angle)]
    ])
    T = np.array([[60], [2], [0]], dtype=np.float64)  # 60mm基线

    image_size = (640, 480)

    # 立体校正
    R1, R2, P1, P2, Q, roi1, roi2 = cv2.stereoRectify(
        K_left, dist_left,
        K_right, dist_right,
        image_size, R, T,
        alpha=0  # 0: 只保留有效像素
    )

    print("校正变换矩阵:")
    print(f"\nR1 (左相机旋转):\n{R1}")
    print(f"\nR2 (右相机旋转):\n{R2}")
    print(f"\nP1 (左投影矩阵):\n{P1}")
    print(f"\nP2 (右投影矩阵):\n{P2}")
    print(f"\nQ (视差-深度映射矩阵):\n{Q}")

    # 创建校正映射
    map1_left, map2_left = cv2.initUndistortRectifyMap(
        K_left, dist_left, R1, P1, image_size, cv2.CV_32FC1
    )
    map1_right, map2_right = cv2.initUndistortRectifyMap(
        K_right, dist_right, R2, P2, image_size, cv2.CV_32FC1
    )

    # 创建测试图像
    def create_stereo_test_images():
        """创建模拟的立体图像对"""
        # 左图像
        img_left = np.ones((480, 640), dtype=np.uint8) * 180
        cv2.rectangle(img_left, (200, 150), (350, 300), 80, -1)
        cv2.circle(img_left, (450, 200), 50, 120, -1)
        cv2.putText(img_left, 'LEFT', (50, 400), cv2.FONT_HERSHEY_SIMPLEX, 1, 50, 2)

        # 右图像（模拟视差）
        img_right = np.ones((480, 640), dtype=np.uint8) * 180
        cv2.rectangle(img_right, (170, 150), (320, 300), 80, -1)  # 左移30像素
        cv2.circle(img_right, (420, 200), 50, 120, -1)  # 左移30像素
        cv2.putText(img_right, 'RIGHT', (50, 400), cv2.FONT_HERSHEY_SIMPLEX, 1, 50, 2)

        return img_left, img_right

    img_left, img_right = create_stereo_test_images()

    # 应用校正
    img_left_rect = cv2.remap(img_left, map1_left, map2_left, cv2.INTER_LINEAR)
    img_right_rect = cv2.remap(img_right, map1_right, map2_right, cv2.INTER_LINEAR)

    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))

    # 原始图像
    combined_orig = np.hstack([img_left, img_right])
    # 绘制水平线检查极线是否对齐
    for y in range(0, 480, 40):
        cv2.line(combined_orig, (0, y), (1280, y), 150, 1)

    axes[0, 0].imshow(combined_orig, cmap='gray')
    axes[0, 0].set_title('原始立体图像对（可能不对齐）')
    axes[0, 0].axis('off')

    # 校正后图像
    combined_rect = np.hstack([img_left_rect, img_right_rect])
    for y in range(0, 480, 40):
        cv2.line(combined_rect, (0, y), (1280, y), 150, 1)

    axes[0, 1].imshow(combined_rect, cmap='gray')
    axes[0, 1].set_title('校正后立体图像对（极线对齐）')
    axes[0, 1].axis('off')

    # 可视化校正映射
    axes[1, 0].imshow(map1_left, cmap='jet')
    axes[1, 0].set_title('左相机X映射')
    axes[1, 0].axis('off')

    axes[1, 1].imshow(map1_right, cmap='jet')
    axes[1, 1].set_title('右相机X映射')
    axes[1, 1].axis('off')

    plt.tight_layout()
    plt.savefig('stereo_rectification.png', dpi=150, bbox_inches='tight')
    plt.show()

    return map1_left, map2_left, map1_right, map2_right, Q

maps = stereo_rectification()
```

## 4. 视差计算

### 4.1 代码示例3：使用SGBM计算视差

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def compute_disparity():
    """计算立体视差图"""

    # 创建立体图像对
    def create_depth_scene():
        """创建带深度信息的场景"""
        # 左图像
        img_left = np.ones((480, 640), dtype=np.uint8) * 200

        # 添加不同深度的物体
        # 近处物体（大视差）
        cv2.rectangle(img_left, (100, 150), (200, 300), 80, -1)
        # 中等距离物体
        cv2.rectangle(img_left, (280, 180), (380, 280), 120, -1)
        # 远处物体（小视差）
        cv2.rectangle(img_left, (450, 200), (530, 260), 160, -1)

        # 添加纹理以便匹配
        for i in range(0, 480, 20):
            for j in range(0, 640, 20):
                if (i + j) % 40 == 0:
                    cv2.circle(img_left, (j, i), 3, int(img_left[i, j]) - 30, -1)

        # 右图像（根据深度生成视差）
        img_right = np.ones((480, 640), dtype=np.uint8) * 200

        # 近处物体视差大（40像素）
        cv2.rectangle(img_right, (60, 150), (160, 300), 80, -1)
        # 中等距离（20像素视差）
        cv2.rectangle(img_right, (260, 180), (360, 280), 120, -1)
        # 远处（8像素视差）
        cv2.rectangle(img_right, (442, 200), (522, 260), 160, -1)

        # 添加相同纹理
        for i in range(0, 480, 20):
            for j in range(0, 640, 20):
                if (i + j) % 40 == 0:
                    cv2.circle(img_right, (j, i), 3, int(img_right[i, j]) - 30, -1)

        return img_left, img_right

    img_left, img_right = create_depth_scene()

    # SGBM参数
    min_disp = 0
    num_disp = 64  # 必须是16的倍数
    block_size = 5

    stereo = cv2.StereoSGBM_create(
        minDisparity=min_disp,
        numDisparities=num_disp,
        blockSize=block_size,
        P1=8 * 1 * block_size ** 2,
        P2=32 * 1 * block_size ** 2,
        disp12MaxDiff=1,
        uniquenessRatio=10,
        speckleWindowSize=100,
        speckleRange=32,
        preFilterCap=63,
        mode=cv2.STEREO_SGBM_MODE_SGBM_3WAY
    )

    # 计算视差
    disparity = stereo.compute(img_left, img_right).astype(np.float32) / 16.0

    print(f"视差范围: {disparity.min():.1f} - {disparity.max():.1f}")
    print(f"有效视差像素比例: {np.sum(disparity > 0) / disparity.size * 100:.1f}%")

    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))

    axes[0, 0].imshow(img_left, cmap='gray')
    axes[0, 0].set_title('左图像')
    axes[0, 0].axis('off')

    axes[0, 1].imshow(img_right, cmap='gray')
    axes[0, 1].set_title('右图像')
    axes[0, 1].axis('off')

    # 视差图
    disp_vis = disparity.copy()
    disp_vis[disp_vis < 0] = 0
    im = axes[1, 0].imshow(disp_vis, cmap='jet')
    axes[1, 0].set_title('视差图（暖色=近，冷色=远）')
    axes[1, 0].axis('off')
    plt.colorbar(im, ax=axes[1, 0], fraction=0.046, label='视差(像素)')

    # 3D效果图
    disp_normalized = cv2.normalize(disp_vis, None, 0, 255, cv2.NORM_MINMAX)
    disp_colored = cv2.applyColorMap(disp_normalized.astype(np.uint8), cv2.COLORMAP_JET)
    axes[1, 1].imshow(cv2.cvtColor(disp_colored, cv2.COLOR_BGR2RGB))
    axes[1, 1].set_title('彩色视差图')
    axes[1, 1].axis('off')

    plt.tight_layout()
    plt.savefig('disparity_map.png', dpi=150, bbox_inches='tight')
    plt.show()

    return disparity

disparity = compute_disparity()
```

## 5. 深度计算

### 5.1 代码示例5：从视差计算深度

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def disparity_to_depth():
    """从视差图计算深度"""

    # 相机参数
    focal_length = 800  # 像素
    baseline = 60  # mm

    # 创建模拟视差图
    disparity = np.zeros((480, 640), dtype=np.float32)

    # 不同深度区域
    # 区域1: 近处 (视差40像素 -> 深度1200mm)
    disparity[100:250, 50:200] = 40

    # 区域2: 中等距离 (视差20像素 -> 深度2400mm)
    disparity[150:300, 250:400] = 20

    # 区域3: 远处 (视差10像素 -> 深度4800mm)
    disparity[180:280, 450:580] = 10

    # 背景 (视差5像素 -> 深度9600mm)
    disparity[disparity == 0] = 5

    # 计算深度: Z = f * B / d
    depth = np.zeros_like(disparity)
    valid = disparity > 0
    depth[valid] = (focal_length * baseline) / disparity[valid]

    print("深度统计:")
    print(f"  最小深度: {depth[valid].min():.1f} mm")
    print(f"  最大深度: {depth[valid].max():.1f} mm")
    print(f"  平均深度: {depth[valid].mean():.1f} mm")

    # 验证计算
    print("\n验证公式 Z = f * B / d:")
    for d, name in [(40, '近处'), (20, '中等'), (10, '远处'), (5, '背景')]:
        z = focal_length * baseline / d
        print(f"  {name}: 视差={d}px -> 深度={z:.0f}mm")

    # 可视化
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))

    im1 = axes[0].imshow(disparity, cmap='jet')
    axes[0].set_title('视差图')
    axes[0].axis('off')
    plt.colorbar(im1, ax=axes[0], fraction=0.046, label='视差(像素)')

    im2 = axes[1].imshow(depth, cmap='jet_r')  # 反转颜色：红色近，蓝色远
    axes[1].set_title('深度图')
    axes[1].axis('off')
    plt.colorbar(im2, ax=axes[1], fraction=0.046, label='深度(mm)')

    # 深度直方图
    axes[2].hist(depth.flatten(), bins=50, edgecolor='black', alpha=0.7)
    axes[2].set_xlabel('深度 (mm)')
    axes[2].set_ylabel('像素数')
    axes[2].set_title('深度分布')

    plt.tight_layout()
    plt.savefig('depth_calculation.png', dpi=150, bbox_inches='tight')
    plt.show()

    return depth

depth = disparity_to_depth()
```

### 5.2 代码示例6：使用Q矩阵重投影

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

def reproject_to_3d():
    """使用Q矩阵将视差图重投影为3D点云"""

    # Q矩阵（从stereoRectify获取）
    # 典型的Q矩阵结构
    fx = 800
    fy = 800
    cx = 320
    cy = 240
    baseline = 60  # mm

    Q = np.array([
        [1, 0, 0, -cx],
        [0, 1, 0, -cy],
        [0, 0, 0, fx],
        [0, 0, -1/baseline, 0]
    ], dtype=np.float64)

    print("Q矩阵:")
    print(Q)

    # 创建模拟视差图
    disparity = np.zeros((480, 640), dtype=np.float32)

    # 创建一个倾斜平面
    for y in range(480):
        for x in range(640):
            # 视差随y增加（模拟地面）
            disparity[y, x] = 10 + y * 0.05

    # 添加一个物体
    cv2.rectangle(disparity, (200, 150), (350, 300), 30, -1)

    # 重投影到3D
    points_3d = cv2.reprojectImageTo3D(disparity, Q)

    print(f"\n3D点云形状: {points_3d.shape}")
    print(f"X范围: {points_3d[:,:,0].min():.1f} - {points_3d[:,:,0].max():.1f}")
    print(f"Y范围: {points_3d[:,:,1].min():.1f} - {points_3d[:,:,1].max():.1f}")
    print(f"Z范围: {points_3d[:,:,2].min():.1f} - {points_3d[:,:,2].max():.1f}")

    # 过滤无效点
    mask = disparity > 0
    points = points_3d[mask]

    # 降采样以便可视化
    sample_rate = 50
    points_sample = points[::sample_rate]

    # 可视化
    fig = plt.figure(figsize=(15, 5))

    # 视差图
    ax1 = fig.add_subplot(131)
    im = ax1.imshow(disparity, cmap='jet')
    ax1.set_title('视差图')
    ax1.axis('off')
    plt.colorbar(im, ax=ax1, fraction=0.046)

    # 深度图
    ax2 = fig.add_subplot(132)
    depth = points_3d[:, :, 2]
    depth[depth < 0] = 0
    depth[depth > 10000] = 10000
    im2 = ax2.imshow(depth, cmap='jet_r')
    ax2.set_title('深度图 (Z坐标)')
    ax2.axis('off')
```

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

class StereoVisionSystem:
    """立体视觉系统"""

    def __init__(self, K_left, K_right, dist_left, dist_right, R, T):
        self.K_left = K_left
        self.K_right = K_right
        self.dist_left = dist_left
        self.dist_right = dist_right
        self.R = R
        self.T = T

        self.image_size = (640, 480)

        # 立体校正
        self.R1, self.R2, self.P1, self.P2, self.Q, _, _ = cv2.stereoRectify(
            K_left, dist_left, K_right, dist_right,
            self.image_size, R, T, alpha=0
        )

        # 创建校正映射
        self.map1_left, self.map2_left = cv2.initUndistortRectifyMap(
            K_left, dist_left, self.R1, self.P1, self.image_size, cv2.CV_32FC1
        )
        self.map1_right, self.map2_right = cv2.initUndistortRectifyMap(
            K_right, dist_right, self.R2, self.P2, self.image_size, cv2.CV_32FC1
        )

        # 创建SGBM匹配器
        self.stereo = cv2.StereoSGBM_create(
            minDisparity=0,
            numDisparities=64,
            blockSize=5,
            P1=8 * 5 ** 2,
            P2=32 * 5 ** 2,
            disp12MaxDiff=1,
            uniquenessRatio=10,
            speckleWindowSize=100,
            speckleRange=32
        )

    def rectify(self, img_left, img_right):
        """校正立体图像对"""
        rect_left = cv2.remap(img_left, self.map1_left, self.map2_left, cv2.INTER_LINEAR)
        rect_right = cv2.remap(img_right, self.map1_right, self.map2_right, cv2.INTER_LINEAR)
        return rect_left, rect_right

    def compute_disparity(self, rect_left, rect_right):
        """计算视差图"""
        disparity = self.stereo.compute(rect_left, rect_right).astype(np.float32) / 16.0
        return disparity

    def compute_depth(self, disparity):
        """计算深度图"""
        points_3d = cv2.reprojectImageTo3D(disparity, self.Q)
        depth = points_3d[:, :, 2]
        depth[depth < 0] = 0
        depth[depth > 10000] = 10000
        return depth, points_3d

    def process(self, img_left, img_right):
        """完整处理流程"""
        # 1. 校正
        rect_left, rect_right = self.rectify(img_left, img_right)

        # 2. 计算视差
        disparity = self.compute_disparity(rect_left, rect_right)

        # 3. 计算深度
        depth, points_3d = self.compute_depth(disparity)

        return {
            'rectified_left': rect_left,
            'rectified_right': rect_right,
            'disparity': disparity,
            'depth': depth,
            'points_3d': points_3d
        }

# 创建立体视觉系统
```

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def disparity_post_processing():
    """视差图后处理优化"""

    # 创建带噪声的视差图
    disparity = np.zeros((300, 400), dtype=np.float32)
    disparity[:] = 20

    # 添加物体
    cv2.rectangle(disparity, (100, 80), (200, 180), 40, -1)
    cv2.rectangle(disparity, (250, 100), (350, 200), 30, -1)

    # 添加噪声和孔洞
    noise = np.random.randn(*disparity.shape) * 3
    disparity_noisy = disparity + noise

    # 添加孔洞（无效匹配）
    holes = np.random.random(disparity.shape) < 0.05
    disparity_noisy[holes] = -1

    # 后处理1: 中值滤波
    disparity_median = cv2.medianBlur(disparity_noisy.astype(np.float32), 5)

    # 后处理2: 双边滤波（保边平滑）
    disparity_bilateral = cv2.bilateralFilter(
        disparity_noisy.astype(np.float32), 9, 75, 75
    )

    # 后处理3: 形态学操作填充孔洞
    kernel = np.ones((5, 5), np.uint8)
    disparity_morph = cv2.morphologyEx(
        disparity_noisy.astype(np.float32),
        cv2.MORPH_CLOSE, kernel
    )

    # 后处理4: WLS滤波（加权最小二乘）
    # 需要左右视差一致性检查
    disparity_wls = disparity_median.copy()  # 简化演示

    # 可视化
    fig, axes = plt.subplots(2, 3, figsize=(15, 10))

    axes[0, 0].imshow(disparity, cmap='jet')

    axes[0, 0].axis('off')

    axes[0, 1].imshow(disparity_noisy, cmap='jet')
    axes[0, 1].set_title('带噪声和孔洞')
    axes[0, 1].axis('off')

    axes[0, 2].imshow(disparity_median, cmap='jet')
    axes[0, 2].set_title('中值滤波')
    axes[0, 2].axis('off')

    axes[1, 0].imshow(disparity_bilateral, cmap='jet')
    axes[1, 0].set_title('双边滤波')
    axes[1, 0].axis('off')

    axes[1, 1].imshow(disparity_morph, cmap='jet')
    axes[1, 1].set_title('形态学处理')
    axes[1, 1].axis('off')

    # 误差对比
    errors = {
        '带噪声': np.mean(np.abs(disparity_noisy[disparity_noisy > 0] - disparity[disparity_noisy > 0])),
        '中值': np.mean(np.abs(disparity_median - disparity)),
        '双边': np.mean(np.abs(disparity_bilateral - disparity)),
        '形态学': np.mean(np.abs(disparity_morph - disparity)),
    }

    ax = axes[1, 2]
    ax.bar(errors.keys(), errors.values(), color='steelblue', edgecolor='black')
    ax.set_ylabel('平均误差')
    ax.set_title('后处理效果对比')
```
