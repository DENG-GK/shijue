# 棋盘格标定

## 1. 棋盘格标定概述

哼，笨蛋，棋盘格标定是最常用的相机标定方法！本小姐今天就来教你怎么用OpenCV进行实际的相机标定！(￣▽￣)／

### 1.1 为什么选择棋盘格

| 优点 | 说明 |
|------|------|
| 角点易检测 | 黑白交界处对比度高 |
| 精度高 | 可以亚像素精确化 |
| 制作简单 | 打印即可使用 |
| OpenCV支持 | 内置检测函数 |

### 1.2 标定板规格

```
     ← 内角点列数 →
   ┌─┬─┬─┬─┬─┬─┬─┬─┐  ↑
   ├─┼─┼─┼─┼─┼─┼─┼─┤  │
   ├─┼─┼─┼─┼─┼─┼─┼─┤  内角点行数
   ├─┼─┼─┼─┼─┼─┼─┼─┤  │
   ├─┼─┼─┼─┼─┼─┼─┼─┤  ↓
   └─┴─┴─┴─┴─┴─┴─┴─┘

   9列×6行 = 54个内角点
   方格边长：通常20-30mm
```

## 2. 角点检测

### 2.1 cv2.findChessboardCorners

```python
retval, corners = cv2.findChessboardCorners(image, patternSize, flags)
```

**参数：**
- `image`：输入图像（灰度图）
- `patternSize`：内角点数量 (columns, rows)
- `flags`：检测标志

**返回：**
- `retval`：是否找到所有角点
- `corners`：角点坐标数组

### 2.2 代码示例1：生成棋盘格图像

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def generate_chessboard(rows=7, cols=9, square_size=50):
    """
    生成棋盘格图像

    参数:
        rows: 行数（方格数）
        cols: 列数（方格数）
        square_size: 方格边长（像素）
    """
    height = rows * square_size
    width = cols * square_size

    board = np.zeros((height, width), dtype=np.uint8)

    for i in range(rows):
        for j in range(cols):
            if (i + j) % 2 == 0:
                y = i * square_size
                x = j * square_size
                board[y:y+square_size, x:x+square_size] = 255

    return board

# 生成棋盘格
board = generate_chessboard(7, 9, 50)

# 内角点数量
inner_corners = (8, 6)  # 9-1列, 7-1行

# 检测角点
ret, corners = cv2.findChessboardCorners(board, inner_corners)

print(f"角点检测结果: {ret}")
print(f"检测到角点数: {len(corners) if corners is not None else 0}")
print(f"期望角点数: {inner_corners[0] * inner_corners[1]}")

# 绘制角点
board_color = cv2.cvtColor(board, cv2.COLOR_GRAY2BGR)
if ret:
    cv2.drawChessboardCorners(board_color, inner_corners, corners, ret)

# 可视化
plt.figure(figsize=(12, 8))
plt.imshow(cv2.cvtColor(board_color, cv2.COLOR_BGR2RGB))
plt.title(f'棋盘格角点检测 ({inner_corners[0]}×{inner_corners[1]}={inner_corners[0]*inner_corners[1]}个角点)')
```

### 2.3 代码示例2：真实图像角点检测

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def detect_chessboard_corners(image, pattern_size, show_result=True):
    """
    在图像中检测棋盘格角点

    参数:
        image: 输入图像
        pattern_size: 内角点数量 (cols, rows)
    """
    # 转灰度
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image

    # 检测角点
    flags = cv2.CALIB_CB_ADAPTIVE_THRESH + cv2.CALIB_CB_NORMALIZE_IMAGE
    ret, corners = cv2.findChessboardCorners(gray, pattern_size, flags)

    if ret and show_result:
        # 亚像素精确化
        criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001)
        corners_refined = cv2.cornerSubPix(gray, corners, (11, 11), (-1, -1), criteria)

        # 绘制
        image_draw = image.copy()
        if len(image_draw.shape) == 2:
            image_draw = cv2.cvtColor(image_draw, cv2.COLOR_GRAY2BGR)
        cv2.drawChessboardCorners(image_draw, pattern_size, corners_refined, ret)

        return ret, corners_refined, image_draw
    else:
        return ret, corners, None

# 创建模拟的"真实"棋盘格图像（带透视变换）
def create_realistic_chessboard():
    # 生成标准棋盘格
    board = generate_chessboard(7, 9, 60)

    # 添加边框
    board_with_border = np.zeros((board.shape[0]+100, board.shape[1]+100), dtype=np.uint8)
    board_with_border[:] = 200  # 灰色背景
    board_with_border[50:-50, 50:-50] = board

    # 透视变换模拟拍摄角度
    h, w = board_with_border.shape
    src_pts = np.float32([[0, 0], [w, 0], [w, h], [0, h]])
    dst_pts = np.float32([[50, 30], [w-30, 50], [w-50, h-20], [30, h-40]])

    M = cv2.getPerspectiveTransform(src_pts, dst_pts)
    warped = cv2.warpPerspective(board_with_border, M, (w, h))

    # 添加噪声
    noise = np.random.randint(0, 15, warped.shape, dtype=np.uint8)
    warped = cv2.add(warped, noise)

    return warped

# 生成模拟图像
simulated_image = create_realistic_chessboard()

# 检测角点
pattern_size = (8, 6)
ret, corners, result_image = detect_chessboard_corners(simulated_image, pattern_size)

if ret:
    print(f"成功检测到 {len(corners)} 个角点")

    # 显示前5个角点坐标
    print("\n前5个角点坐标:")
    for i in range(min(5, len(corners))):
        print(f"  角点{i+1}: ({corners[i][0][0]:.2f}, {corners[i][0][1]:.2f})")
else:
    print("角点检测失败")

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(14, 6))

axes[0].imshow(simulated_image, cmap='gray')
axes[0].set_title('原始图像')
axes[0].axis('off')

if result_image is not None:
    axes[1].imshow(cv2.cvtColor(result_image, cv2.COLOR_BGR2RGB))
    axes[1].set_title(f'角点检测结果 ({len(corners)}个角点)')
else:
    axes[1].imshow(simulated_image, cmap='gray')
    axes[1].set_title('检测失败')
axes[1].axis('off')

plt.tight_layout()
plt.savefig('realistic_detection.png', dpi=150, bbox_inches='tight')
```

## 3. 亚像素精确化

### 3.1 cv2.cornerSubPix

角点检测的初始精度通常只有像素级，需要亚像素精确化提高精度。

```python
corners = cv2.cornerSubPix(image, corners, winSize, zeroZone, criteria)
```

### 3.2 代码示例3：亚像素精确化效果

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def compare_subpixel_refinement():
    """对比亚像素精确化前后的效果"""

    # 生成棋盘格
    board = generate_chessboard(6, 8, 80)

    # 添加轻微模糊模拟真实图像
    board = cv2.GaussianBlur(board, (3, 3), 0.5)

    # 透视变换
    h, w = board.shape
    src_pts = np.float32([[0, 0], [w, 0], [w, h], [0, h]])
    dst_pts = np.float32([[20, 15], [w-15, 25], [w-25, h-10], [15, h-20]])
    M = cv2.getPerspectiveTransform(src_pts, dst_pts)
    board = cv2.warpPerspective(board, M, (w, h))

    pattern_size = (7, 5)

    # 检测角点（初始精度）
    ret, corners_raw = cv2.findChessboardCorners(board, pattern_size)

    if not ret:
        print("角点检测失败")
        return

    # 亚像素精确化
    criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.001)
    corners_refined = cv2.cornerSubPix(
        board, corners_raw.copy(),
        winSize=(11, 11),
        zeroZone=(-1, -1),
        criteria=criteria
    )

    # 计算精确化前后的差异
    differences = corners_refined - corners_raw
    diff_magnitudes = np.linalg.norm(differences.reshape(-1, 2), axis=1)

    print(f"角点数: {len(corners_raw)}")
    print(f"平均位移: {np.mean(diff_magnitudes):.4f} 像素")
    print(f"最大位移: {np.max(diff_magnitudes):.4f} 像素")

    # 可视化
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))

    # 原始检测
    board_raw = cv2.cvtColor(board, cv2.COLOR_GRAY2BGR)
    for corner in corners_raw:
        x, y = corner[0]
        cv2.circle(board_raw, (int(x), int(y)), 8, (0, 0, 255), 2)

    axes[0].imshow(cv2.cvtColor(board_raw, cv2.COLOR_BGR2RGB))
    axes[0].set_title('原始角点检测')
    axes[0].axis('off')

    # 亚像素精确化后
    board_refined = cv2.cvtColor(board, cv2.COLOR_GRAY2BGR)
    for corner in corners_refined:
        x, y = corner[0]
        cv2.circle(board_refined, (int(x), int(y)), 8, (0, 255, 0), 2)

    axes[1].imshow(cv2.cvtColor(board_refined, cv2.COLOR_BGR2RGB))
    axes[1].set_title('亚像素精确化后')
    axes[1].axis('off')

    # 局部放大对比
    # 选择一个角点区域放大
    idx = 15
    cx_raw, cy_raw = corners_raw[idx][0]
    cx_ref, cy_ref = corners_refined[idx][0]

    size = 30
    x1, y1 = int(cx_raw - size), int(cy_raw - size)
    x2, y2 = int(cx_raw + size), int(cy_raw + size)

    crop = board[max(0,y1):min(h,y2), max(0,x1):min(w,x2)]
    crop_color = cv2.cvtColor(crop, cv2.COLOR_GRAY2BGR)
    crop_color = cv2.resize(crop_color, None, fx=4, fy=4, interpolation=cv2.INTER_NEAREST)

    # 在放大图上标记
    scale = 4
    raw_x = (cx_raw - x1) * scale
    raw_y = (cy_raw - y1) * scale
    ref_x = (cx_ref - x1) * scale
    ref_y = (cy_ref - y1) * scale

    cv2.circle(crop_color, (int(raw_x), int(raw_y)), 8, (0, 0, 255), 2)
    cv2.circle(crop_color, (int(ref_x), int(ref_y)), 8, (0, 255, 0), 2)
    cv2.line(crop_color, (int(raw_x), int(raw_y)), (int(ref_x), int(ref_y)), (255, 0, 0), 2)

    axes[2].imshow(cv2.cvtColor(crop_color, cv2.COLOR_BGR2RGB))
    axes[2].set_title(f'局部放大 (位移: {diff_magnitudes[idx]:.3f}像素)\n红=原始, 绿=精确化')
    axes[2].axis('off')

    plt.tight_layout()
    plt.savefig('subpixel_refinement.png', dpi=150, bbox_inches='tight')
    plt.show()

    return diff_magnitudes

diffs = compare_subpixel_refinement()
```

## 4. 完整标定流程

### 4.1 代码示例4：模拟完整标定

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

class CameraCalibrator:
    """相机标定器"""

    def __init__(self, pattern_size, square_size):
        """
        参数:
            pattern_size: 内角点数量 (cols, rows)
            square_size: 方格边长 (mm)
        """
        self.pattern_size = pattern_size
        self.square_size = square_size

        # 生成3D点坐标
        self.objp = np.zeros((pattern_size[0] * pattern_size[1], 3), np.float32)
        self.objp[:, :2] = np.mgrid[0:pattern_size[0], 0:pattern_size[1]].T.reshape(-1, 2)
        self.objp *= square_size

        self.object_points = []
        self.image_points = []
        self.image_size = None

    def add_image(self, image):
        """添加标定图像"""
        if len(image.shape) == 3:
            gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            gray = image

        if self.image_size is None:
            self.image_size = gray.shape[::-1]

        # 检测角点
        flags = cv2.CALIB_CB_ADAPTIVE_THRESH + cv2.CALIB_CB_NORMALIZE_IMAGE
        ret, corners = cv2.findChessboardCorners(gray, self.pattern_size, flags)

        if ret:
            # 亚像素精确化
            criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001)
            corners = cv2.cornerSubPix(gray, corners, (11, 11), (-1, -1), criteria)

            self.object_points.append(self.objp)
            self.image_points.append(corners)

            return True, corners
        return False, None

    def calibrate(self):
        """执行标定"""
        if len(self.object_points) < 3:
            raise ValueError("至少需要3张有效图像")

        ret, K, dist, rvecs, tvecs = cv2.calibrateCamera(
            self.object_points,
            self.image_points,
            self.image_size,
            None, None
        )

        return {
            'rms_error': ret,
            'camera_matrix': K,
            'dist_coeffs': dist,
            'rvecs': rvecs,
            'tvecs': tvecs
        }

def generate_calibration_images(num_images=12):
    """生成模拟标定图像"""
    images = []

    # 基础棋盘格
    board = generate_chessboard(7, 9, 50)
    h, w = board.shape

    # 添加边框
    padded = np.zeros((h+100, w+100), dtype=np.uint8)
    padded[:] = 180
    padded[50:-50, 50:-50] = board
    h, w = padded.shape

    for i in range(num_images):
        # 随机透视变换
        dx = np.random.randint(-30, 30, 4)
        dy = np.random.randint(-30, 30, 4)

        src_pts = np.float32([[0, 0], [w, 0], [w, h], [0, h]])
        dst_pts = np.float32([
            [dx[0], dy[0]],
            [w+dx[1], dy[1]],
            [w+dx[2], h+dy[2]],
            [dx[3], h+dy[3]]
        ])

        M = cv2.getPerspectiveTransform(src_pts, dst_pts)
        warped = cv2.warpPerspective(padded, M, (w, h))

        # 添加噪声
        noise = np.random.randint(0, 10, warped.shape, dtype=np.uint8)
        warped = cv2.add(warped, noise)

        # 调整亮度
        brightness = np.random.randint(-20, 20)
        warped = np.clip(warped.astype(np.int32) + brightness, 0, 255).astype(np.uint8)

        images.append(warped)

    return images

# 生成标定图像
print("生成标定图像...")
images = generate_calibration_images(15)

# 创建标定器
calibrator = CameraCalibrator(pattern_size=(8, 6), square_size=50)

# 添加图像
successful = 0
for i, img in enumerate(images):
    ret, corners = calibrator.add_image(img)
    if ret:
        successful += 1
        print(f"图像 {i+1}: 检测成功")
    else:
        print(f"图像 {i+1}: 检测失败")

print(f"\n成功检测: {successful}/{len(images)} 张图像")

# 执行标定
if successful >= 3:
    result = calibrator.calibrate()

    print("\n===== 标定结果 =====")
    print(f"重投影误差: {result['rms_error']:.4f} 像素")
    print(f"\n相机内参矩阵 K:")
    print(result['camera_matrix'])
    print(f"\n畸变系数:")
    print(result['dist_coeffs'].flatten())

    # 可视化部分标定图像
    fig, axes = plt.subplots(2, 4, figsize=(16, 8))

    for i in range(min(8, len(images))):
        ax = axes[i // 4, i % 4]
        img_color = cv2.cvtColor(images[i], cv2.COLOR_GRAY2BGR)

        if i < len(calibrator.image_points):
            cv2.drawChessboardCorners(
                img_color,
                calibrator.pattern_size,
                calibrator.image_points[i],
                True
            )

        ax.imshow(cv2.cvtColor(img_color, cv2.COLOR_BGR2RGB))
        ax.set_title(f'图像 {i+1}')
        ax.axis('off')

    plt.suptitle(f'标定图像 (重投影误差: {result["rms_error"]:.4f} 像素)', fontsize=14)
    plt.tight_layout()
    plt.savefig('calibration_images.png', dpi=150, bbox_inches='tight')
    plt.show()
```

### 4.2 代码示例5：标定结果分析

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def analyze_calibration_result(calibrator, result):
    """分析标定结果"""

    K = result['camera_matrix']
    dist = result['dist_coeffs']
    rvecs = result['rvecs']
    tvecs = result['tvecs']

    print("=" * 50)
    print("标定结果详细分析")
    print("=" * 50)

    # 1. 内参分析
    print("\n[1] 内参分析:")
    fx, fy = K[0, 0], K[1, 1]
    cx, cy = K[0, 2], K[1, 2]
    print(f"  焦距: fx={fx:.2f}, fy={fy:.2f} 像素")
    print(f"  主点: cx={cx:.2f}, cy={cy:.2f} 像素")
    print(f"  焦距比: fy/fx = {fy/fx:.4f} (理想值: 1.0)")

    # 2. 畸变分析
    print("\n[2] 畸变系数:")
    dist_flat = dist.flatten()
    print(f"  k1 = {dist_flat[0]:.6f} (径向畸变)")
    print(f"  k2 = {dist_flat[1]:.6f} (径向畸变)")
    print(f"  p1 = {dist_flat[2]:.6f} (切向畸变)")
    print(f"  p2 = {dist_flat[3]:.6f} (切向畸变)")
    if len(dist_flat) > 4:
        print(f"  k3 = {dist_flat[4]:.6f} (径向畸变)")

    # 3. 外参分析
    print("\n[3] 外参分析 (相机到棋盘格的变换):")
    for i, (rvec, tvec) in enumerate(zip(rvecs, tvecs)):
        R, _ = cv2.Rodrigues(rvec)
        distance = np.linalg.norm(tvec)
        print(f"  图像{i+1}: 距离={distance:.1f}mm")

    # 4. 计算每张图像的重投影误差
    print("\n[4] 每张图像的重投影误差:")
    errors = []
    for i in range(len(calibrator.object_points)):
        imgpts, _ = cv2.projectPoints(
            calibrator.object_points[i],
            rvecs[i], tvecs[i],
            K, dist
        )
        error = cv2.norm(calibrator.image_points[i], imgpts, cv2.NORM_L2)
        error /= len(imgpts)
        errors.append(error)
        print(f"  图像{i+1}: {error:.4f} 像素")

    print(f"\n  平均误差: {np.mean(errors):.4f} 像素")
    print(f"  最大误差: {np.max(errors):.4f} 像素")
    print(f"  最小误差: {np.min(errors):.4f} 像素")

    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))

    # 重投影误差柱状图
    axes[0, 0].bar(range(1, len(errors)+1), errors, color='steelblue', edgecolor='black')
    axes[0, 0].axhline(y=np.mean(errors), color='r', linestyle='--',
                       label=f'平均: {np.mean(errors):.4f}')
    axes[0, 0].set_xlabel('图像编号')
    axes[0, 0].set_ylabel('重投影误差 (像素)')
    axes[0, 0].set_title('各图像重投影误差')
    axes[0, 0].legend()

    # 相机位置分布
    positions = np.array([-R.T @ t for R, t in
                         zip([cv2.Rodrigues(r)[0] for r in rvecs], tvecs)])

    axes[0, 1].scatter(positions[:, 0], positions[:, 1], c='blue', s=100)
    for i, pos in enumerate(positions):
        axes[0, 1].annotate(str(i+1), (pos[0]+5, pos[1]+5))
    axes[0, 1].set_xlabel('X (mm)')
    axes[0, 1].set_ylabel('Y (mm)')
    axes[0, 1].set_title('相机位置分布 (俯视图)')
    axes[0, 1].set_aspect('equal')
    axes[0, 1].grid(True, alpha=0.3)

    # 畸变系数可视化
    dist_labels = ['k1', 'k2', 'p1', 'p2', 'k3'][:len(dist_flat)]
    colors = ['red' if d < 0 else 'green' for d in dist_flat]
    axes[1, 0].bar(dist_labels, dist_flat, color=colors, edgecolor='black')
    axes[1, 0].axhline(y=0, color='k', linewidth=0.5)
    axes[1, 0].set_ylabel('畸变系数值')
    axes[1, 0].set_title('畸变系数')

    # 内参可视化
    params = ['fx', 'fy', 'cx', 'cy']
    values = [fx, fy, cx, cy]
    axes[1, 1].bar(params, values, color='purple', edgecolor='black', alpha=0.7)
    axes[1, 1].set_ylabel('像素')
    axes[1, 1].set_title('相机内参')

    plt.tight_layout()
    plt.savefig('calibration_analysis.png', dpi=150, bbox_inches='tight')
    plt.show()

    return errors

# 使用之前的标定结果

    errors = analyze_calibration_result(calibrator, result)
```

```python
def save_calibration(result, filename='camera_calibration.npz'):
    """保存标定结果到文件"""

    np.savez(
        filename,
        camera_matrix=result['camera_matrix'],
        dist_coeffs=result['dist_coeffs'],
        rms_error=result['rms_error']
    )

    print(f"标定结果已保存到: {filename}")

    # 同时保存可读的JSON格式
    json_filename = filename.replace('.npz', '.json')
    json_data = {
        'camera_matrix': result['camera_matrix'].tolist(),
        'dist_coeffs': result['dist_coeffs'].flatten().tolist(),
        'rms_error': float(result['rms_error']),
        'fx': float(result['camera_matrix'][0, 0]),
        'fy': float(result['camera_matrix'][1, 1]),
        'cx': float(result['camera_matrix'][0, 2]),
        'cy': float(result['camera_matrix'][1, 2])
    }

    with open(json_filename, 'w') as f:
        json.dump(json_data, f, indent=2)

    print(f"可读格式已保存到: {json_filename}")

def load_calibration(filename='camera_calibration.npz'):
    """从文件加载标定结果"""

    if not os.path.exists(filename):
        raise FileNotFoundError(f"标定文件不存在: {filename}")

    data = np.load(filename)

    result = {
        'camera_matrix': data['camera_matrix'],
        'dist_coeffs': data['dist_coeffs'],
        'rms_error': float(data['rms_error'])
    }

    print(f"已加载标定结果: {filename}")
    print(f"重投影误差: {result['rms_error']:.4f}")

    return result

# 示例：创建模拟标定结果并保存
```

```python
import numpy as np

def check_calibration_quality(result, image_size):
    """检查标定质量"""

    K = result['camera_matrix']
    dist = result['dist_coeffs']
    rms = result['rms_error']

    print("=" * 50)
    print("标定质量检查报告")
    print("=" * 50)

    quality_score = 100

    # 1. 重投影误差检查
    print("\n[1] 重投影误差检查:")
    if rms < 0.3:
        print(f"  ✓ 优秀 ({rms:.4f} < 0.3 像素)")
    elif rms < 0.5:
        print(f"  ✓ 良好 ({rms:.4f} < 0.5 像素)")
        quality_score -= 10
    elif rms < 1.0:
        print(f"  △ 一般 ({rms:.4f} < 1.0 像素)")
        quality_score -= 25
    else:
        print(f"  ✗ 较差 ({rms:.4f} >= 1.0 像素)")
        quality_score -= 50

    # 2. 焦距比检查
    fx, fy = K[0, 0], K[1, 1]
    ratio = fy / fx
    print("\n[2] 焦距比检查:")
    if 0.99 < ratio < 1.01:
        print(f"  ✓ 正常 (fy/fx = {ratio:.4f})")
    else:
        print(f"  △ 略有偏差 (fy/fx = {ratio:.4f})")
        quality_score -= 10

    # 3. 主点位置检查
    cx, cy = K[0, 2], K[1, 2]
    w, h = image_size
    cx_expected, cy_expected = w/2, h/2
    cx_diff = abs(cx - cx_expected) / w * 100
    cy_diff = abs(cy - cy_expected) / h * 100

    print("\n[3] 主点位置检查:")
    if cx_diff < 5 and cy_diff < 5:
        print(f"  ✓ 正常 (偏离中心 < 5%)")
    elif cx_diff < 10 and cy_diff < 10:
        print(f"  △ 略有偏移 (偏离中心 < 10%)")
        quality_score -= 10
    else:
        print(f"  ✗ 偏移较大 (偏离中心 >= 10%)")
        quality_score -= 20

    # 4. 畸变系数检查
    dist_flat = dist.flatten()
    print("\n[4] 畸变系数检查:")
    if abs(dist_flat[0]) < 0.5:
        print(f"  ✓ k1正常 ({dist_flat[0]:.4f})")
    else:
        print(f"  △ k1较大 ({dist_flat[0]:.4f})")
        quality_score -= 10

    # 5. 总评
    print("\n" + "=" * 50)
    print(f"综合质量评分: {quality_score}/100")
    if quality_score >= 90:
        print("评级: 优秀 ⭐⭐⭐⭐⭐")
    elif quality_score >= 75:
        print("评级: 良好 ⭐⭐⭐⭐")
    elif quality_score >= 60:
        print("评级: 一般 ⭐⭐⭐")
    else:
        print("评级: 需要改进 ⭐⭐")
        print("\n建议:")
        print("  - 增加标定图像数量")
        print("  - 确保棋盘格覆盖更多图像区域")
        print("  - 检查棋盘格是否平整")
        print("  - 减少运动模糊")

    return quality_score
```

### 7.1 代码示例8：圆点阵列标定

```python
import numpy as np
import cv2
import matplotlib.pyplot as plt

def create_circle_grid(rows=5, cols=7, circle_radius=15, spacing=50):
    """创建圆点阵列标定板"""

    height = (rows + 1) * spacing
    width = (cols + 1) * spacing

    board = np.ones((height, width), dtype=np.uint8) * 255

    for i in range(rows):
        for j in range(cols):
            cy = (i + 1) * spacing
            cx = (j + 1) * spacing
            cv2.circle(board, (cx, cy), circle_radius, 0, -1)

    return board

def detect_circle_grid(image, pattern_size, is_symmetric=True):
    """检测圆点阵列"""

    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image

    # 设置检测参数
    params = cv2.SimpleBlobDetector_Params()
    params.minThreshold = 10
    params.maxThreshold = 200
    params.filterByArea = True
    params.minArea = 100
    params.filterByCircularity = True
    params.minCircularity = 0.8
    params.filterByConvexity = True
    params.minConvexity = 0.87
    params.filterByInertia = True
    params.minInertiaRatio = 0.5

    detector = cv2.SimpleBlobDetector_create(params)

    if is_symmetric:
        flags = cv2.CALIB_CB_SYMMETRIC_GRID
    else:
        flags = cv2.CALIB_CB_ASYMMETRIC_GRID

    ret, centers = cv2.findCirclesGrid(gray, pattern_size, None,
                                        flags=flags, blobDetector=detector)

    return ret, centers

# 创建圆点阵列
circle_board = create_circle_grid(5, 7, 12, 40)

# 检测
pattern_size = (7, 5)  # (cols, rows)
ret, centers = detect_circle_grid(circle_board, pattern_size)

print(f"圆点检测结果: {ret}")
if centers is not None:
    print(f"检测到圆点数: {len(centers)}")

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(12, 5))

axes[0].imshow(circle_board, cmap='gray')
axes[0].set_title('圆点阵列标定板')
axes[0].axis('off')

board_color = cv2.cvtColor(circle_board, cv2.COLOR_GRAY2BGR)
if ret:
    cv2.drawChessboardCorners(board_color, pattern_size, centers, ret)

axes[1].imshow(cv2.cvtColor(board_color, cv2.COLOR_BGR2RGB))
axes[1].set_title(f'圆心检测结果')
axes[1].axis('off')

plt.tight_layout()
plt.savefig('circle_grid.png', dpi=150, bbox_inches='tight')
plt.show()
```
