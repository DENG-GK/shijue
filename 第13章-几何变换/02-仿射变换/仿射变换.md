# 仿射变换（Affine Transformation）

## 1. 仿射变换概述

仿射变换是一种保持点、直线和平面关系的线性变换。它包括平移、旋转、缩放、错切（剪切）及其组合，是图像处理中最常用的几何变换之一。

### 1.1 仿射变换的定义

仿射变换可以用以下公式表示：

$$\begin{bmatrix} x' \\ y' \end{bmatrix} = \begin{bmatrix} a_{11} & a_{12} \\ a_{21} & a_{22} \end{bmatrix} \begin{bmatrix} x \\ y \end{bmatrix} + \begin{bmatrix} t_x \\ t_y \end{bmatrix}$$

或使用齐次坐标：

$$\begin{bmatrix} x' \\ y' \\ 1 \end{bmatrix} = \begin{bmatrix} a_{11} & a_{12} & t_x \\ a_{21} & a_{22} & t_y \\ 0 & 0 & 1 \end{bmatrix} \begin{bmatrix} x \\ y \\ 1 \end{bmatrix}$$

### 1.2 仿射变换的特性

| 特性 | 说明 |
|------|------|
| **保持共线性** | 变换后直线仍是直线 |
| **保持平行性** | 平行线变换后仍平行 |
| **保持比例** | 线段上的比例关系保持 |
| **6个自由度** | 需要6个参数确定变换 |

### 1.3 仿射变换矩阵结构

```
      ┌                    ┐
      │ a₁₁  a₁₂  tₓ │
M =   │ a₂₁  a₂₂  tᵧ │
      └                    ┘

其中：
- [a₁₁ a₁₂; a₂₁ a₂₂]: 线性变换部分（旋转、缩放、错切）
- [tₓ; tᵧ]: 平移部分
```

```python
dst = cv2.warpAffine(src, M, dsize, flags, borderMode, borderValue)
```

- `M`：2×3变换矩阵
- `dsize`：输出图像尺寸 (width, height)
- `flags`：插值方法
- `borderMode`：边界模式
- `borderValue`：边界填充值

### 示例1：基本仿射变换

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def basic_affine_transform():
    """
    Demonstrate basic affine transformation
    """
    # Create test image
    image = np.zeros((300, 400, 3), dtype=np.uint8)
    image[:, :] = [220, 220, 220]
    cv2.rectangle(image, (50, 50), (350, 250), (0, 128, 255), -1)
    cv2.putText(image, 'AFFINE', (100, 165), cv2.FONT_HERSHEY_SIMPLEX, 2, (255, 255, 255), 3)
    cv2.circle(image, (80, 80), 20, (255, 0, 0), -1)

    h, w = image.shape[:2]

    # Define source points (triangle)
    src_pts = np.float32([
        [50, 50],    # Top-left
        [350, 50],   # Top-right
        [50, 250]    # Bottom-left
    ])

    # Different destination points for different effects
    transformations = {
        'Original': np.float32([[50, 50], [350, 50], [50, 250]]),
        'Translation': np.float32([[100, 80], [400, 80], [100, 280]]),
        'Rotation': np.float32([[100, 150], [300, 50], [0, 250]]),
        'Scale': np.float32([[25, 25], [375, 25], [25, 275]]),
        'Shear X': np.float32([[100, 50], [400, 50], [50, 250]]),
        'Shear Y': np.float32([[50, 100], [350, 50], [50, 300]]),
    }

    results = {}
    for name, dst_pts in transformations.items():
        if name == 'Original':
            results[name] = image.copy()
        else:
            M = cv2.getAffineTransform(src_pts, dst_pts)
            results[name] = cv2.warpAffine(image, M, (w, h))

    # Display
    plt.figure(figsize=(18, 10))

    for i, (name, result) in enumerate(results.items()):
        plt.subplot(2, 3, i+1)
        plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB))
        plt.title(name)
        plt.axis('off')

        # Draw triangles
        if name != 'Original':
            pts = transformations[name].astype(np.int32)
            for j in range(3):
                cv2.circle(result, tuple(pts[j]), 5, (0, 255, 0), -1)

    plt.tight_layout()
    plt.savefig('basic_affine.png', dpi=150, bbox_inches='tight')
    plt.show()

    print("Basic affine transformations completed!")
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def affine_from_three_points():
    """
    Define affine transformation using three corresponding points
    """
    # Create test image with visible markers
    image = np.zeros((300, 400, 3), dtype=np.uint8)
    image[:, :] = [200, 200, 200]

    # Draw a triangle with markers at vertices
    pts = np.array([[100, 50], [300, 50], [200, 250]], np.int32)
    cv2.fillPoly(image, [pts], (0, 150, 255))

    # Mark the three control points
    colors = [(255, 0, 0), (0, 255, 0), (0, 0, 255)]
    labels = ['P1', 'P2', 'P3']
    for i, (pt, color, label) in enumerate(zip(pts, colors, labels)):
        cv2.circle(image, tuple(pt), 10, color, -1)
        cv2.putText(image, label, (pt[0]-15, pt[1]-15),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)

    h, w = image.shape[:2]

    # Source points
    src_pts = pts.astype(np.float32)

    # Different destination configurations
    dest_configs = {
        'Original': pts.astype(np.float32),
        'Move P1': np.float32([[150, 80], [300, 50], [200, 250]]),
        'Move P2': np.float32([[100, 50], [350, 100], [200, 250]]),
        'Move P3': np.float32([[100, 50], [300, 50], [150, 200]]),
        'Move All': np.float32([[50, 100], [350, 80], [180, 280]]),
        'Flip': np.float32([[300, 50], [100, 50], [200, 250]]),
    }

    # Apply transformations
    results = {'Original': image.copy()}
    matrices = {'Original': None}

    for name, dst_pts in dest_configs.items():
        if name != 'Original':
            M = cv2.getAffineTransform(src_pts, dst_pts)
            result = cv2.warpAffine(image, M, (w, h),
                                     borderMode=cv2.BORDER_CONSTANT,
                                     borderValue=(128, 128, 128))

            # Mark destination points on result
            for pt, color in zip(dst_pts.astype(np.int32), colors):
                cv2.circle(result, tuple(pt), 8, color, 2)

            results[name] = result
            matrices[name] = M

    # Display
    plt.figure(figsize=(18, 10))

    for i, (name, result) in enumerate(results.items()):
        plt.subplot(2, 3, i+1)
        plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB))
        plt.title(name)
        plt.axis('off')

    plt.tight_layout()
    plt.savefig('three_point_affine.png', dpi=150, bbox_inches='tight')
    plt.show()

    # Print some matrices
    print("\nAffine Transformation Matrices:")
    print("="*60)
    for name, M in matrices.items():
        if M is not None:
            print(f"\n{name}:")
            print(f"  [{M[0,0]:.4f}  {M[0,1]:.4f}  {M[0,2]:.2f}]")
            print(f"  [{M[1,0]:.4f}  {M[1,1]:.4f}  {M[1,2]:.2f}]")
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def build_affine_matrices():
    """
    Manually build different types of affine transformation matrices
    """
    # Create test image
    image = np.zeros((200, 300, 3), dtype=np.uint8)
    image[:, :] = [220, 220, 220]
    cv2.rectangle(image, (50, 30), (250, 170), (0, 128, 255), -1)
    cv2.circle(image, (80, 60), 15, (255, 0, 0), -1)  # Marker
    cv2.putText(image, 'M', (120, 120), cv2.FONT_HERSHEY_SIMPLEX, 2, (255, 255, 255), 3)

    h, w = image.shape[:2]
    cx, cy = w // 2, h // 2  # Center

    def create_translation_matrix(tx, ty):
        """Create translation matrix"""
        return np.float32([
            [1, 0, tx],
            [0, 1, ty]
        ])

    def create_rotation_matrix(angle, center):
        """Create rotation matrix around center"""
        return cv2.getRotationMatrix2D(center, angle, 1.0)

    def create_scale_matrix(sx, sy, center):
        """Create scale matrix around center"""
        # Scale around center = translate to origin, scale, translate back
        M = np.float32([
            [sx, 0, center[0] * (1 - sx)],
            [0, sy, center[1] * (1 - sy)]
        ])
        return M

    def create_shear_matrix(shx, shy):
        """Create shear matrix"""
        return np.float32([
            [1, shx, 0],
            [shy, 1, 0]
        ])

    def create_flip_matrix(flip_x, flip_y, center):
        """Create flip matrix"""
        sx = -1 if flip_x else 1
        sy = -1 if flip_y else 1
        return np.float32([
            [sx, 0, center[0] * (1 - sx)],
            [0, sy, center[1] * (1 - sy)]
        ])

    # Apply different transformations
    transformations = {
        'Original': np.float32([[1, 0, 0], [0, 1, 0]]),
        'Translate (50, 20)': create_translation_matrix(50, 20),
        'Rotate 30°': create_rotation_matrix(30, (cx, cy)),
        'Scale (1.3, 0.7)': create_scale_matrix(1.3, 0.7, (cx, cy)),
        'Shear X (0.3)': create_shear_matrix(0.3, 0),
        'Shear Y (0.3)': create_shear_matrix(0, 0.3),
        'Flip Horizontal': create_flip_matrix(True, False, (cx, cy)),
        'Flip Vertical': create_flip_matrix(False, True, (cx, cy)),
    }

    # Display
    plt.figure(figsize=(20, 10))

    for i, (name, M) in enumerate(transformations.items()):
        result = cv2.warpAffine(image, M, (w, h))

        plt.subplot(2, 4, i+1)
        plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB))
        plt.title(f'{name}\n[{M[0,0]:.2f} {M[0,1]:.2f} {M[0,2]:.1f}]\n[{M[1,0]:.2f} {M[1,1]:.2f} {M[1,2]:.1f}]',
                  fontsize=9)
        plt.axis('off')

    plt.tight_layout()
    plt.savefig('build_affine_matrices.png', dpi=150, bbox_inches='tight')
    plt.show()
```

### 示例4：组合多个仿射变换

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def combine_affine_transforms():
    """
    Combine multiple affine transformations into one
    """
    # Create test image
    image = np.zeros((300, 400, 3), dtype=np.uint8)
    image[:, :] = [200, 200, 200]
    cv2.rectangle(image, (100, 75), (300, 225), (0, 150, 255), -1)
    cv2.circle(image, (130, 105), 15, (255, 0, 0), -1)
    cv2.putText(image, 'COMBO', (130, 170), cv2.FONT_HERSHEY_SIMPLEX, 1.2, (255, 255, 255), 2)

    h, w = image.shape[:2]
    cx, cy = w // 2, h // 2

    # Individual transformation matrices (as 3x3 for multiplication)
    def to_3x3(M):
        return np.vstack([M, [0, 0, 1]])

    def to_2x3(M):
        return M[:2, :]

    # Step 1: Translate to origin
    T1 = to_3x3(np.float32([[1, 0, -cx], [0, 1, -cy]]))

    # Step 2: Scale
    S = to_3x3(np.float32([[1.2, 0, 0], [0, 0.8, 0]]))

    # Step 3: Rotate
    angle = np.radians(25)
    R = to_3x3(np.float32([
        [np.cos(angle), -np.sin(angle), 0],
        [np.sin(angle), np.cos(angle), 0]
    ]))

    # Step 4: Translate back + offset
    T2 = to_3x3(np.float32([[1, 0, cx + 30], [0, 1, cy + 20]]))

    # Combine: T2 @ R @ S @ T1
    M_combined = T2 @ R @ S @ T1

    # Apply step by step for visualization
    results = {
        'Original': np.float32([[1, 0, 0], [0, 1, 0]]),
        'Step 1: Center': to_2x3(T1),
        'Step 2: +Scale': to_2x3(S @ T1),
        'Step 3: +Rotate': to_2x3(R @ S @ T1),
        'Step 4: +Translate': to_2x3(M_combined),
    }

    # Apply and display
    plt.figure(figsize=(20, 8))

    for i, (name, M) in enumerate(results.items()):
        result = cv2.warpAffine(image, M, (w, h),
                                 borderMode=cv2.BORDER_CONSTANT,
                                 borderValue=(128, 128, 128))

        plt.subplot(1, 5, i+1)
        plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB))
        plt.title(name, fontsize=10)
        plt.axis('off')

    plt.tight_layout()
    plt.savefig('combine_affine.png', dpi=150, bbox_inches='tight')
    plt.show()

    # Print combined matrix
    print("\nCombined Transformation Matrix:")
    print("M = T2 @ R @ S @ T1")
    print(f"\n[{M_combined[0,0]:.4f}  {M_combined[0,1]:.4f}  {M_combined[0,2]:.2f}]")
    print(f"[{M_combined[1,0]:.4f}  {M_combined[1,1]:.4f}  {M_combined[1,2]:.2f}]")
    print(f"[{M_combined[2,0]:.4f}  {M_combined[2,1]:.4f}  {M_combined[2,2]:.2f}]")
```

### 示例5：图像校正（倾斜校正）

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def skew_correction():
    """
    Correct skewed/tilted images using affine transformation
    """
    # Create a skewed document image
    image = np.ones((300, 400, 3), dtype=np.uint8) * 255

    # Add text lines (simulating document)
    for i in range(8):
        y = 40 + i * 30
        cv2.line(image, (30, y), (370, y), (0, 0, 0), 2)

    # Add some text
    cv2.putText(image, 'DOCUMENT', (100, 160), cv2.FONT_HERSHEY_SIMPLEX, 1.5, (0, 0, 0), 2)

    h, w = image.shape[:2]

    # Apply artificial skew
    skew_angle = 15
    skew_M = cv2.getRotationMatrix2D((w//2, h//2), skew_angle, 1.0)
    skewed = cv2.warpAffine(image, skew_M, (w, h), borderValue=(200, 200, 200))

    # Method 1: Detect skew angle using Hough lines
    def detect_skew_angle(img):
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        edges = cv2.Canny(gray, 50, 150, apertureSize=3)

        lines = cv2.HoughLines(edges, 1, np.pi/180, 100)
        if lines is None:
            return 0

        angles = []
        for rho, theta in lines[:, 0]:
            angle = np.degrees(theta) - 90
            if -45 < angle < 45:  # Filter reasonable angles
                angles.append(angle)

        if angles:
            return np.median(angles)
        return 0

    detected_angle = detect_skew_angle(skewed)

    # Correct the skew
    correction_M = cv2.getRotationMatrix2D((w//2, h//2), detected_angle, 1.0)
    corrected = cv2.warpAffine(skewed, correction_M, (w, h), borderValue=(255, 255, 255))

    # Display
    plt.figure(figsize=(16, 8))

    plt.subplot(2, 3, 1)
    plt.imshow(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
    plt.title('Original Document')
    plt.axis('off')

    plt.subplot(2, 3, 2)
    plt.imshow(cv2.cvtColor(skewed, cv2.COLOR_BGR2RGB))
    plt.title(f'Skewed ({skew_angle}°)')
    plt.axis('off')

    plt.subplot(2, 3, 3)
    plt.imshow(cv2.cvtColor(corrected, cv2.COLOR_BGR2RGB))
    plt.title(f'Corrected (detected: {detected_angle:.1f}°)')
    plt.axis('off')

    # Show edge detection for angle measurement
    plt.subplot(2, 3, 4)
    gray = cv2.cvtColor(skewed, cv2.COLOR_BGR2GRAY)
    edges = cv2.Canny(gray, 50, 150)
    plt.imshow(edges, cmap='gray')
    plt.title('Edge Detection')
    plt.axis('off')

    # Show detected lines
    plt.subplot(2, 3, 5)
    line_img = skewed.copy()
    lines = cv2.HoughLines(edges, 1, np.pi/180, 100)
    if lines is not None:
        for rho, theta in lines[:10, 0]:
            a = np.cos(theta)
            b = np.sin(theta)
            x0 = a * rho
            y0 = b * rho
            x1 = int(x0 + 1000 * (-b))
            y1 = int(y0 + 1000 * (a))
            x2 = int(x0 - 1000 * (-b))
            y2 = int(y0 - 1000 * (a))
            cv2.line(line_img, (x1, y1), (x2, y2), (0, 255, 0), 1)
    plt.imshow(cv2.cvtColor(line_img, cv2.COLOR_BGR2RGB))
    plt.title('Detected Lines')
    plt.axis('off')

    plt.tight_layout()
    plt.savefig('skew_correction.png', dpi=150, bbox_inches='tight')
    plt.show()

    print(f"Original skew angle: {skew_angle}°")
    print(f"Detected skew angle: {detected_angle:.2f}°")

skew_correction()
```

```python
def mirror_effects():
    """
    Create various mirror and flip effects
    """

    # Different mirror effects
    # 1. Horizontal mirror (left-right)
    h_mirror = cv2.flip(image, 1)

    # 2. Vertical mirror (top-bottom)
    v_mirror = cv2.flip(image, 0)

    # 3. Both axes (180° rotation)
    both_mirror = cv2.flip(image, -1)

    # 4. Create reflection effect (image + mirror below)
    reflection = np.vstack([image, cv2.flip(image, 0)])
    # Add gradient fade to reflection
    for i in range(h):
        alpha = 1 - (i / h) * 0.7
        reflection[h + i, :] = (reflection[h + i, :] * alpha).astype(np.uint8)

    # 5. Create kaleidoscope effect
    def create_kaleidoscope(img):
        h, w = img.shape[:2]
        # Create 4 quadrants
        top_left = img
        top_right = cv2.flip(img, 1)
        bottom_left = cv2.flip(img, 0)
        bottom_right = cv2.flip(img, -1)

        top = np.hstack([top_left, top_right])
        bottom = np.hstack([bottom_left, bottom_right])
        return np.vstack([top, bottom])

    kaleidoscope = create_kaleidoscope(image)

    # Display
    plt.figure(figsize=(18, 12))

    plt.imshow(cv2.cvtColor(both_mirror, cv2.COLOR_BGR2RGB))
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt
import random

def data_augmentation():
    """
    Use affine transformations for data augmentation
    """
    # Create sample image
    image = np.zeros((200, 200, 3), dtype=np.uint8)
    image[:, :] = [100, 150, 200]
    cv2.circle(image, (100, 100), 60, (255, 100, 50), -1)
    cv2.rectangle(image, (60, 60), (140, 140), (50, 200, 100), 3)
    cv2.putText(image, 'A', (75, 115), cv2.FONT_HERSHEY_SIMPLEX, 1.5, (255, 255, 255), 2)

    h, w = image.shape[:2]
    cx, cy = w // 2, h // 2

    def random_affine_transform(img, max_rotation=15, max_scale=0.15,
                                 max_shear=0.1, max_translate=0.1):
        """Apply random affine transformation"""
        h, w = img.shape[:2]
        cx, cy = w // 2, h // 2

        # Random parameters
        angle = random.uniform(-max_rotation, max_rotation)
        scale = random.uniform(1 - max_scale, 1 + max_scale)
        shear_x = random.uniform(-max_shear, max_shear)
        shear_y = random.uniform(-max_shear, max_shear)
        tx = random.uniform(-max_translate, max_translate) * w
        ty = random.uniform(-max_translate, max_translate) * h

        # Build transformation
        # Center
        T1 = np.array([[1, 0, -cx], [0, 1, -cy], [0, 0, 1]], dtype=np.float32)

        # Shear
        Sh = np.array([[1, shear_x, 0], [shear_y, 1, 0], [0, 0, 1]], dtype=np.float32)

        # Scale
        S = np.array([[scale, 0, 0], [0, scale, 0], [0, 0, 1]], dtype=np.float32)

        # Rotate
        rad = np.radians(angle)
        R = np.array([
            [np.cos(rad), -np.sin(rad), 0],
            [np.sin(rad), np.cos(rad), 0],
            [0, 0, 1]
        ], dtype=np.float32)

        # Translate back + offset
        T2 = np.array([[1, 0, cx + tx], [0, 1, cy + ty], [0, 0, 1]], dtype=np.float32)

        # Combine
        M = T2 @ R @ S @ Sh @ T1

        # Apply
        result = cv2.warpAffine(img, M[:2, :], (w, h),
                                 borderMode=cv2.BORDER_REFLECT_101)

        return result

    # Generate augmented samples
    n_augmented = 15
    augmented = [image]  # Original first

    for _ in range(n_augmented):
        aug = random_affine_transform(image)
        augmented.append(aug)

    augmented.append(cv2.flip(image, 1))
    augmented.append(cv2.flip(image, 0))

    # Display

    for i, img in enumerate(augmented[:18]):
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def affine_estimation_least_squares():
    """
    Estimate affine transformation from multiple point correspondences
    using least squares (more than 3 points)
    """
    # Create test image
    image = np.zeros((300, 400, 3), dtype=np.uint8)
    image[:, :] = [200, 200, 200]

    # Define feature points
    src_pts = np.array([
        [50, 50], [350, 50], [350, 250], [50, 250],
        [200, 100], [200, 200], [100, 150], [300, 150]
    ], dtype=np.float32)

    # Mark points on source
    for i, pt in enumerate(src_pts):
        cv2.circle(image, tuple(pt.astype(int)), 8, (0, 255, 0), -1)
        cv2.putText(image, str(i), (int(pt[0])+10, int(pt[1])),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 0), 1)

    h, w = image.shape[:2]

    # True transformation
    true_M = np.float32([
        [0.85, -0.2, 50],
        [0.15, 0.9, 30]
    ])

    # Transform points
    ones = np.ones((len(src_pts), 1))
    src_h = np.hstack([src_pts, ones])
    dst_pts = (src_h @ true_M.T).astype(np.float32)

    # Add noise to destination points
    noise = np.random.normal(0, 2, dst_pts.shape).astype(np.float32)
    dst_pts_noisy = dst_pts + noise

    # Estimate transformation using different methods

    # Method 1: Using only 3 points
    M_3pts = cv2.getAffineTransform(src_pts[:3], dst_pts_noisy[:3])

    # Method 2: Using estimateAffine2D with all points (robust estimation)
    M_robust, inliers = cv2.estimateAffine2D(src_pts, dst_pts_noisy,
                                               method=cv2.RANSAC)

    # Method 3: Using estimateAffinePartial2D (only rotation, translation, scale)
    M_partial, _ = cv2.estimateAffinePartial2D(src_pts, dst_pts_noisy)

    # Apply transformations
    transformed_true = cv2.warpAffine(image, true_M, (w, h))
    transformed_3pts = cv2.warpAffine(image, M_3pts, (w, h))
    transformed_robust = cv2.warpAffine(image, M_robust, (w, h))
    transformed_partial = cv2.warpAffine(image, M_partial, (w, h))

    # Display
    plt.figure(figsize=(18, 10))

    plt.subplot(2, 4, 1)
    plt.imshow(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
    plt.title('Source with Points')
    plt.axis('off')

    plt.subplot(2, 4, 2)
    plt.imshow(cv2.cvtColor(transformed_true, cv2.COLOR_BGR2RGB))
    plt.title('True Transform')
    plt.axis('off')

    plt.subplot(2, 4, 3)
    plt.imshow(cv2.cvtColor(transformed_3pts, cv2.COLOR_BGR2RGB))
    plt.title('3-Point Estimate')
    plt.axis('off')

    plt.subplot(2, 4, 4)
    plt.imshow(cv2.cvtColor(transformed_robust, cv2.COLOR_BGR2RGB))
    plt.title('RANSAC Estimate (All Points)')
    plt.axis('off')

    plt.subplot(2, 4, 5)
    plt.imshow(cv2.cvtColor(transformed_partial, cv2.COLOR_BGR2RGB))
    plt.title('Partial Affine\n(Rotation+Scale+Translation)')
    plt.axis('off')

    # Show point correspondences
    plt.subplot(2, 4, 6)
    for i in range(len(src_pts)):
        plt.plot([src_pts[i, 0], dst_pts_noisy[i, 0]],
                 [src_pts[i, 1], dst_pts_noisy[i, 1]], 'b-', alpha=0.5)
        plt.plot(src_pts[i, 0], src_pts[i, 1], 'go', markersize=8)
        plt.plot(dst_pts_noisy[i, 0], dst_pts_noisy[i, 1], 'rx', markersize=8)
    plt.legend(['Correspondence', 'Source', 'Target'], loc='upper right')
    plt.title('Point Correspondences')
    plt.axis('equal')
    plt.grid(True, alpha=0.3)

    # Error comparison
    plt.subplot(2, 4, 7)
    errors = {
        'True': 0,
        '3-Point': np.linalg.norm(true_M - M_3pts),
        'RANSAC': np.linalg.norm(true_M - M_robust),
        'Partial': np.linalg.norm(true_M - M_partial),
    }
    plt.bar(errors.keys(), errors.values(), color=['green', 'red', 'blue', 'orange'])
    plt.ylabel('Matrix Frobenius Norm Error')
    plt.title('Estimation Error')

    plt.tight_layout()
    plt.savefig('affine_estimation.png', dpi=150, bbox_inches='tight')
    plt.show()

    # Print matrices
    print("\nTransformation Matrices Comparison:")
    print("="*60)
    print("\nTrue:")
    print(true_M)
    print("\n3-Point:")
    print(M_3pts)
    print("\nRANSAC (All Points):")
    print(M_robust)
    print("\nPartial Affine:")
    print(M_partial)
```

```python
import cv2
import numpy as np

def create_affine_tool():
    """
    Create interactive affine transformation tool
    """
    # Create test image
    image = np.zeros((300, 400, 3), dtype=np.uint8)
    image[:, :] = [220, 220, 220]
    cv2.rectangle(image, (100, 75), (300, 225), (0, 128, 255), -1)
    cv2.putText(image, 'AFFINE', (130, 165), cv2.FONT_HERSHEY_SIMPLEX, 1.2, (255, 255, 255), 2)
    cv2.circle(image, (130, 105), 15, (255, 0, 0), -1)

    h, w = image.shape[:2]
    cx, cy = w // 2, h // 2

    # Parameters
    params = {
        'rotation': 0,
        'scale_x': 100,
        'scale_y': 100,
        'shear_x': 0,
        'shear_y': 0,
        'trans_x': 50,
        'trans_y': 50,
    }

    def update_transform(x=None):
        # Get parameters
        angle = params['rotation'] - 180
        sx = params['scale_x'] / 100.0
        sy = params['scale_y'] / 100.0
        shx = (params['shear_x'] - 50) / 100.0
        shy = (params['shear_y'] - 50) / 100.0
        tx = params['trans_x'] - 50
        ty = params['trans_y'] - 50

        # Build transformation
        # Center
        T1 = np.array([[1, 0, -cx], [0, 1, -cy], [0, 0, 1]], dtype=np.float32)

        # Shear
        Sh = np.array([[1, shx, 0], [shy, 1, 0], [0, 0, 1]], dtype=np.float32)

        # Scale
        S = np.array([[sx, 0, 0], [0, sy, 0], [0, 0, 1]], dtype=np.float32)

        # Rotate
        rad = np.radians(angle)
        R = np.array([
            [np.cos(rad), -np.sin(rad), 0],
            [np.sin(rad), np.cos(rad), 0],
            [0, 0, 1]
        ], dtype=np.float32)

        # Translate back + offset
        T2 = np.array([[1, 0, cx + tx], [0, 1, cy + ty], [0, 0, 1]], dtype=np.float32)

        # Combine
        M = T2 @ R @ S @ Sh @ T1

        # Apply
        result = cv2.warpAffine(image, M[:2, :], (w, h),
                                 borderMode=cv2.BORDER_CONSTANT,
                                 borderValue=(128, 128, 128))

        # Add info overlay
        info = f"Rot:{angle:.0f} Sx:{sx:.2f} Sy:{sy:.2f} ShX:{shx:.2f} ShY:{shy:.2f} Tx:{tx:.0f} Ty:{ty:.0f}"
        cv2.putText(result, info, (10, 20), cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 0, 0), 1)

        cv2.imshow('Affine Transform Tool', result)

    def on_rotation(val):
        params['rotation'] = val
        update_transform()

    def on_scale_x(val):
        params['scale_x'] = max(10, val)
        update_transform()

    def on_scale_y(val):
        params['scale_y'] = max(10, val)
        update_transform()

    def on_shear_x(val):
        params['shear_x'] = val
        update_transform()

    def on_shear_y(val):
        params['shear_y'] = val
        update_transform()

    def on_trans_x(val):
        params['trans_x'] = val
        update_transform()

    def on_trans_y(val):
        params['trans_y'] = val
        update_transform()

    # Create window
    cv2.namedWindow('Affine Transform Tool', cv2.WINDOW_NORMAL)
    cv2.createTrackbar('Rotation', 'Affine Transform Tool', 180, 360, on_rotation)
    cv2.createTrackbar('Scale X', 'Affine Transform Tool', 100, 200, on_scale_x)
    cv2.createTrackbar('Scale Y', 'Affine Transform Tool', 100, 200, on_scale_y)
    cv2.createTrackbar('Shear X', 'Affine Transform Tool', 50, 100, on_shear_x)
    cv2.createTrackbar('Shear Y', 'Affine Transform Tool', 50, 100, on_shear_y)
    cv2.createTrackbar('Trans X', 'Affine Transform Tool', 50, 100, on_trans_x)
    cv2.createTrackbar('Trans Y', 'Affine Transform Tool', 50, 100, on_trans_y)

    update_transform()

    print("Interactive Affine Transform Tool")
    print("="*40)
    print("Controls:")
    print("  Rotation: -180 to 180 degrees")
    print("  Scale X/Y: 0.1 to 2.0")
    print("  Shear X/Y: -0.5 to 0.5")
    print("  Trans X/Y: -50 to 50 pixels")
    print("Press 'r' to reset, 'q' to quit")

    while True:
        key = cv2.waitKey(1) & 0xFF
        if key == ord('q'):

            cv2.setTrackbarPos('Rotation', 'Affine Transform Tool', 180)

            cv2.setTrackbarPos('Trans X', 'Affine Transform Tool', 50)
```

## 6. 总结

### 6.1 仿射变换核心概念

| 概念 | 说明 |
|------|------|
| **2×3矩阵** | 6个参数，6个自由度 |
| **保持平行性** | 平行线变换后仍平行 |
| **3点确定** | 需要3对对应点 |
| **可组合** | 多个变换可合并为一个矩阵 |

### 6.2 关键函数

```python
# 从三点计算仿射矩阵
M = cv2.getAffineTransform(src_pts, dst_pts)

M = cv2.getRotationMatrix2D(center, angle, scale)

# 应用仿射变换
result = cv2.warpAffine(src, M, dsize)

M_inv = cv2.invertAffineTransform(M)
```
