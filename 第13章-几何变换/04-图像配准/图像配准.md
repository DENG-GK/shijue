# 图像配准（Image Registration）

## 1. 图像配准概述

图像配准是将不同时间、不同视角或不同传感器获取的图像对齐到同一坐标系下的过程。它是图像融合、变化检测、医学影像分析等应用的基础。

### 1.1 什么是图像配准

图像配准的目标是找到一个变换 $T$，使得：

$$I_{aligned} = T(I_{source})$$

使得 $I_{aligned}$ 与 $I_{target}$ 最大程度地对齐。

### 1.2 配准的分类

```
图像配准分类
├── 按变换类型
│   ├── 刚性配准（平移+旋转）
│   ├── 仿射配准（+缩放+错切）
│   └── 非刚性配准（局部变形）
├── 按方法
│   ├── 基于特征的配准
│   ├── 基于区域的配准
│   └── 基于深度学习的配准
└── 按应用
    ├── 医学影像配准
    ├── 遥感图像配准
    ├── 全景拼接
    └── 运动补偿
```

### 1.3 配准流程

```
源图像        →   特征检测   →   特征匹配   →   变换估计   →   图像变换   →   配准图像
目标图像      →   特征检测   ↗
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def orb_registration():
    """
    Image registration using ORB features
    """
    # Create reference image
    reference = np.ones((300, 400, 3), dtype=np.uint8) * 200
    cv2.rectangle(reference, (50, 50), (350, 250), (0, 128, 255), -1)
    cv2.circle(reference, (100, 100), 30, (255, 0, 0), -1)
    cv2.circle(reference, (300, 100), 30, (0, 255, 0), -1)
    cv2.circle(reference, (200, 200), 30, (0, 0, 255), -1)
    cv2.putText(reference, 'REF', (160, 160), cv2.FONT_HERSHEY_SIMPLEX, 1, (255, 255, 255), 2)

    # Add some texture for feature detection
    for i in range(20):
        x, y = np.random.randint(60, 340), np.random.randint(60, 240)
        cv2.circle(reference, (x, y), 5, (150, 150, 150), -1)

    h, w = reference.shape[:2]

    # Create transformed image (source to be registered)
    angle = 15
    scale = 0.9
    tx, ty = 30, 20

    center = (w // 2, h // 2)
    M_true = cv2.getRotationMatrix2D(center, angle, scale)
    M_true[0, 2] += tx
    M_true[1, 2] += ty

    source = cv2.warpAffine(reference, M_true, (w, h), borderValue=(128, 128, 128))

    # Convert to grayscale
    gray_ref = cv2.cvtColor(reference, cv2.COLOR_BGR2GRAY)
    gray_src = cv2.cvtColor(source, cv2.COLOR_BGR2GRAY)

    # Detect ORB features
    orb = cv2.ORB_create(nfeatures=500)

    kp_ref, desc_ref = orb.detectAndCompute(gray_ref, None)
    kp_src, desc_src = orb.detectAndCompute(gray_src, None)

    print(f"Reference keypoints: {len(kp_ref)}")
    print(f"Source keypoints: {len(kp_src)}")

    # Match features
    bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
    matches = bf.match(desc_src, desc_ref)
    matches = sorted(matches, key=lambda x: x.distance)

    print(f"Matches found: {len(matches)}")

    # Extract matched points
    src_pts = np.float32([kp_src[m.queryIdx].pt for m in matches]).reshape(-1, 1, 2)
    ref_pts = np.float32([kp_ref[m.trainIdx].pt for m in matches]).reshape(-1, 1, 2)

    # Estimate transformation
    M_estimated, mask = cv2.estimateAffine2D(src_pts, ref_pts, method=cv2.RANSAC)

    # Apply transformation
    aligned = cv2.warpAffine(source, M_estimated, (w, h))

    # Calculate error
    error = cv2.absdiff(reference, aligned)

    # Display
    plt.figure(figsize=(20, 12))

    plt.subplot(2, 4, 1)
    plt.imshow(cv2.cvtColor(reference, cv2.COLOR_BGR2RGB))
    plt.title('Reference Image')
    plt.axis('off')

    plt.subplot(2, 4, 2)
    plt.imshow(cv2.cvtColor(source, cv2.COLOR_BGR2RGB))
    plt.title('Source (Transformed)')
    plt.axis('off')

    plt.subplot(2, 4, 3)
    # Draw matches
    match_img = cv2.drawMatches(source, kp_src, reference, kp_ref,
                                  matches[:20], None,
                                  flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
    plt.imshow(cv2.cvtColor(match_img, cv2.COLOR_BGR2RGB))
    plt.title(f'Feature Matches (top 20/{len(matches)})')
    plt.axis('off')

    plt.subplot(2, 4, 4)
    plt.imshow(cv2.cvtColor(aligned, cv2.COLOR_BGR2RGB))
    plt.title('Aligned Result')
    plt.axis('off')

    plt.subplot(2, 4, 5)
    # Overlay before alignment
    overlay_before = cv2.addWeighted(reference, 0.5, source, 0.5, 0)
    plt.imshow(cv2.cvtColor(overlay_before, cv2.COLOR_BGR2RGB))
    plt.title('Before Alignment (Overlay)')
    plt.axis('off')

    plt.subplot(2, 4, 6)
    # Overlay after alignment
    overlay_after = cv2.addWeighted(reference, 0.5, aligned, 0.5, 0)
    plt.imshow(cv2.cvtColor(overlay_after, cv2.COLOR_BGR2RGB))
    plt.title('After Alignment (Overlay)')
    plt.axis('off')

    plt.subplot(2, 4, 7)
    error_enhanced = cv2.convertScaleAbs(error, alpha=3)
    plt.imshow(cv2.cvtColor(error_enhanced, cv2.COLOR_BGR2RGB))
    plt.title(f'Error (3x)\nMean: {np.mean(error):.2f}')
    plt.axis('off')

    plt.subplot(2, 4, 8)
    # Compare matrices
    M_true_3x3 = np.vstack([M_true, [0, 0, 1]])
    M_est_3x3 = np.vstack([M_estimated, [0, 0, 1]])

    info = f"True Transform:\n"
    info += f"[{M_true[0,0]:.3f}  {M_true[0,1]:.3f}  {M_true[0,2]:.1f}]\n"
    info += f"[{M_true[1,0]:.3f}  {M_true[1,1]:.3f}  {M_true[1,2]:.1f}]\n\n"
    info += f"Estimated Transform:\n"
    info += f"[{M_estimated[0,0]:.3f}  {M_estimated[0,1]:.3f}  {M_estimated[0,2]:.1f}]\n"
    info += f"[{M_estimated[1,0]:.3f}  {M_estimated[1,1]:.3f}  {M_estimated[1,2]:.1f}]\n\n"
    info += f"Matrix Error: {np.linalg.norm(M_true - M_estimated):.4f}"

    plt.text(0.1, 0.5, info, fontsize=10, family='monospace',
             verticalalignment='center', transform=plt.gca().transAxes)
    plt.axis('off')
    plt.title('Transform Comparison')

    plt.tight_layout()
    plt.savefig('orb_registration.png', dpi=150, bbox_inches='tight')
    plt.show()
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def sift_registration():
    """
    Image registration using SIFT features
    Note: Requires opencv-contrib-python
    """
    # Create test images
    reference = np.ones((300, 400, 3), dtype=np.uint8) * 220

    # Add rich texture for feature detection
    for i in range(30):
        x, y = np.random.randint(20, 380), np.random.randint(20, 280)
        size = np.random.randint(10, 30)
        color = tuple(np.random.randint(50, 200, 3).tolist())
        shape = np.random.choice(['circle', 'rect'])
        if shape == 'circle':
            cv2.circle(reference, (x, y), size, color, -1)
        else:
            cv2.rectangle(reference, (x-size, y-size), (x+size, y+size), color, -1)

    h, w = reference.shape[:2]

    # Create perspective transformed version
    src_pts = np.float32([[0, 0], [w, 0], [w, h], [0, h]])
    dst_pts = np.float32([[30, 20], [w-20, 40], [w-40, h-30], [20, h-50]])

    M_true = cv2.getPerspectiveTransform(src_pts, dst_pts)
    source = cv2.warpPerspective(reference, M_true, (w, h), borderValue=(128, 128, 128))

    # Convert to grayscale
    gray_ref = cv2.cvtColor(reference, cv2.COLOR_BGR2GRAY)
    gray_src = cv2.cvtColor(source, cv2.COLOR_BGR2GRAY)

    # Try SIFT, fall back to ORB if not available
    try:
        detector = cv2.SIFT_create()
        norm = cv2.NORM_L2
        print("Using SIFT detector")
    except:
        detector = cv2.ORB_create(nfeatures=1000)
        norm = cv2.NORM_HAMMING
        print("Using ORB detector (SIFT not available)")

    # Detect and compute
    kp_ref, desc_ref = detector.detectAndCompute(gray_ref, None)
    kp_src, desc_src = detector.detectAndCompute(gray_src, None)

    print(f"Reference keypoints: {len(kp_ref)}")
    print(f"Source keypoints: {len(kp_src)}")

    # Match using FLANN or BFMatcher
    if norm == cv2.NORM_L2:
        # FLANN for SIFT
        FLANN_INDEX_KDTREE = 1
        index_params = dict(algorithm=FLANN_INDEX_KDTREE, trees=5)
        search_params = dict(checks=50)
        flann = cv2.FlannBasedMatcher(index_params, search_params)
        matches = flann.knnMatch(desc_src, desc_ref, k=2)

        # Apply ratio test
        good_matches = []
        for m, n in matches:
            if m.distance < 0.7 * n.distance:
                good_matches.append(m)
    else:
        # BFMatcher for ORB
        bf = cv2.BFMatcher(norm, crossCheck=True)
        good_matches = bf.match(desc_src, desc_ref)
        good_matches = sorted(good_matches, key=lambda x: x.distance)[:50]

    print(f"Good matches: {len(good_matches)}")

    if len(good_matches) >= 4:
        # Extract points
        src_pts = np.float32([kp_src[m.queryIdx].pt for m in good_matches]).reshape(-1, 1, 2)
        ref_pts = np.float32([kp_ref[m.trainIdx].pt for m in good_matches]).reshape(-1, 1, 2)

        # Estimate homography
        H_estimated, mask = cv2.findHomography(src_pts, ref_pts, cv2.RANSAC, 5.0)

        # Apply transformation
        aligned = cv2.warpPerspective(source, H_estimated, (w, h))

        # Calculate inliers
        inliers = mask.ravel().sum()
        print(f"Inliers: {inliers}/{len(good_matches)}")

    # Display
    plt.figure(figsize=(18, 10))

    plt.subplot(2, 3, 1)
    ref_kp = cv2.drawKeypoints(reference, kp_ref, None, color=(0, 255, 0))
    plt.imshow(cv2.cvtColor(ref_kp, cv2.COLOR_BGR2RGB))
    plt.title(f'Reference ({len(kp_ref)} keypoints)')
    plt.axis('off')

    plt.subplot(2, 3, 2)
    src_kp = cv2.drawKeypoints(source, kp_src, None, color=(0, 255, 0))
    plt.imshow(cv2.cvtColor(src_kp, cv2.COLOR_BGR2RGB))
    plt.title(f'Source ({len(kp_src)} keypoints)')
    plt.axis('off')

    plt.subplot(2, 3, 3)
    if len(good_matches) >= 4:
        plt.imshow(cv2.cvtColor(aligned, cv2.COLOR_BGR2RGB))
        plt.title('Aligned')
    plt.axis('off')

    plt.subplot(2, 3, 4)
    match_img = cv2.drawMatches(source, kp_src, reference, kp_ref,
                                  good_matches[:30], None,
                                  flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
    plt.imshow(cv2.cvtColor(match_img, cv2.COLOR_BGR2RGB))
    plt.title('Feature Matches')
    plt.axis('off')

    plt.subplot(2, 3, 5)
    overlay_before = cv2.addWeighted(reference, 0.5, source, 0.5, 0)
    plt.imshow(cv2.cvtColor(overlay_before, cv2.COLOR_BGR2RGB))
    plt.title('Before')
    plt.axis('off')

    plt.subplot(2, 3, 6)
    if len(good_matches) >= 4:
        overlay_after = cv2.addWeighted(reference, 0.5, aligned, 0.5, 0)
        plt.imshow(cv2.cvtColor(overlay_after, cv2.COLOR_BGR2RGB))
        plt.title('After')
    plt.axis('off')

    plt.tight_layout()
    plt.savefig('sift_registration.png', dpi=150, bbox_inches='tight')
    plt.show()

sift_registration()
```

### 示例3：相位相关法

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def phase_correlation_registration():
    """
    Image registration using phase correlation
    Best for pure translation
    """
    # Create reference image
    reference = np.zeros((256, 256), dtype=np.uint8)
    cv2.rectangle(reference, (50, 50), (200, 200), 200, -1)
    cv2.circle(reference, (128, 128), 40, 150, -1)

    # Add texture
    for i in range(20):
        x, y = np.random.randint(60, 190), np.random.randint(60, 190)
        cv2.circle(reference, (x, y), 5, 100, -1)

    # Create translated version
    true_shift = (30, 20)  # (x, y) shift
    M = np.float32([[1, 0, true_shift[0]], [0, 1, true_shift[1]]])
    source = cv2.warpAffine(reference, M, (256, 256))

    # Phase correlation
    def phase_correlate(ref, src):
        """
        Estimate translation using phase correlation
        """
        # Convert to float
        ref_f = ref.astype(np.float64)
        src_f = src.astype(np.float64)

        # Apply window to reduce edge effects
        rows, cols = ref_f.shape
        hann = cv2.createHanningWindow((cols, rows), cv2.CV_64F)
        ref_f = ref_f * hann
        src_f = src_f * hann

        # FFT
        ref_fft = np.fft.fft2(ref_f)
        src_fft = np.fft.fft2(src_f)

        # Cross-power spectrum
        cross = ref_fft * np.conj(src_fft)
        cross_magnitude = np.abs(cross)
        cross_magnitude[cross_magnitude == 0] = 1  # Avoid division by zero
        normalized_cross = cross / cross_magnitude

        # Inverse FFT
        correlation = np.fft.ifft2(normalized_cross)
        correlation = np.abs(correlation)

        # Find peak
        peak_loc = np.unravel_index(np.argmax(correlation), correlation.shape)

        # Convert to shift (handle wrap-around)
        shift_y = peak_loc[0]
        shift_x = peak_loc[1]

        if shift_y > rows // 2:
            shift_y -= rows
        if shift_x > cols // 2:
            shift_x -= cols

        return (-shift_x, -shift_y), correlation

    # Estimate shift
    estimated_shift, correlation = phase_correlate(reference, source)

    print(f"True shift: {true_shift}")
    print(f"Estimated shift: {estimated_shift}")

    # Apply correction
    M_correct = np.float32([[1, 0, estimated_shift[0]], [0, 1, estimated_shift[1]]])
    aligned = cv2.warpAffine(source, M_correct, (256, 256))

    # OpenCV's built-in phase correlation
    shift_cv, response = cv2.phaseCorrelate(
        reference.astype(np.float64),
        source.astype(np.float64)
    )
    print(f"OpenCV shift: {shift_cv}")

    # Display
    plt.figure(figsize=(18, 10))

    plt.subplot(2, 4, 1)
    plt.imshow(reference, cmap='gray')
    plt.title('Reference')
    plt.axis('off')

    plt.subplot(2, 4, 2)
    plt.imshow(source, cmap='gray')
    plt.title(f'Source (shifted {true_shift})')
    plt.axis('off')

    plt.subplot(2, 4, 3)
    plt.imshow(np.fft.fftshift(correlation), cmap='hot')
    plt.colorbar()
    plt.title('Phase Correlation')
    plt.axis('off')

    plt.subplot(2, 4, 4)
    plt.imshow(aligned, cmap='gray')
    plt.title(f'Aligned (shift: {estimated_shift})')
    plt.axis('off')

    plt.subplot(2, 4, 5)
    overlay_before = np.stack([reference, source, np.zeros_like(reference)], axis=2)
    plt.imshow(overlay_before)
    plt.title('Before (R=ref, G=src)')
    plt.axis('off')

    plt.subplot(2, 4, 6)
    overlay_after = np.stack([reference, aligned, np.zeros_like(reference)], axis=2)
    plt.imshow(overlay_after)
    plt.title('After (should be yellow)')
    plt.axis('off')

    plt.subplot(2, 4, 7)
    error_before = cv2.absdiff(reference, source)
    plt.imshow(error_before, cmap='hot')
    plt.title(f'Error Before\nSum: {np.sum(error_before)}')
    plt.axis('off')

    plt.subplot(2, 4, 8)
    error_after = cv2.absdiff(reference, aligned)
    plt.imshow(error_after, cmap='hot')
    plt.title(f'Error After\nSum: {np.sum(error_after)}')
    plt.axis('off')

    plt.tight_layout()
    plt.savefig('phase_correlation.png', dpi=150, bbox_inches='tight')
    plt.show()

phase_correlation_registration()
```

### 示例4：模板匹配配准

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def template_matching_registration():
    """
    Image registration using template matching
    """
    # Create reference image with distinct features
    reference = np.ones((300, 400, 3), dtype=np.uint8) * 200

    # Add distinct markers
    markers = [
        ((80, 80), (255, 0, 0)),
        ((320, 80), (0, 255, 0)),
        ((200, 220), (0, 0, 255)),
    ]

    for (x, y), color in markers:
        cv2.circle(reference, (x, y), 25, color, -1)
        cv2.circle(reference, (x, y), 10, (255, 255, 255), -1)

    h, w = reference.shape[:2]

    # Create transformed source
    angle = 5
    scale = 0.95
    tx, ty = 15, 10

    center = (w // 2, h // 2)
    M_true = cv2.getRotationMatrix2D(center, angle, scale)
    M_true[0, 2] += tx
    M_true[1, 2] += ty

    source = cv2.warpAffine(reference, M_true, (w, h), borderValue=(180, 180, 180))

    # Template matching for each marker
    def find_marker(image, template_center, template_size=60):
        """Find marker in image using template matching"""
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

        # Extract template from reference
        x, y = template_center
        half = template_size // 2
        template = gray[y-half:y+half, x-half:x+half]

        if template.shape[0] < 10 or template.shape[1] < 10:
            return None

        # Template matching
        result = cv2.matchTemplate(cv2.cvtColor(source, cv2.COLOR_BGR2GRAY),
                                    template, cv2.TM_CCOEFF_NORMED)
        _, max_val, _, max_loc = cv2.minMaxLoc(result)

        # Return center of matched region
        matched_center = (max_loc[0] + half, max_loc[1] + half)

        return matched_center, max_val

    # Find markers in source
    ref_points = []
    src_points = []

    for (x, y), _ in markers:
        result = find_marker(reference, (x, y))
        if result:
            matched, confidence = result
            print(f"Marker at ({x}, {y}) → found at {matched} (conf: {confidence:.3f})")
            ref_points.append([x, y])
            src_points.append(list(matched))

    # Estimate transformation
    ref_pts = np.float32(ref_points)
    src_pts = np.float32(src_points)

    # Use affine or similarity transform
    M_estimated, inliers = cv2.estimateAffinePartial2D(src_pts, ref_pts)

    # Apply transformation
    aligned = cv2.warpAffine(source, M_estimated, (w, h))

    # Display
    plt.figure(figsize=(18, 12))

    plt.subplot(2, 3, 1)
    ref_marked = reference.copy()
    for (x, y), _ in markers:
        cv2.rectangle(ref_marked, (x-30, y-30), (x+30, y+30), (0, 255, 255), 2)
    plt.imshow(cv2.cvtColor(ref_marked, cv2.COLOR_BGR2RGB))
    plt.title('Reference (with templates)')
    plt.axis('off')

    plt.subplot(2, 3, 2)
    src_marked = source.copy()
    for pt in src_points:
        cv2.circle(src_marked, tuple(int(x) for x in pt), 8, (0, 255, 255), 2)
    plt.imshow(cv2.cvtColor(src_marked, cv2.COLOR_BGR2RGB))
    plt.title('Source (matched locations)')
    plt.axis('off')

    plt.subplot(2, 3, 3)
    plt.imshow(cv2.cvtColor(aligned, cv2.COLOR_BGR2RGB))
    plt.title('Aligned')
    plt.axis('off')

    plt.subplot(2, 3, 4)
    overlay_before = cv2.addWeighted(reference, 0.5, source, 0.5, 0)
    plt.imshow(cv2.cvtColor(overlay_before, cv2.COLOR_BGR2RGB))
    plt.title('Before')
    plt.axis('off')

    plt.subplot(2, 3, 5)
    overlay_after = cv2.addWeighted(reference, 0.5, aligned, 0.5, 0)
    plt.imshow(cv2.cvtColor(overlay_after, cv2.COLOR_BGR2RGB))
    plt.title('After')
    plt.axis('off')

    plt.subplot(2, 3, 6)
    # Show point correspondences
    for i in range(len(ref_points)):
        plt.plot([ref_points[i][0], src_points[i][0]],
                 [ref_points[i][1], src_points[i][1]], 'b-', linewidth=2)
        plt.plot(ref_points[i][0], ref_points[i][1], 'go', markersize=12)
        plt.plot(src_points[i][0], src_points[i][1], 'rx', markersize=12)
    plt.gca().invert_yaxis()
    plt.legend(['Correspondence', 'Reference', 'Source'])
    plt.title('Point Correspondences')
    plt.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig('template_registration.png', dpi=150, bbox_inches='tight')
    plt.show()
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def ecc_registration():
    """
    Image registration using Enhanced Correlation Coefficient (ECC)
    This is an iterative refinement method
    """
    # Create reference image
    reference = np.zeros((256, 256, 3), dtype=np.uint8)
    reference[:, :] = [200, 200, 200]

    # Add various features
    cv2.rectangle(reference, (40, 40), (216, 216), (0, 128, 255), -1)
    cv2.circle(reference, (128, 100), 30, (255, 0, 0), -1)
    cv2.ellipse(reference, (128, 180), (50, 25), 0, 0, 360, (0, 255, 0), -1)

    # Add texture
    for i in range(30):
        x, y = np.random.randint(50, 200), np.random.randint(50, 200)
        cv2.circle(reference, (x, y), 3, (100, 100, 100), -1)

    h, w = reference.shape[:2]

    # Create transformed source
    angle = 8
    scale = 0.92
    tx, ty = 12, 8

    center = (w // 2, h // 2)
    M_true = cv2.getRotationMatrix2D(center, angle, scale)
    M_true[0, 2] += tx
    M_true[1, 2] += ty

    source = cv2.warpAffine(reference, M_true, (w, h), borderValue=(128, 128, 128))

    # Convert to grayscale

    # ECC parameters
    criteria = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 1000, 1e-6)
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

class ImageRegistrationSystem:
    """
    Comprehensive image registration system
    """

    def __init__(self):
        self.reference = None
        self.source = None
        self.transform_matrix = None
        self.aligned = None

    def set_images(self, reference, source):
        """Set reference and source images"""
        self.reference = reference
        self.source = source

    def register_features(self, method='orb', transform='affine'):
        """Feature-based registration"""
        gray_ref = cv2.cvtColor(self.reference, cv2.COLOR_BGR2GRAY)
        gray_src = cv2.cvtColor(self.source, cv2.COLOR_BGR2GRAY)

        # Detect features
        if method == 'orb':
            detector = cv2.ORB_create(nfeatures=1000)
            norm = cv2.NORM_HAMMING
        else:
            try:
                detector = cv2.SIFT_create()
                norm = cv2.NORM_L2
            except:
                detector = cv2.ORB_create(nfeatures=1000)
                norm = cv2.NORM_HAMMING

        kp_ref, desc_ref = detector.detectAndCompute(gray_ref, None)
        kp_src, desc_src = detector.detectAndCompute(gray_src, None)

        # Match features
        bf = cv2.BFMatcher(norm, crossCheck=True)
        matches = bf.match(desc_src, desc_ref)
        matches = sorted(matches, key=lambda x: x.distance)[:100]

        # Extract points
        src_pts = np.float32([kp_src[m.queryIdx].pt for m in matches]).reshape(-1, 1, 2)
        ref_pts = np.float32([kp_ref[m.trainIdx].pt for m in matches]).reshape(-1, 1, 2)

        # Estimate transform
        if transform == 'affine':
            self.transform_matrix, mask = cv2.estimateAffine2D(src_pts, ref_pts, method=cv2.RANSAC)
            h, w = self.reference.shape[:2]
            self.aligned = cv2.warpAffine(self.source, self.transform_matrix, (w, h))
        else:
            self.transform_matrix, mask = cv2.findHomography(src_pts, ref_pts, cv2.RANSAC)
            h, w = self.reference.shape[:2]
            self.aligned = cv2.warpPerspective(self.source, self.transform_matrix, (w, h))

        return len(matches), mask.sum() if mask is not None else 0

    def register_ecc(self, motion_type=cv2.MOTION_AFFINE):
        """ECC-based registration"""
        gray_ref = cv2.cvtColor(self.reference, cv2.COLOR_BGR2GRAY).astype(np.float32)
        gray_src = cv2.cvtColor(self.source, cv2.COLOR_BGR2GRAY).astype(np.float32)

        criteria = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 1000, 1e-6)
        warp_matrix = np.eye(2, 3, dtype=np.float32)

        cc, self.transform_matrix = cv2.findTransformECC(
            gray_ref, gray_src, warp_matrix, motion_type, criteria
        )

        h, w = self.reference.shape[:2]
        self.aligned = cv2.warpAffine(self.source, self.transform_matrix, (w, h),
                                       flags=cv2.INTER_LINEAR + cv2.WARP_INVERSE_MAP)

        return cc

    def register_phase(self):
        """Phase correlation registration (translation only)"""
        gray_ref = cv2.cvtColor(self.reference, cv2.COLOR_BGR2GRAY).astype(np.float64)
        gray_src = cv2.cvtColor(self.source, cv2.COLOR_BGR2GRAY).astype(np.float64)

        shift, response = cv2.phaseCorrelate(gray_ref, gray_src)

        self.transform_matrix = np.float32([[1, 0, shift[0]], [0, 1, shift[1]]])

        h, w = self.reference.shape[:2]
        self.aligned = cv2.warpAffine(self.source, self.transform_matrix, (w, h))

        return shift, response

    def calculate_metrics(self):
        """Calculate registration quality metrics"""
        if self.aligned is None:
            return None

        # MSE
        mse = np.mean((self.reference.astype(float) - self.aligned.astype(float)) ** 2)

        # PSNR
        psnr = cv2.PSNR(self.reference, self.aligned)

        # SSIM (simplified)
        gray_ref = cv2.cvtColor(self.reference, cv2.COLOR_BGR2GRAY)
        gray_aligned = cv2.cvtColor(self.aligned, cv2.COLOR_BGR2GRAY)

        c1 = (0.01 * 255) ** 2
        c2 = (0.03 * 255) ** 2

        mu_ref = cv2.GaussianBlur(gray_ref.astype(float), (11, 11), 1.5)
        mu_aligned = cv2.GaussianBlur(gray_aligned.astype(float), (11, 11), 1.5)

        mu_ref_sq = mu_ref ** 2
        mu_aligned_sq = mu_aligned ** 2
        mu_ref_aligned = mu_ref * mu_aligned

        sigma_ref_sq = cv2.GaussianBlur(gray_ref.astype(float) ** 2, (11, 11), 1.5) - mu_ref_sq
        sigma_aligned_sq = cv2.GaussianBlur(gray_aligned.astype(float) ** 2, (11, 11), 1.5) - mu_aligned_sq
        sigma_ref_aligned = cv2.GaussianBlur(gray_ref.astype(float) * gray_aligned.astype(float), (11, 11), 1.5) - mu_ref_aligned

        ssim_map = ((2 * mu_ref_aligned + c1) * (2 * sigma_ref_aligned + c2)) / \
                   ((mu_ref_sq + mu_aligned_sq + c1) * (sigma_ref_sq + sigma_aligned_sq + c2))
        ssim = np.mean(ssim_map)

        return {'mse': mse, 'psnr': psnr, 'ssim': ssim}

    def visualize(self, title="Registration Result"):
        """Visualize registration result"""
        if self.aligned is None:
            print("No registration performed yet")
            return

        plt.figure(figsize=(16, 8))

        plt.subplot(2, 3, 1)
        plt.imshow(cv2.cvtColor(self.reference, cv2.COLOR_BGR2RGB))
        plt.title('Reference')
        plt.axis('off')

        plt.subplot(2, 3, 2)
        plt.imshow(cv2.cvtColor(self.source, cv2.COLOR_BGR2RGB))
        plt.title('Source')
        plt.axis('off')

        plt.subplot(2, 3, 3)
        plt.imshow(cv2.cvtColor(self.aligned, cv2.COLOR_BGR2RGB))
        plt.title('Aligned')
        plt.axis('off')

        plt.subplot(2, 3, 4)
        overlay_before = cv2.addWeighted(self.reference, 0.5, self.source, 0.5, 0)
        plt.imshow(cv2.cvtColor(overlay_before, cv2.COLOR_BGR2RGB))
        plt.title('Before')
        plt.axis('off')

        plt.subplot(2, 3, 5)
        overlay_after = cv2.addWeighted(self.reference, 0.5, self.aligned, 0.5, 0)
        plt.imshow(cv2.cvtColor(overlay_after, cv2.COLOR_BGR2RGB))
        plt.title('After')
        plt.axis('off')

        plt.subplot(2, 3, 6)
        metrics = self.calculate_metrics()
        if metrics:
            info = f"Quality Metrics:\n\n"
            info += f"MSE: {metrics['mse']:.4f}\n"
            info += f"PSNR: {metrics['psnr']:.2f} dB\n"
            info += f"SSIM: {metrics['ssim']:.4f}"
            plt.text(0.1, 0.5, info, fontsize=12, family='monospace',
                     verticalalignment='center', transform=plt.gca().transAxes)
        plt.axis('off')
        plt.title('Metrics')

        plt.suptitle(title)
        plt.tight_layout()
        plt.savefig('registration_result.png', dpi=150, bbox_inches='tight')
        plt.show()
```

```python
import cv2

def registration_error_analysis():
    """
    Analyze registration error in detail
    """
    # Create reference with known features
    reference = np.ones((300, 400, 3), dtype=np.uint8) * 200

    # Add grid of markers
    markers = []
    for i in range(4):
        for j in range(5):
            x = 50 + j * 75
            y = 50 + i * 65
            markers.append((x, y))
            cv2.circle(reference, (x, y), 10, (0, 0, 255), -1)

    h, w = reference.shape[:2]

    # Create transformed source
    angle = 5
    scale = 0.95
    tx, ty = 10, 8

    M_true = cv2.getRotationMatrix2D((w//2, h//2), angle, scale)
    M_true[0, 2] += tx
    M_true[1, 2] += ty

    source = cv2.warpAffine(reference, M_true, (w, h), borderValue=(180, 180, 180))

    # Perform registration
    gray_ref = cv2.cvtColor(reference, cv2.COLOR_BGR2GRAY)
    gray_src = cv2.cvtColor(source, cv2.COLOR_BGR2GRAY)

    orb = cv2.ORB_create(nfeatures=500)
    kp_ref, desc_ref = orb.detectAndCompute(gray_ref, None)
    kp_src, desc_src = orb.detectAndCompute(gray_src, None)

    bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
    matches = bf.match(desc_src, desc_ref)

    src_pts = np.float32([kp_src[m.queryIdx].pt for m in matches]).reshape(-1, 1, 2)
    ref_pts = np.float32([kp_ref[m.trainIdx].pt for m in matches]).reshape(-1, 1, 2)

    M_estimated, mask = cv2.estimateAffine2D(src_pts, ref_pts, method=cv2.RANSAC)

    aligned = cv2.warpAffine(source, M_estimated, (w, h))

    # Calculate errors
    # 1. Matrix error
    matrix_error = np.linalg.norm(M_true - M_estimated, 'fro')

    # 2. Point-wise error
    markers_arr = np.array(markers, dtype=np.float32)
    ones = np.ones((len(markers), 1), dtype=np.float32)
    markers_h = np.hstack([markers_arr, ones])

    # Transform markers
    true_transformed = (M_true @ markers_h.T).T
    est_transformed = (M_estimated @ markers_h.T).T

    point_errors = np.sqrt(np.sum((true_transformed - est_transformed) ** 2, axis=1))

    # 3. Image error
    error_img = cv2.absdiff(reference, aligned)
    mse = np.mean(error_img.astype(float) ** 2)
    psnr = cv2.PSNR(reference, aligned)

    # Display
    plt.figure(figsize=(20, 12))

    plt.subplot(2, 4, 1)
    plt.imshow(cv2.cvtColor(reference, cv2.COLOR_BGR2RGB))
    plt.title('Reference')
    plt.axis('off')

    plt.subplot(2, 4, 2)
    plt.imshow(cv2.cvtColor(source, cv2.COLOR_BGR2RGB))
    plt.title('Source')
    plt.axis('off')

    plt.subplot(2, 4, 3)
    plt.imshow(cv2.cvtColor(aligned, cv2.COLOR_BGR2RGB))
    plt.title('Aligned')
    plt.axis('off')

    plt.subplot(2, 4, 4)
    error_enhanced = cv2.convertScaleAbs(error_img, alpha=5)
    plt.imshow(cv2.cvtColor(error_enhanced, cv2.COLOR_BGR2RGB))
    plt.title(f'Error (5x)\nMSE: {mse:.2f}')
    plt.axis('off')

    # Point error visualization
    plt.subplot(2, 4, 5)
    for i, (marker, err) in enumerate(zip(markers, point_errors)):
        color = plt.cm.hot(err / max(point_errors))[:3]
        plt.scatter(marker[0], marker[1], c=[color], s=100)
        plt.annotate(f'{err:.2f}', (marker[0]+5, marker[1]+5), fontsize=8)
    plt.xlim(0, w)
    plt.ylim(h, 0)
    plt.title('Point-wise Error')
    plt.colorbar(plt.cm.ScalarMappable(cmap='hot'), label='Error (pixels)')

    # Error histogram
    plt.subplot(2, 4, 6)
    plt.hist(point_errors, bins=10, color='steelblue', edgecolor='black')
    plt.xlabel('Error (pixels)')
    plt.ylabel('Count')
    plt.title(f'Error Distribution\nMean: {np.mean(point_errors):.2f}px')

    # Matrix comparison
    plt.subplot(2, 4, 7)
    info = f"True Transform:\n"
    info += f"[{M_true[0,0]:.4f}  {M_true[0,1]:.4f}  {M_true[0,2]:.2f}]\n"
    info += f"[{M_true[1,0]:.4f}  {M_true[1,1]:.4f}  {M_true[1,2]:.2f}]\n\n"

    info += f"[{M_estimated[0,0]:.4f}  {M_estimated[0,1]:.4f}  {M_estimated[0,2]:.2f}]\n"
    info += f"[{M_estimated[1,0]:.4f}  {M_estimated[1,1]:.4f}  {M_estimated[1,2]:.2f}]\n\n"
    info += f"Frobenius Norm Error: {matrix_error:.6f}"
    plt.text(0.1, 0.5, info, fontsize=10, family='monospace',
             verticalalignment='center', transform=plt.gca().transAxes)
    plt.axis('off')
    plt.title('Matrix Error')

    # Summary
```

## 8. 总结

### 8.1 配准方法对比

| 方法 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| 特征匹配 | 鲁棒、通用 | 需要纹理 | 大多数场景 |
| 相位相关 | 快速、精确 | 仅平移 | 小位移 |
| ECC | 精确、亚像素 | 需初值 | 精细配准 |
| 模板匹配 | 简单 | 有限变换 | 已知模板 |
