# 透视变换（Perspective Transformation）

## 1. 透视变换概述

透视变换（也称为单应性变换、Homography）是一种更通用的几何变换，能够模拟相机从不同角度观察平面物体的效果。它比仿射变换更灵活，可以处理透视失真的校正问题。

### 1.1 透视变换的定义

透视变换使用3×3矩阵，可以表示为：

$$\begin{bmatrix} x' \\ y' \\ w' \end{bmatrix} = \begin{bmatrix} h_{11} & h_{12} & h_{13} \\ h_{21} & h_{22} & h_{23} \\ h_{31} & h_{32} & h_{33} \end{bmatrix} \begin{bmatrix} x \\ y \\ 1 \end{bmatrix}$$

最终坐标需要归一化：
$$x_{final} = \frac{x'}{w'}, \quad y_{final} = \frac{y'}{w'}$$

### 1.2 透视变换 vs 仿射变换

| 特性 | 仿射变换 | 透视变换 |
|------|----------|----------|
| 矩阵大小 | 2×3 | 3×3 |
| 自由度 | 6 | 8 |
| 所需点对 | 3对 | 4对 |
| 保持平行 | ✓ | ✗ |
| 保持直线 | ✓ | ✓ |
| 透视校正 | ✗ | ✓ |

### 1.3 透视变换的应用

```
透视变换应用
├── 文档扫描校正
├── 鸟瞰图转换
├── 增强现实（AR）
├── 全景拼接
├── 车道线检测
└── 二维码/条形码读取
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def basic_perspective_transform():
    """
    Demonstrate basic perspective transformation
    """
    # Create test image
    image = np.zeros((400, 500, 3), dtype=np.uint8)
    image[:, :] = [220, 220, 220]

    # Draw a rectangle with text
    cv2.rectangle(image, (50, 50), (450, 350), (0, 128, 255), -1)
    cv2.putText(image, 'PERSPECTIVE', (80, 220), cv2.FONT_HERSHEY_SIMPLEX, 1.8, (255, 255, 255), 3)

    # Mark corners
    corners = [(50, 50), (450, 50), (450, 350), (50, 350)]
    for i, corner in enumerate(corners):
        cv2.circle(image, corner, 10, (255, 0, 0), -1)
        cv2.putText(image, str(i+1), (corner[0]+15, corner[1]+5),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 0), 2)

    h, w = image.shape[:2]

    # Source points (rectangle corners)
    src_pts = np.float32([[50, 50], [450, 50], [450, 350], [50, 350]])

    # Different destination configurations
    transforms = {
        'Original': src_pts,
        'Top narrow': np.float32([[100, 50], [400, 50], [450, 350], [50, 350]]),
        'Bottom narrow': np.float32([[50, 50], [450, 50], [400, 350], [100, 350]]),
        'Left tilt': np.float32([[50, 80], [450, 30], [450, 320], [50, 370]]),
        'Right tilt': np.float32([[50, 30], [450, 80], [450, 370], [50, 320]]),
        'Bird\'s eye': np.float32([[100, 0], [400, 0], [500, 400], [0, 400]]),
    }

    results = {}
    for name, dst_pts in transforms.items():
        if name == 'Original':
            results[name] = image.copy()
        else:
            M = cv2.getPerspectiveTransform(src_pts, dst_pts)
            results[name] = cv2.warpPerspective(image, M, (w, h))

    # Display
    plt.figure(figsize=(18, 10))

    for i, (name, result) in enumerate(results.items()):
        plt.subplot(2, 3, i+1)
        plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB))
        plt.title(name)
        plt.axis('off')

    plt.tight_layout()
    plt.savefig('basic_perspective.png', dpi=150, bbox_inches='tight')
    plt.show()

    print("Basic perspective transformation completed!")
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def perspective_math():
    """
    Visualize the mathematics of perspective transformation
    """
    # Create simple grid
    grid_size = 200
    grid = np.ones((grid_size, grid_size, 3), dtype=np.uint8) * 255

    # Draw grid lines
    for i in range(0, grid_size, 20):
        cv2.line(grid, (i, 0), (i, grid_size-1), (200, 200, 200), 1)
        cv2.line(grid, (0, i), (grid_size-1, i), (200, 200, 200), 1)

    # Add diagonal and center markers
    cv2.line(grid, (0, 0), (grid_size-1, grid_size-1), (0, 0, 255), 1)
    cv2.circle(grid, (grid_size//2, grid_size//2), 5, (255, 0, 0), -1)

    # Source corners
    src_pts = np.float32([
        [0, 0],
        [grid_size-1, 0],
        [grid_size-1, grid_size-1],
        [0, grid_size-1]
    ])

    # Perspective distortion (simulating camera angle)
    dst_pts = np.float32([
        [30, 30],
        [170, 20],
        [190, 180],
        [10, 170]
    ])

    # Get transformation matrix
    M = cv2.getPerspectiveTransform(src_pts, dst_pts)

    # Apply transformation
    warped = cv2.warpPerspective(grid, M, (grid_size, grid_size))

    # Visualize points transformation
    plt.figure(figsize=(18, 10))

    # Original grid
    plt.subplot(2, 3, 1)
    plt.imshow(cv2.cvtColor(grid, cv2.COLOR_BGR2RGB))
    for i, pt in enumerate(src_pts):
        plt.plot(pt[0], pt[1], 'go', markersize=10)
        plt.annotate(f'P{i+1}', (pt[0]+5, pt[1]+5), fontsize=10)
    plt.title('Original Grid')
    plt.axis('off')

    # Transformed grid
    plt.subplot(2, 3, 2)
    plt.imshow(cv2.cvtColor(warped, cv2.COLOR_BGR2RGB))
    for i, pt in enumerate(dst_pts):
        plt.plot(pt[0], pt[1], 'ro', markersize=10)
        plt.annotate(f'P{i+1}\'', (pt[0]+5, pt[1]+5), fontsize=10)
    plt.title('Transformed Grid')
    plt.axis('off')

    # Show transformation matrix
    plt.subplot(2, 3, 3)
    matrix_text = "Transformation Matrix H:\n\n"
    for i in range(3):
        matrix_text += f"[{M[i,0]:8.4f}  {M[i,1]:8.4f}  {M[i,2]:8.4f}]\n"
    matrix_text += "\n\nFor point (x, y):\n"
    matrix_text += "[x']   [h11 h12 h13] [x]\n"
    matrix_text += "[y'] = [h21 h22 h23] [y]\n"
    matrix_text += "[w']   [h31 h32 h33] [1]\n"
    matrix_text += "\nx_final = x'/w'\n"
    matrix_text += "y_final = y'/w'"
    plt.text(0.1, 0.5, matrix_text, fontsize=10, family='monospace',
             verticalalignment='center', transform=plt.gca().transAxes)
    plt.axis('off')
    plt.title('Transformation Matrix')

    # Point mapping visualization
    plt.subplot(2, 3, 4)
    # Show how a sample point transforms
    test_points = np.float32([
        [50, 50], [150, 50], [100, 100], [50, 150], [150, 150]
    ])

    for pt in test_points:
        # Transform point using matrix
        pt_h = np.array([pt[0], pt[1], 1])
        transformed_h = M @ pt_h
        transformed = transformed_h[:2] / transformed_h[2]

        plt.plot([pt[0], transformed[0]], [pt[1], transformed[1]], 'b-', alpha=0.5)
        plt.plot(pt[0], pt[1], 'go', markersize=8)
        plt.plot(transformed[0], transformed[1], 'ro', markersize=8)

    plt.xlim(-10, 210)
    plt.ylim(-10, 210)
    plt.gca().invert_yaxis()
    plt.legend(['Mapping', 'Original', 'Transformed'], loc='upper right')
    plt.title('Point Transformation')
    plt.grid(True, alpha=0.3)

    # Inverse transformation
    plt.subplot(2, 3, 5)
    M_inv = np.linalg.inv(M)
    recovered = cv2.warpPerspective(warped, M_inv, (grid_size, grid_size))
    plt.imshow(cv2.cvtColor(recovered, cv2.COLOR_BGR2RGB))
    plt.title('Recovered (Inverse)')
    plt.axis('off')

    # Error
    plt.subplot(2, 3, 6)
    error = cv2.absdiff(grid, recovered)
    plt.imshow(cv2.cvtColor(error * 10, cv2.COLOR_BGR2RGB))
    plt.title(f'Error (10x)\nMax: {error.max()}')
    plt.axis('off')

    plt.tight_layout()
    plt.savefig('perspective_math.png', dpi=150, bbox_inches='tight')
    plt.show()
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def document_perspective_correction():
    """
    Simulate document scanning with perspective correction
    """
    # Create a "document" image
    doc = np.ones((400, 300, 3), dtype=np.uint8) * 255

    # Add content to document
    cv2.rectangle(doc, (20, 20), (280, 380), (0, 0, 0), 2)

    # Add text lines
    for i in range(15):
        y = 50 + i * 22
        cv2.line(doc, (40, y), (260, y), (100, 100, 100), 1)

    cv2.putText(doc, 'DOCUMENT', (70, 100), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 0), 2)
    cv2.putText(doc, 'Page 1', (110, 350), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 0), 1)

    # Simulate "photo of document on desk" - perspective distortion
    h, w = doc.shape[:2]

    # Original corners
    src_pts = np.float32([[0, 0], [w, 0], [w, h], [0, h]])

    # Distorted corners (as if photographed at angle)
    dst_pts = np.float32([[50, 80], [380, 30], [420, 380], [30, 420]])

    # Create larger canvas for the "photo"
    canvas_size = (450, 500)
    canvas = np.ones((canvas_size[1], canvas_size[0], 3), dtype=np.uint8) * 180

    # Add some texture to canvas (desk surface)
    noise = np.random.randint(0, 20, canvas.shape, dtype=np.uint8)
    canvas = cv2.add(canvas, noise)

    # Apply perspective to document
    M_distort = cv2.getPerspectiveTransform(src_pts, dst_pts)
    distorted_doc = cv2.warpPerspective(doc, M_distort, canvas_size)

    # Composite onto canvas
    mask = cv2.warpPerspective(np.ones((h, w), dtype=np.uint8) * 255, M_distort, canvas_size)
    canvas[mask > 0] = distorted_doc[mask > 0]

    # Now correct the perspective
    # In real scenario, corners would be detected
    detected_corners = dst_pts  # Simulating detected corners

    # Target: standard document aspect ratio
    target_h, target_w = 400, 300
    target_pts = np.float32([[0, 0], [target_w, 0], [target_w, target_h], [0, target_h]])

    # Correction transformation
    M_correct = cv2.getPerspectiveTransform(detected_corners, target_pts)
    corrected = cv2.warpPerspective(canvas, M_correct, (target_w, target_h))

    # Display
    plt.figure(figsize=(18, 10))

    plt.subplot(2, 3, 1)
    plt.imshow(cv2.cvtColor(doc, cv2.COLOR_BGR2RGB))
    plt.title('Original Document')
    plt.axis('off')

    plt.subplot(2, 3, 2)
    # Mark corners on distorted image
    canvas_marked = canvas.copy()
    for i, pt in enumerate(detected_corners.astype(int)):
        cv2.circle(canvas_marked, tuple(pt), 8, (0, 255, 0), -1)
        cv2.putText(canvas_marked, str(i+1), (pt[0]+10, pt[1]+10),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
    plt.imshow(cv2.cvtColor(canvas_marked, cv2.COLOR_BGR2RGB))
    plt.title('Photo of Document\n(with detected corners)')
    plt.axis('off')

    plt.subplot(2, 3, 3)
    plt.imshow(cv2.cvtColor(corrected, cv2.COLOR_BGR2RGB))
    plt.title('Corrected Document')
    plt.axis('off')

    # Show the process
    plt.subplot(2, 3, 4)
    # Overlay original and corrected
    doc_resized = cv2.resize(doc, (target_w, target_h))
    overlay = cv2.addWeighted(doc_resized, 0.5, corrected, 0.5, 0)
    plt.imshow(cv2.cvtColor(overlay, cv2.COLOR_BGR2RGB))
    plt.title('Original vs Corrected\n(Overlay)')
    plt.axis('off')

    # Corner mapping diagram
    plt.subplot(2, 3, 5)
    for i in range(4):
        src = detected_corners[i]
        dst = target_pts[i]
        plt.plot([src[0], dst[0] + 500], [src[1], dst[1]], 'b-', linewidth=1.5)
        plt.plot(src[0], src[1], 'ro', markersize=10)
        plt.plot(dst[0] + 500, dst[1], 'go', markersize=10)
        plt.annotate(f'C{i+1}', (src[0]-20, src[1]), fontsize=10)

    plt.xlim(-50, 850)
    plt.ylim(500, -50)
    plt.axvline(x=450, color='gray', linestyle='--', alpha=0.5)
    plt.text(200, 470, 'Distorted', ha='center', fontsize=12)
    plt.text(650, 470, 'Corrected', ha='center', fontsize=12)
    plt.title('Corner Mapping')
    plt.axis('equal')

    plt.tight_layout()
    plt.savefig('document_correction.png', dpi=150, bbox_inches='tight')
    plt.show()
```

### 示例4：自动文档角点检测

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def auto_document_detection():
    """
    Automatic document corner detection and correction
    """
    # Create realistic document scenario
    canvas = np.ones((600, 800, 3), dtype=np.uint8) * 150

    # Add wood texture pattern
    for i in range(0, 800, 10):
        intensity = 140 + np.random.randint(-10, 10)
        cv2.line(canvas, (i, 0), (i, 599), (intensity, intensity-10, intensity-20), 2)

    # Create document
    doc = np.ones((350, 250, 3), dtype=np.uint8) * 252
    cv2.rectangle(doc, (15, 15), (235, 335), (0, 0, 0), 1)
    for i in range(12):
        y = 40 + i * 25
        cv2.line(doc, (30, y), (220, y), (150, 150, 150), 1)
    cv2.putText(doc, 'INVOICE', (60, 70), cv2.FONT_HERSHEY_SIMPLEX, 0.9, (0, 0, 0), 2)

    # Perspective warp document onto canvas
    h, w = doc.shape[:2]
    src_pts = np.float32([[0, 0], [w, 0], [w, h], [0, h]])
    dst_pts = np.float32([[150, 100], [550, 80], [600, 480], [100, 520]])

    M = cv2.getPerspectiveTransform(src_pts, dst_pts)
    warped_doc = cv2.warpPerspective(doc, M, (800, 600))
    mask = cv2.warpPerspective(np.ones((h, w), dtype=np.uint8) * 255, M, (800, 600))
    canvas[mask > 0] = warped_doc[mask > 0]

    # Detection process
    def detect_document_corners(img):
        """Detect document corners using edge detection and contours"""
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        blurred = cv2.GaussianBlur(gray, (5, 5), 0)
        edges = cv2.Canny(blurred, 50, 150)

        # Find contours
        contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

        # Find largest quadrilateral
        max_area = 0
        best_corners = None

        for contour in contours:
            area = cv2.contourArea(contour)
            if area > max_area:
                peri = cv2.arcLength(contour, True)
                approx = cv2.approxPolyDP(contour, 0.02 * peri, True)

                if len(approx) == 4:
                    max_area = area
                    best_corners = approx.reshape(4, 2)

        return best_corners

    def order_corners(pts):
        """Order corners: top-left, top-right, bottom-right, bottom-left"""
        rect = np.zeros((4, 2), dtype=np.float32)

        s = pts.sum(axis=1)
        rect[0] = pts[np.argmin(s)]  # Top-left
        rect[2] = pts[np.argmax(s)]  # Bottom-right

        diff = np.diff(pts, axis=1)
        rect[1] = pts[np.argmin(diff)]  # Top-right
        rect[3] = pts[np.argmax(diff)]  # Bottom-left

        return rect

    # Detect corners
    detected = detect_document_corners(canvas)

    if detected is not None:
        ordered_corners = order_corners(detected)

        # Calculate target dimensions
        w1 = np.linalg.norm(ordered_corners[0] - ordered_corners[1])
        w2 = np.linalg.norm(ordered_corners[2] - ordered_corners[3])
        h1 = np.linalg.norm(ordered_corners[0] - ordered_corners[3])
        h2 = np.linalg.norm(ordered_corners[1] - ordered_corners[2])

        target_w = int(max(w1, w2))
        target_h = int(max(h1, h2))

        target_pts = np.float32([[0, 0], [target_w, 0], [target_w, target_h], [0, target_h]])

        # Correct perspective
        M_correct = cv2.getPerspectiveTransform(ordered_corners, target_pts)
        corrected = cv2.warpPerspective(canvas, M_correct, (target_w, target_h))

    # Display
    plt.figure(figsize=(18, 12))

    plt.subplot(2, 3, 1)
    plt.imshow(cv2.cvtColor(canvas, cv2.COLOR_BGR2RGB))
    plt.title('Input Image')
    plt.axis('off')

    plt.subplot(2, 3, 2)
    gray = cv2.cvtColor(canvas, cv2.COLOR_BGR2GRAY)
    edges = cv2.Canny(cv2.GaussianBlur(gray, (5, 5), 0), 50, 150)
    plt.imshow(edges, cmap='gray')
    plt.title('Edge Detection')
    plt.axis('off')

    plt.subplot(2, 3, 3)
    canvas_corners = canvas.copy()
    if detected is not None:
        for i, pt in enumerate(ordered_corners.astype(int)):
            cv2.circle(canvas_corners, tuple(pt), 10, (0, 255, 0), -1)
            cv2.putText(canvas_corners, str(i+1), (pt[0]+15, pt[1]+5),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
        # Draw polygon
        cv2.polylines(canvas_corners, [ordered_corners.astype(int)], True, (0, 255, 0), 2)
    plt.imshow(cv2.cvtColor(canvas_corners, cv2.COLOR_BGR2RGB))
    plt.title('Detected Corners')
    plt.axis('off')

    plt.subplot(2, 3, 4)
    if detected is not None:
        plt.imshow(cv2.cvtColor(corrected, cv2.COLOR_BGR2RGB))
        plt.title(f'Corrected\n{corrected.shape[1]}x{corrected.shape[0]}')
    plt.axis('off')

    plt.subplot(2, 3, 5)
    plt.imshow(cv2.cvtColor(doc, cv2.COLOR_BGR2RGB))
    plt.title('Original Document')
    plt.axis('off')

    plt.subplot(2, 3, 6)
    # Quality comparison
    info_text = """
    Document Detection Pipeline:

    1. Convert to grayscale
    2. Apply Gaussian blur
    3. Edge detection (Canny)
    4. Find contours
    5. Approximate to polygon
    6. Select largest quadrilateral
    7. Order corners
    8. Calculate perspective transform
    9. Warp to correct perspective

    Tips:
    - Good lighting improves detection
    - High contrast edges help
    - Document should be largest object
    """
    plt.text(0.1, 0.5, info_text, fontsize=9, family='monospace',
             verticalalignment='center', transform=plt.gca().transAxes)
    plt.axis('off')
    plt.title('Processing Pipeline')

    plt.tight_layout()
    plt.savefig('auto_document_detection.png', dpi=150, bbox_inches='tight')
    plt.show()

auto_document_detection()
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def birds_eye_view():
    """
    Transform road perspective to bird's eye view
    """
    # Create simulated road scene
    image = np.zeros((400, 600, 3), dtype=np.uint8)

    # Sky
    image[:150, :] = [200, 180, 150]

    # Road (trapezoid due to perspective)
    road_pts = np.array([[200, 150], [400, 150], [600, 400], [0, 400]], np.int32)
    cv2.fillPoly(image, [road_pts], (80, 80, 80))

    # Lane markings (dashed lines with perspective)
    for i in range(3):
        # Left and right lane markers
        for offset in [-100, 100]:
            y_start = 160 + i * 80
            y_end = y_start + 40

            # Calculate x positions based on perspective
            x_top = 300 + offset * (1 - (y_start - 150) / 250 * 0.6)
            x_bottom = 300 + offset * (1 - (y_end - 150) / 250 * 0.6)

            pts = np.array([
                [x_top - 5, y_start],
                [x_top + 5, y_start],
                [x_bottom + 8, y_end],
                [x_bottom - 8, y_end]
            ], np.int32)
            cv2.fillPoly(image, [pts], (255, 255, 255))

    # Center line
    for i in range(4):
        y_start = 155 + i * 60
        y_end = y_start + 30
        scale_top = 1 - (y_start - 150) / 250 * 0.6
        scale_bottom = 1 - (y_end - 150) / 250 * 0.6

        pts = np.array([
            [298, y_start], [302, y_start],
            [302, y_end], [298, y_end]
        ], np.int32)
        cv2.fillPoly(image, [pts], (255, 255, 0))

    h, w = image.shape[:2]

    # Define perspective transform points
    # Source: trapezoid in perspective view
    src_pts = np.float32([
        [200, 150],   # Top-left
        [400, 150],   # Top-right
        [550, 380],   # Bottom-right
        [50, 380]     # Bottom-left
    ])

    # Destination: rectangle (bird's eye view)
    dst_pts = np.float32([
        [150, 0],
        [450, 0],
        [450, 400],
        [150, 400]
    ])

    # Calculate transformation
    M_bev = cv2.getPerspectiveTransform(src_pts, dst_pts)
    bev = cv2.warpPerspective(image, M_bev, (w, h))

    # Inverse transformation
    M_inv = cv2.getPerspectiveTransform(dst_pts, src_pts)
    recovered = cv2.warpPerspective(bev, M_inv, (w, h))

    # Display
    plt.figure(figsize=(18, 10))

    plt.subplot(2, 3, 1)
    img_marked = image.copy()
    for i, pt in enumerate(src_pts.astype(int)):
        cv2.circle(img_marked, tuple(pt), 8, (0, 255, 0), -1)
    cv2.polylines(img_marked, [src_pts.astype(int)], True, (0, 255, 0), 2)
    plt.imshow(cv2.cvtColor(img_marked, cv2.COLOR_BGR2RGB))
    plt.title('Original View\n(Source Region)')
    plt.axis('off')

    plt.subplot(2, 3, 2)
    bev_marked = bev.copy()
```

```python
    true_H = np.array([
        [0.9, -0.1, 30],
        [0.15, 0.85, 20],
        [0.0002, 0.0001, 1]
    ], dtype=np.float32)

    # Transform image
    transformed = cv2.warpPerspective(image, true_H, (w, h))

    # Transform points
    src_pts = np.array(feature_positions, dtype=np.float32)
    ones = np.ones((len(src_pts), 1), dtype=np.float32)
    src_h = np.hstack([src_pts, ones])

    dst_h = (true_H @ src_h.T).T
    dst_pts = dst_h[:, :2] / dst_h[:, 2:3]

    # Add noise to simulate detection
    noise = np.random.normal(0, 2, dst_pts.shape).astype(np.float32)
    dst_pts_noisy = dst_pts + noise

    # Estimate homography using different methods
    # Method 1: cv2.findHomography with RANSAC
    H_ransac, mask_ransac = cv2.findHomography(src_pts, dst_pts_noisy, cv2.RANSAC, 5.0)

    # Method 2: cv2.findHomography with LMEDS
    H_lmeds, mask_lmeds = cv2.findHomography(src_pts, dst_pts_noisy, cv2.LMEDS)

    # Method 3: cv2.getPerspectiveTransform (4 points only)
    H_4pts = cv2.getPerspectiveTransform(src_pts[:4], dst_pts_noisy[:4])

    result_ransac = cv2.warpPerspective(image, H_ransac, (w, h))

    def matrix_error(H_est, H_true):
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

class PerspectiveTransformer:
    """
    Complete perspective transformation utility class
    """

    def __init__(self):
        self.src_pts = None
        self.dst_pts = None
        self.H = None
        self.H_inv = None

    def set_source_points(self, points):
        """Set source corner points (4 points)"""
        points = np.array(points, dtype=np.float32)
        if points.shape != (4, 2):
            raise ValueError("Exactly 4 points required")
        self.src_pts = self._order_points(points)

    def set_destination_rectangle(self, width=None, height=None, margin=0):
        """Set destination as rectangle"""
        if self.src_pts is None:
            raise ValueError("Set source points first")

        if width is None:
            # Calculate from source points
            w1 = np.linalg.norm(self.src_pts[0] - self.src_pts[1])
            w2 = np.linalg.norm(self.src_pts[2] - self.src_pts[3])
            width = int(max(w1, w2))

        if height is None:
            h1 = np.linalg.norm(self.src_pts[0] - self.src_pts[3])
            h2 = np.linalg.norm(self.src_pts[1] - self.src_pts[2])
            height = int(max(h1, h2))

        self.dst_pts = np.float32([
            [margin, margin],
            [width - margin, margin],
            [width - margin, height - margin],
            [margin, height - margin]
        ])

        self._compute_homography()
        return width, height

    def _order_points(self, pts):
        """Order points: top-left, top-right, bottom-right, bottom-left"""
        rect = np.zeros((4, 2), dtype=np.float32)

        s = pts.sum(axis=1)
        rect[0] = pts[np.argmin(s)]
        rect[2] = pts[np.argmax(s)]

        diff = np.diff(pts, axis=1)
        rect[1] = pts[np.argmin(diff)]
        rect[3] = pts[np.argmax(diff)]

        return rect

    def _compute_homography(self):
        """Compute homography matrices"""
        self.H = cv2.getPerspectiveTransform(self.src_pts, self.dst_pts)
        self.H_inv = np.linalg.inv(self.H)

    def warp(self, image, output_size=None):
        """Apply perspective transformation"""
        if self.H is None:
            raise ValueError("Compute transformation first")

        if output_size is None:
            output_size = (int(self.dst_pts[:, 0].max()),
                           int(self.dst_pts[:, 1].max()))

        return cv2.warpPerspective(image, self.H, output_size)

    def warp_inverse(self, image, output_size):
        """Apply inverse transformation"""
        if self.H_inv is None:
            raise ValueError("Compute transformation first")

        return cv2.warpPerspective(image, self.H_inv, output_size)

    def transform_point(self, point):
        """Transform a single point"""
        pt = np.array([point[0], point[1], 1])
        transformed = self.H @ pt
        return transformed[:2] / transformed[2]

    def transform_points(self, points):
        """Transform multiple points"""
        return np.array([self.transform_point(p) for p in points])
```

## 7. 总结

### 7.1 透视变换核心概念

| 概念 | 说明 |
|------|------|
| **3×3矩阵** | 8个自由度（第9个固定为1） |
| **齐次坐标** | 需要归一化 |
| **4点确定** | 需要4对非共线对应点 |
| **保持直线** | 直线变换后仍是直线 |
| **不保持平行** | 平行线可能相交 |
