# 直方图均衡化

## 1. 什么是直方图均衡化？

### 1.1 基本概念

直方图均衡化（Histogram Equalization）是一种**增强图像对比度**的技术，通过重新分配像素值，使直方图更加均匀分布。

> **核心思想**：让图像的灰度级分布更加均匀，充分利用0-255的整个灰度范围。

```
均衡化前：直方图集中在某个区域
频率 │    ████
     │   ██████
     │  ████████
     │ ██████████
     └─────────────── 灰度值
           ↑集中区域

均衡化后：直方图分布更均匀
频率 │ ██ ██ ██ ██ ██
     │████████████████
     │████████████████
     │████████████████
     └─────────────── 灰度值
       ↑ 分布更均匀 ↑
```

### 1.2 为什么需要直方图均衡化？

| 问题场景 | 均衡化效果 |
|---------|-----------|
| **欠曝图像** | 提升整体亮度，显示暗部细节 |
| **过曝图像** | 降低整体亮度，恢复高光细节 |
| **低对比度** | 增强对比度，让细节更清晰 |
| **雾霾图像** | 去除灰蒙蒙的感觉 |

### 1.3 可视化对比

```python
"""
示例1：直方图均衡化的效果展示
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def show_equalization_effect(image, title="Image"):
    """展示直方图均衡化前后的对比"""
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # 直方图均衡化
    equalized = cv2.equalizeHist(gray)

    # 计算直方图
    hist_before = cv2.calcHist([gray], [0], None, [256], [0, 256])
    hist_after = cv2.calcHist([equalized], [0], None, [256], [0, 256])

    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))

    # 原图
    axes[0, 0].imshow(gray, cmap='gray', vmin=0, vmax=255)
    axes[0, 0].set_title(f'Original: {title}', fontsize=11)
    axes[0, 0].axis('off')

    # 原图直方图
    axes[0, 1].plot(hist_before, color='blue')
    axes[0, 1].fill_between(range(256), hist_before.flatten(), alpha=0.3)
    axes[0, 1].set_title('Original Histogram', fontsize=11)
    axes[0, 1].set_xlim([0, 255])

    # 均衡化后
    axes[1, 0].imshow(equalized, cmap='gray', vmin=0, vmax=255)
    axes[1, 0].set_title('After Equalization', fontsize=11)
    axes[1, 0].axis('off')

    # 均衡化后直方图
    axes[1, 1].plot(hist_after, color='green')
    axes[1, 1].fill_between(range(256), hist_after.flatten(), alpha=0.3, color='green')
    axes[1, 1].set_title('Equalized Histogram', fontsize=11)
    axes[1, 1].set_xlim([0, 255])

    plt.tight_layout()
    plt.show()

    return gray, equalized

# 创建低对比度图像

print(f"原图 - 范围: [{original.min()}, {original.max()}], 标准差: {original.std():.1f}")
print(f"均衡化后 - 范围: [{equalized.min()}, {equalized.max()}], 标准差: {equalized.std():.1f}")
```

## 2. 直方图均衡化原理

### 2.1 数学原理

直方图均衡化使用**累积分布函数（CDF）**来映射像素值：

```
步骤1: 计算归一化直方图 p(i) = n(i) / N

步骤2: 计算累积分布函数 CDF(i) = Σ p(j), j = 0 到 i

步骤3: 映射到新的灰度值 new_value = round(CDF(i) × (L-1))
        其中 L = 256（灰度级数）
```

### 2.2 手动实现

```python
"""
示例2：手动实现直方图均衡化
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def equalize_histogram_manual(image):
    """
    手动实现直方图均衡化

    步骤：
    1. 计算直方图
    2. 计算累积分布函数（CDF）
    3. 归一化CDF
    4. 映射像素值
    """
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # 步骤1：计算直方图
    hist = np.zeros(256)
    for pixel in gray.flatten():
        hist[pixel] += 1

    # 步骤2：计算CDF
    cdf = np.cumsum(hist)

    # 步骤3：归一化CDF（映射到0-255）
    cdf_min = cdf[cdf > 0].min()  # 第一个非零值
    cdf_normalized = (cdf - cdf_min) / (gray.size - cdf_min) * 255

    # 处理可能的NaN
    cdf_normalized = np.nan_to_num(cdf_normalized, nan=0)

    # 步骤4：映射像素值
    equalized = cdf_normalized[gray].astype(np.uint8)

    return equalized, hist, cdf, cdf_normalized

# 创建测试图像
test_img = np.random.normal(80, 25, (300, 400)).clip(0, 255).astype(np.uint8)

# 手动实现

# OpenCV实现

# 可视化

# CDF
axes[0, 2].plot(cdf / cdf.max() * 255, color='red', linewidth=2, label='CDF')
axes[0, 2].plot(cdf_norm, color='green', linewidth=2, linestyle='--', label='Normalized CDF')
axes[0, 2].set_title('Cumulative Distribution Function', fontsize=11)
axes[0, 2].set_xlim([0, 255])
```

```python
"""
示例3：可视化像素映射函数
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def visualize_mapping(image):
    """可视化直方图均衡化的映射函数"""
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # 计算直方图和CDF
    hist = cv2.calcHist([gray], [0], None, [256], [0, 256]).flatten()
    cdf = np.cumsum(hist)
    cdf_normalized = (cdf - cdf.min()) / (cdf.max() - cdf.min()) * 255

    # 创建映射表
    mapping = cdf_normalized.astype(np.uint8)

    # 可视化
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))

    # 映射函数
    axes[0].plot(range(256), mapping, 'b-', linewidth=2)
    axes[0].plot([0, 255], [0, 255], 'r--', linewidth=1, label='Identity (no change)')
    axes[0].set_title('Pixel Value Mapping Function', fontsize=12)
    axes[0].set_xlabel('Input Pixel Value')
    axes[0].set_ylabel('Output Pixel Value')
    axes[0].set_xlim([0, 255])
    axes[0].set_ylim([0, 255])
    axes[0].legend()
    axes[0].grid(True, alpha=0.3)
    axes[0].set_aspect('equal')

    # 直方图对比
    equalized = cv2.equalizeHist(gray)
    hist_eq = cv2.calcHist([equalized], [0], None, [256], [0, 256]).flatten()

    axes[1].plot(hist, 'b-', alpha=0.7, label='Original')
    axes[1].plot(hist_eq, 'g-', alpha=0.7, label='Equalized')
    axes[1].set_title('Histogram Comparison', fontsize=12)
    axes[1].set_xlabel('Pixel Value')
    axes[1].set_ylabel('Frequency')
    axes[1].set_xlim([0, 255])
    axes[1].legend()

    plt.tight_layout()
    plt.show()

    return mapping

# 创建测试图像（暗图像）
```

```python
"""
示例4：cv2.equalizeHist() 基本使用
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def basic_equalization_demo():
    """直方图均衡化基本演示"""
    # 创建不同类型的测试图像
    images = {
        'Dark (Underexposed)': np.random.normal(50, 20, (200, 300)).clip(0, 255).astype(np.uint8),
        'Bright (Overexposed)': np.random.normal(200, 20, (200, 300)).clip(0, 255).astype(np.uint8),

        axes[row, 2].axis('off')
```

```python
"""
示例5：彩色图像均衡化的错误与正确方法
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def create_color_image():
    """创建测试彩色图像"""
    img = np.zeros((300, 400, 3), dtype=np.uint8)

    # 创建渐变背景
    for i in range(300):
        for j in range(400):
            img[i, j] = [int(50 + 100 * i / 300),
                        int(80 + 80 * j / 400),
                        int(60 + 60 * (i + j) / 700)]

    # 添加彩色形状
    cv2.circle(img, (100, 150), 50, (200, 100, 50), -1)
    cv2.rectangle(img, (200, 80), (320, 220), (50, 180, 120), -1)

    return img

def equalize_color_wrong(image):
    """错误方法：直接对每个通道均衡化"""
    b, g, r = cv2.split(image)
    b_eq = cv2.equalizeHist(b)
    g_eq = cv2.equalizeHist(g)
    r_eq = cv2.equalizeHist(r)
    return cv2.merge([b_eq, g_eq, r_eq])

def equalize_color_correct(image):
    """正确方法：在YCrCb/HSV空间的亮度通道均衡化"""
    # 方法1：YCrCb空间
    ycrcb = cv2.cvtColor(image, cv2.COLOR_BGR2YCrCb)
    ycrcb[:, :, 0] = cv2.equalizeHist(ycrcb[:, :, 0])
    result_ycrcb = cv2.cvtColor(ycrcb, cv2.COLOR_YCrCb2BGR)

    # 方法2：HSV空间
    hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
    hsv[:, :, 2] = cv2.equalizeHist(hsv[:, :, 2])
    result_hsv = cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    return result_ycrcb, result_hsv

# 创建测试图像
color_img = create_color_image()

# 不同方法
wrong_result = equalize_color_wrong(color_img)
correct_ycrcb, correct_hsv = equalize_color_correct(color_img)

# 可视化
fig, axes = plt.subplots(2, 2, figsize=(12, 10))

axes[0, 0].imshow(cv2.cvtColor(color_img, cv2.COLOR_BGR2RGB))
axes[0, 0].set_title('Original Image', fontsize=12)
axes[0, 0].axis('off')

axes[0, 1].imshow(cv2.cvtColor(wrong_result, cv2.COLOR_BGR2RGB))
axes[0, 1].set_title('Wrong: BGR Channels Separately\n(Color distortion!)', fontsize=11)
axes[0, 1].axis('off')

axes[1, 0].imshow(cv2.cvtColor(correct_ycrcb, cv2.COLOR_BGR2RGB))
axes[1, 0].set_title('Correct: YCrCb (Y channel only)\n(Recommended)', fontsize=11)
axes[1, 0].axis('off')

axes[1, 1].imshow(cv2.cvtColor(correct_hsv, cv2.COLOR_BGR2RGB))
axes[1, 1].set_title('Correct: HSV (V channel only)', fontsize=11)
axes[1, 1].axis('off')

plt.tight_layout()
plt.show()

print("彩色图像直方图均衡化：")
```

### 4.2 完整的彩色图像均衡化函数

```python
"""
示例6：完整的彩色图像直方图均衡化
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def equalize_color_image(image, method='ycrcb'):
    """
    彩色图像直方图均衡化

    Parameters:
    -----------
    image : numpy.ndarray
        输入彩色图像 (BGR格式)
    method : str
        'ycrcb' - 在YCrCb空间均衡化Y通道 (推荐)
        'hsv' - 在HSV空间均衡化V通道
        'lab' - 在LAB空间均衡化L通道

    Returns:
    --------
    result : numpy.ndarray
        均衡化后的彩色图像
    """
    if method == 'ycrcb':
        # YCrCb方法（推荐）
        ycrcb = cv2.cvtColor(image, cv2.COLOR_BGR2YCrCb)
        ycrcb[:, :, 0] = cv2.equalizeHist(ycrcb[:, :, 0])
        result = cv2.cvtColor(ycrcb, cv2.COLOR_YCrCb2BGR)

    elif method == 'hsv':
        # HSV方法
        hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
        hsv[:, :, 2] = cv2.equalizeHist(hsv[:, :, 2])
        result = cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    elif method == 'lab':
        # LAB方法
        lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)
        lab[:, :, 0] = cv2.equalizeHist(lab[:, :, 0])
        result = cv2.cvtColor(lab, cv2.COLOR_LAB2BGR)

    else:
        raise ValueError(f"Unknown method: {method}")

    return result

# 创建低对比度彩色图像
def create_low_contrast_color():
    img = np.zeros((300, 400, 3), dtype=np.uint8)
    base = 100
    range_val = 50

    for i in range(300):
        for j in range(400):
            img[i, j] = [base + int(range_val * np.sin(i/30)),
                        base + int(range_val * np.cos(j/40)),
                        base + int(range_val * np.sin((i+j)/50))]

    cv2.circle(img, (150, 150), 60, (140, 100, 80), -1)
    cv2.rectangle(img, (220, 80), (350, 220), (80, 130, 110), -1)

    return img

color_img = create_low_contrast_color()

# 三种方法
result_ycrcb = equalize_color_image(color_img, 'ycrcb')
result_hsv = equalize_color_image(color_img, 'hsv')
result_lab = equalize_color_image(color_img, 'lab')

# 可视化
fig, axes = plt.subplots(2, 2, figsize=(12, 10))

axes[0, 0].imshow(cv2.cvtColor(color_img, cv2.COLOR_BGR2RGB))
axes[0, 0].set_title('Original (Low Contrast)', fontsize=12)
axes[0, 0].axis('off')

axes[0, 1].imshow(cv2.cvtColor(result_ycrcb, cv2.COLOR_BGR2RGB))
axes[0, 1].set_title('YCrCb Method', fontsize=12)
axes[0, 1].axis('off')

axes[1, 0].imshow(cv2.cvtColor(result_hsv, cv2.COLOR_BGR2RGB))
axes[1, 0].set_title('HSV Method', fontsize=12)
axes[1, 0].axis('off')

axes[1, 1].imshow(cv2.cvtColor(result_lab, cv2.COLOR_BGR2RGB))
axes[1, 1].set_title('LAB Method', fontsize=12)
axes[1, 1].axis('off')

plt.tight_layout()
plt.show()

print("三种方法对比：")
print("- YCrCb: 最常用，效果稳定")
print("- HSV: 效果类似，但可能在极端情况下产生伪影")
print("- LAB: 感知上更均匀，适合需要精确颜色保持的场景")
```

```python
    fig, axes = plt.subplots(3, 3, figsize=(12, 12))

    # 1. 噪声放大
    noisy = np.random.normal(128, 10, (200, 300)).clip(0, 255).astype(np.uint8)
    # 添加椒盐噪声
    salt_pepper = noisy.copy()

    axes[0, 0].imshow(salt_pepper, cmap='gray')

    # 2. 过度增强
    normal = np.random.normal(128, 40, (200, 300)).clip(0, 255).astype(np.uint8)
```

```python
"""

import cv2
import numpy as np
import matplotlib.pyplot as plt

def enhance_medical_image(image):
    """医学图像增强流程"""
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # 1. 降噪
    denoised = cv2.GaussianBlur(gray, (3, 3), 0)

    # 2. 直方图均衡化
    equalized = cv2.equalizeHist(denoised)

    # 3. 锐化（可选）
    kernel = np.array([[-1, -1, -1],
                       [-1,  9, -1],
                       [-1, -1, -1]])
    sharpened = cv2.filter2D(equalized, -1, kernel)

    return denoised, equalized, sharpened
```

```python
"""
示例10：夜景/低光照图像增强
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def enhance_night_image(image):
    """夜景图像增强"""
    # 转换到YCrCb
    if len(image.shape) == 3:
        ycrcb = cv2.cvtColor(image, cv2.COLOR_BGR2YCrCb)
        y, cr, cb = cv2.split(ycrcb)
    else:
        y = image.copy()

    # 对Y通道进行处理
    # 1. 轻微降噪
    y_denoised = cv2.GaussianBlur(y, (3, 3), 0)

    # 2. 直方图均衡化
    y_equalized = cv2.equalizeHist(y_denoised)

    # 3. 混合原始和均衡化结果（避免过度增强）
    alpha = 0.7
    y_blended = cv2.addWeighted(y_equalized, alpha, y, 1 - alpha, 0)

    if len(image.shape) == 3:
        # 合并通道
        enhanced = cv2.merge([y_blended, cr, cb])
        enhanced = cv2.cvtColor(enhanced, cv2.COLOR_YCrCb2BGR)
    else:
        enhanced = y_blended

    return enhanced

# 创建模拟夜景图像
def create_night_image():
    img = np.zeros((300, 400, 3), dtype=np.uint8)

    # 深色背景
    img[:] = [20, 25, 30]

    # 添加一些"灯光"
    lights = [(80, 100), (200, 80), (320, 120), (150, 200), (280, 180)]
    for x, y in lights:
        # 创建光晕效果
        for r in range(50, 0, -5):
            intensity = 30 + (50 - r) * 2
            cv2.circle(img, (x, y), r, (intensity, intensity + 10, intensity + 20), -1)

    # 添加一些建筑轮廓
    cv2.rectangle(img, (50, 150), (120, 280), (40, 35, 30), -1)
    cv2.rectangle(img, (180, 120), (240, 280), (35, 40, 35), -1)
    cv2.rectangle(img, (280, 100), (370, 280), (30, 35, 40), -1)

    return img

night_img = create_night_image()
enhanced = enhance_night_image(night_img)

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(12, 5))
```
