# 对比度增强（Contrast Enhancement）

## 1. 对比度增强概述

对比度增强是图像处理中最基本也是最重要的技术之一，它通过调整图像中像素值的分布范围来改善图像的视觉效果。良好的对比度能够使图像中的细节更加清晰可辨。

### 1.1 什么是对比度

对比度（Contrast）是指图像中最亮和最暗区域之间的差异程度。高对比度图像具有明显的明暗区分，而低对比度图像则显得灰蒙蒙、缺乏层次感。

**对比度的度量方式**：

1. **Michelson对比度**（适用于周期图案）：
   $$C = \frac{I_{max} - I_{min}}{I_{max} + I_{min}}$$

2. **RMS对比度**（均方根对比度）：
   $$C_{RMS} = \sqrt{\frac{1}{MN}\sum_{i,j}(I_{ij} - \bar{I})^2}$$

3. **Weber对比度**（适用于单一目标）：
   $$C = \frac{I - I_b}{I_b}$$

### 1.2 对比度增强方法分类

```
对比度增强方法
├── 点运算（逐像素操作）
│   ├── 线性变换
│   ├── 分段线性变换

│   └── 伽马变换

│   ├── 直方图均衡化
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def linear_contrast_stretch(image, out_min=0, out_max=255):
    """
    Linear contrast stretching (min-max normalization)

    Parameters:
        image: Input image
        out_min: Output minimum value
        out_max: Output maximum value

    Returns:
        Contrast stretched image
    """
    # Get input range
    in_min = np.min(image)
    in_max = np.max(image)

    # Avoid division by zero
    if in_max == in_min:
        return np.full_like(image, (out_min + out_max) // 2)

    # Linear mapping
    stretched = (image - in_min) * (out_max - out_min) / (in_max - in_min) + out_min

    return np.clip(stretched, out_min, out_max).astype(np.uint8)
```

### 示例2：百分比裁剪拉伸

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def percentile_contrast_stretch(image, low_percentile=2, high_percentile=98):
    """
    Contrast stretching with percentile clipping
    This method is more robust to outliers

    Parameters:
        image: Input image
        low_percentile: Lower percentile for clipping
        high_percentile: Upper percentile for clipping

    Returns:
        Contrast stretched image
    """
    # Calculate percentile values
    p_low = np.percentile(image, low_percentile)
    p_high = np.percentile(image, high_percentile)

    # Clip and stretch
    clipped = np.clip(image, p_low, p_high)
    stretched = (clipped - p_low) * 255 / (p_high - p_low)

    return stretched.astype(np.uint8)

# Read image
image = cv2.imread('image.jpg', cv2.IMREAD_GRAYSCALE)
if image is None:
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def piecewise_linear_transform(image, points):
    """
    Piecewise linear transformation

    Parameters:
        image: Input image
        points: List of (input, output) control points
                e.g., [(0, 0), (50, 30), (200, 220), (255, 255)]

    Returns:
        Transformed image
    """
    # Sort points by input value
    points = sorted(points, key=lambda x: x[0])

    # Build lookup table
    lut = np.zeros(256, dtype=np.uint8)

    for i in range(len(points) - 1):
        x1, y1 = points[i]
        x2, y2 = points[i + 1]

        # Linear interpolation between points
        for x in range(x1, x2 + 1):
            if x2 != x1:
                y = y1 + (y2 - y1) * (x - x1) / (x2 - x1)
            else:
                y = y1
            lut[x] = int(np.clip(y, 0, 255))

    # Apply LUT
    return cv2.LUT(image, lut), lut

# Read image
image = cv2.imread('image.jpg', cv2.IMREAD_GRAYSCALE)
if image is None:
    image = np.random.randint(40, 220, (300, 400), dtype=np.uint8)

transformations = [
    [(0, 0), (255, 255)],  # Identity

    [(0, 30), (128, 128), (255, 225)],  # Reduce dynamic range

# Display

plt.axis('off')

# Transformed images
```

```python
def intensity_level_slicing(image, low, high, highlight_value=255, method='highlight'):
    """
    Intensity level slicing to highlight specific range

    Parameters:
        image: Input image
        low: Lower bound of range to highlight
        high: Upper bound of range to highlight
        highlight_value: Value for highlighted pixels
        method: 'highlight' or 'binary'

    Returns:
        Sliced image
    """
    if method == 'binary':
        # Binary: Only show highlighted range
        result = np.zeros_like(image)
        mask = (image >= low) & (image <= high)
        result[mask] = highlight_value
    else:
        # Highlight: Preserve background, highlight range
        result = image.copy()
        mask = (image >= low) & (image <= high)
        result[mask] = highlight_value

    return result
```

```python
import cv2
import numpy as np

def log_transform(image, c=None):
    """
    Logarithmic transformation for contrast enhancement
    Formula: s = c * log(1 + r)

    Parameters:
        image: Input image
        c: Scaling constant (auto-calculated if None)

    Returns:
        Transformed image
    """
    # Convert to float
    img_float = image.astype(np.float64)

    # Apply log transform
    log_img = np.log1p(img_float)  # log(1 + x)

    # Calculate c if not provided (to use full range)
    if c is None:
        c = 255 / np.log1p(255)

    # Scale
    result = c * log_img

    return np.clip(result, 0, 255).astype(np.uint8)

def inverse_log_transform(image, c=None):
    """
    Inverse logarithmic transformation
    Formula: s = c * (exp(r) - 1)

    Parameters:
        image: Input image
        c: Scaling constant

    Returns:
        Transformed image
    """
    img_float = image.astype(np.float64) / 255.0

    # Apply inverse log (exponential)
    exp_img = np.expm1(img_float * np.log(256))  # exp(x) - 1

    if c is None:
        c = 255 / np.expm1(np.log(256))

    result = c * exp_img

    return np.clip(result, 0, 255).astype(np.uint8)
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def sigmoid_contrast(image, gain=10, cutoff=0.5):
    """
    S-curve (sigmoid) contrast enhancement

    Parameters:
        image: Input image
        gain: Steepness of the curve (higher = more contrast)
        cutoff: Center point of the sigmoid (0-1)

    Returns:
        Enhanced image
    """
    # Normalize to 0-1
    normalized = image / 255.0

    # Apply sigmoid
    # s = 1 / (1 + exp(-gain * (r - cutoff)))
    enhanced = 1 / (1 + np.exp(-gain * (normalized - cutoff)))

    # Scale back to 0-255
    return (enhanced * 255).astype(np.uint8)

def tanh_contrast(image, alpha=3):
    """
    Hyperbolic tangent contrast enhancement

    Parameters:
        image: Input image
        alpha: Steepness parameter

    Returns:
        Enhanced image
    """
    # Normalize to -1 to 1
    normalized = (image / 255.0) * 2 - 1

    # Apply tanh
    enhanced = np.tanh(alpha * normalized)

    # Scale back to 0-255
    return ((enhanced + 1) / 2 * 255).astype(np.uint8)
```

```python
def local_contrast_enhancement(image, kernel_size=31, clip_limit=2.0):
    """
    Local Contrast Enhancement using local statistics

    Parameters:
        image: Input grayscale image
        kernel_size: Size of local region
        clip_limit: Limit for contrast enhancement

    Returns:
        Enhanced image
    """
    # Ensure odd kernel size
    if kernel_size % 2 == 0:
        kernel_size += 1

    # Calculate local mean
    local_mean = cv2.blur(image.astype(np.float64), (kernel_size, kernel_size))

    # Calculate local variance
    local_sq_mean = cv2.blur(image.astype(np.float64)**2, (kernel_size, kernel_size))
    local_var = local_sq_mean - local_mean**2
    local_std = np.sqrt(np.maximum(local_var, 0))

    # Global statistics
    global_mean = np.mean(image)
    global_std = np.std(image)

    # Avoid division by zero
    local_std = np.maximum(local_std, 1e-5)

    # Local contrast enhancement
    enhanced = global_mean + (image - local_mean) * (global_std / local_std)

    # Clip enhancement factor
    enhanced = np.clip(enhanced, 0, 255).astype(np.uint8)

    return enhanced

def unsharp_mask(image, sigma=1.0, strength=1.5):

    enhanced = cv2.addWeighted(image, 1 + strength, blurred, -strength, 0)

    return enhanced
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def single_scale_retinex(image, sigma=30):
    """
    Single Scale Retinex (SSR)

    Parameters:
        image: Input image
        sigma: Gaussian blur sigma

    Returns:
        Enhanced image
    """
    # Convert to float and add small value to avoid log(0)
    img_float = image.astype(np.float64) + 1

    # Log of original image
    log_img = np.log(img_float)

    # Gaussian blur (illumination estimation)
    blur = cv2.GaussianBlur(img_float, (0, 0), sigma)
    log_blur = np.log(blur)

    # Retinex: log(image) - log(blur) = log(reflectance)
    retinex = log_img - log_blur

    # Normalize to 0-255
    retinex = (retinex - retinex.min()) / (retinex.max() - retinex.min()) * 255

    return retinex.astype(np.uint8)

def multi_scale_retinex(image, sigmas=[15, 80, 250]):
    """
    Multi-Scale Retinex (MSR)

    Parameters:
        image: Input image
        sigmas: List of sigma values for different scales

    Returns:
        Enhanced image
    """
    img_float = image.astype(np.float64) + 1
    log_img = np.log(img_float)

    retinex = np.zeros_like(img_float)

    for sigma in sigmas:
        blur = cv2.GaussianBlur(img_float, (0, 0), sigma)
        retinex += log_img - np.log(blur)

    retinex /= len(sigmas)

    # Normalize
    retinex = (retinex - retinex.min()) / (retinex.max() - retinex.min()) * 255

    return retinex.astype(np.uint8)

def msrcr(image, sigmas=[15, 80, 250], alpha=125, beta=46, G=192, b=-30):
    """
    Multi-Scale Retinex with Color Restoration (MSRCR)

    Parameters:
        image: Input BGR image
        sigmas: Sigma values for MSR
        alpha, beta, G, b: Color restoration parameters

    Returns:
        Enhanced image
    """
    img_float = image.astype(np.float64) + 1

    # Process each channel
    result = np.zeros_like(img_float)

    for i in range(3):
        channel = img_float[:, :, i]
        log_channel = np.log(channel)

        retinex = np.zeros_like(channel)
        for sigma in sigmas:
            blur = cv2.GaussianBlur(channel, (0, 0), sigma)
            retinex += log_channel - np.log(blur)
        retinex /= len(sigmas)

        # Color restoration
        sum_channels = np.sum(img_float, axis=2) + 1e-5
        color_restoration = beta * (np.log(alpha * channel / sum_channels))

        result[:, :, i] = G * (retinex * color_restoration + b)

    # Normalize
    result = (result - result.min()) / (result.max() - result.min()) * 255

    return result.astype(np.uint8)
```

```python
        Parameters:
            image: Input image

            Enhanced image

        if len(image.shape) == 3:
            gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            gray = image

        # Analyze image
        mean_val = np.mean(gray)
        std_val = np.std(gray)

        if method == 'auto':
            # Determine enhancement method based on statistics
            if mean_val < 80:
                method = 'dark'
            elif mean_val > 180:
                method = 'bright'
            elif std_val < 40:
                method = 'low_contrast'
            else:
                method = 'normal'

        if method == 'dark':
            enhanced = self._enhance_dark(image)
        elif method == 'bright':
            enhanced = self._enhance_bright(image)
        elif method == 'low_contrast':
            enhanced = self._enhance_low_contrast(image)
```

```python
            result = np.zeros_like(image)
            for i in range(3):
                channel = image[:, :, i]
                p2, p98 = np.percentile(channel, (2, 98))
                result[:, :, i] = np.clip((channel - p2) * 255 / (p98 - p2 + 1), 0, 255)
            return result.astype(np.uint8)
        else:
            p2, p98 = np.percentile(image, (2, 98))
            return np.clip((image - p2) * 255 / (p98 - p2 + 1), 0, 255).astype(np.uint8)

    def _apply_auto(self, image):
        """Auto-select enhancement method"""
        if len(image.shape) == 3:
            gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            gray = image

        mean_val = np.mean(gray)

        if mean_val < 80:
            # Dark image - use gamma
            enhanced = self._apply_gamma(image, gamma=0.5)
            return self._apply_clahe(enhanced)
        elif mean_val > 180:
            # Bright image - use inverse gamma
            return self._apply_gamma(image, gamma=1.8)
        else:
            # Normal - use CLAHE
            return self._apply_clahe(image)

    def process_batch(self, input_dir, method='auto', extensions=('.jpg', '.png', '.bmp')):
        """
        Process all images in a directory

        Parameters:
            input_dir: Input directory
            method: Enhancement method
            extensions: File extensions to process

        Returns:
            Processing statistics
        """
        input_path = Path(input_dir)
        stats = {'processed': 0, 'failed': 0, 'files': []}

        # Get all image files
        image_files = []
        for ext in extensions:
            image_files.extend(input_path.glob(f'*{ext}'))
            image_files.extend(input_path.glob(f'*{ext.upper()}'))

        print(f"Found {len(image_files)} images to process")

        for img_path in image_files:
            try:
                # Read image
                image = cv2.imread(str(img_path))
                if image is None:
                    stats['failed'] += 1
                    continue

                # Enhance
                enhanced = self.enhance_single(image, method)

                # Save
                output_path = os.path.join(self.output_dir, img_path.name)
                cv2.imwrite(output_path, enhanced)

                stats['processed'] += 1
                stats['files'].append(img_path.name)

            except Exception as e:
                print(f"Error processing {img_path.name}: {e}")
                stats['failed'] += 1

        return stats

    def create_comparison(self, image_path, methods=None):
        """
        Create comparison image with all methods

        Parameters:
            image_path: Path to input image
            methods: List of methods to compare

        Returns:
            Comparison image
        """
        if methods is None:
            methods = ['original', 'clahe', 'gamma', 'linear', 'auto']

        image = cv2.imread(image_path)
        if image is None:
            # Create test image
            image = np.random.randint(50, 200, (300, 400, 3), dtype=np.uint8)

        results = {'original': image}

        for method in methods:
            if method != 'original':
                results[method] = self.enhance_single(image, method)

        # Create figure
        n_methods = len(methods)
        fig, axes = plt.subplots(2, (n_methods + 1) // 2, figsize=(15, 8))
        axes = axes.flatten()

        for i, method in enumerate(methods):
            img_rgb = cv2.cvtColor(results[method], cv2.COLOR_BGR2RGB)
            axes[i].imshow(img_rgb)
            axes[i].set_title(method.upper())
            axes[i].axis('off')

        # Hide unused axes
        for i in range(n_methods, len(axes)):
            axes[i].axis('off')

        plt.tight_layout()
        comparison_path = os.path.join(self.output_dir, 'comparison.png')
        plt.savefig(comparison_path, dpi=150, bbox_inches='tight')
        plt.close()

        return comparison_path
```

```python
def evaluate_enhancement(original, enhanced):
    """
    Evaluate enhancement quality using various metrics

    Parameters:
        original: Original image
        enhanced: Enhanced image

    Returns:
        Dictionary of metrics
    """
    # Convert to grayscale if needed
    if len(original.shape) == 3:
        orig_gray = cv2.cvtColor(original, cv2.COLOR_BGR2GRAY)
        enh_gray = cv2.cvtColor(enhanced, cv2.COLOR_BGR2GRAY)
    else:
        orig_gray = original
        enh_gray = enhanced

    metrics = {}

    # Contrast (standard deviation)
    metrics['contrast_orig'] = np.std(orig_gray)
    metrics['contrast_enh'] = np.std(enh_gray)
    metrics['contrast_gain'] = metrics['contrast_enh'] / (metrics['contrast_orig'] + 1e-5)

    # Dynamic range
    metrics['range_orig'] = np.max(orig_gray) - np.min(orig_gray)
    metrics['range_enh'] = np.max(enh_gray) - np.min(enh_gray)

    # Entropy (information content)
    def calc_entropy(img):
        hist = cv2.calcHist([img], [0], None, [256], [0, 256])
        hist = hist / hist.sum()
        hist = hist[hist > 0]
        return -np.sum(hist * np.log2(hist))

    metrics['entropy_enh'] = calc_entropy(enh_gray)

    metrics['mean_orig'] = np.mean(orig_gray)
    metrics['mean_enh'] = np.mean(enh_gray)

        sobelx = cv2.Sobel(img, cv2.CV_64F, 1, 0, ksize=3)
        sobely = cv2.Sobel(img, cv2.CV_64F, 0, 1, ksize=3)
        return np.mean(np.sqrt(sobelx**2 + sobely**2))

    metrics['edges_orig'] = edge_strength(orig_gray)
    metrics['edges_enh'] = edge_strength(enh_gray)

    return metrics
```

## 8. 总结

### 8.1 对比度增强方法汇总

| 方法 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| **线性拉伸** | 简单快速 | 对异常值敏感 | 正常分布图像 |
| **百分比拉伸** | 鲁棒性好 | 可能丢失细节 | 有噪声的图像 |
| **直方图均衡化** | 自动优化 | 可能过度增强 | 低对比度图像 |
| **CLAHE** | 局部自适应 | 参数敏感 | 光照不均图像 |
| **伽马变换** | 可控性强 | 需要选择参数 | 曝光问题图像 |
| **S曲线** | 增强中间调 | 可能饱和 | 一般增强需求 |
| **Retinex** | 自然效果 | 计算复杂 | 复杂光照场景 |

### 8.2 选择指南

```
图像特征分析
├── 暗图像（均值 < 80）
│   └── 推荐：伽马变换（γ < 1）+ CLAHE
├── 亮图像（均值 > 180）
│   └── 推荐：伽马变换（γ > 1）
├── 低对比度（标准差 < 40）
│   └── 推荐：直方图均衡化或CLAHE
├── 光照不均
│   └── 推荐：CLAHE或Retinex
└── 正常图像
    └── 推荐：轻度CLAHE或S曲线
```

### 8.3 最佳实践

1. **先分析后处理**：了解图像特征再选择方法
2. **保守增强**：避免过度处理导致失真
3. **彩色图像**：使用LAB空间处理亮度通道
4. **局部优先**：优先考虑局部自适应方法
5. **评估结果**：使用客观指标验证增强效果
6. **保留原图**：始终保留原始图像备份
