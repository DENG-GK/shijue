# 伽马变换（Gamma Transformation）

## 1. 伽马变换概述

伽马变换（Gamma Transformation）是一种重要的非线性图像增强技术，广泛应用于图像亮度校正、对比度调整和显示设备校准。它通过幂次变换来调整图像的灰度级分布，能够有效地增强图像的暗部或亮部细节。

### 1.1 什么是伽马变换

伽马变换的数学公式为：

$$s = c \cdot r^{\gamma}$$

其中：
- **r**：输入像素值（归一化到0-1范围）
- **s**：输出像素值
- **c**：常数（通常为1）
- **γ（gamma）**：伽马值，控制变换曲线的形状

### 1.2 伽马值的影响

| γ值范围 | 变换效果 | 应用场景 |
|---------|----------|----------|
| γ < 1 | 提亮暗部，压缩亮部 | 增强曝光不足的图像 |
| γ = 1 | 无变化（线性映射） | 原始图像 |
| γ > 1 | 压缩暗部，提亮亮部 | 增强过曝图像 |

### 1.3 伽马变换曲线

```
输出
  |        γ<1
1 |       ___----
  |    _--
  |  _/        γ=1
  | /        ___----
  |/      _--
  |    _/        γ>1
  | _/
  |/______________ 输入
0               1
```

## 2. 基础伽马变换实现

### 示例1：基本伽马变换

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def gamma_transform(image, gamma, c=1.0):
    """
    Apply gamma transformation to an image

    Parameters:
        image: Input image (uint8)
        gamma: Gamma value (γ < 1 brightens, γ > 1 darkens)
        c: Constant multiplier (default 1.0)

    Returns:
        Transformed image (uint8)
    """
    # Normalize to 0-1 range
    normalized = image / 255.0

    # Apply gamma transformation
    transformed = c * np.power(normalized, gamma)

    # Clip values to valid range and convert back to uint8
    transformed = np.clip(transformed * 255, 0, 255).astype(np.uint8)

    return transformed

# Read image
image = cv2.imread('image.jpg', cv2.IMREAD_GRAYSCALE)

# Apply different gamma values
gamma_values = [0.3, 0.5, 1.0, 1.5, 2.5]
results = []

for gamma in gamma_values:
    result = gamma_transform(image, gamma)
    results.append(result)

# Display results
plt.figure(figsize=(15, 8))

for i, (gamma, result) in enumerate(zip(gamma_values, results)):
    plt.subplot(2, 3, i+1)
    plt.imshow(result, cmap='gray')
    plt.title(f'Gamma = {gamma}')
    plt.axis('off')

plt.tight_layout()
plt.savefig('gamma_comparison.png', dpi=150, bbox_inches='tight')
plt.show()

print("Basic gamma transformation completed!")
```

### 示例2：使用查找表（LUT）加速伽马变换

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def gamma_transform_lut(image, gamma):
    """
    Apply gamma transformation using Look-Up Table (LUT)
    This method is much faster for large images

    Parameters:
        image: Input image (uint8)
        gamma: Gamma value

    Returns:
        Transformed image (uint8)
    """
    # Build LUT
    inv_gamma = 1.0 / gamma
    table = np.array([((i / 255.0) ** inv_gamma) * 255
                      for i in range(256)]).astype(np.uint8)

    # Apply LUT
    return cv2.LUT(image, table)

# Read image
image = cv2.imread('image.jpg', cv2.IMREAD_GRAYSCALE)

# Compare speed
import time

# Method 1: Direct calculation
start = time.time()
for _ in range(100):

# Method 2: LUT
start = time.time()
for _ in range(100):
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def gamma_transform_luminance(image, gamma):
    """
    Apply gamma transformation only to luminance channel
    This preserves color saturation better than direct RGB transformation

    Parameters:
        image: Input BGR image
        gamma: Gamma value

    Returns:
        Transformed image
    """
    # Convert to LAB color space
    lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)
    l, a, b = cv2.split(lab)

    # Apply gamma to L channel
    inv_gamma = 1.0 / gamma
    table = np.array([((i / 255.0) ** inv_gamma) * 255
                      for i in range(256)]).astype(np.uint8)
    l_corrected = cv2.LUT(l, table)

    # Merge channels
    lab_corrected = cv2.merge([l_corrected, a, b])

    # Convert back to BGR
    return cv2.cvtColor(lab_corrected, cv2.COLOR_LAB2BGR)

# Read color image
```

### 示例6：基于图像均值的自适应伽马校正

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def auto_gamma_correction(image, target_mean=128):
    """
    Automatically determine gamma value based on image mean

    Parameters:
        image: Input image
        target_mean: Desired mean intensity (default 128)

    Returns:
        Corrected image and calculated gamma value
    """
    # Calculate current mean
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image

    current_mean = np.mean(gray)

    # Calculate gamma
    # target = current^gamma => gamma = log(target/255) / log(current/255)
    if current_mean > 0 and current_mean < 255:
        gamma = np.log(target_mean / 255.0) / np.log(current_mean / 255.0)
        gamma = np.clip(gamma, 0.1, 10.0)  # Limit gamma range
    else:
        gamma = 1.0

    # Apply gamma correction
    inv_gamma = 1.0 / gamma
    table = np.array([((i / 255.0) ** inv_gamma) * 255
                      for i in range(256)]).astype(np.uint8)

    corrected = cv2.LUT(image, table)

    return corrected, gamma
```

### 示例7：基于直方图的自适应伽马校正（AGC）

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def adaptive_gamma_correction(image, clip_limit=0.01):
    """
    Adaptive Gamma Correction based on Weighted Distribution

    Parameters:
        image: Input grayscale image
        clip_limit: Histogram clipping limit

    Returns:
        Corrected image
    """
    # Ensure grayscale
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # Calculate histogram
    hist = cv2.calcHist([gray], [0], None, [256], [0, 256])
    hist = hist.flatten() / hist.sum()

    # Clip histogram
    clip_value = clip_limit
    hist = np.clip(hist, 0, clip_value)
    hist = hist / hist.sum()

    # Calculate CDF
    cdf = np.cumsum(hist)

    # Calculate weighted mean
    intensity = np.arange(256)
    mean_intensity = np.sum(intensity * hist)

    # Calculate gamma based on mean intensity
    # Lower mean -> lower gamma (brighten)
    # Higher mean -> higher gamma (darken)
    gamma = -np.log2(mean_intensity / 255.0 + 1e-8)
    gamma = np.clip(gamma, 0.1, 4.0)

    # Apply adaptive gamma based on local intensity
    # Build modified LUT
    lut = np.zeros(256, dtype=np.float32)
    for i in range(256):
        # Local gamma adjustment
        local_gamma = gamma * (1 - cdf[i]) + 1 * cdf[i]
        lut[i] = np.power(i / 255.0, local_gamma) * 255

    lut = np.clip(lut, 0, 255).astype(np.uint8)

    return cv2.LUT(gray, lut), gamma

# Test

    # Create test image

    image[:, 200:] = 200  # Bright region
    # Add gradient
    for i in range(300):
```

```python
def enhance_low_light_image(image, gamma=0.4, denoise=True):
    """
    Enhance low-light images using gamma correction with optional denoising

    Parameters:
        image: Input low-light image
        gamma: Gamma value for brightening (< 1)
        denoise: Whether to apply denoising

    Returns:
        Enhanced image
    """
    # Convert to LAB
    lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)

    lab_enhanced = cv2.merge([l_enhanced, a, b])
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def create_gamma_calibration_pattern():
    """
    Create a gamma calibration pattern for monitor calibration
    """
    width = 800
    height = 600

    # Create pattern
    pattern = np.zeros((height, width), dtype=np.uint8)

    # Top section: Gray gradient
    gradient = np.tile(np.linspace(0, 255, width), (100, 1)).astype(np.uint8)
    pattern[:100, :] = gradient

    # Middle section: Checkerboard patterns for different gamma values
    y_start = 120
    box_height = 80
    box_width = width // 5

    gamma_values = [1.8, 2.0, 2.2, 2.4, 2.6]

    for i, gamma in enumerate(gamma_values):
        x_start = i * box_width

        # Create checkerboard
        for y in range(y_start, y_start + box_height):
            for x in range(x_start, x_start + box_width):
                if (x // 4 + y // 4) % 2 == 0:
                    pattern[y, x] = 0
                else:
                    pattern[y, x] = 255

        # Create solid gray that should match at specific gamma
        target_gray = int(255 * np.power(0.5, 1/gamma))
        solid_y = y_start + box_height + 10
        pattern[solid_y:solid_y+50, x_start:x_start+box_width] = target_gray

    # Bottom section: Gamma test strips
    strip_y = 300
    strip_height = 60

    for i in range(5):
        gamma = 1.8 + i * 0.2
        x_start = i * box_width

        # Create gradient with this gamma applied
        inv_gamma = 1.0 / gamma
        for x in range(box_width):
            val = int(255 * np.power(x / box_width, inv_gamma))
            pattern[strip_y:strip_y+strip_height, x_start+x] = val

    return pattern, gamma_values
```

```python
def gamma_based_tone_mapping(hdr_image, gamma=2.2, exposure=1.0):
    """
    Simple gamma-based tone mapping for HDR images

    Parameters:
        hdr_image: HDR image (float32, may have values > 1)
        gamma: Output gamma
        exposure: Exposure adjustment

    Returns:
        LDR image (uint8)
    """
    # Apply exposure
    exposed = hdr_image * exposure

    # Clamp negative values
    exposed = np.maximum(exposed, 0)

    # Simple Reinhard tone mapping
    mapped = exposed / (1 + exposed)

    # Apply gamma correction
    inv_gamma = 1.0 / gamma
    corrected = np.power(mapped, inv_gamma)

    # Convert to 8-bit
    ldr = np.clip(corrected * 255, 0, 255).astype(np.uint8)

    return ldr
```

```python
def enhance_medical_image(image, gamma=0.7, clahe_clip=2.0):
    """
    Enhance medical images using gamma correction and CLAHE

    Parameters:
        image: Input medical image (grayscale)
        gamma: Gamma value for initial correction
        clahe_clip: CLAHE clip limit

    Returns:
        Enhanced image
    """

    clahe = cv2.createCLAHE(clipLimit=clahe_clip, tileGridSize=(8, 8))
```

```python
def compare_enhancement_methods(image):
    """
    Compare gamma correction with histogram equalization
    """
    # Ensure grayscale
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    gamma = 0.5

    table = np.array([((i / 255.0) ** inv_gamma) * 255
                      for i in range(256)]).astype(np.uint8)

    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
```

## 7. 总结

### 7.1 伽马变换的特点

| 特点 | 说明 |
|------|------|
| **非线性变换** | 使用幂函数进行像素映射 |
| **可控性强** | 通过调整γ值精确控制变换程度 |
| **保持顺序** | 像素相对亮暗关系保持不变 |
| **全局操作** | 对所有像素应用相同的变换 |

### 7.2 γ值选择指南

| 场景 | 推荐γ值 | 说明 |
|------|---------|------|
| 低光图像增强 | 0.3 - 0.6 | 提亮暗部细节 |
| 轻度曝光不足 | 0.6 - 0.9 | 温和提亮 |
| 正常图像 | 1.0 | 无变化 |
| 轻度过曝 | 1.2 - 1.5 | 温和压暗 |
| 严重过曝 | 1.5 - 3.0 | 恢复亮部细节 |
| 显示器校准 | 2.2 | sRGB标准 |

### 7.3 最佳实践

1. **使用LUT加速**：对于大图像，使用查找表显著提升性能
2. **LAB空间处理**：处理彩色图像时，在LAB空间的L通道操作能更好保持色彩
3. **结合其他技术**：与CLAHE结合可获得更好的局部对比度
4. **自适应处理**：根据图像统计特征自动选择γ值
5. **避免过度处理**：极端γ值可能导致细节丢失或噪声放大
