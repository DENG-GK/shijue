# 图像直方图

## 1. 什么是图像直方图？

### 1.1 直方图的概念

图像直方图（Histogram）是一种统计图表，用于**统计图像中各个灰度级（或颜色值）出现的频率**。

> **简单理解**：直方图告诉我们图像中有多少像素是黑色的、有多少是白色的、有多少是灰色的。

```
直方图示意：

频率 │
     │    ██
     │    ██  ██
     │ ██ ██  ██ ██
     │ ██ ██  ██ ██ ██
     │ ██ ██  ██ ██ ██ ██
     └─────────────────────── 灰度值
       0  50 100 150 200 255
      黑        灰        白
```

### 1.2 直方图的作用

| 作用 | 说明 |
|------|------|
| **评估曝光** | 判断图像是过曝、欠曝还是正常 |
| **分析对比度** | 直方图越宽，对比度越高 |
| **图像增强** | 根据直方图选择增强方法 |
| **阈值选择** | 帮助选择合适的二值化阈值 |
| **图像匹配** | 直方图可用于图像相似度比较 |

### 1.3 直方图的类型

```python
"""
示例1：理解不同类型图像的直方图
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def create_sample_images():
    """创建不同类型的示例图像"""
    # 1. 暗图像（欠曝）
    dark = np.random.normal(50, 20, (200, 300)).clip(0, 255).astype(np.uint8)

    # 2. 亮图像（过曝）
    bright = np.random.normal(200, 20, (200, 300)).clip(0, 255).astype(np.uint8)

    # 3. 低对比度图像
    low_contrast = np.random.normal(128, 15, (200, 300)).clip(0, 255).astype(np.uint8)

    # 4. 高对比度图像
    high_contrast = np.zeros((200, 300), dtype=np.uint8)
    high_contrast[:, :150] = np.random.normal(50, 20, (200, 150)).clip(0, 255)
    high_contrast[:, 150:] = np.random.normal(200, 20, (200, 150)).clip(0, 255)

    # 5. 正常图像
    normal = np.random.normal(128, 50, (200, 300)).clip(0, 255).astype(np.uint8)

    return {
        'Dark (Underexposed)': dark,
        'Bright (Overexposed)': bright,
        'Low Contrast': low_contrast,
        'High Contrast': high_contrast,
        'Normal': normal
    }

images = create_sample_images()

# 可视化
fig, axes = plt.subplots(2, 5, figsize=(20, 8))

for i, (name, img) in enumerate(images.items()):
    # 显示图像
    axes[0, i].imshow(img, cmap='gray', vmin=0, vmax=255)
    axes[0, i].set_title(name, fontsize=10)
    axes[0, i].axis('off')

    # 显示直方图
    hist = cv2.calcHist([img], [0], None, [256], [0, 256])
    axes[1, i].plot(hist, color='blue')
    axes[1, i].fill_between(range(256), hist.flatten(), alpha=0.3)
    axes[1, i].set_xlim([0, 256])
    axes[1, i].set_xlabel('Pixel Value')
    if i == 0:
        axes[1, i].set_ylabel('Frequency')

plt.tight_layout()
plt.show()

print("直方图特征分析：")
print("=" * 60)
print("暗图像:     直方图集中在左侧（低灰度值）")
print("亮图像:     直方图集中在右侧（高灰度值）")
print("低对比度:   直方图集中在中间，范围窄")
print("高对比度:   直方图分布在两端，范围宽")
print("正常图像:   直方图分布均匀，覆盖大部分范围")
```

## 2. OpenCV计算直方图

### 2.1 cv2.calcHist() 函数

```python
hist = cv2.calcHist(images, channels, mask, histSize, ranges)
```

**参数说明：**

| 参数 | 类型 | 说明 |
|------|------|------|
| `images` | list | 输入图像列表，用方括号包裹 `[img]` |
| `channels` | list | 通道索引，灰度图为 `[0]`，彩色图 `[0]`/`[1]`/`[2]` |
| `mask` | ndarray/None | 掩码，None表示计算整幅图像 |
| `histSize` | list | 直方图的bin数量，如 `[256]` |
| `ranges` | list | 像素值范围，通常为 `[0, 256]` |

**返回值：**

| 返回值 | 类型 | 说明 |
|--------|------|------|
| `hist` | ndarray | 形状为 (histSize, 1) 的数组 |

### 2.2 基本使用

```python
"""
示例2：计算和显示灰度直方图
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def compute_histogram(image):
    """计算灰度图像的直方图"""
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # 使用OpenCV计算直方图
    hist = cv2.calcHist([gray], [0], None, [256], [0, 256])

    return hist

# 创建测试图像
img = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (200, 250), 100, -1)
cv2.circle(img, (300, 150), 80, 200, -1)
noise = np.random.normal(0, 10, img.shape)
img = np.clip(img + noise, 0, 255).astype(np.uint8)

# 计算直方图
hist = compute_histogram(img)

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(12, 4))

axes[0].imshow(img, cmap='gray')
axes[0].set_title('Image', fontsize=12)
axes[0].axis('off')

axes[1].plot(hist, color='blue', linewidth=1)
axes[1].fill_between(range(256), hist.flatten(), alpha=0.3)
axes[1].set_title('Histogram', fontsize=12)
axes[1].set_xlabel('Pixel Value (0-255)')
axes[1].set_ylabel('Frequency')
axes[1].set_xlim([0, 255])
axes[1].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

# 打印统计信息
print(f"直方图形状: {hist.shape}")
print(f"像素总数: {hist.sum():.0f}")
print(f"图像尺寸: {img.shape[0]} x {img.shape[1]} = {img.size}")
print(f"最大频率: {hist.max():.0f} (灰度值 {hist.argmax()})")
```

### 2.3 使用NumPy计算直方图

```python
"""
示例3：使用NumPy计算直方图
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def compare_histogram_methods(image):
    """对比OpenCV和NumPy计算直方图的方法"""
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # 方法1：OpenCV
    hist_cv = cv2.calcHist([gray], [0], None, [256], [0, 256])

    # 方法2：NumPy histogram
    hist_np, bins = np.histogram(gray.flatten(), bins=256, range=[0, 256])

    # 方法3：NumPy bincount（最快）
    hist_bc = np.bincount(gray.flatten(), minlength=256)

    return hist_cv.flatten(), hist_np, hist_bc

# 创建测试图像
img = np.random.randint(0, 256, (500, 500), dtype=np.uint8)

# 性能对比
import time

methods = ['cv2.calcHist', 'np.histogram', 'np.bincount']
times = []

for method in methods:
    start = time.time()
    for _ in range(100):
        if method == 'cv2.calcHist':
            cv2.calcHist([img], [0], None, [256], [0, 256])
        elif method == 'np.histogram':
            np.histogram(img.flatten(), bins=256, range=[0, 256])
        else:
            np.bincount(img.flatten(), minlength=256)
    times.append((time.time() - start) * 10)

# 验证结果一致性
hist_cv, hist_np, hist_bc = compare_histogram_methods(img)

print("直方图计算方法对比：")
print("=" * 50)
for method, t in zip(methods, times):
    print(f"{method:15s}: {t:.2f} ms (100次平均)")
print("=" * 50)
print(f"\n结果一致性验证:")
print(f"OpenCV vs NumPy:   {np.allclose(hist_cv, hist_np)}")
print(f"OpenCV vs bincount: {np.allclose(hist_cv, hist_bc)}")
```

```python
"""
示例4：彩色图像的三通道直方图
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def color_histogram(image):
    """计算彩色图像的三通道直方图"""
    # OpenCV读取的是BGR格式
    colors = ('b', 'g', 'r')
    channel_names = ('Blue', 'Green', 'Red')

    histograms = []
    for i, (color, name) in enumerate(zip(colors, channel_names)):
        hist = cv2.calcHist([image], [i], None, [256], [0, 256])
        histograms.append((name, color, hist))

    return histograms

# 创建彩色测试图像
def create_color_image():
    img = np.zeros((300, 400, 3), dtype=np.uint8)

    # 红色区域
    img[50:150, 50:150] = [0, 0, 200]

    # 绿色区域
    img[50:150, 150:250] = [0, 200, 0]

    # 蓝色区域
    img[50:150, 250:350] = [200, 0, 0]

    # 混合区域
    img[150:250, 100:200] = [100, 150, 200]  # 偏橙色
    img[150:250, 200:300] = [200, 100, 50]   # 偏青色

    # 添加噪声
    noise = np.random.normal(0, 10, img.shape)
    img = np.clip(img + noise, 0, 255).astype(np.uint8)

    return img

color_img = create_color_image()
histograms = color_histogram(color_img)

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# 显示图像（转换为RGB用于matplotlib显示）
axes[0].imshow(cv2.cvtColor(color_img, cv2.COLOR_BGR2RGB))
axes[0].set_title('Color Image (RGB)', fontsize=12)
axes[0].axis('off')

# 显示三通道直方图
for name, color, hist in histograms:
    axes[1].plot(hist, color=color, label=name, linewidth=1.5)

axes[1].set_title('Color Histogram (BGR Channels)', fontsize=12)
axes[1].set_xlabel('Pixel Value')
axes[1].set_ylabel('Frequency')
axes[1].set_xlim([0, 255])
axes[1].legend()
axes[1].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
```

```python
"""
示例5：HSV颜色空间的直方图
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def hsv_histogram(image):
    """计算HSV空间的直方图"""
    hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)

    # H通道：色调（0-179）
    hist_h = cv2.calcHist([hsv], [0], None, [180], [0, 180])

    # S通道：饱和度（0-255）
    hist_s = cv2.calcHist([hsv], [1], None, [256], [0, 256])

    # V通道：明度（0-255）
    hist_v = cv2.calcHist([hsv], [2], None, [256], [0, 256])

    return hist_h, hist_s, hist_v, hsv

# 创建彩色图像
color_img = create_color_image()

hist_h, hist_s, hist_v, hsv = hsv_histogram(color_img)

# 可视化
fig, axes = plt.subplots(2, 3, figsize=(15, 8))

axes[0, 1].axis('off')
```

```python
"""

import numpy as np
import matplotlib.pyplot as plt

def roi_histogram(image, mask):

        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

        gray = image.copy()

    hist_full = cv2.calcHist([gray], [0], None, [256], [0, 256])

mask = np.zeros(img.shape, dtype=np.uint8)
```

### 4.2 直方图归一化

```python
"""
示例7：直方图归一化
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def normalize_histogram(image, norm_type='probability'):
    """
    归一化直方图

    Parameters:
    -----------
    norm_type : str
        'probability' - 归一化为概率分布（和为1）
        'max' - 归一化到[0, 1]范围（最大值为1）
    """
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    hist = cv2.calcHist([gray], [0], None, [256], [0, 256])

    if norm_type == 'probability':
        # 归一化为概率分布
        hist_norm = hist / hist.sum()
    else:
        # 归一化到[0, 1]
        hist_norm = hist / hist.max()

    return hist, hist_norm

# 创建测试图像
img = np.random.randint(0, 256, (300, 400), dtype=np.uint8)

hist, hist_prob = normalize_histogram(img, 'probability')
_, hist_max = normalize_histogram(img, 'max')

# 可视化
fig, axes = plt.subplots(1, 3, figsize=(15, 4))

axes[0].plot(hist)
axes[0].set_title('Original Histogram\n(Frequency)', fontsize=11)
axes[0].set_xlabel('Pixel Value')
axes[0].set_ylabel('Count')

axes[1].plot(hist_prob)
axes[1].set_title('Probability Histogram\n(Sum = 1)', fontsize=11)
axes[1].set_xlabel('Pixel Value')
axes[1].set_ylabel('Probability')

axes[2].plot(hist_max)
axes[2].set_title('Normalized Histogram\n(Max = 1)', fontsize=11)
axes[2].set_xlabel('Pixel Value')
axes[2].set_ylabel('Normalized Value')

plt.tight_layout()
plt.show()

print(f"原始直方图和: {hist.sum():.0f}")
print(f"概率直方图和: {hist_prob.sum():.4f}")
print(f"最大值归一化最大值: {hist_max.max():.4f}")
```

### 4.3 直方图统计量

```python
"""
示例8：从直方图提取统计信息
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def histogram_statistics(image):
    """从直方图计算统计量"""
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # 计算直方图
    hist = cv2.calcHist([gray], [0], None, [256], [0, 256]).flatten()

    # 归一化为概率分布
    prob = hist / hist.sum()

    # 像素值
    values = np.arange(256)

    # 统计量
    mean = np.sum(values * prob)
    variance = np.sum(((values - mean) ** 2) * prob)
    std = np.sqrt(variance)

    # 众数（最常见的灰度值）
    mode = np.argmax(hist)

    # 中位数
    cumsum = np.cumsum(prob)
    median = np.searchsorted(cumsum, 0.5)

    # 偏度和峰度
    skewness = np.sum(((values - mean) / std) ** 3 * prob)
    kurtosis = np.sum(((values - mean) / std) ** 4 * prob) - 3

    stats = {
        'mean': mean,
        'std': std,
        'variance': variance,
        'mode': mode,
        'median': median,
        'skewness': skewness,
        'kurtosis': kurtosis,
        'min': gray.min(),
        'max': gray.max()
    }

    return hist, stats

# 创建不同特征的图像
images = {
    'Dark': np.random.normal(60, 20, (200, 300)).clip(0, 255).astype(np.uint8),
    'Bright': np.random.normal(200, 20, (200, 300)).clip(0, 255).astype(np.uint8),
    'High Contrast': np.random.normal(128, 60, (200, 300)).clip(0, 255).astype(np.uint8),
    'Low Contrast': np.random.normal(128, 15, (200, 300)).clip(0, 255).astype(np.uint8),
}

# 分析每个图像
print("图像直方图统计量分析：")
print("=" * 80)
print(f"{'Image':15s} {'Mean':>8s} {'Std':>8s} {'Mode':>6s} {'Median':>8s} {'Skewness':>10s}")
print("=" * 80)

for name, img in images.items():
    hist, stats = histogram_statistics(img)
    print(f"{name:15s} {stats['mean']:8.1f} {stats['std']:8.1f} "
          f"{stats['mode']:6d} {stats['median']:8d} {stats['skewness']:10.2f}")

print("=" * 80)
print("\n统计量含义：")
print("- Mean: 平均灰度值，反映整体亮度")
print("- Std: 标准差，反映对比度")
print("- Mode: 众数，最常见的灰度值")
print("- Median: 中位数，中间灰度值")
print("- Skewness: 偏度，正值表示右偏（暗图像），负值表示左偏（亮图像）")
```

```python
"""
示例9：2D直方图（H-S直方图）
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def compute_2d_histogram(image):
    """计算H-S 2D直方图"""
    hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)

    # 计算2D直方图
    # H: 0-180, S: 0-256
    hist_2d = cv2.calcHist([hsv], [0, 1], None, [180, 256], [0, 180, 0, 256])

    return hist_2d, hsv

# 创建彩色测试图像
def create_colorful_image():
    img = np.zeros((300, 400, 3), dtype=np.uint8)

    # 不同颜色的区域
    colors = [
        ([0, 0, 255], (50, 50)),      # 红色
        ([0, 255, 0], (150, 50)),     # 绿色
        ([255, 0, 0], (250, 50)),     # 蓝色
        ([0, 255, 255], (100, 150)),  # 黄色
        ([255, 0, 255], (200, 150)),  # 洋红
        ([255, 255, 0], (300, 150)),  # 青色
    ]

    for color, center in colors:
        cv2.circle(img, center, 40, color, -1)

    return img

color_img = create_colorful_image()
hist_2d, hsv = compute_2d_histogram(color_img)

# 可视化
fig, axes = plt.subplots(1, 3, figsize=(15, 4))

axes[0].imshow(cv2.cvtColor(color_img, cv2.COLOR_BGR2RGB))
axes[0].set_title('Color Image', fontsize=12)
axes[0].axis('off')

# 2D直方图
```

## 6. 直方图比较

### 6.1 直方图相似度度量

```python
"""
示例10：直方图比较与相似度计算
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def compare_histograms(img1, img2):
    """比较两幅图像的直方图"""
    # 转换为灰度图
    if len(img1.shape) == 3:
        gray1 = cv2.cvtColor(img1, cv2.COLOR_BGR2GRAY)
    else:
        gray1 = img1

    if len(img2.shape) == 3:
        gray2 = cv2.cvtColor(img2, cv2.COLOR_BGR2GRAY)
    else:
        gray2 = img2

    # 计算直方图
    hist1 = cv2.calcHist([gray1], [0], None, [256], [0, 256])
    hist2 = cv2.calcHist([gray2], [0], None, [256], [0, 256])

    # 归一化
    cv2.normalize(hist1, hist1, 0, 1, cv2.NORM_MINMAX)
    cv2.normalize(hist2, hist2, 0, 1, cv2.NORM_MINMAX)

    # 四种比较方法
    methods = {
        'Correlation': cv2.HISTCMP_CORREL,           # 相关性（越大越相似）
        'Chi-Square': cv2.HISTCMP_CHISQR,            # 卡方（越小越相似）
        'Intersection': cv2.HISTCMP_INTERSECT,       # 交集（越大越相似）
        'Bhattacharyya': cv2.HISTCMP_BHATTACHARYYA,  # 巴氏距离（越小越相似）
    }

    results = {}
    for name, method in methods.items():
        score = cv2.compareHist(hist1, hist2, method)
        results[name] = score

    return hist1, hist2, results

# 创建测试图像
img_original = np.random.normal(128, 40, (200, 300)).clip(0, 255).astype(np.uint8)
img_similar = img_original + np.random.normal(0, 10, img_original.shape)
img_similar = img_similar.clip(0, 255).astype(np.uint8)
img_different = np.random.normal(80, 20, (200, 300)).clip(0, 255).astype(np.uint8)

# 比较
hist_orig, hist_sim, results_sim = compare_histograms(img_original, img_similar)
hist_orig, hist_diff, results_diff = compare_histograms(img_original, img_different)

# 可视化
fig, axes = plt.subplots(2, 3, figsize=(15, 10))

# 图像
axes[0, 0].imshow(img_original, cmap='gray')
axes[0, 0].set_title('Original Image', fontsize=11)
axes[0, 0].axis('off')

axes[0, 1].imshow(img_similar, cmap='gray')
axes[0, 1].set_title('Similar Image', fontsize=11)
axes[0, 1].axis('off')

axes[0, 2].imshow(img_different, cmap='gray')
axes[0, 2].set_title('Different Image', fontsize=11)
axes[0, 2].axis('off')

# 直方图
axes[1, 0].plot(hist_orig, label='Original')
axes[1, 0].set_title('Original Histogram', fontsize=11)
axes[1, 0].set_xlim([0, 255])

axes[1, 1].plot(hist_orig, label='Original', alpha=0.7)
axes[1, 1].plot(hist_sim, label='Similar', alpha=0.7)
axes[1, 1].set_title('Original vs Similar', fontsize=11)
axes[1, 1].set_xlim([0, 255])
axes[1, 1].legend()

axes[1, 2].plot(hist_orig, label='Original', alpha=0.7)
axes[1, 2].plot(hist_diff, label='Different', alpha=0.7)
axes[1, 2].set_title('Original vs Different', fontsize=11)
axes[1, 2].set_xlim([0, 255])
axes[1, 2].legend()

plt.tight_layout()
plt.show()

# 打印比较结果

    print(f"{method:<20s} {results_sim[method]:>15.4f} {results_diff[method]:>15.4f}")
print("=" * 60)
print("\n说明：")
print("- Correlation: 相关性，范围[-1, 1]，越大越相似")
print("- Chi-Square: 卡方距离，越小越相似")
print("- Intersection: 交集，越大越相似")
print("- Bhattacharyya: 巴氏距离，范围[0, 1]，越小越相似")
```
