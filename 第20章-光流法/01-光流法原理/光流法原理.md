## 一、光流基础概念

### 1.1 什么是光流

**光流（Optical Flow）** 是指图像中像素点的运动模式：

- **定义**：描述图像中每个像素点在连续帧之间的位移向量
- **本质**：二维速度场，表示像素的视在运动
- **假设**：相邻帧之间亮度保持不变（亮度恒定假设）

### 1.2 光流的物理意义

想象一下：
- 摄像机固定，物体移动 → 产生光流
- 摄像机移动，物体固定 → 也产生光流
- 光流反映的是图像平面上的运动投影

### 1.3 亮度恒定假设

光流估计的核心假设：

$$I(x, y, t) = I(x + dx, y + dy, t + dt)$$

对右边进行泰勒展开并化简：

$$\frac{\partial I}{\partial x}u + \frac{\partial I}{\partial y}v + \frac{\partial I}{\partial t} = 0$$

或者写成：

$$I_x u + I_y v + I_t = 0$$

其中：
- $u = \frac{dx}{dt}$ 是x方向速度
- $v = \frac{dy}{dt}$ 是y方向速度
- $I_x, I_y$ 是空间梯度
- $I_t$ 是时间梯度

### 1.4 光流分类

| 类型 | 描述 | 代表算法 |
|------|------|----------|
| **稀疏光流** | 只计算特征点的光流 | Lucas-Kanade |
| **密集光流** | 计算每个像素的光流 | Farneback, Horn-Schunck |

```python
"""
示例2：计算图像梯度
梯度是光流计算的基础，本小姐来演示一下！
"""
import cv2
import numpy as np

def compute_gradients(img):
    """计算图像的空间梯度"""
    # Sobel算子计算梯度
    Ix = cv2.Sobel(img, cv2.CV_64F, 1, 0, ksize=3)
    Iy = cv2.Sobel(img, cv2.CV_64F, 0, 1, ksize=3)

    return Ix, Iy

def compute_temporal_gradient(img1, img2):
    """计算时间梯度"""
    It = img2.astype(np.float64) - img1.astype(np.float64)
    return It

# 创建测试图像
```

### 示例4：简单光流计算

```python
"""
示例4：简单光流计算
本小姐来实现一个简化版的光流计算！
"""
import cv2
import numpy as np

def simple_optical_flow(img1, img2, window_size=5):
    """
    简化版Lucas-Kanade光流
    仅用于教学演示
    """
    # 转换为浮点数
    I1 = img1.astype(np.float64)
    I2 = img2.astype(np.float64)

    # 计算梯度
    Ix = cv2.Sobel(I1, cv2.CV_64F, 1, 0, ksize=3)
    Iy = cv2.Sobel(I1, cv2.CV_64F, 0, 1, ksize=3)
    It = I2 - I1

    # 初始化光流场
    h, w = img1.shape
    u = np.zeros((h, w))
    v = np.zeros((h, w))

    # 半窗口大小
    half_win = window_size // 2

    # 对每个像素计算光流（简化版，实际中应使用特征点）
    step = 10  # 降采样以加速

    for y in range(half_win, h - half_win, step):
        for x in range(half_win, w - half_win, step):
            # 提取窗口
            Ix_win = Ix[y-half_win:y+half_win+1, x-half_win:x+half_win+1].flatten()
            Iy_win = Iy[y-half_win:y+half_win+1, x-half_win:x+half_win+1].flatten()
            It_win = It[y-half_win:y+half_win+1, x-half_win:x+half_win+1].flatten()

            # 构建矩阵 A^T A 和 A^T b
            A = np.vstack([Ix_win, Iy_win]).T
            b = -It_win

            # 最小二乘求解
            ATA = A.T @ A
            ATb = A.T @ b

            # 检查矩阵是否可逆
            if np.linalg.det(ATA) > 1e-6:
                flow = np.linalg.solve(ATA, ATb)
                u[y, x] = flow[0]
                v[y, x] = flow[1]

    return u, v

# 测试
cap = cv2.VideoCapture(0)

ret, frame1 = cap.read()
cv2.waitKey(100)
ret, frame2 = cap.read()

if ret:
    gray1 = cv2.cvtColor(frame1, cv2.COLOR_BGR2GRAY)
    gray2 = cv2.cvtColor(frame2, cv2.COLOR_BGR2GRAY)

    # 计算简单光流
    u, v = simple_optical_flow(gray1, gray2)

    # 可视化
    magnitude = np.sqrt(u**2 + v**2)
    magnitude = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)

    cv2.imshow('Frame 1', gray1)
    cv2.imshow('Flow Magnitude', magnitude)

    print("光流计算完成！")
    print(f"最大位移：{np.max(magnitude):.2f}")

    cv2.waitKey(0)

cap.release()
cv2.destroyAllWindows()
```

```python
"""
示例5：光流可视化方法
本小姐教你如何优雅地展示光流！
"""
import cv2
import numpy as np

def draw_flow_arrows(img, flow, step=16, scale=1, color=(0, 255, 0)):
    """用箭头绘制光流"""
    h, w = img.shape[:2]
    result = img.copy()

    for y in range(0, h, step):
        for x in range(0, w, step):
            fx, fy = flow[y, x]
            end_x = int(x + fx * scale)
            end_y = int(y + fy * scale)

            # 只绘制有意义的光流
            if abs(fx) > 0.5 or abs(fy) > 0.5:
                cv2.arrowedLine(result, (x, y), (end_x, end_y),
                               color, 1, tipLength=0.3)

    return result

def flow_to_color(flow):
    """将光流转换为彩色图像（HSV编码）"""
    h, w = flow.shape[:2]
    fx, fy = flow[:, :, 0], flow[:, :, 1]

    # 计算幅度和角度
    magnitude = np.sqrt(fx**2 + fy**2)
    angle = np.arctan2(fy, fx)

    # 创建HSV图像
    hsv = np.zeros((h, w, 3), dtype=np.uint8)
    hsv[:, :, 0] = (angle + np.pi) * 180 / (2 * np.pi)  # 色调表示方向
    hsv[:, :, 1] = 255  # 饱和度
    hsv[:, :, 2] = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX)  # 亮度表示幅度

    # 转换为BGR
    bgr = cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    return bgr

def create_flow_legend():
    """创建光流颜色图例"""
    size = 100
    legend = np.zeros((size * 2, size * 2, 3), dtype=np.uint8)

    for y in range(-size, size):
        for x in range(-size, size):
            angle = np.arctan2(y, x)
            magnitude = np.sqrt(x**2 + y**2)

            if magnitude < size:
                hue = int((angle + np.pi) * 180 / (2 * np.pi))
                sat = 255
                val = int(magnitude * 255 / size)

                hsv_pixel = np.array([[[hue, sat, val]]], dtype=np.uint8)
                bgr_pixel = cv2.cvtColor(hsv_pixel, cv2.COLOR_HSV2BGR)

                legend[y + size, x + size] = bgr_pixel[0, 0]

    return legend

# 演示
cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

# 创建图例
legend = create_flow_legend()
cv2.imshow('Flow Color Legend', legend)

print("光流可视化演示")
print("- 箭头显示：方向和大小")
print("- 颜色显示：色调=方向，亮度=幅度")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算密集光流
    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        pyr_scale=0.5, levels=3, winsize=15,
        iterations=3, poly_n=5, poly_sigma=1.2, flags=0
    )

    # 箭头可视化
    arrows = draw_flow_arrows(frame, flow, step=20, scale=2)

    # 颜色可视化
    colors = flow_to_color(flow)

    cv2.imshow('Flow Arrows', arrows)
    cv2.imshow('Flow Colors', colors)

    prev_gray = gray

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例6：光流算法对比

```python
"""
示例6：光流算法对比
对比不同光流算法的效果！
"""
import cv2
import numpy as np

def compare_flow_algorithms(prev_gray, gray):
    """对比不同的光流算法"""
    results = {}

    # 1. Farneback密集光流（默认参数）
    flow_farneback = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        pyr_scale=0.5, levels=3, winsize=15,
        iterations=3, poly_n=5, poly_sigma=1.2, flags=0
    )
    results['Farneback'] = flow_farneback

    # 2. Farneback（高精度参数）
    flow_farneback_hq = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        pyr_scale=0.5, levels=5, winsize=21,
        iterations=5, poly_n=7, poly_sigma=1.5, flags=0
    )
    results['Farneback HQ'] = flow_farneback_hq

    return results

def flow_magnitude_display(flow):
    """显示光流幅度"""
    magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)
    magnitude = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX)
    return magnitude.astype(np.uint8)

# 测试
cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 对比算法
    results = compare_flow_algorithms(prev_gray, gray)

    # 显示结果
    cv2.imshow('Original', frame)
```

```python
"""
示例8：光流金字塔
多尺度处理提高光流精度！
"""
import cv2
import numpy as np

def build_pyramid(img, levels):
    """构建图像金字塔"""
    pyramid = [img]
    for i in range(levels - 1):
        img = cv2.pyrDown(img)
        pyramid.append(img)
    return pyramid

def visualize_pyramid(pyramid):
    """可视化图像金字塔"""
    # 找到最大高度
    max_height = pyramid[0].shape[0]
    total_width = sum(p.shape[1] for p in pyramid) + len(pyramid) * 10

    # 创建画布
    canvas = np.zeros((max_height, total_width), dtype=np.uint8)

    x_offset = 0
    for i, level in enumerate(pyramid):
        h, w = level.shape
        canvas[:h, x_offset:x_offset + w] = level
        x_offset += w + 10

    return canvas

# 演示
```

### 示例9：光流一致性检验

```python
"""
示例9：光流一致性检验
验证光流估计的准确性！
"""
import cv2
import numpy as np

def forward_backward_check(prev_gray, gray, threshold=1.0):
    """
    前向-后向一致性检验
    计算前向光流和后向光流，检查是否一致
    """
    # 前向光流
    flow_forward = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        pyr_scale=0.5, levels=3, winsize=15,
        iterations=3, poly_n=5, poly_sigma=1.2, flags=0
    )

    # 后向光流
    flow_backward = cv2.calcOpticalFlowFarneback(
        gray, prev_gray, None,
        pyr_scale=0.5, levels=3, winsize=15,
        iterations=3, poly_n=5, poly_sigma=1.2, flags=0
    )

    # 检验一致性
    h, w = prev_gray.shape
    y, x = np.mgrid[0:h, 0:w].astype(np.float32)

    # 使用前向光流映射到新位置
    x_new = x + flow_forward[:, :, 0]
    y_new = y + flow_forward[:, :, 1]

    # 在新位置采样后向光流
    flow_backward_sampled = cv2.remap(
        flow_backward, x_new, y_new,
        cv2.INTER_LINEAR
    )

    # 计算不一致性
    diff = flow_forward + flow_backward_sampled
    inconsistency = np.sqrt(diff[:, :, 0]**2 + diff[:, :, 1]**2)

    # 创建有效性掩码
    valid_mask = inconsistency < threshold

    return flow_forward, valid_mask, inconsistency

# 测试
cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

print("光流一致性检验")

print("红色区域：光流不可靠")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 一致性检验
    flow, valid_mask, inconsistency = forward_backward_check(prev_gray, gray)

    # 可视化
    result = frame.copy()

    # 有效区域显示为绿色
    valid_overlay = np.zeros_like(frame)
    valid_overlay[:, :, 1] = (valid_mask * 100).astype(np.uint8)

    # 无效区域显示为红色
    invalid_overlay = np.zeros_like(frame)
    invalid_overlay[:, :, 2] = ((~valid_mask) * 100).astype(np.uint8)

    result = cv2.addWeighted(result, 1, valid_overlay, 0.3, 0)
    result = cv2.addWeighted(result, 1, invalid_overlay, 0.3, 0)

    # 统计
    valid_ratio = np.sum(valid_mask) / valid_mask.size * 100
    cv2.putText(result, f'Valid Flow: {valid_ratio:.1f}%', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)

    cv2.imshow('Consistency Check', result)

    prev_gray = gray

    if cv2.waitKey(30) & 0xFF == ord('q'):
```

### 示例10：光流应用综合演示

```python
"""
示例10：光流应用综合演示
展示光流的多种应用场景！
"""
import cv2
import numpy as np

class OpticalFlowDemo:
    def __init__(self):
        self.cap = cv2.VideoCapture(0)
        self.prev_gray = None
        self.mode = 0
        self.modes = ['Flow Arrows', 'Flow Colors', 'Motion Heatmap',
                      'Edge Flow', 'Zoom Detection']

    def compute_flow(self, prev_gray, gray):
        """计算光流"""
        return cv2.calcOpticalFlowFarneback(
            prev_gray, gray, None,
            pyr_scale=0.5, levels=3, winsize=15,
            iterations=3, poly_n=5, poly_sigma=1.2, flags=0
        )

    def mode_flow_arrows(self, frame, flow):
        """箭头显示模式"""
        result = frame.copy()
        step = 16
        h, w = frame.shape[:2]

        for y in range(0, h, step):
            for x in range(0, w, step):
                fx, fy = flow[y, x]
                if abs(fx) > 1 or abs(fy) > 1:
                    end_x = int(x + fx * 2)
                    end_y = int(y + fy * 2)
                    cv2.arrowedLine(result, (x, y), (end_x, end_y),
                                   (0, 255, 0), 1, tipLength=0.3)
        return result

    def mode_flow_colors(self, frame, flow):
        """颜色编码模式"""
        h, w = flow.shape[:2]
        magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)
        angle = np.arctan2(flow[:, :, 1], flow[:, :, 0])

        hsv = np.zeros((h, w, 3), dtype=np.uint8)
        hsv[:, :, 0] = (angle + np.pi) * 180 / (2 * np.pi)
        hsv[:, :, 1] = 255
        hsv[:, :, 2] = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX)

        return cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    def mode_motion_heatmap(self, frame, flow):
        """运动热图模式"""
        magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)
        mag_norm = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)
        heatmap = cv2.applyColorMap(mag_norm, cv2.COLORMAP_JET)
        return cv2.addWeighted(frame, 0.5, heatmap, 0.5, 0)

    def mode_edge_flow(self, frame, flow):
        """边缘光流模式"""
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        edges = cv2.Canny(gray, 50, 150)
        edges_3ch = cv2.cvtColor(edges, cv2.COLOR_GRAY2BGR)

        magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)
        mag_norm = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)
        heatmap = cv2.applyColorMap(mag_norm, cv2.COLORMAP_HOT)

        mask = edges > 0
        result = frame.copy()
        result[mask] = heatmap[mask]

        return result

    def mode_zoom_detection(self, frame, flow):
        """缩放检测模式"""
        h, w = flow.shape[:2]
        cx, cy = w // 2, h // 2

        # 计算从中心向外的径向分量
        y, x = np.mgrid[0:h, 0:w]
        dx = x - cx
        dy = y - cy
        r = np.sqrt(dx**2 + dy**2) + 1e-6

        # 径向光流分量
        radial = (flow[:, :, 0] * dx + flow[:, :, 1] * dy) / r

        # 判断缩放
        mean_radial = np.mean(radial[h//4:3*h//4, w//4:3*w//4])

        result = frame.copy()

        if mean_radial > 0.5:
            zoom_type = "ZOOM IN"
            color = (0, 255, 0)
        elif mean_radial < -0.5:
            zoom_type = "ZOOM OUT"
            color = (0, 0, 255)
        else:
            zoom_type = "STABLE"
            color = (255, 255, 0)

        cv2.putText(result, zoom_type, (w//2 - 60, h//2),
                    cv2.FONT_HERSHEY_SIMPLEX, 1, color, 2)
        cv2.circle(result, (cx, cy), 50, color, 2)

        return result

    def run(self):
        print("光流综合演示")
        print("按 'm' 切换模式")
        print("按 'q' 退出")

        ret, prev_frame = self.cap.read()
        self.prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

        while True:
            ret, frame = self.cap.read()
            if not ret:
                break

            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            flow = self.compute_flow(self.prev_gray, gray)

            # 根据模式处理
            if self.mode == 0:
                result = self.mode_flow_arrows(frame, flow)
            elif self.mode == 1:
                result = self.mode_flow_colors(frame, flow)
            elif self.mode == 2:
                result = self.mode_motion_heatmap(frame, flow)
            elif self.mode == 3:
                result = self.mode_edge_flow(frame, flow)
            elif self.mode == 4:
                result = self.mode_zoom_detection(frame, flow)

            # 显示模式名称
            cv2.putText(result, f'Mode: {self.modes[self.mode]}', (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)

            cv2.imshow('Optical Flow Demo', result)

            self.prev_gray = gray

            key = cv2.waitKey(30) & 0xFF
            if key == ord('q'):
                break
            elif key == ord('m'):
                self.mode = (self.mode + 1) % len(self.modes)
                print(f"切换到: {self.modes[self.mode]}")

        self.cap.release()
        cv2.destroyAllWindows()

# 运行演示
demo = OpticalFlowDemo()
demo.run()
```
