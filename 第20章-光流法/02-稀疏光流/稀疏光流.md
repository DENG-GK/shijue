# 稀疏光流

> 哼，密集光流计算量太大了！本小姐哈雷酱这就来教你稀疏光流，只追踪关键特征点，既高效又实用！这可是目标跟踪的核心技术呢！(￣▽￣)／

## 一、稀疏光流概述

### 1.1 什么是稀疏光流

**稀疏光流**只在特定的特征点上计算光流：

- **特征点**：角点、边缘点等容易追踪的点
- **优势**：计算量小，速度快
- **代表算法**：Lucas-Kanade (LK) 光流

### 1.2 Lucas-Kanade算法

LK算法假设：
1. **亮度恒定**：像素亮度不变
2. **时间连续**：运动幅度小
3. **空间一致**：邻域内运动相同

对于窗口内的所有像素，建立超定方程组，用最小二乘法求解：

$$\begin{bmatrix} u \\ v \end{bmatrix} = (A^T A)^{-1} A^T b$$

其中 $A$ 包含空间梯度，$b$ 包含时间梯度。

### 1.3 金字塔LK光流

为了处理大位移运动，使用图像金字塔：

1. 从最粗层开始估计
2. 将估计结果传递到下一层
3. 逐层细化直到最精细层

## 二、代码示例

### 示例1：基础LK光流

```python
"""
示例1：基础LK光流
本小姐来演示最经典的Lucas-Kanade光流！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

# Shi-Tomasi角点检测参数
feature_params = dict(
    maxCorners=100,      # 最多检测100个角点
    qualityLevel=0.3,    # 角点质量阈值
    minDistance=7,       # 角点最小距离
    blockSize=7          # 检测窗口大小
)

# LK光流参数
lk_params = dict(
    winSize=(15, 15),              # 搜索窗口大小
    maxLevel=2,                     # 金字塔层数
    criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 10, 0.03)
)

# 随机颜色
colors = np.random.randint(0, 255, (100, 3))

# 读取第一帧
ret, old_frame = cap.read()
old_gray = cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)

# 检测初始特征点
p0 = cv2.goodFeaturesToTrack(old_gray, mask=None, **feature_params)

# 创建用于绘制轨迹的掩码
mask = np.zeros_like(old_frame)

print("基础LK光流追踪")
print("按 'r' 重新检测特征点")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    if p0 is not None and len(p0) > 0:
        # 计算光流
        p1, st, err = cv2.calcOpticalFlowPyrLK(
            old_gray, frame_gray, p0, None, **lk_params
        )

        # 选择好的点
        if p1 is not None:
            good_new = p1[st == 1]
            good_old = p0[st == 1]

            # 绘制轨迹
            for i, (new, old) in enumerate(zip(good_new, good_old)):
                a, b = new.ravel().astype(int)
                c, d = old.ravel().astype(int)
                mask = cv2.line(mask, (a, b), (c, d), colors[i % 100].tolist(), 2)
                frame = cv2.circle(frame, (a, b), 5, colors[i % 100].tolist(), -1)

            # 更新
            p0 = good_new.reshape(-1, 1, 2)

    img = cv2.add(frame, mask)

    # 显示特征点数量
    num_points = len(p0) if p0 is not None else 0
    cv2.putText(img, f'Points: {num_points}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('LK Optical Flow', img)

    old_gray = frame_gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        # 重新检测特征点
        p0 = cv2.goodFeaturesToTrack(old_gray, mask=None, **feature_params)
        mask = np.zeros_like(old_frame)
        print(f"重新检测到 {len(p0)} 个特征点")

cap.release()
cv2.destroyAllWindows()
```

### 示例2：特征点自动补充

```python
"""
示例2：特征点自动补充
当特征点丢失时自动检测新的特征点！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

feature_params = dict(
    maxCorners=100,
    qualityLevel=0.3,
    minDistance=7,
    blockSize=7
)

lk_params = dict(
    winSize=(15, 15),
    maxLevel=2,
    criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 10, 0.03)
)

colors = np.random.randint(0, 255, (100, 3))

ret, old_frame = cap.read()
old_gray = cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)
p0 = cv2.goodFeaturesToTrack(old_gray, mask=None, **feature_params)

mask = np.zeros_like(old_frame)

MIN_POINTS = 20  # 最小特征点数量

print("特征点自动补充演示")
print(f"当特征点少于 {MIN_POINTS} 时自动补充")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    if p0 is not None and len(p0) > 0:
        p1, st, err = cv2.calcOpticalFlowPyrLK(
            old_gray, frame_gray, p0, None, **lk_params
        )

        if p1 is not None:
            good_new = p1[st == 1]
            good_old = p0[st == 1]

            for i, (new, old) in enumerate(zip(good_new, good_old)):
                a, b = new.ravel().astype(int)
                c, d = old.ravel().astype(int)
                mask = cv2.line(mask, (a, b), (c, d), colors[i % 100].tolist(), 2)
                frame = cv2.circle(frame, (a, b), 5, colors[i % 100].tolist(), -1)

            p0 = good_new.reshape(-1, 1, 2)

    # 检查是否需要补充特征点
    if p0 is None or len(p0) < MIN_POINTS:
        # 创建掩码，避免在现有特征点附近检测
        detect_mask = np.ones_like(frame_gray) * 255

        if p0 is not None and len(p0) > 0:
            for pt in p0:
                x, y = pt.ravel().astype(int)
                cv2.circle(detect_mask, (x, y), 10, 0, -1)

        # 检测新特征点
        new_points = cv2.goodFeaturesToTrack(
            frame_gray, mask=detect_mask, **feature_params
        )

        if new_points is not None:
            if p0 is not None and len(p0) > 0:
                p0 = np.vstack([p0, new_points])
            else:
                p0 = new_points
            print(f"补充特征点，当前共 {len(p0)} 个")

    img = cv2.add(frame, mask)

    num_points = len(p0) if p0 is not None else 0
    cv2.putText(img, f'Points: {num_points}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    status = "OK" if num_points >= MIN_POINTS else "Replenishing..."
    color = (0, 255, 0) if num_points >= MIN_POINTS else (0, 0, 255)
    cv2.putText(img, status, (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2)

    cv2.imshow('Auto Replenish', img)

    old_gray = frame_gray.copy()

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例3：双向光流验证

```python
"""
示例3：双向光流验证
用后向光流验证前向光流的准确性！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

feature_params = dict(
    maxCorners=100,
    qualityLevel=0.3,
    minDistance=7,
    blockSize=7
)

lk_params = dict(
    winSize=(15, 15),
    maxLevel=2,
    criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 10, 0.03)
)

ret, old_frame = cap.read()
old_gray = cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)
p0 = cv2.goodFeaturesToTrack(old_gray, mask=None, **feature_params)

FORWARD_BACKWARD_THRESHOLD = 1.0  # 验证阈值

print("双向光流验证")
print("绿色：验证通过的点")
print("红色：验证失败的点")
print("按 'r' 重置，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    if p0 is not None and len(p0) > 0:
        # 前向光流
        p1, st1, err1 = cv2.calcOpticalFlowPyrLK(
            old_gray, frame_gray, p0, None, **lk_params
        )

        # 后向光流
        p0_back, st2, err2 = cv2.calcOpticalFlowPyrLK(
            frame_gray, old_gray, p1, None, **lk_params
        )

        # 计算前后一致性误差
        if p0_back is not None:
            diff = np.abs(p0 - p0_back).reshape(-1, 2).max(axis=1)
            good = diff < FORWARD_BACKWARD_THRESHOLD

            # 筛选可靠的点
            good_mask = (st1.flatten() == 1) & (st2.flatten() == 1) & good

            result = frame.copy()

            for i, (pt_new, pt_old) in enumerate(zip(p1, p0)):
                if good_mask[i]:
                    # 验证通过 - 绿色
                    a, b = pt_new.ravel().astype(int)
                    c, d = pt_old.ravel().astype(int)
                    cv2.circle(result, (a, b), 5, (0, 255, 0), -1)
                    cv2.line(result, (c, d), (a, b), (0, 255, 0), 2)
                elif st1[i][0] == 1:
                    # 验证失败 - 红色
                    a, b = pt_new.ravel().astype(int)
                    cv2.circle(result, (a, b), 5, (0, 0, 255), -1)

            # 只保留验证通过的点
            if np.sum(good_mask) > 0:
                p0 = p1[good_mask].reshape(-1, 1, 2)
            else:
                p0 = None

            # 显示统计
            valid_count = np.sum(good_mask)
            total_count = len(good_mask)
            cv2.putText(result, f'Valid: {valid_count}/{total_count}', (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

            cv2.imshow('Forward-Backward Check', result)

    else:
        cv2.imshow('Forward-Backward Check', frame)

    old_gray = frame_gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        p0 = cv2.goodFeaturesToTrack(old_gray, mask=None, **feature_params)

cap.release()
cv2.destroyAllWindows()
```

### 示例4：ROI区域追踪

```python
"""
示例4：ROI区域追踪
只在指定区域内进行特征点追踪！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

feature_params = dict(
    maxCorners=50,
    qualityLevel=0.3,
    minDistance=7,
    blockSize=7
)

lk_params = dict(
    winSize=(15, 15),
    maxLevel=2,
    criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 10, 0.03)
)

ret, old_frame = cap.read()
h, w = old_frame.shape[:2]

# 定义ROI区域（中心区域）
roi_x = w // 4
roi_y = h // 4
roi_w = w // 2
roi_h = h // 2

old_gray = cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)

# 创建ROI掩码
roi_mask = np.zeros_like(old_gray)
roi_mask[roi_y:roi_y+roi_h, roi_x:roi_x+roi_w] = 255

# 在ROI内检测特征点
p0 = cv2.goodFeaturesToTrack(old_gray, mask=roi_mask, **feature_params)

colors = np.random.randint(0, 255, (50, 3))

print("ROI区域追踪")
print("只追踪矩形区域内的特征点")
print("按 'r' 重置，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    result = frame.copy()

    # 绘制ROI区域
    cv2.rectangle(result, (roi_x, roi_y), (roi_x + roi_w, roi_y + roi_h),
                  (255, 255, 0), 2)

    if p0 is not None and len(p0) > 0:
        p1, st, err = cv2.calcOpticalFlowPyrLK(
            old_gray, frame_gray, p0, None, **lk_params
        )

        if p1 is not None:
            good_new = p1[st == 1]
            good_old = p0[st == 1]

            # 过滤掉离开ROI的点
            in_roi = []
            for i, pt in enumerate(good_new):
                x, y = pt.ravel()
                if roi_x <= x <= roi_x + roi_w and roi_y <= y <= roi_y + roi_h:
                    in_roi.append(i)

            for idx in in_roi:
                new = good_new[idx]
                old = good_old[idx]
                a, b = new.ravel().astype(int)
                c, d = old.ravel().astype(int)
                cv2.line(result, (c, d), (a, b), colors[idx % 50].tolist(), 2)
                cv2.circle(result, (a, b), 5, colors[idx % 50].tolist(), -1)

            # 只保留ROI内的点
            if len(in_roi) > 0:
                p0 = good_new[in_roi].reshape(-1, 1, 2)
            else:
                p0 = None

    # 自动补充特征点
    if p0 is None or len(p0) < 10:
        p0 = cv2.goodFeaturesToTrack(frame_gray, mask=roi_mask, **feature_params)
        if p0 is not None:
            print(f"补充特征点：{len(p0)}")

    num_points = len(p0) if p0 is not None else 0
    cv2.putText(result, f'Points in ROI: {num_points}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('ROI Tracking', result)

    old_gray = frame_gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        p0 = cv2.goodFeaturesToTrack(frame_gray, mask=roi_mask, **feature_params)

cap.release()
cv2.destroyAllWindows()
```

### 示例5：鼠标点击追踪

```python
"""
示例5：鼠标点击追踪
点击任意位置开始追踪该点！
"""
import cv2
import numpy as np

# 全局变量
points_to_track = []
frame_gray = None

def mouse_callback(event, x, y, flags, param):
    global points_to_track
    if event == cv2.EVENT_LBUTTONDOWN:
        points_to_track.append([[x, y]])
        print(f"添加追踪点：({x}, {y})")

cap = cv2.VideoCapture(0)
cv2.namedWindow('Click to Track')
cv2.setMouseCallback('Click to Track', mouse_callback)

lk_params = dict(
    winSize=(21, 21),
    maxLevel=3,
    criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 30, 0.01)
)

ret, old_frame = cap.read()
old_gray = cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)

# 轨迹历史
trajectories = {}
next_id = 0

print("鼠标点击追踪")
print("点击画面添加追踪点")
print("按 'c' 清除所有点")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 添加新点
    for pt in points_to_track:
        trajectories[next_id] = [tuple(pt[0])]
        next_id += 1
    points_to_track = []

    result = frame.copy()

    if trajectories:
        # 准备追踪点
        p0 = np.array([list(traj[-1]) for traj in trajectories.values()],
                      dtype=np.float32).reshape(-1, 1, 2)

        # 计算光流
        p1, st, err = cv2.calcOpticalFlowPyrLK(
            old_gray, frame_gray, p0, None, **lk_params
        )

        # 更新轨迹
        if p1 is not None:
            ids_to_remove = []
            for i, (obj_id, traj) in enumerate(trajectories.items()):
                if st[i][0] == 1:
                    x, y = p1[i].ravel()
                    traj.append((int(x), int(y)))
                    # 限制轨迹长度
                    if len(traj) > 50:
                        traj.pop(0)
                else:
                    ids_to_remove.append(obj_id)

            # 移除丢失的点
            for obj_id in ids_to_remove:
                del trajectories[obj_id]
                print(f"点 {obj_id} 丢失")

        # 绘制轨迹和点
        for obj_id, traj in trajectories.items():
            # 绘制轨迹
            for i in range(1, len(traj)):
                thickness = int(np.sqrt(i / 2.0) + 1)
                cv2.line(result, traj[i-1], traj[i], (0, 255, 0), thickness)

            # 绘制当前点
            if traj:
                cv2.circle(result, traj[-1], 8, (0, 0, 255), -1)
                cv2.putText(result, str(obj_id), (traj[-1][0] + 10, traj[-1][1]),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

    cv2.putText(result, f'Tracking: {len(trajectories)} points', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(result, 'Click to add point', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 1)

    cv2.imshow('Click to Track', result)

    old_gray = frame_gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('c'):
        trajectories.clear()
        print("清除所有追踪点")

cap.release()
cv2.destroyAllWindows()
```

### 示例6：速度估计

```python
"""
示例6：速度估计
根据光流估计特征点的运动速度！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)
fps = cap.get(cv2.CAP_PROP_FPS)
if fps == 0:
    fps = 30

feature_params = dict(
    maxCorners=50,
    qualityLevel=0.3,
    minDistance=10,
    blockSize=7
)

lk_params = dict(
    winSize=(15, 15),
    maxLevel=2,
    criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 10, 0.03)
)

ret, old_frame = cap.read()
old_gray = cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)
p0 = cv2.goodFeaturesToTrack(old_gray, mask=None, **feature_params)

print("速度估计")
print("显示每个特征点的移动速度")
print("按 'r' 重置，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    result = frame.copy()

    if p0 is not None and len(p0) > 0:
        p1, st, err = cv2.calcOpticalFlowPyrLK(
            old_gray, frame_gray, p0, None, **lk_params
        )

        if p1 is not None:
            good_new = p1[st == 1]
            good_old = p0[st == 1]

            velocities = []

            for i, (new, old) in enumerate(zip(good_new, good_old)):
                x_new, y_new = new.ravel()
                x_old, y_old = old.ravel()

                # 计算位移
                dx = x_new - x_old
                dy = y_new - y_old

                # 计算速度（像素/帧）
                speed = np.sqrt(dx**2 + dy**2)
                # 转换为像素/秒
                speed_per_sec = speed * fps

                velocities.append(speed_per_sec)

                # 根据速度选择颜色
                if speed_per_sec < 50:
                    color = (0, 255, 0)  # 慢 - 绿色
                elif speed_per_sec < 150:
                    color = (0, 255, 255)  # 中 - 黄色
                else:
                    color = (0, 0, 255)  # 快 - 红色

                # 绘制点和速度箭头
                cv2.circle(result, (int(x_new), int(y_new)), 5, color, -1)

                # 绘制速度方向箭头
                arrow_scale = 3
                end_x = int(x_new + dx * arrow_scale)
                end_y = int(y_new + dy * arrow_scale)
                cv2.arrowedLine(result, (int(x_new), int(y_new)),
                               (end_x, end_y), color, 2, tipLength=0.3)

            p0 = good_new.reshape(-1, 1, 2)

            # 显示统计信息
            if velocities:
                avg_speed = np.mean(velocities)
                max_speed = np.max(velocities)
                cv2.putText(result, f'Avg Speed: {avg_speed:.1f} px/s', (10, 30),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
                cv2.putText(result, f'Max Speed: {max_speed:.1f} px/s', (10, 60),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    # 自动补充特征点
    if p0 is None or len(p0) < 20:
        p0 = cv2.goodFeaturesToTrack(frame_gray, mask=None, **feature_params)

    cv2.imshow('Velocity Estimation', result)

    old_gray = frame_gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        p0 = cv2.goodFeaturesToTrack(frame_gray, mask=None, **feature_params)

cap.release()
cv2.destroyAllWindows()
```

"""
示例7：相机运动估计
通过光流估计相机的平移和旋转！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

feature_params = dict(
    maxCorners=200,
    qualityLevel=0.01,
    minDistance=10,
    blockSize=7
)

lk_params = dict(
    winSize=(21, 21),
    maxLevel=3,
    criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 30, 0.01)
)

ret, old_frame = cap.read()
h, w = old_frame.shape[:2]
old_gray = cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)
p0 = cv2.goodFeaturesToTrack(old_gray, mask=None, **feature_params)

print("相机运动估计")
print("显示相机的平移和旋转估计")
print("按 'r' 重置，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    result = frame.copy()

    if p0 is not None and len(p0) > 10:
        p1, st, err = cv2.calcOpticalFlowPyrLK(
            old_gray, frame_gray, p0, None, **lk_params
        )

        if p1 is not None:
            good_new = p1[st == 1]
            good_old = p0[st == 1]

            if len(good_new) >= 10:
                # 计算平均运动
                flow_vectors = good_new - good_old

                # 平移估计（平均位移）
                mean_dx = np.mean(flow_vectors[:, 0])
                mean_dy = np.mean(flow_vectors[:, 1])

                # 旋转估计（基于径向分量）
                cx, cy = w / 2, h / 2
                rotation = 0
                count = 0

                for old, new in zip(good_old, good_new):
                    # 相对于中心的位置
                    rx_old = old[0] - cx
                    ry_old = old[1] - cy
                    rx_new = new[0] - cx
                    ry_new = new[1] - cy

                    # 计算角度变化
                    angle_old = np.arctan2(ry_old, rx_old)
                    angle_new = np.arctan2(ry_new, rx_new)
                    diff = angle_new - angle_old

                    # 处理跨越±π的情况
                    if diff > np.pi:
                        diff -= 2 * np.pi
                    elif diff < -np.pi:
                        diff += 2 * np.pi

                    rotation += diff
                    count += 1

                if count > 0:
                    rotation = rotation / count * 180 / np.pi  # 转换为度

                # 缩放估计
                dist_old = np.sqrt((good_old[:, 0] - cx)**2 + (good_old[:, 1] - cy)**2)
                dist_new = np.sqrt((good_new[:, 0] - cx)**2 + (good_new[:, 1] - cy)**2)
                scale = np.mean(dist_new / (dist_old + 1e-6))

                # 显示运动信息
                cv2.putText(result, f'Translation X: {mean_dx:.2f}', (10, 30),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
                cv2.putText(result, f'Translation Y: {mean_dy:.2f}', (10, 60),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
                cv2.putText(result, f'Rotation: {rotation:.2f} deg', (10, 90),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
                cv2.putText(result, f'Scale: {scale:.3f}', (10, 120),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

                # 运动类型判断
                if abs(mean_dx) > 5 or abs(mean_dy) > 5:
                    motion = "MOVING"
                elif abs(rotation) > 1:
                    motion = "ROTATING"
                elif abs(scale - 1) > 0.02:
                    motion = "ZOOMING"
                else:
                    motion = "STABLE"

                cv2.putText(result, motion, (w - 150, 30),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 0, 255), 2)

                # 绘制特征点
                for pt in good_new:
                    cv2.circle(result, tuple(pt.astype(int)), 3, (0, 255, 0), -1)

            p0 = good_new.reshape(-1, 1, 2)

    # 自动补充
    if p0 is None or len(p0) < 50:
        p0 = cv2.goodFeaturesToTrack(frame_gray, mask=None, **feature_params)

    cv2.imshow('Camera Motion', result)

    old_gray = frame_gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        p0 = cv2.goodFeaturesToTrack(frame_gray, mask=None, **feature_params)

cap.release()
cv2.destroyAllWindows()

"""
示例8：目标跟踪器
使用光流实现简单的目标跟踪！
"""
import cv2
import numpy as np

class OpticalFlowTracker:
    def __init__(self):
        self.lk_params = dict(
            winSize=(21, 21),
            maxLevel=3,
            criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 30, 0.01)
        )
        self.feature_params = dict(
            maxCorners=100,
            qualityLevel=0.1,
            minDistance=5,
            blockSize=7
        )
        self.prev_gray = None
        self.points = None
        self.bbox = None

    def init(self, frame, bbox):
        """初始化跟踪器"""
        self.bbox = bbox
        x, y, w, h = bbox

        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        self.prev_gray = gray

        # 创建ROI掩码
        mask = np.zeros_like(gray)
        mask[y:y+h, x:x+w] = 255

        # 在ROI内检测特征点
        self.points = cv2.goodFeaturesToTrack(gray, mask=mask, **self.feature_params)

        return self.points is not None and len(self.points) > 0

    def update(self, frame):
        """更新跟踪"""
        if self.points is None or len(self.points) == 0:
            return False, self.bbox

        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        # 计算光流
        new_points, st, err = cv2.calcOpticalFlowPyrLK(
            self.prev_gray, gray, self.points, None, **self.lk_params
        )

        if new_points is None:
            return False, self.bbox

        # 选择有效点
        good_new = new_points[st == 1]
        good_old = self.points[st == 1]

        if len(good_new) < 4:
            return False, self.bbox

        # 计算变换
        dx = np.median(good_new[:, 0] - good_old[:, 0])
        dy = np.median(good_new[:, 1] - good_old[:, 1])

        # 更新边界框
        x, y, w, h = self.bbox
        x = int(x + dx)
        y = int(y + dy)
        self.bbox = (x, y, w, h)

        # 更新
        self.points = good_new.reshape(-1, 1, 2)
        self.prev_gray = gray

        return True, self.bbox

# 使用
cap = cv2.VideoCapture(0)

tracker = None
selecting = False
start_point = None
bbox = None

def mouse_callback(event, x, y, flags, param):
    global selecting, start_point, bbox

    if event == cv2.EVENT_LBUTTONDOWN:
        selecting = True
        start_point = (x, y)

    elif event == cv2.EVENT_MOUSEMOVE and selecting:
        pass

    elif event == cv2.EVENT_LBUTTONUP:
        selecting = False
        x1, y1 = start_point
        x2, y2 = x, y
        bbox = (min(x1, x2), min(y1, y2), abs(x2 - x1), abs(y2 - y1))

cv2.namedWindow('Tracker')
cv2.setMouseCallback('Tracker', mouse_callback)

print("光流目标跟踪器")
print("用鼠标框选目标区域")
print("按 'r' 重新选择")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    result = frame.copy()

    # 显示选择框
    if selecting and start_point:
        cv2.rectangle(result, start_point,
                      (cv2.getWindowImageRect('Tracker')[2], cv2.getWindowImageRect('Tracker')[3]),
                      (0, 255, 0), 2)

    # 初始化跟踪器
    if bbox is not None and tracker is None:
        tracker = OpticalFlowTracker()
        if tracker.init(frame, bbox):
            print("跟踪器初始化成功")
        else:
            print("初始化失败，请重新选择")
            tracker = None
        bbox = None

    # 更新跟踪
    if tracker is not None:
        success, box = tracker.update(frame)

        if success:
            x, y, w, h = box
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

            # 绘制特征点
            if tracker.points is not None:
                for pt in tracker.points:
                    cv2.circle(result, tuple(pt.ravel().astype(int)), 3, (0, 0, 255), -1)

            cv2.putText(result, 'Tracking', (x, y - 10),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
        else:
            cv2.putText(result, 'Lost', (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

    cv2.imshow('Tracker', result)

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        tracker = None
        bbox = None

cap.release()
cv2.destroyAllWindows()

### 示例10：综合跟踪系统

```python
"""
示例10：综合跟踪系统
结合多种技术的完整跟踪系统！
"""
import cv2
import numpy as np
from collections import deque

class AdvancedTracker:
    def __init__(self):
        self.feature_params = dict(
            maxCorners=200,
            qualityLevel=0.01,
            minDistance=10,
            blockSize=7
        )
        self.lk_params = dict(
            winSize=(21, 21),
            maxLevel=3,
            criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 30, 0.01)
        )

        self.prev_gray = None
        self.points = None
        self.point_ids = []
        self.next_id = 0
        self.trajectories = {}
        self.max_trajectory_len = 30

    def update(self, frame):
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        # 初始化
        if self.prev_gray is None:
            self.prev_gray = gray
            self.points = cv2.goodFeaturesToTrack(gray, mask=None, **self.feature_params)
            if self.points is not None:
                self.point_ids = list(range(len(self.points)))
                self.next_id = len(self.points)
                for i, pt in enumerate(self.points):
                    self.trajectories[i] = deque([tuple(pt.ravel())], maxlen=self.max_trajectory_len)
            return frame

        result = frame.copy()

        if self.points is not None and len(self.points) > 0:
            # 前向光流
            p1, st1, _ = cv2.calcOpticalFlowPyrLK(
                self.prev_gray, gray, self.points, None, **self.lk_params
            )

            # 后向光流验证
            p0_back, st2, _ = cv2.calcOpticalFlowPyrLK(
                gray, self.prev_gray, p1, None, **self.lk_params
            )

            # 前后一致性检验
            diff = np.abs(self.points - p0_back).reshape(-1, 2).max(axis=1)
            good = (st1.flatten() == 1) & (st2.flatten() == 1) & (diff < 1.0)

            # 更新轨迹
            new_points = []
            new_ids = []

            for i, (pt_new, pt_old, is_good, pt_id) in enumerate(
                    zip(p1, self.points, good, self.point_ids)):

                if is_good:
                    pt_tuple = tuple(pt_new.ravel())
                    new_points.append(pt_new)
                    new_ids.append(pt_id)

                    if pt_id in self.trajectories:
                        self.trajectories[pt_id].append(pt_tuple)
                else:
                    # 移除丢失点的轨迹
                    if pt_id in self.trajectories:
                        del self.trajectories[pt_id]

            self.points = np.array(new_points).reshape(-1, 1, 2) if new_points else None
            self.point_ids = new_ids

            # 绘制轨迹
            for pt_id, traj in self.trajectories.items():
                color = self._id_to_color(pt_id)

                # 绘制轨迹线
                for i in range(1, len(traj)):
                    thickness = int(np.sqrt(i) + 1)
                    pt1 = tuple(map(int, traj[i-1]))
                    pt2 = tuple(map(int, traj[i]))
                    cv2.line(result, pt1, pt2, color, thickness)

                # 绘制当前点
                if traj:
                    cv2.circle(result, tuple(map(int, traj[-1])), 5, color, -1)

        # 补充特征点
        if self.points is None or len(self.points) < 50:
            # 创建掩码避免重复检测
            mask = np.ones_like(gray) * 255
            if self.points is not None:
                for pt in self.points:
                    x, y = pt.ravel().astype(int)
                    cv2.circle(mask, (x, y), 15, 0, -1)

            new_pts = cv2.goodFeaturesToTrack(gray, mask=mask, **self.feature_params)

            if new_pts is not None:
                for pt in new_pts:
                    if self.points is None:
                        self.points = pt.reshape(1, 1, 2)
                    else:
                        self.points = np.vstack([self.points, pt.reshape(1, 1, 2)])

                    self.point_ids.append(self.next_id)
                    self.trajectories[self.next_id] = deque([tuple(pt.ravel())],
                                                           maxlen=self.max_trajectory_len)
                    self.next_id += 1

        # 显示统计
        num_points = len(self.points) if self.points is not None else 0
        cv2.putText(result, f'Tracking: {num_points} points', (10, 30),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.putText(result, f'Trajectories: {len(self.trajectories)}', (10, 60),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

        self.prev_gray = gray

        return result

    def _id_to_color(self, pt_id):
        """将ID转换为唯一颜色"""
        np.random.seed(pt_id)
        return tuple(map(int, np.random.randint(0, 255, 3)))

# 运行
cap = cv2.VideoCapture(0)
tracker = AdvancedTracker()

print("综合跟踪系统")
print("- 双向光流验证")
print("- 自动特征点补充")
print("- 轨迹可视化")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    result = tracker.update(frame)

    cv2.imshow('Advanced Tracker', result)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```
