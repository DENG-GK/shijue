# 密集光流

> 哼，稀疏光流只能追踪特征点，太局限了！本小姐哈雷酱这就来教你密集光流，计算每个像素的运动向量，获得完整的运动场！(￣▽￣)／

## 一、密集光流概述

### 1.1 什么是密集光流

**密集光流（Dense Optical Flow）** 计算图像中每个像素的光流向量：

- **输出**：与图像同尺寸的二维向量场
- **优势**：获得完整的运动信息
- **劣势**：计算量大

### 1.2 常用算法

| 算法 | 特点 | 速度 |
|------|------|------|
| **Farneback** | 多项式展开，平滑 | 中等 |
| **DIS** | 快速密集光流 | 快 |
| **DeepFlow** | 深度学习方法 | 慢 |
| **TVL1** | 变分方法 | 慢 |

### 1.3 Farneback算法原理

Gunnar Farneback方法：
1. 使用多项式展开近似每个像素邻域
2. 通过匹配多项式系数估计位移
3. 多尺度金字塔处理

## 二、代码示例

### 示例1：基础Farneback光流

```python
"""
示例1：基础Farneback光流
本小姐来演示最常用的密集光流算法！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

print("Farneback密集光流")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算密集光流
    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray,
        None,                    # 初始光流估计
        pyr_scale=0.5,          # 金字塔缩放比例
        levels=3,                # 金字塔层数
        winsize=15,              # 窗口大小
        iterations=3,            # 每层迭代次数
        poly_n=5,                # 多项式展开邻域大小
        poly_sigma=1.2,          # 多项式展开高斯标准差
        flags=0
    )

    # 计算幅度和角度
    magnitude, angle = cv2.cartToPolar(flow[:, :, 0], flow[:, :, 1])

    # HSV可视化
    hsv = np.zeros_like(frame)
    hsv[:, :, 0] = angle * 180 / np.pi / 2  # 色调 = 方向
    hsv[:, :, 1] = 255                       # 饱和度
    hsv[:, :, 2] = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX)  # 亮度 = 幅度

    flow_rgb = cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    cv2.imshow('Original', frame)
    cv2.imshow('Dense Flow', flow_rgb)

    prev_gray = gray

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例2：光流箭头可视化

```python
"""
示例2：光流箭头可视化
用箭头更直观地展示光流方向！
"""
import cv2
import numpy as np

def draw_flow(img, flow, step=16):
    """绘制光流箭头"""
    h, w = img.shape[:2]
    y, x = np.mgrid[step//2:h:step, step//2:w:step].reshape(2, -1).astype(int)
    fx, fy = flow[y, x].T

    # 创建线段端点
    lines = np.vstack([x, y, x + fx, y + fy]).T.reshape(-1, 2, 2)
    lines = np.int32(lines + 0.5)

    # 绘制
    vis = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

    for (x1, y1), (x2, y2) in lines:
        # 根据速度选择颜色
        speed = np.sqrt((x2 - x1)**2 + (y2 - y1)**2)
        if speed < 2:
            continue
        elif speed < 5:
            color = (0, 255, 0)   # 绿色 - 慢
        elif speed < 10:
            color = (0, 255, 255) # 黄色 - 中
        else:
            color = (0, 0, 255)   # 红色 - 快

        cv2.arrowedLine(vis, (x1, y1), (x2, y2), color, 1, tipLength=0.3)

    return vis

cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

print("光流箭头可视化")
print("绿色=慢，黄色=中，红色=快")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        0.5, 3, 15, 3, 5, 1.2, 0
    )

    # 绘制箭头
    vis = draw_flow(gray, flow, step=16)

    cv2.imshow('Flow Arrows', vis)

    prev_gray = gray

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例3：DIS快速密集光流

```python
"""
示例3：DIS快速密集光流
比Farneback更快的密集光流算法！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

# 创建DIS光流计算器
# cv2.DISOPTICAL_FLOW_PRESET_ULTRAFAST  - 最快
# cv2.DISOPTICAL_FLOW_PRESET_FAST       - 快速
# cv2.DISOPTICAL_FLOW_PRESET_MEDIUM     - 中等
dis = cv2.DISOpticalFlow_create(cv2.DISOPTICAL_FLOW_PRESET_MEDIUM)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

import time

print("DIS快速密集光流")
print("按 'p' 切换预设")
print("按 'q' 退出")

presets = [
    ('ULTRAFAST', cv2.DISOPTICAL_FLOW_PRESET_ULTRAFAST),
    ('FAST', cv2.DISOPTICAL_FLOW_PRESET_FAST),
    ('MEDIUM', cv2.DISOPTICAL_FLOW_PRESET_MEDIUM),
]
current_preset = 2

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算光流
    start_time = time.time()
    flow = dis.calc(prev_gray, gray, None)
    elapsed = (time.time() - start_time) * 1000

    # 可视化
    magnitude, angle = cv2.cartToPolar(flow[:, :, 0], flow[:, :, 1])

    hsv = np.zeros((frame.shape[0], frame.shape[1], 3), dtype=np.uint8)
    hsv[:, :, 0] = angle * 180 / np.pi / 2
    hsv[:, :, 1] = 255
    hsv[:, :, 2] = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX)

    flow_rgb = cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    # 显示信息
    cv2.putText(flow_rgb, f'Preset: {presets[current_preset][0]}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)
    cv2.putText(flow_rgb, f'Time: {elapsed:.1f}ms', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)

    cv2.imshow('Original', frame)
    cv2.imshow('DIS Flow', flow_rgb)

    prev_gray = gray

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('p'):
        current_preset = (current_preset + 1) % len(presets)
        dis = cv2.DISOpticalFlow_create(presets[current_preset][1])
        print(f"切换到: {presets[current_preset][0]}")

cap.release()
cv2.destroyAllWindows()
```

### 示例5：运动区域分割

```python
"""
示例5：运动区域分割
基于光流分割运动和静止区域！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

MOTION_THRESHOLD = 2.0  # 运动阈值

print("运动区域分割")
print("按 '+'/'-' 调节阈值")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算光流
    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        0.5, 3, 15, 3, 5, 1.2, 0
    )

    # 计算光流幅度
    magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)

    # 创建运动掩码
    motion_mask = (magnitude > MOTION_THRESHOLD).astype(np.uint8) * 255

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    motion_mask = cv2.morphologyEx(motion_mask, cv2.MORPH_CLOSE, kernel)
    motion_mask = cv2.morphologyEx(motion_mask, cv2.MORPH_OPEN, kernel)

    # 查找运动轮廓
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    result = frame.copy()

    # 绘制运动区域
    for contour in contours:
        if cv2.contourArea(contour) > 500:
            x, y, w, h = cv2.boundingRect(contour)
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    # 提取运动前景
    motion_fg = cv2.bitwise_and(frame, frame, mask=motion_mask)

    # 统计
    motion_ratio = np.sum(motion_mask > 0) / motion_mask.size * 100

    cv2.putText(result, f'Threshold: {MOTION_THRESHOLD:.1f}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(result, f'Motion Area: {motion_ratio:.1f}%', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Original', result)
    cv2.imshow('Motion Mask', motion_mask)
    cv2.imshow('Motion Foreground', motion_fg)

    prev_gray = gray

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key in [ord('+'), ord('=')]:
        MOTION_THRESHOLD += 0.5
        print(f"阈值: {MOTION_THRESHOLD}")
    elif key == ord('-'):
        MOTION_THRESHOLD = max(0.5, MOTION_THRESHOLD - 0.5)
        print(f"阈值: {MOTION_THRESHOLD}")

cap.release()
cv2.destroyAllWindows()
```

### 示例6：视频稳定

```python
"""
示例6：视频稳定
使用密集光流实现简单的视频稳定！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

h, w = prev_gray.shape

# 累积变换
accumulated_transform = np.eye(3, dtype=np.float32)

# 平滑系数
SMOOTHING = 0.95

print("视频稳定")
print("按 's' 开启/关闭稳定")
print("按 'r' 重置")
print("按 'q' 退出")

stabilization_on = True

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算光流
    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        0.5, 3, 15, 3, 5, 1.2, 0
    )

    # 估计全局运动（平均光流）
    mean_flow_x = np.mean(flow[:, :, 0])
    mean_flow_y = np.mean(flow[:, :, 1])

    # 创建平移变换矩阵
    transform = np.array([
        [1, 0, -mean_flow_x],
        [0, 1, -mean_flow_y],
        [0, 0, 1]
    ], dtype=np.float32)

    # 平滑累积变换
    accumulated_transform = SMOOTHING * accumulated_transform + (1 - SMOOTHING) * transform

    if stabilization_on:
        # 应用变换
        stabilized = cv2.warpAffine(
            frame,
            accumulated_transform[:2, :],
            (w, h)
        )
    else:
        stabilized = frame.copy()

    # 显示对比
    comparison = np.hstack([frame, stabilized])

    # 添加标签
    cv2.putText(comparison, 'Original', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(comparison, 'Stabilized' if stabilization_on else 'OFF', (w + 10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    # 显示运动信息
    cv2.putText(comparison, f'dX: {mean_flow_x:.2f}', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)
    cv2.putText(comparison, f'dY: {mean_flow_y:.2f}', (10, 80),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

    cv2.imshow('Video Stabilization', comparison)

    prev_gray = gray

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('s'):
        stabilization_on = not stabilization_on
        print(f"稳定: {'开启' if stabilization_on else '关闭'}")
    elif key == ord('r'):
        accumulated_transform = np.eye(3, dtype=np.float32)
        print("变换已重置")

cap.release()
cv2.destroyAllWindows()
```

### 示例7：光流分层显示

```python
"""
示例7：光流分层显示
按速度分层显示光流！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

# 速度层级阈值
SLOW_THRESHOLD = 2
MEDIUM_THRESHOLD = 5
FAST_THRESHOLD = 10

print("光流分层显示")
print("蓝=慢，绿=中，红=快，白=很快")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        0.5, 3, 15, 3, 5, 1.2, 0
    )

    magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)

    # 创建分层显示
    layers = np.zeros_like(frame)

    # 慢速 - 蓝色
    slow_mask = (magnitude >= SLOW_THRESHOLD) & (magnitude < MEDIUM_THRESHOLD)
    layers[slow_mask] = (255, 0, 0)

    # 中速 - 绿色
    medium_mask = (magnitude >= MEDIUM_THRESHOLD) & (magnitude < FAST_THRESHOLD)
    layers[medium_mask] = (0, 255, 0)

    # 快速 - 红色
    fast_mask = (magnitude >= FAST_THRESHOLD)
    layers[fast_mask] = (0, 0, 255)

    # 很快 - 白色
    veryfast_mask = magnitude >= FAST_THRESHOLD * 2
    layers[veryfast_mask] = (255, 255, 255)

    # 叠加到原图
    alpha = 0.5
    overlay = cv2.addWeighted(frame, 1, layers, alpha, 0)

    # 统计各层像素数
    total_pixels = magnitude.size
    slow_count = np.sum(slow_mask) / total_pixels * 100
    medium_count = np.sum(medium_mask) / total_pixels * 100
    fast_count = np.sum(fast_mask) / total_pixels * 100

    # 显示统计
    cv2.putText(overlay, f'Slow (blue): {slow_count:.1f}%', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 0, 0), 2)
    cv2.putText(overlay, f'Medium (green): {medium_count:.1f}%', (10, 55),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
    cv2.putText(overlay, f'Fast (red): {fast_count:.1f}%', (10, 80),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 255), 2)

    cv2.imshow('Layered Flow', overlay)
    cv2.imshow('Layers Only', layers)

    prev_gray = gray

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例8：光流运动补偿

```python
"""
示例8：光流运动补偿
用光流预测下一帧！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

h, w = prev_gray.shape

print("光流运动补偿")
print("左=实际帧，右=预测帧")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算光流
    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        0.5, 3, 15, 3, 5, 1.2, 0
    )

    # 使用光流对前一帧进行变形，预测当前帧
    # 创建映射坐标
    y, x = np.mgrid[0:h, 0:w].astype(np.float32)
    map_x = x + flow[:, :, 0]
    map_y = y + flow[:, :, 1]

    # 变形
    predicted = cv2.remap(prev_gray, map_x, map_y, cv2.INTER_LINEAR)

    # 计算预测误差
    error = cv2.absdiff(gray, predicted)
    mean_error = np.mean(error)

    # 显示
    actual_color = cv2.cvtColor(gray, cv2.COLOR_GRAY2BGR)
    predicted_color = cv2.cvtColor(predicted, cv2.COLOR_GRAY2BGR)
    error_color = cv2.applyColorMap(error, cv2.COLORMAP_JET)

    # 添加标签
    cv2.putText(actual_color, 'Actual', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(predicted_color, 'Predicted', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(error_color, f'Error: {mean_error:.1f}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)

    # 拼接显示
    top = np.hstack([actual_color, predicted_color])
    bottom = np.hstack([error_color, frame])
    comparison = np.vstack([top, bottom])

    # 缩放以适应屏幕
    comparison = cv2.resize(comparison, (w, h))

    cv2.imshow('Motion Compensation', comparison)

    prev_gray = gray

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

```python
"""
示例9：光流场纹理
创建艺术化的光流纹理效果！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

h, w = prev_gray.shape

# 创建噪声纹理
np.random.seed(42)
noise = np.random.rand(h, w).astype(np.float32)

print("光流场纹理")
print("按 'r' 重置纹理")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算光流
    flow = cv2.calcOpticalFlowFarneback(
        prev_gray, gray, None,
        0.5, 3, 15, 3, 5, 1.2, 0
    )

    # 使用光流推动纹理
    y, x = np.mgrid[0:h, 0:w].astype(np.float32)
    map_x = x + flow[:, :, 0] * 0.5
    map_y = y + flow[:, :, 1] * 0.5

    # 变形纹理
    noise = cv2.remap(noise, map_x, map_y, cv2.INTER_LINEAR)

    # 添加一点新噪声（防止完全消失）
    noise = noise * 0.99 + np.random.rand(h, w).astype(np.float32) * 0.01

    # 归一化显示
    texture = (noise * 255).astype(np.uint8)

    # 应用颜色映射
    colored = cv2.applyColorMap(texture, cv2.COLORMAP_PLASMA)

    # 可视化光流幅度
    magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)
    mag_norm = cv2.normalize(magnitude, None, 0, 1, cv2.NORM_MINMAX)

    # 根据运动强度混合
    alpha = mag_norm[:, :, np.newaxis]
    blended = (frame * (1 - alpha) + colored * alpha).astype(np.uint8)

    cv2.imshow('Flow Texture', colored)
    cv2.imshow('Blended', blended)

    prev_gray = gray

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        noise = np.random.rand(h, w).astype(np.float32)
        print("纹理已重置")

cap.release()
cv2.destroyAllWindows()
```

### 示例10：光流综合分析系统

```python
"""
示例10：光流综合分析系统
完整的光流分析和可视化系统！
"""
import cv2
import numpy as np
import time

class DenseFlowAnalyzer:
    def __init__(self):
        self.prev_gray = None
        self.flow_history = []
        self.max_history = 10

        # Farneback参数
        self.farneback_params = dict(
            pyr_scale=0.5,
            levels=3,
            winsize=15,
            iterations=3,
            poly_n=5,
            poly_sigma=1.2,
            flags=0
        )

    def compute_flow(self, gray):
        if self.prev_gray is None:
            self.prev_gray = gray
            return None

        flow = cv2.calcOpticalFlowFarneback(
            self.prev_gray, gray, None, **self.farneback_params
        )

        self.prev_gray = gray
        return flow

    def analyze_flow(self, flow):
        """分析光流"""
        if flow is None:
            return {}

        fx, fy = flow[:, :, 0], flow[:, :, 1]
        magnitude = np.sqrt(fx**2 + fy**2)
        angle = np.arctan2(fy, fx)

        h, w = flow.shape[:2]
        cx, cy = w / 2, h / 2

        # 基本统计
        stats = {
            'mean_magnitude': np.mean(magnitude),
            'max_magnitude': np.max(magnitude),
            'mean_angle': np.mean(angle) * 180 / np.pi,
            'motion_area': np.sum(magnitude > 1) / magnitude.size * 100,
        }

        # 全局运动
        stats['global_dx'] = np.mean(fx)
        stats['global_dy'] = np.mean(fy)

        # 旋转检测
        y, x = np.mgrid[0:h, 0:w].astype(float)
        dx, dy = x - cx, y - cy
        r = np.sqrt(dx**2 + dy**2) + 1e-6

        tangential = (fx * (-dy) + fy * dx) / r
        stats['rotation'] = np.mean(tangential[magnitude > 1]) if np.sum(magnitude > 1) > 100 else 0

        # 缩放检测
        radial = (fx * dx + fy * dy) / r
        stats['zoom'] = np.mean(radial[magnitude > 1]) if np.sum(magnitude > 1) > 100 else 0

        return stats

    def visualize_hsv(self, flow):
        """HSV可视化"""
        if flow is None:
            return None

        h, w = flow.shape[:2]
        magnitude, angle = cv2.cartToPolar(flow[:, :, 0], flow[:, :, 1])

        hsv = np.zeros((h, w, 3), dtype=np.uint8)
        hsv[:, :, 0] = angle * 180 / np.pi / 2
        hsv[:, :, 1] = 255
        hsv[:, :, 2] = cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX)

        return cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    def visualize_arrows(self, frame, flow, step=16):
        """箭头可视化"""
        if flow is None:
            return frame.copy()

        vis = frame.copy()
        h, w = flow.shape[:2]

        for y in range(step // 2, h, step):
            for x in range(step // 2, w, step):
                fx, fy = flow[y, x]
                if abs(fx) > 1 or abs(fy) > 1:
                    cv2.arrowedLine(vis, (x, y),
                                   (int(x + fx * 2), int(y + fy * 2)),
                                   (0, 255, 0), 1, tipLength=0.3)
        return vis

def main():
    cap = cv2.VideoCapture(0)
    analyzer = DenseFlowAnalyzer()

    modes = ['HSV Color', 'Arrows', 'Magnitude', 'Segmentation']
    current_mode = 0

    print("光流综合分析系统")
    print("按 'm' 切换显示模式")
    print("按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        # 计算光流
        start_time = time.time()
        flow = analyzer.compute_flow(gray)
        elapsed = (time.time() - start_time) * 1000

        # 分析光流
        stats = analyzer.analyze_flow(flow)

        # 可视化
        if current_mode == 0:
            vis = analyzer.visualize_hsv(flow)
            if vis is None:
                vis = frame.copy()
        elif current_mode == 1:
            vis = analyzer.visualize_arrows(frame, flow)
        elif current_mode == 2:
            if flow is not None:
                magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)
                vis = cv2.applyColorMap(
                    cv2.normalize(magnitude, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8),
                    cv2.COLORMAP_JET
                )
            else:
                vis = frame.copy()
        else:
            if flow is not None:
                magnitude = np.sqrt(flow[:, :, 0]**2 + flow[:, :, 1]**2)
                mask = (magnitude > 2).astype(np.uint8) * 255
                kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
                mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
                vis = cv2.bitwise_and(frame, frame, mask=mask)
            else:
                vis = frame.copy()

        # 显示信息
        y_offset = 25
        cv2.putText(vis, f'Mode: {modes[current_mode]}', (10, y_offset),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2)

        if stats:
            y_offset += 25
            cv2.putText(vis, f"Time: {elapsed:.1f}ms", (10, y_offset),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)
            y_offset += 20
            cv2.putText(vis, f"Mean Mag: {stats['mean_magnitude']:.2f}", (10, y_offset),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)
            y_offset += 20
            cv2.putText(vis, f"Motion: {stats['motion_area']:.1f}%", (10, y_offset),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)
            y_offset += 20
            cv2.putText(vis, f"dX: {stats['global_dx']:.2f}, dY: {stats['global_dy']:.2f}",
                        (10, y_offset), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

            # 运动类型判断
            motion_type = "STABLE"
            if stats['motion_area'] > 5:
                if abs(stats['rotation']) > 0.5:
                    motion_type = "ROTATING"
                elif abs(stats['zoom']) > 0.5:
                    motion_type = "ZOOMING" if stats['zoom'] > 0 else "ZOOMING OUT"
                elif abs(stats['global_dx']) > 2 or abs(stats['global_dy']) > 2:
                    motion_type = "PANNING"
                else:
                    motion_type = "MOTION"

            h, w = vis.shape[:2]
            cv2.putText(vis, motion_type, (w - 150, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 0, 255), 2)

        cv2.imshow('Dense Flow Analyzer', vis)
        cv2.imshow('Original', frame)

        key = cv2.waitKey(30) & 0xFF
        if key == ord('q'):
            break
        elif key == ord('m'):
            current_mode = (current_mode + 1) % len(modes)
            print(f"切换到: {modes[current_mode]}")

    cap.release()
    cv2.destroyAllWindows()

if __name__ == '__main__':
    main()
```

## 三、Farneback参数说明

| 参数 | 说明 | 建议值 |
|------|------|--------|
| `pyr_scale` | 金字塔缩放 | 0.5 |
| `levels` | 金字塔层数 | 3~5 |
| `winsize` | 窗口大小 | 15~21 |
| `iterations` | 迭代次数 | 3~5 |
| `poly_n` | 多项式邻域 | 5或7 |
| `poly_sigma` | 高斯标准差 | 1.1~1.5 |
