# 亚像素级角点检测

## 1. 亚像素精度概述

在许多计算机视觉应用中，像素级的角点定位精度可能不够。亚像素级角点检测可以将角点位置精确到小数点后若干位，显著提高测量和匹配的精度。

### 1.1 为什么需要亚像素精度

1. **三维重建**：精确的特征点位置对深度估计至关重要
2. **相机标定**：亚像素精度可以显著提高标定质量
3. **精密测量**：工业检测中的高精度要求
4. **图像配准**：提高配准的准确性
5. **运动估计**：光流计算的精度提升

### 1.2 亚像素精细化原理

亚像素精细化通常基于以下假设：
- 角点周围的灰度分布是平滑的
- 可以用数学模型拟合局部区域

常用方法：
1. **梯度法**：基于角点周围梯度方向的收敛点
2. **曲面拟合**：拟合局部灰度曲面，找极值点
3. **相关法**：基于亚像素偏移的相关匹配

### 1.3 cv2.cornerSubPix() 函数

```python
cv2.cornerSubPix(image, corners, winSize, zeroZone, criteria)
```

**参数说明**：

| 参数 | 说明 |
|------|------|
| image | 输入图像（灰度） |
| corners | 初始角点位置（会被修改） |
| winSize | 搜索窗口半径 |
| zeroZone | 死区半径（-1,-1表示无死区） |
| criteria | 终止条件 |

**终止条件**：

```python
criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, max_iter, epsilon)
```

### 示例1：基本亚像素精细化

```python
"""
示例1：基本亚像素精细化
使用cornerSubPix()提高角点精度
"""
import cv2
import numpy as np

# 创建带有明确角点的测试图像
img = np.zeros((400, 500), dtype=np.uint8)
cv2.rectangle(img, (100, 100), (250, 250), 200, -1)
cv2.rectangle(img, (280, 120), (420, 280), 200, -1)

# 添加轻微模糊使角点更真实
img = cv2.GaussianBlur(img, (3, 3), 0)

# 检测初始角点（像素级）
corners = cv2.goodFeaturesToTrack(
    img,
    maxCorners=20,
    qualityLevel=0.01,
    minDistance=30
)

print("初始角点（像素级）:")
for i, corner in enumerate(corners):
    x, y = corner.ravel()
    print(f"  角点 {i}: ({x:.2f}, {y:.2f})")

# 亚像素精细化
corners_refined = corners.copy()

# 定义终止条件
criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.001)

# 亚像素精细化
cv2.cornerSubPix(
    img,
    corners_refined,
    winSize=(5, 5),
    zeroZone=(-1, -1),
    criteria=criteria
)

print("\n亚像素精细化后:")
for i, corner in enumerate(corners_refined):
    x, y = corner.ravel()
    orig_x, orig_y = corners[i].ravel()
    dx = x - orig_x
    dy = y - orig_y
    print(f"  角点 {i}: ({x:.4f}, {y:.4f}) 偏移: ({dx:.4f}, {dy:.4f})")

# 可视化
result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

for corner in corners:
    x, y = corner.ravel().astype(int)
    cv2.circle(result, (x, y), 8, (255, 0, 0), 2)  # 蓝色：原始

for corner in corners_refined:
    x, y = corner.ravel()
    cv2.circle(result, (int(x), int(y)), 4, (0, 0, 255), -1)  # 红色：精细化

cv2.putText(result, "Blue=Original, Red=Refined", (10, 30),
           cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

cv2.imshow("Subpixel Refinement", result)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例2：精细化精度分析

```python
"""
示例2：精细化精度分析
分析亚像素精细化的精度提升
"""
import cv2
import numpy as np

def create_synthetic_corner(size=100, corner_pos=(50.3, 50.7)):
    """创建具有已知亚像素位置的合成角点"""
    img = np.zeros((size, size), dtype=np.float32)

    # 创建带有精确位置的角点
    for y in range(size):
        for x in range(size):
            if x > corner_pos[0] and y > corner_pos[1]:
                img[y, x] = 200

    # 添加亚像素效果（抗锯齿）
    img = cv2.GaussianBlur(img, (3, 3), 0.5)

    return img.astype(np.uint8), corner_pos

# 创建多个测试用例
test_cases = [
    (50.0, 50.0),   # 整数位置
    (50.3, 50.3),   # 小偏移
    (50.5, 50.5),   # 半像素
    (50.7, 50.7),   # 大偏移
    (50.25, 50.75), # 不对称偏移
]

print("亚像素精细化精度测试:")
print("=" * 70)
print(f"{'真实位置':>20} {'原始检测':>20} {'精细化后':>20} {'误差':>10}")
print("-" * 70)

for true_x, true_y in test_cases:
    # 创建合成图像
    img, _ = create_synthetic_corner(corner_pos=(true_x, true_y))

    # 检测角点
    corners = cv2.goodFeaturesToTrack(img, maxCorners=1, qualityLevel=0.01, minDistance=10)

    if corners is not None:
        orig_x, orig_y = corners[0].ravel()

        # 精细化
        corners_refined = corners.copy()
        criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.0001)
        cv2.cornerSubPix(img, corners_refined, (5, 5), (-1, -1), criteria)

        ref_x, ref_y = corners_refined[0].ravel()

        # 计算误差
        orig_error = np.sqrt((orig_x - true_x)**2 + (orig_y - true_y)**2)
        ref_error = np.sqrt((ref_x - true_x)**2 + (ref_y - true_y)**2)

        print(f"({true_x:.2f}, {true_y:.2f})"
              f"   ({orig_x:.2f}, {orig_y:.2f})"
              f"   ({ref_x:.4f}, {ref_y:.4f})"
              f"   {ref_error:.6f}")

print("-" * 70)
print("误差单位：像素")
```

```python
"""
示例3：窗口大小的影响
分析winSize参数对精细化结果的影响
"""
import cv2
import numpy as np

# 创建测试图像
img = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img, (80, 80), (180, 180), 200, -1)
cv2.rectangle(img, (220, 100), (360, 220), 200, -1)

# 添加噪声和模糊
noise = np.random.normal(0, 5, img.shape).astype(np.int16)
img = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)
img = cv2.GaussianBlur(img, (3, 3), 0)

# 检测初始角点
corners = cv2.goodFeaturesToTrack(img, maxCorners=10, qualityLevel=0.01, minDistance=20)

if corners is None:
    print("未检测到角点")
else:
    print("不同窗口大小的影响:")
    print("-" * 60)
    print(f"{'窗口大小':>10} {'角点0位置':>25} {'与原始偏移':>20}")
    print("-" * 60)

    # 原始位置
    orig_pos = corners[0].ravel().copy()
    print(f"{'原始':>10} ({orig_pos[0]:.4f}, {orig_pos[1]:.4f})")

    # 不同窗口大小
    window_sizes = [(3, 3), (5, 5), (7, 7), (11, 11), (15, 15)]
    criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.001)

    results = []
    for win_size in window_sizes:
        corners_copy = corners.copy()
        cv2.cornerSubPix(img, corners_copy, win_size, (-1, -1), criteria)

        pos = corners_copy[0].ravel()
        offset = np.sqrt((pos[0] - orig_pos[0])**2 + (pos[1] - orig_pos[1])**2)

        print(f"{win_size}".rjust(10) + f" ({pos[0]:.4f}, {pos[1]:.4f})" + f" {offset:.6f}".rjust(20))

        results.append((win_size, corners_copy))

    # 可视化
    result_imgs = []
    for win_size, refined_corners in results:
        canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

        # 绘制原始位置
        for corner in corners:
            x, y = corner.ravel().astype(int)
            cv2.circle(canvas, (x, y), 6, (255, 0, 0), 2)

        # 绘制精细化位置
        for corner in refined_corners:
            x, y = corner.ravel()
            cv2.circle(canvas, (int(x), int(y)), 3, (0, 0, 255), -1)

        cv2.putText(canvas, f"win={win_size[0]}", (10, 25),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

        result_imgs.append(canvas)

    # 组合显示
    row1 = np.hstack(result_imgs[:3])
    row2 = np.hstack(result_imgs[3:] + [np.zeros_like(result_imgs[0])])
    combined = np.vstack([row1, row2])

    cv2.imshow("Window Size Comparison", combined)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
```

```python
"""
示例5：终止条件的影响
分析不同终止条件对精细化的影响
"""
import cv2
import numpy as np
import time

# 创建测试图像
img = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (150, 150), 200, -1)
cv2.rectangle(img, (180, 80), (350, 220), 200, -1)

# 添加模糊
img = cv2.GaussianBlur(img, (5, 5), 1)

# 检测角点
corners = cv2.goodFeaturesToTrack(img, maxCorners=10, qualityLevel=0.01, minDistance=20)

if corners is not None:
    print("终止条件影响分析:")
    print("=" * 80)
    print(f"{'条件类型':>15} {'max_iter':>10} {'epsilon':>12} {'时间(ms)':>12} {'位置':>20}")
    print("-" * 80)

    # 不同终止条件
    conditions = [
        # (类型, max_iter, epsilon)
        (cv2.TERM_CRITERIA_MAX_ITER, 10, 0),
        (cv2.TERM_CRITERIA_MAX_ITER, 100, 0),
        (cv2.TERM_CRITERIA_EPS, 100, 0.1),
        (cv2.TERM_CRITERIA_EPS, 100, 0.01),
        (cv2.TERM_CRITERIA_EPS, 100, 0.001),
        (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.01),
        (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.001),
    ]

    condition_names = {
        cv2.TERM_CRITERIA_MAX_ITER: "MAX_ITER",
        cv2.TERM_CRITERIA_EPS: "EPS",
        cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER: "EPS+ITER",
    }

    for cond_type, max_iter, epsilon in conditions:
        corners_copy = corners.copy()
        criteria = (cond_type, max_iter, epsilon)

        start = time.time()
        for _ in range(100):  # 运行100次取平均
            corners_test = corners.copy()
            cv2.cornerSubPix(img, corners_test, (5, 5), (-1, -1), criteria)
        elapsed = (time.time() - start) / 100 * 1000

        cv2.cornerSubPix(img, corners_copy, (5, 5), (-1, -1), criteria)
        pos = corners_copy[0].ravel()

        name = condition_names.get(cond_type, "UNKNOWN")
        print(f"{name:>15} {max_iter:>10} {epsilon:>12.4f} {elapsed:>12.4f} ({pos[0]:.4f}, {pos[1]:.4f})")

    print("-" * 80)
    print("结论: EPS+ITER组合可以在精度和速度之间取得平衡")
```

### 示例6：相机标定中的亚像素精度

```python
"""
示例6：相机标定中的亚像素精度
模拟棋盘格角点的亚像素精细化
"""
import cv2
import numpy as np

def create_checkerboard(rows, cols, square_size, img_size):
    """创建棋盘格图像"""
    img = np.ones(img_size, dtype=np.uint8) * 255

    # 计算起始位置（居中）
    board_width = cols * square_size
    board_height = rows * square_size
    start_x = (img_size[1] - board_width) // 2
    start_y = (img_size[0] - board_height) // 2

    # 绘制棋盘格
    for i in range(rows):
        for j in range(cols):
            if (i + j) % 2 == 0:
                x1 = start_x + j * square_size
                y1 = start_y + i * square_size
                x2 = x1 + square_size
                y2 = y1 + square_size
                cv2.rectangle(img, (x1, y1), (x2, y2), 0, -1)

    return img, (start_x, start_y)

# 创建棋盘格
rows, cols = 7, 9
square_size = 40
img_size = (400, 500)

img, offset = create_checkerboard(rows, cols, square_size, img_size)

# 添加轻微透视变换模拟真实拍摄
h, w = img.shape
pts1 = np.float32([[0, 0], [w, 0], [0, h], [w, h]])
pts2 = np.float32([[10, 5], [w-5, 10], [5, h-10], [w-10, h-5]])
M = cv2.getPerspectiveTransform(pts1, pts2)
img = cv2.warpPerspective(img, M, (w, h), borderValue=200)

# 添加噪声
noise = np.random.normal(0, 3, img.shape).astype(np.int16)
img = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)

# 检测棋盘格角点
pattern_size = (cols - 1, rows - 1)  # 内角点数
ret, corners = cv2.findChessboardCorners(img, pattern_size, None)

if ret:
    print(f"检测到 {len(corners)} 个棋盘格角点")

    # 原始角点位置
    corners_orig = corners.copy()

    # 亚像素精细化
    criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001)
    cv2.cornerSubPix(img, corners, (11, 11), (-1, -1), criteria)

    # 计算偏移
    offsets = np.sqrt(np.sum((corners - corners_orig) ** 2, axis=2))
    print(f"\n亚像素偏移统计:")
    print(f"  平均偏移: {offsets.mean():.4f} 像素")
    print(f"  最大偏移: {offsets.max():.4f} 像素")
    print(f"  最小偏移: {offsets.min():.4f} 像素")
    print(f"  标准差: {offsets.std():.4f} 像素")

    # 可视化
    result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

    # 绘制原始位置（蓝色小圆）
    for corner in corners_orig:
        x, y = corner.ravel().astype(int)
        cv2.circle(result, (x, y), 5, (255, 0, 0), 1)

    # 绘制精细化位置（红色点）
    cv2.drawChessboardCorners(result, pattern_size, corners, ret)

    # 放大显示一个角点
    sample_idx = len(corners) // 2
    sample_orig = corners_orig[sample_idx].ravel()
    sample_ref = corners[sample_idx].ravel()

    # 在右侧显示放大视图
    zoom_size = 60
    zoom_scale = 4
    x, y = sample_ref.astype(int)

    if y - zoom_size//2 >= 0 and y + zoom_size//2 < h and x - zoom_size//2 >= 0 and x + zoom_size//2 < w:
        zoom_region = img[y-zoom_size//2:y+zoom_size//2, x-zoom_size//2:x+zoom_size//2]
        zoom_region = cv2.resize(zoom_region, None, fx=zoom_scale, fy=zoom_scale, interpolation=cv2.INTER_NEAREST)
        zoom_color = cv2.cvtColor(zoom_region, cv2.COLOR_GRAY2BGR)

        # 在放大图上标记位置
        center = zoom_size//2 * zoom_scale
        cv2.circle(zoom_color, (center, center), 3, (0, 0, 255), -1)

        cv2.imshow("Zoomed Corner", zoom_color)

    cv2.imshow("Chessboard Corners", result)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
else:
    print("未检测到棋盘格角点")
```

### 示例7：亚像素精细化工具类

```python
"""
示例7：亚像素精细化工具类
封装亚像素角点精细化功能
"""
import cv2
import numpy as np
from typing import Tuple, Optional
from dataclasses import dataclass

@dataclass
class SubpixelCorner:
    """亚像素角点"""
    x: float
    y: float
    original_x: float
    original_y: float
    offset: float

class SubpixelRefiner:
    """亚像素精细化器"""

    def __init__(self,
                 win_size: Tuple[int, int] = (5, 5),
                 zero_zone: Tuple[int, int] = (-1, -1),
                 max_iter: int = 100,
                 epsilon: float = 0.001):
        """
        初始化精细化器

        Args:
            win_size: 搜索窗口大小
            zero_zone: 死区大小
            max_iter: 最大迭代次数
            epsilon: 精度阈值
        """
        self.win_size = win_size
        self.zero_zone = zero_zone
        self.criteria = (
            cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER,
            max_iter,
            epsilon
        )

    def refine(self,
               img: np.ndarray,
               corners: np.ndarray) -> Tuple[np.ndarray, list]:
        """
        精细化角点位置

        Args:
            img: 灰度图像
            corners: 初始角点位置 (N, 1, 2)

        Returns:
            refined_corners: 精细化后的角点
            corner_info: 角点详细信息列表
        """
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img

        # 保存原始位置
        original = corners.copy()

        # 精细化
        refined = corners.copy()
        cv2.cornerSubPix(gray, refined, self.win_size, self.zero_zone, self.criteria)

        # 生成详细信息
        corner_info = []
        for orig, ref in zip(original, refined):
            ox, oy = orig.ravel()
            rx, ry = ref.ravel()
            offset = np.sqrt((rx - ox)**2 + (ry - oy)**2)

            corner_info.append(SubpixelCorner(
                x=float(rx),
                y=float(ry),
                original_x=float(ox),
                original_y=float(oy),
                offset=float(offset)
            ))

        return refined, corner_info

    def refine_with_stats(self,
                          img: np.ndarray,
                          corners: np.ndarray) -> dict:
        """
        精细化并返回统计信息

        Returns:
            包含精细化结果和统计信息的字典
        """
        refined, corner_info = self.refine(img, corners)

        offsets = [c.offset for c in corner_info]

        return {
            'refined_corners': refined,
            'corner_info': corner_info,
            'stats': {
                'count': len(corner_info),
                'mean_offset': np.mean(offsets),
                'max_offset': np.max(offsets),
                'min_offset': np.min(offsets),
                'std_offset': np.std(offsets)
            }
        }

    def visualize(self,
                  img: np.ndarray,
                  corners: np.ndarray,
                  refined: np.ndarray,
                  show_offset: bool = True) -> np.ndarray:
        """
        可视化精细化结果
        """
        result = img.copy() if len(img.shape) == 3 else cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

        for orig, ref in zip(corners, refined):
            ox, oy = orig.ravel().astype(int)
            rx, ry = ref.ravel()

            # 原始位置（蓝色空心圆）
            cv2.circle(result, (ox, oy), 6, (255, 0, 0), 1)

            # 精细化位置（红色实心圆）
            cv2.circle(result, (int(rx), int(ry)), 3, (0, 0, 255), -1)

            # 连线（可选）
            if show_offset:
                cv2.line(result, (ox, oy), (int(rx), int(ry)), (0, 255, 0), 1)

        return result
```

```python
"""
示例8：高精度角点检测流程
完整的高精度角点检测工作流
"""
import cv2
import numpy as np

class HighPrecisionCornerDetector:
    """高精度角点检测器"""

    def __init__(self):
        # 初始检测参数
        self.max_corners = 100
        self.quality_level = 0.01
        self.min_distance = 10

        # 亚像素精细化参数
        self.subpix_win = (5, 5)
        self.subpix_zero_zone = (-1, -1)
        self.subpix_criteria = (
            cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER,
            100,
            0.001
        )

    def detect(self, img, mask=None):
        """
        完整的高精度角点检测
        """
        # 预处理
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img.copy()

        # 轻微模糊减少噪声
        gray = cv2.GaussianBlur(gray, (3, 3), 0)

        # 初始检测
        corners = cv2.goodFeaturesToTrack(
            gray,
            maxCorners=self.max_corners,
            qualityLevel=self.quality_level,
            minDistance=self.min_distance,
            mask=mask
        )

        if corners is None:
            return None, None

        # 保存原始位置
        corners_original = corners.copy()

        # 亚像素精细化
        cv2.cornerSubPix(
            gray,
            corners,
            self.subpix_win,
            self.subpix_zero_zone,
            self.subpix_criteria
        )

        return corners, corners_original

    def detect_with_validation(self, img, mask=None):
        """
        带验证的角点检测
        """
        corners, corners_original = self.detect(img, mask)

        if corners is None:
            return None

        # 计算偏移
        offsets = np.sqrt(np.sum((corners - corners_original) ** 2, axis=2)).ravel()

        # 过滤异常偏移（可能是误检）
        valid_mask = offsets < 2.0  # 偏移超过2像素可能有问题

        valid_corners = corners[valid_mask]
        valid_original = corners_original[valid_mask]
        valid_offsets = offsets[valid_mask]

        return {
            'corners': valid_corners,
            'original': valid_original,
            'offsets': valid_offsets,
            'removed': np.sum(~valid_mask)
        }
```

```python
"""
示例9：精度评估与对比
评估亚像素精细化的效果
"""
import cv2
import numpy as np

def evaluate_subpixel_accuracy(img, true_corners, detected_corners):
    """
    评估亚像素精度

    Args:
        true_corners: 真实角点位置 [(x, y), ...]
        detected_corners: 检测到的角点 (N, 1, 2)

    Returns:
        匹配信息和误差统计
    """
    if detected_corners is None or len(detected_corners) == 0:
        return None

    # 将检测角点转换为列表
    detected = [tuple(c.ravel()) for c in detected_corners]

    # 匹配：为每个真实角点找最近的检测角点
    matches = []
    for tx, ty in true_corners:
        min_dist = float('inf')
        best_match = None

        for dx, dy in detected:
            dist = np.sqrt((dx - tx)**2 + (dy - ty)**2)
            if dist < min_dist:
                min_dist = dist
                best_match = (dx, dy)

        if min_dist < 20:  # 阈值
            matches.append({
                'true': (tx, ty),
                'detected': best_match,
                'error': min_dist
            })

    if not matches:
        return None

    errors = [m['error'] for m in matches]

    return {
        'matches': matches,
        'stats': {
            'matched': len(matches),
            'mean_error': np.mean(errors),
            'max_error': np.max(errors),
            'rmse': np.sqrt(np.mean(np.array(errors)**2))
        }
    }
```

```python
"""
示例10：综合应用示例
完整的亚像素角点检测应用
"""
import cv2
import numpy as np
import time

class PrecisionCornerSystem:
    """精密角点检测系统"""

    def __init__(self):
        # 检测参数
        self.detector_params = {
            'maxCorners': 100,
            'qualityLevel': 0.01,
            'minDistance': 10,
            'blockSize': 3
        }

        # 精细化参数
        self.subpix_params = {
            'winSize': (5, 5),
            'zeroZone': (-1, -1),
            'criteria': (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.001)
        }

        # 性能统计
        self.stats = {
            'detection_time': 0,
            'refinement_time': 0,
            'total_corners': 0
        }

    def process(self, img, use_subpixel=True):
        """
        处理图像并返回角点

        Returns:
            result: 包含角点和统计信息的字典
        """
        # 预处理
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img.copy()

        # 检测
        t1 = time.time()
        corners = cv2.goodFeaturesToTrack(gray, **self.detector_params)
        t2 = time.time()
        self.stats['detection_time'] = (t2 - t1) * 1000

        if corners is None:
            return None

        corners_original = corners.copy()

        # 精细化
        if use_subpixel:
            t1 = time.time()
            cv2.cornerSubPix(gray, corners, **self.subpix_params)
            t2 = time.time()
            self.stats['refinement_time'] = (t2 - t1) * 1000

        self.stats['total_corners'] = len(corners)

        # 计算偏移
        offsets = np.sqrt(np.sum((corners - corners_original) ** 2, axis=2)).ravel()

        return {
            'corners': corners,
            'original': corners_original,
            'offsets': offsets,
            'stats': self.stats.copy()
        }

    def visualize(self, img, result, show_info=True):
        """可视化结果"""
        vis = img.copy() if len(img.shape) == 3 else cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

        if result is None:
            cv2.putText(vis, "No corners detected", (10, 30),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
            return vis

        # 绘制角点
        for orig, ref in zip(result['original'], result['corners']):
            ox, oy = orig.ravel().astype(int)
            rx, ry = ref.ravel()

            # 原始位置
            cv2.circle(vis, (ox, oy), 5, (255, 0, 0), 1)
            # 精细化位置
            cv2.circle(vis, (int(rx), int(ry)), 2, (0, 0, 255), -1)

        # 显示信息
        if show_info:
            info = [
                f"Corners: {result['stats']['total_corners']}",
                f"Detection: {result['stats']['detection_time']:.2f}ms",
                f"Refinement: {result['stats']['refinement_time']:.2f}ms",
                f"Mean offset: {result['offsets'].mean():.4f}px"
            ]

            for i, text in enumerate(info):
                cv2.putText(vis, text, (10, 25 + i*25),
                           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

        return vis
```

### 3.1 参数选择建议

| 参数 | 推荐值 | 说明 |
|------|--------|------|
| winSize | (5,5) 或 (7,7) | 根据特征尺度选择 |
| zeroZone | (-1,-1) | 通常不需要死区 |

| epsilon | 0.001-0.01 | 精度要求 |

### 3.2 使用注意事项

1. **图像质量**：高质量图像可获得更好的亚像素精度
2. **初始位置**：初始检测应该足够接近真实位置
3. **窗口大小**：太小可能不收敛，太大可能引入误差
4. **噪声影响**：适当预处理可以提高精度
