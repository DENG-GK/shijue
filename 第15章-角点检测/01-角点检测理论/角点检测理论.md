# 角点检测理论

## 1. 角点概述

角点是图像中具有独特几何特征的点，是计算机视觉中最重要的特征之一。角点检测在图像配准、目标跟踪、三维重建等领域有广泛应用。

### 1.1 什么是角点

**角点的定义**：
- 两条或多条边的交点
- 在任意方向上移动都会引起显著灰度变化的点
- 局部区域中具有两个主方向的点

**角点的特性**：

| 特性 | 说明 |
|------|------|
| 独特性 | 在局部区域内唯一 |
| 稳定性 | 对光照、视角变化鲁棒 |
| 可重复性 | 不同图像中能重复检测 |
| 信息量 | 包含丰富的几何信息 |

### 1.2 角点vs边缘vs平坦区域

| 区域类型 | 灰度变化特点 | 特征值特点 |
|----------|--------------|------------|
| 平坦区域 | 任何方向变化都小 | λ₁≈λ₂≈0 |
| 边缘 | 沿边方向变化小 | λ₁>>λ₂或λ₂>>λ₁ |
| 角点 | 任何方向变化都大 | λ₁和λ₂都较大 |

### 1.3 角点检测的应用

1. **图像配准**：不同视角图像的对齐
2. **目标跟踪**：视频中物体的跟踪
3. **三维重建**：立体视觉和SfM
4. **图像拼接**：全景图像生成
5. **物体识别**：基于特征的识别
6. **运动估计**：光流计算

## 2. 角点检测原理

### 2.1 基本思想

角点检测的核心思想：在角点处，向任意方向移动一个小窗口，窗口内的灰度值都会发生显著变化。

**数学表达**：

设图像为 $I(x, y)$，窗口移动 $(u, v)$ 后的灰度变化为：

$$E(u, v) = \sum_{(x,y) \in W} w(x,y) [I(x+u, y+v) - I(x, y)]^2$$

其中：
- $W$ 是窗口区域
- $w(x,y)$ 是权重函数（通常为高斯函数）

### 2.2 自相关矩阵

对 $E(u, v)$ 进行泰勒展开：

$$E(u, v) \approx [u, v] M \begin{bmatrix} u \\ v \end{bmatrix}$$

其中 $M$ 是自相关矩阵（也称结构张量）：

$$M = \sum_{(x,y) \in W} w(x,y) \begin{bmatrix} I_x^2 & I_x I_y \\ I_x I_y & I_y^2 \end{bmatrix}$$

$I_x$ 和 $I_y$ 分别是图像在 $x$ 和 $y$ 方向的梯度。

### 2.3 特征值分析

矩阵 $M$ 的特征值 $\lambda_1$ 和 $\lambda_2$ 反映了局部区域的结构：

- **平坦区域**：$\lambda_1 \approx \lambda_2 \approx 0$
- **边缘**：$\lambda_1 >> \lambda_2$ 或 $\lambda_2 >> \lambda_1$
- **角点**：$\lambda_1$ 和 $\lambda_2$ 都较大

### 示例1：理解灰度变化

```python
"""
示例1：理解灰度变化
可视化不同区域移动窗口时的灰度变化
"""
import cv2
import numpy as np
import matplotlib.pyplot as plt

def compute_ssd(img, x, y, window_size=5, max_shift=5):
    """计算窗口移动后的SSD（Sum of Squared Differences）"""
    half = window_size // 2
    h, w = img.shape

    # 提取参考窗口
    ref_window = img[y-half:y+half+1, x-half:x+half+1].astype(float)

    # 计算不同移动量的SSD
    ssd_map = np.zeros((2*max_shift+1, 2*max_shift+1))

    for du in range(-max_shift, max_shift+1):
        for dv in range(-max_shift, max_shift+1):
            ny, nx = y + dv, x + du
            if ny-half >= 0 and ny+half+1 <= h and nx-half >= 0 and nx+half+1 <= w:
                shifted = img[ny-half:ny+half+1, nx-half:nx+half+1].astype(float)
                ssd = np.sum((ref_window - shifted) ** 2)
                ssd_map[dv+max_shift, du+max_shift] = ssd

    return ssd_map

# 创建测试图像
img = np.zeros((200, 200), dtype=np.uint8)

# 创建不同区域
# 平坦区域
img[50:150, 50:150] = 128

# 边缘
cv2.rectangle(img, (20, 20), (80, 80), 200, -1)

# 角点
cv2.rectangle(img, (120, 20), (180, 80), 200, -1)

# 选择测试点
test_points = [
    (100, 100, "Flat Region"),    # 平坦区域中心
    (50, 50, "Edge"),              # 边缘上
    (120, 20, "Corner"),           # 角点
]

fig, axes = plt.subplots(1, 4, figsize=(16, 4))

# 显示原图
axes[0].imshow(img, cmap='gray')
axes[0].set_title('Original Image')
for x, y, name in test_points:
    axes[0].plot(x, y, 'ro', markersize=8)
    axes[0].annotate(name, (x+5, y+5), color='red')

# 显示各点的SSD图
for i, (x, y, name) in enumerate(test_points):
    ssd_map = compute_ssd(img, x, y)
    im = axes[i+1].imshow(ssd_map, cmap='hot')
    axes[i+1].set_title(f'{name}\nSSD Map')
    plt.colorbar(im, ax=axes[i+1])

plt.tight_layout()
plt.savefig('ssd_analysis.png', dpi=150)
plt.show()

print("分析结果：")
print("- 平坦区域：SSD在所有方向都很小")
print("- 边缘：SSD沿边方向小，垂直方向大")
print("- 角点：SSD在所有方向都大")
```

### 示例3：特征值分析

```python
"""
示例3：特征值分析
计算自相关矩阵的特征值来分析图像结构
"""
import cv2
import numpy as np

def compute_eigenvalues(img, block_size=3, ksize=3):
    """计算图像每个像素位置的特征值"""
    img_float = img.astype(np.float32)

    # 计算梯度
    Ix = cv2.Sobel(img_float, cv2.CV_32F, 1, 0, ksize=ksize)
    Iy = cv2.Sobel(img_float, cv2.CV_32F, 0, 1, ksize=ksize)

    # 计算自相关矩阵元素
    Ixx = Ix ** 2
    Iyy = Iy ** 2
    Ixy = Ix * Iy

    # 使用窗口求和
    kernel = np.ones((block_size, block_size), np.float32)
    Sxx = cv2.filter2D(Ixx, -1, kernel)
    Syy = cv2.filter2D(Iyy, -1, kernel)
    Sxy = cv2.filter2D(Ixy, -1, kernel)

    # 计算特征值
    # 对于2x2矩阵 [[a, b], [b, c]]
    # 特征值 = (a+c)/2 ± sqrt(((a-c)/2)^2 + b^2)
    trace = Sxx + Syy
    det = Sxx * Syy - Sxy ** 2
    discriminant = np.sqrt(np.maximum((trace/2)**2 - det, 0))

    lambda1 = trace/2 + discriminant
    lambda2 = trace/2 - discriminant

    return lambda1, lambda2

# 创建测试图像
img = np.zeros((300, 300), dtype=np.uint8)

# 不同区域
img[100:200, 100:200] = 150  # 矩形
cv2.circle(img, (80, 80), 40, 200, -1)  # 圆

# 计算特征值
lambda1, lambda2 = compute_eigenvalues(img)

# 分类像素
min_eigenvalue = np.minimum(lambda1, lambda2)
max_eigenvalue = np.maximum(lambda1, lambda2)

# 阈值
threshold = 1000

# 角点：两个特征值都大
corners = (min_eigenvalue > threshold).astype(np.uint8) * 255

# 边缘：一个大一个小
edges = ((max_eigenvalue > threshold) & (min_eigenvalue < threshold/10)).astype(np.uint8) * 255

# 可视化
canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

# 标记角点（红色）
canvas[corners > 0] = [0, 0, 255]

# 标记边缘（绿色）
canvas[edges > 0] = [0, 255, 0]

# 创建特征值可视化
lambda1_vis = ((lambda1 / lambda1.max()) * 255).astype(np.uint8)
lambda2_vis = ((lambda2 / lambda2.max()) * 255).astype(np.uint8)
min_vis = ((min_eigenvalue / min_eigenvalue.max()) * 255).astype(np.uint8)

# 显示结果
result = np.zeros((300, 1200, 3), dtype=np.uint8)

result[:, :300] = canvas
cv2.putText(result, "Classification", (10, 25), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2)

result[:, 300:600] = cv2.cvtColor(lambda1_vis, cv2.COLOR_GRAY2BGR)
cv2.putText(result, "Lambda1 (max)", (310, 25), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

result[:, 600:900] = cv2.cvtColor(lambda2_vis, cv2.COLOR_GRAY2BGR)
cv2.putText(result, "Lambda2 (min)", (610, 25), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

result[:, 900:1200] = cv2.cvtColor(min_vis, cv2.COLOR_GRAY2BGR)
cv2.putText(result, "min(L1,L2)", (910, 25), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

cv2.imshow("Eigenvalue Analysis", result)
cv2.waitKey(0)
cv2.destroyAllWindows()

print("特征值分析:")
print(f"Lambda1 最大值: {lambda1.max():.1f}")
print(f"Lambda2 最大值: {lambda2.max():.1f}")
print("红色=角点, 绿色=边缘")
```

### 示例4：Harris响应函数

```python
"""
示例4：Harris响应函数
计算Harris角点响应值
"""
import cv2
import numpy as np

def harris_response(img, block_size=2, ksize=3, k=0.04):
    """手动计算Harris角点响应"""
    img_float = img.astype(np.float32)

    # 计算梯度
    Ix = cv2.Sobel(img_float, cv2.CV_32F, 1, 0, ksize=ksize)
    Iy = cv2.Sobel(img_float, cv2.CV_32F, 0, 1, ksize=ksize)

    # 计算自相关矩阵元素
    Ixx = Ix ** 2
    Iyy = Iy ** 2
    Ixy = Ix * Iy

    # 高斯加权求和
    Sxx = cv2.GaussianBlur(Ixx, (block_size*2+1, block_size*2+1), 0)
    Syy = cv2.GaussianBlur(Iyy, (block_size*2+1, block_size*2+1), 0)
    Sxy = cv2.GaussianBlur(Ixy, (block_size*2+1, block_size*2+1), 0)

    # Harris响应
    # R = det(M) - k * trace(M)^2
    det = Sxx * Syy - Sxy ** 2
    trace = Sxx + Syy
    R = det - k * trace ** 2

    return R

# 创建测试图像
img = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (150, 150), 200, -1)
cv2.rectangle(img, (200, 50), (350, 200), 200, -1)
cv2.circle(img, (100, 250), 40, 200, -1)

# 计算Harris响应
R = harris_response(img)

# OpenCV的Harris实现对比
R_cv = cv2.cornerHarris(img.astype(np.float32), 2, 3, 0.04)

# 归一化显示
R_norm = cv2.normalize(R, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)
R_cv_norm = cv2.normalize(R_cv, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)

# 阈值检测角点
threshold = 0.01 * R.max()
corners = R > threshold

# 可视化
canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
canvas[corners] = [0, 0, 255]

# 用膨胀使角点更明显
corners_dilated = cv2.dilate(corners.astype(np.uint8), None)
canvas[corners_dilated > 0] = [0, 0, 255]

# 显示结果
result = np.zeros((300, 1200, 3), dtype=np.uint8)

result[:, :400] = canvas
cv2.putText(result, "Corners Detected", (10, 25),
           cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

result[:, 400:800, :] = cv2.cvtColor(R_norm, cv2.COLOR_GRAY2BGR)
cv2.putText(result, "Harris Response (Manual)", (410, 25),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

result[:, 800:1200, :] = cv2.cvtColor(R_cv_norm, cv2.COLOR_GRAY2BGR)
cv2.putText(result, "Harris Response (OpenCV)", (810, 25),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

cv2.imshow("Harris Response", result)
cv2.waitKey(0)
cv2.destroyAllWindows()

print("Harris响应分析:")
print(f"响应值范围: [{R.min():.2e}, {R.max():.2e}]")
print(f"检测到的角点数: {np.sum(corners)}")
```

### 示例5：不同k值的影响

```python
"""
示例5：不同k值的影响
分析Harris角点检测中k参数的影响
"""
import cv2
import numpy as np

# 创建测试图像
img = np.zeros((250, 350), dtype=np.uint8)
cv2.rectangle(img, (30, 30), (130, 130), 200, -1)
cv2.rectangle(img, (180, 30), (320, 170), 200, -1)

# 不同的k值
k_values = [0.01, 0.04, 0.06, 0.10, 0.15, 0.25]

print("不同k值的影响:")
print("-" * 50)
print(f"{'k值':>8} {'角点数':>10} {'响应范围':>25}")
print("-" * 50)

results = []
for k in k_values:
    # 计算Harris响应
    R = cv2.cornerHarris(img.astype(np.float32), 2, 3, k)

    # 阈值
    threshold = 0.01 * R.max()
    corners = R > threshold

    # 统计
    corner_count = np.sum(corners)
    print(f"{k:>8.2f} {corner_count:>10} [{R.min():.2e}, {R.max():.2e}]")

    # 可视化
    canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
    canvas[cv2.dilate(corners.astype(np.uint8), None) > 0] = [0, 0, 255]

    # 添加标签
    cv2.putText(canvas, f"k={k}", (10, 20),
               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
    cv2.putText(canvas, f"n={corner_count}", (10, 40),
               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 0), 1)

    results.append(canvas)

# 组合显示
row1 = np.hstack(results[:3])
row2 = np.hstack(results[3:])
combined = np.vstack([row1, row2])

cv2.imshow("Different k Values", combined)
cv2.waitKey(0)
cv2.destroyAllWindows()

print("-" * 50)
print("说明: k值越大，角点检测越严格")
print("典型k值范围: 0.04-0.06")
```

"""
示例7：角点检测算法比较
比较Harris、Shi-Tomasi和FAST的检测结果
"""
import cv2
import numpy as np

img = cv2.imread("image.jpg", cv2.IMREAD_GRAYSCALE)
if img is None:
    # 创建合成图像

    for i in range(0, 400, 50):

    # 添加一些圆形

shi_tomasi_corners = np.zeros(img.shape, dtype=np.uint8)

### 示例8：角点检测评估

```python
"""
示例8：角点检测评估
评估角点检测的重复性和准确性
"""
import cv2
import numpy as np

def detect_corners_harris(img, threshold=0.01):
    """Harris角点检测"""
    R = cv2.cornerHarris(img.astype(np.float32), 2, 3, 0.04)
    corners = np.where(R > threshold * R.max())
    return list(zip(corners[1], corners[0]))  # (x, y)

def detect_corners_shi_tomasi(img, max_corners=100):
    """Shi-Tomasi角点检测"""
    corners = cv2.goodFeaturesToTrack(img, max_corners, 0.01, 10)
    if corners is None:
        return []
    return [(int(c[0][0]), int(c[0][1])) for c in corners]

def compute_repeatability(corners1, corners2, threshold=5):
    """计算重复性"""
    if len(corners1) == 0 or len(corners2) == 0:
        return 0.0

    matched = 0
    for c1 in corners1:
        for c2 in corners2:
            dist = np.sqrt((c1[0] - c2[0])**2 + (c1[1] - c2[1])**2)
            if dist < threshold:
                matched += 1
                break

    return matched / len(corners1)

# 创建测试图像
img = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (150, 150), 200, -1)
cv2.rectangle(img, (200, 80), (350, 220), 200, -1)
cv2.circle(img, (100, 250), 40, 200, -1)

# 原始图像检测
corners_original = detect_corners_shi_tomasi(img)

# 测试不同变换
transformations = []

# 1. 旋转
center = (200, 150)
M_rot = cv2.getRotationMatrix2D(center, 15, 1.0)
img_rotated = cv2.warpAffine(img, M_rot, (400, 300))
corners_rotated = detect_corners_shi_tomasi(img_rotated)
transformations.append(("Rotation 15°", img_rotated, corners_rotated))

# 2. 缩放
img_scaled = cv2.resize(img, None, fx=0.8, fy=0.8)
img_scaled = cv2.copyMakeBorder(img_scaled, 30, 30, 40, 40, cv2.BORDER_CONSTANT)
corners_scaled = detect_corners_shi_tomasi(img_scaled)
transformations.append(("Scale 0.8x", img_scaled, corners_scaled))

# 3. 添加噪声
noise = np.random.normal(0, 10, img.shape).astype(np.int16)
img_noisy = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)
corners_noisy = detect_corners_shi_tomasi(img_noisy)
transformations.append(("Gaussian Noise", img_noisy, corners_noisy))

# 4. 亮度变化
img_bright = cv2.convertScaleAbs(img, alpha=1.2, beta=30)
corners_bright = detect_corners_shi_tomasi(img_bright)
transformations.append(("Brightness +30", img_bright, corners_bright))

# 评估
print("角点检测重复性评估:")
print("-" * 60)
print(f"{'变换':>20} {'检测角点数':>12} {'重复率':>12}")
print("-" * 60)
print(f"{'原始图像':>20} {len(corners_original):>12} {'100.0%':>12}")

for name, _, corners in transformations:
    repeatability = compute_repeatability(corners_original, corners)
    print(f"{name:>20} {len(corners):>12} {repeatability*100:>11.1f}%")

# 可视化
results = []

# 原始图像
canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
for x, y in corners_original:
    cv2.circle(canvas, (x, y), 4, (0, 0, 255), -1)
cv2.putText(canvas, "Original", (10, 25), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
results.append(canvas)

# 变换后的图像
for name, transformed, corners in transformations:
    canvas = cv2.cvtColor(transformed, cv2.COLOR_GRAY2BGR)
    for x, y in corners:
        cv2.circle(canvas, (x, y), 4, (0, 0, 255), -1)
    cv2.putText(canvas, name, (10, 25), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)
    results.append(canvas)

# 组合显示
row1 = np.hstack(results[:3])
row2 = np.hstack(results[3:] + [np.zeros_like(results[0])] * (3 - len(results[3:])))
combined = np.vstack([row1, row2])

cv2.imshow("Repeatability Test", combined)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

## 4. 角点检测方法总结

### 4.1 常用算法对比

| 算法 | 原理 | 优点 | 缺点 |
|------|------|------|------|
| Harris | 自相关矩阵 | 旋转不变 | 对尺度敏感 |
| Shi-Tomasi | 最小特征值 | 效果好 | 对尺度敏感 |
| FAST | 像素比较 | 速度快 | 对噪声敏感 |
| ORB | FAST+方向 | 快速+旋转不变 | 精度一般 |

### 4.2 参数选择建议

| 参数 | 建议值 | 说明 |
|------|--------|------|
| blockSize | 2-7 | 特征尺度 |
| ksize | 3 | Sobel核大小 |
| k (Harris) | 0.04-0.06 | 响应函数参数 |
| threshold | 0.01*max | 角点阈值 |

## 5. 总结

本节介绍了角点检测的理论基础：

| 内容 | 要点 |
|------|------|
| 角点定义 | 多方向灰度变化显著的点 |
| 自相关矩阵 | 描述局部结构的2x2矩阵 |
| 特征值分析 | λ₁、λ₂判断区域类型 |
| Harris响应 | R = det(M) - k·trace(M)² |
| 关键参数 | blockSize、k、threshold |
