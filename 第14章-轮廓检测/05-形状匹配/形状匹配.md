# 形状匹配

## 1. 形状匹配概述

形状匹配是计算机视觉中的重要技术，用于判断两个形状的相似程度。OpenCV提供了多种形状匹配方法，包括基于Hu矩的匹配、模板匹配和轮廓匹配等。

### 1.1 应用场景

1. **物体识别**：识别已知形状的物体
2. **图像检索**：基于形状的图像搜索
3. **质量检测**：对比产品与标准形状
4. **手势识别**：匹配预定义的手势模板
5. **文字识别**：字符形状匹配

### 1.2 匹配方法概览

| 方法 | 函数 | 特点 |
|------|------|------|
| Hu矩匹配 | cv2.matchShapes() | 旋转、缩放、平移不变 |
| 模板匹配 | cv2.matchTemplate() | 像素级匹配 |
| 轮廓匹配 | cv2.matchContours() | 点集匹配 |

## 2. Hu矩匹配

### 2.1 Hu矩原理

Hu矩是7个由归一化中心矩计算得到的不变矩，具有：
- **平移不变性**
- **缩放不变性**
- **旋转不变性**

### 2.2 cv2.matchShapes() 函数

```python
result = cv2.matchShapes(contour1, contour2, method, parameter)
```

**参数说明**：

| 参数 | 说明 |
|------|------|
| contour1 | 第一个轮廓 |
| contour2 | 第二个轮廓 |
| method | 匹配方法 |
| parameter | 保留参数（设为0） |

**匹配方法**：

| 方法 | 公式 | 说明 |
|------|------|------|
| cv2.CONTOURS_MATCH_I1 | $\sum\|1/m_i^A - 1/m_i^B\|$ | 倒数差的和 |
| cv2.CONTOURS_MATCH_I2 | $\sum\|m_i^A - m_i^B\|$ | 直接差的和 |
| cv2.CONTOURS_MATCH_I3 | $\max\|m_i^A - m_i^B\|/\|m_i^A\|$ | 最大相对差 |

```python
"""
示例1：基本形状匹配
使用matchShapes()比较两个轮廓的相似度
"""
import cv2
import numpy as np

def create_shape(shape_type, size=100, angle=0):
    """创建指定形状的图像"""
    img = np.zeros((200, 200), dtype=np.uint8)
    center = (100, 100)

    if shape_type == "circle":
        cv2.circle(img, center, size//2, 255, -1)
    elif shape_type == "square":
        half = size // 2
        pts = np.array([[-half, -half], [half, -half], [half, half], [-half, half]])
        # 旋转
        rad = np.radians(angle)
        rot_matrix = np.array([[np.cos(rad), -np.sin(rad)],
                               [np.sin(rad), np.cos(rad)]])
        pts = (pts @ rot_matrix.T + center).astype(int)
        cv2.fillPoly(img, [pts], 255)
    elif shape_type == "triangle":
        pts = []
        for i in range(3):
            a = np.radians(angle + i * 120 - 90)
            pts.append([int(center[0] + size/2 * np.cos(a)),
                       int(center[1] + size/2 * np.sin(a))])
        cv2.fillPoly(img, [np.array(pts)], 255)
    elif shape_type == "star":
        pts = []
        for i in range(5):
            a_out = np.radians(angle + i * 72 - 90)
            a_in = np.radians(angle + (i + 0.5) * 72 - 90)
            pts.append([int(center[0] + size/2 * np.cos(a_out)),
                       int(center[1] + size/2 * np.sin(a_out))])
            pts.append([int(center[0] + size/4 * np.cos(a_in)),
                       int(center[1] + size/4 * np.sin(a_in))])
        cv2.fillPoly(img, [np.array(pts)], 255)

    return img

def get_contour(img):
    """获取图像的主要轮廓"""
    contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    return contours[0] if contours else None

# 创建参考形状
shapes = ["circle", "square", "triangle", "star"]
reference_contours = {}

for shape in shapes:
    img = create_shape(shape)
    contour = get_contour(img)
    reference_contours[shape] = contour

# 创建测试形状（不同大小和旋转）
test_img = create_shape("square", size=80, angle=30)
test_contour = get_contour(test_img)

# 与所有参考形状比较
print("形状匹配结果 (值越小越相似):")
print("-" * 60)
print(f"{'参考形状':>10} {'I1方法':>12} {'I2方法':>12} {'I3方法':>12}")
print("-" * 60)

methods = [
    ("I1", cv2.CONTOURS_MATCH_I1),
    ("I2", cv2.CONTOURS_MATCH_I2),
    ("I3", cv2.CONTOURS_MATCH_I3)
]

results = []
for shape, ref_contour in reference_contours.items():
    scores = []
    for name, method in methods:
        score = cv2.matchShapes(test_contour, ref_contour, method, 0)
        scores.append(score)

    print(f"{shape:>10} {scores[0]:>12.4f} {scores[1]:>12.4f} {scores[2]:>12.4f}")
    results.append((shape, scores[0]))

# 找到最匹配的形状
best_match = min(results, key=lambda x: x[1])
print(f"\n最佳匹配: {best_match[0]} (I1 score: {best_match[1]:.4f})")

# 可视化
canvas = np.zeros((200, 800, 3), dtype=np.uint8)

for i, shape in enumerate(shapes):
    img = create_shape(shape)
    rgb = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
    canvas[:, i*200:(i+1)*200] = rgb
    cv2.putText(canvas, shape, (i*200 + 10, 20),
               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

# 测试形状
test_rgb = cv2.cvtColor(test_img, cv2.COLOR_GRAY2BGR)
cv2.putText(test_rgb, "Test", (10, 20),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 1)

cv2.imshow("Reference Shapes", canvas)
cv2.imshow("Test Shape", test_rgb)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例2：旋转不变性验证
验证matchShapes对旋转的不变性
"""
import cv2
import numpy as np

def create_rotated_shape(angle, size=100):
    """创建旋转后的正方形"""
    img = np.zeros((200, 200), dtype=np.uint8)
    center = (100, 100)
    half = size // 2

    pts = np.array([[-half, -half], [half, -half], [half, half], [-half, half]], dtype=float)

    # 旋转矩阵
    rad = np.radians(angle)
    rot = np.array([[np.cos(rad), -np.sin(rad)],
                    [np.sin(rad), np.cos(rad)]])
    pts = (pts @ rot.T + center).astype(int)

    cv2.fillPoly(img, [pts], 255)
    return img

# 创建参考形状（0度）
ref_img = create_rotated_shape(0)
ref_contours, _ = cv2.findContours(ref_img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
ref_contour = ref_contours[0]

# 不同旋转角度测试
angles = [0, 15, 30, 45, 60, 90, 120, 180]

print("旋转不变性测试 (正方形):")
print("-" * 50)
print(f"{'角度':>8} {'I1 Score':>12} {'I2 Score':>12} {'I3 Score':>12}")
print("-" * 50)

results = []
for angle in angles:
    test_img = create_rotated_shape(angle)
    test_contours, _ = cv2.findContours(test_img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    test_contour = test_contours[0]

    scores = [
        cv2.matchShapes(ref_contour, test_contour, cv2.CONTOURS_MATCH_I1, 0),
        cv2.matchShapes(ref_contour, test_contour, cv2.CONTOURS_MATCH_I2, 0),
        cv2.matchShapes(ref_contour, test_contour, cv2.CONTOURS_MATCH_I3, 0)
    ]

    print(f"{angle:>8}° {scores[0]:>12.6f} {scores[1]:>12.6f} {scores[2]:>12.6f}")
    results.append((angle, test_img, scores[0]))

print("\n结论: Hu矩匹配对旋转具有不变性！")

# 可视化
rows = 2
cols = 4
cell_size = 150
canvas = np.zeros((rows * cell_size, cols * cell_size, 3), dtype=np.uint8)

for idx, (angle, img, score) in enumerate(results):
    row = idx // cols
    col = idx % cols
    x_off = col * cell_size
    y_off = row * cell_size

    # 缩放图像
    small = cv2.resize(img, (cell_size-20, cell_size-20))
    rgb = cv2.cvtColor(small, cv2.COLOR_GRAY2BGR)

    canvas[y_off+10:y_off+cell_size-10, x_off+10:x_off+cell_size-10] = rgb

    cv2.putText(canvas, f"{angle}deg", (x_off+5, y_off+20),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 255, 0), 1)
    cv2.putText(canvas, f"{score:.4f}", (x_off+5, y_off+cell_size-10),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 255, 0), 1)

cv2.imshow("Rotation Invariance", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例4：形状识别系统

```python
"""
示例4：形状识别系统
使用matchShapes进行多形状识别
"""
import cv2
import numpy as np
from typing import Dict, List, Tuple

class ShapeMatcher:
    """形状匹配器"""

    def __init__(self):
        self.templates: Dict[str, np.ndarray] = {}
        self.threshold = 0.15  # 匹配阈值

    def add_template(self, name: str, contour: np.ndarray):
        """添加模板"""
        self.templates[name] = contour.copy()

    def add_template_from_image(self, name: str, img: np.ndarray):
        """从图像添加模板"""
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img

        contours, _ = cv2.findContours(gray, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        if contours:
            # 取最大轮廓
            largest = max(contours, key=cv2.contourArea)
            self.templates[name] = largest

    def match(self, contour: np.ndarray) -> List[Tuple[str, float]]:
        """匹配轮廓，返回排序后的(名称, 分数)列表"""
        results = []
        for name, template in self.templates.items():
            score = cv2.matchShapes(contour, template, cv2.CONTOURS_MATCH_I1, 0)
            results.append((name, score))

        return sorted(results, key=lambda x: x[1])

    def identify(self, contour: np.ndarray) -> Tuple[str, float]:
        """识别轮廓，返回最佳匹配"""
        results = self.match(contour)
        if results and results[0][1] < self.threshold:
            return results[0]
        return ("Unknown", 1.0)


# 创建模板形状
def create_template_shapes():
    """创建标准模板形状"""
    templates = {}

    # 圆形
    img = np.zeros((100, 100), dtype=np.uint8)
    cv2.circle(img, (50, 50), 40, 255, -1)
    templates["circle"] = img

    # 正方形
    img = np.zeros((100, 100), dtype=np.uint8)
    cv2.rectangle(img, (10, 10), (90, 90), 255, -1)
    templates["square"] = img

    # 三角形
    img = np.zeros((100, 100), dtype=np.uint8)
    pts = np.array([[50, 10], [90, 90], [10, 90]])
    cv2.fillPoly(img, [pts], 255)
    templates["triangle"] = img

    # 五角星
    img = np.zeros((100, 100), dtype=np.uint8)
    pts = []
    for i in range(5):
        a_out = -np.pi/2 + i * 2*np.pi/5
        a_in = -np.pi/2 + (i + 0.5) * 2*np.pi/5
        pts.append([int(50 + 40 * np.cos(a_out)), int(50 + 40 * np.sin(a_out))])
        pts.append([int(50 + 18 * np.cos(a_in)), int(50 + 18 * np.sin(a_in))])
    cv2.fillPoly(img, [np.array(pts)], 255)
    templates["star"] = img

    # 十字形
    img = np.zeros((100, 100), dtype=np.uint8)
    cv2.rectangle(img, (35, 10), (65, 90), 255, -1)
    cv2.rectangle(img, (10, 35), (90, 65), 255, -1)
    templates["cross"] = img

    return templates


# 创建测试场景
def create_test_scene():
    """创建包含多个形状的测试场景"""
    img = np.ones((400, 600, 3), dtype=np.uint8) * 240

    shapes_info = []

    # 圆形 (不同大小和位置)
    cv2.circle(img, (80, 100), 50, (100, 100, 100), -1)
    shapes_info.append(("circle", (80, 100)))

    # 旋转的正方形
    center = (220, 100)
    pts = np.array([[-35, -35], [35, -35], [35, 35], [-35, 35]], dtype=float)
    rad = np.radians(30)
    rot = np.array([[np.cos(rad), -np.sin(rad)], [np.sin(rad), np.cos(rad)]])
    pts = (pts @ rot.T + center).astype(int)
    cv2.fillPoly(img, [pts], (100, 100, 100))
    shapes_info.append(("square", center))

    # 三角形
    pts = np.array([[380, 40], [450, 160], [310, 160]])
    cv2.fillPoly(img, [pts], (100, 100, 100))
    shapes_info.append(("triangle", (380, 100)))

    # 星形
    pts = []
    center = (550, 100)
    for i in range(5):
        a_out = -np.pi/2 + i * 2*np.pi/5
        a_in = -np.pi/2 + (i + 0.5) * 2*np.pi/5
        pts.append([int(center[0] + 50 * np.cos(a_out)), int(center[1] + 50 * np.sin(a_out))])
        pts.append([int(center[0] + 22 * np.cos(a_in)), int(center[1] + 22 * np.sin(a_in))])
    cv2.fillPoly(img, [np.array(pts)], (100, 100, 100))
    shapes_info.append(("star", center))

    # 十字形
    cv2.rectangle(img, (60, 220), (100, 380), (100, 100, 100), -1)
    cv2.rectangle(img, (20, 270), (140, 330), (100, 100, 100), -1)
    shapes_info.append(("cross", (80, 300)))

    # 椭圆（未知形状）
    cv2.ellipse(img, (280, 300), (80, 50), 20, 0, 360, (100, 100, 100), -1)
    shapes_info.append(("unknown", (280, 300)))

    return img, shapes_info


# 主程序
if __name__ == "__main__":
    # 创建匹配器
    matcher = ShapeMatcher()
    matcher.threshold = 0.2

    # 添加模板
    templates = create_template_shapes()
    for name, img in templates.items():
        matcher.add_template_from_image(name, img)

    print("已加载模板:", list(matcher.templates.keys()))

    # 创建测试场景
    test_img, shapes_info = create_test_scene()

    # 检测并识别形状
    gray = cv2.cvtColor(test_img, cv2.COLOR_BGR2GRAY)
    _, binary = cv2.threshold(gray, 150, 255, cv2.THRESH_BINARY_INV)
    contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    # 绘制结果
    result = test_img.copy()

    print("\n形状识别结果:")
    print("-" * 50)
    print(f"{'序号':>4} {'识别结果':>12} {'置信度':>10} {'实际':>10}")
    print("-" * 50)

    # 按位置排序
    sorted_contours = sorted(contours, key=lambda c: (cv2.boundingRect(c)[1] // 200,
                                                        cv2.boundingRect(c)[0]))

    for i, contour in enumerate(sorted_contours):
        area = cv2.contourArea(contour)
        if area < 1000:
            continue

        # 识别
        name, score = matcher.identify(contour)
        confidence = max(0, 1 - score / 0.2) * 100

        # 找对应的实际形状
        x, y, w, h = cv2.boundingRect(contour)
        cx, cy = x + w//2, y + h//2

        actual = "unknown"
        for actual_name, pos in shapes_info:
            if abs(pos[0] - cx) < 100 and abs(pos[1] - cy) < 100:
                actual = actual_name
                break

        print(f"{i:>4} {name:>12} {confidence:>9.1f}% {actual:>10}")

        # 绘制结果
        color = (0, 255, 0) if name == actual else (0, 0, 255)
        cv2.drawContours(result, [contour], 0, color, 2)
        cv2.putText(result, f"{name}", (x, y - 10),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

    cv2.imshow("Shape Recognition", result)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
```

```python
"""
示例5：Hu矩详解
计算并分析Hu矩的各个分量
"""
import cv2
import numpy as np

def analyze_hu_moments(contour, name=""):
    """分析轮廓的Hu矩"""
    # 计算矩
    moments = cv2.moments(contour)

    # 计算Hu矩
    hu = cv2.HuMoments(moments)

    # 对数变换（使值更易比较）
    hu_log = -np.sign(hu) * np.log10(np.abs(hu) + 1e-10)

    return hu.flatten(), hu_log.flatten()

# 创建不同形状
shapes = {}

# 圆形
img = np.zeros((200, 200), dtype=np.uint8)
cv2.circle(img, (100, 100), 60, 255, -1)
shapes["Circle"] = img

# 正方形
img = np.zeros((200, 200), dtype=np.uint8)
cv2.rectangle(img, (40, 40), (160, 160), 255, -1)
shapes["Square"] = img

# 矩形
img = np.zeros((200, 200), dtype=np.uint8)
cv2.rectangle(img, (30, 60), (170, 140), 255, -1)
shapes["Rectangle"] = img

# 三角形
img = np.zeros((200, 200), dtype=np.uint8)
pts = np.array([[100, 30], [170, 170], [30, 170]])
cv2.fillPoly(img, [pts], 255)
shapes["Triangle"] = img

# 椭圆
img = np.zeros((200, 200), dtype=np.uint8)
cv2.ellipse(img, (100, 100), (80, 40), 0, 0, 360, 255, -1)
shapes["Ellipse"] = img

# 分析每个形状
print("Hu矩分析 (对数变换后):")
print("=" * 90)
print(f"{'形状':>12}", end="")
for i in range(7):
    print(f"  Hu{i+1:>6}", end="")
print()
print("-" * 90)

hu_data = {}
for name, img in shapes.items():
    contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    if contours:
        hu, hu_log = analyze_hu_moments(contours[0])
        hu_data[name] = hu_log

        print(f"{name:>12}", end="")
        for h in hu_log:
            print(f"  {h:>7.2f}", end="")
        print()

print("\nHu矩特性说明:")
print("-" * 50)
print("Hu1: 与形状的分散程度相关")
print("Hu2: 与形状的对称性相关")
print("Hu3-Hu4: 与形状的扭曲相关")
print("Hu5-Hu7: 与形状的非对称性相关")

# 可视化
canvas = np.zeros((200, 1000, 3), dtype=np.uint8)
for i, (name, img) in enumerate(shapes.items()):
    rgb = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
    canvas[:, i*200:(i+1)*200] = rgb
    cv2.putText(canvas, name, (i*200 + 10, 20),
               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

cv2.imshow("Shapes", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例6：模板匹配与轮廓匹配结合
综合使用模板匹配和轮廓匹配
"""
import cv2
import numpy as np
```

```python
    def add_gesture(self, name: str, template_img: np.ndarray):
        """添加手势模板"""
```

```python
"""
示例9：轮廓匹配工具类
封装完整的形状匹配功能
"""
import cv2
import numpy as np
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass

@dataclass
class MatchResult:
    """匹配结果"""
    name: str
    score: float
    confidence: float
    contour: np.ndarray

class ContourMatcher:
    """轮廓匹配工具类"""

    def __init__(self, method: int = cv2.CONTOURS_MATCH_I1):
        self.templates: Dict[str, np.ndarray] = {}
        self.method = method
        self.threshold = 0.2

    def add_template(self, name: str, contour: np.ndarray):
        """添加模板轮廓"""
        self.templates[name] = contour.copy()

    def add_template_from_image(self, name: str, img: np.ndarray):
        """从图像添加模板"""
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) if len(img.shape) == 3 else img
        _, binary = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)
        contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

        if contours:
            largest = max(contours, key=cv2.contourArea)
            self.templates[name] = largest

    def match_single(self, contour: np.ndarray, template_name: str) -> Optional[float]:
        """与指定模板匹配"""
        if template_name not in self.templates:
            return None
        return cv2.matchShapes(contour, self.templates[template_name], self.method, 0)

    def match_all(self, contour: np.ndarray) -> List[Tuple[str, float]]:
        """与所有模板匹配"""
        results = []
        for name, template in self.templates.items():
            score = cv2.matchShapes(contour, template, self.method, 0)
            results.append((name, score))
        return sorted(results, key=lambda x: x[1])

    def identify(self, contour: np.ndarray) -> MatchResult:
        """识别轮廓"""
        results = self.match_all(contour)

        if not results:
            return MatchResult("Unknown", 1.0, 0.0, contour)

        best_name, best_score = results[0]
        confidence = max(0, (1 - best_score / self.threshold)) * 100 if best_score < self.threshold else 0

        return MatchResult(
            name=best_name if best_score < self.threshold else "Unknown",
            score=best_score,
            confidence=confidence,
            contour=contour
        )

    def find_in_image(self, img: np.ndarray, min_area: int = 500) -> List[MatchResult]:
        """在图像中查找所有匹配的形状"""
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) if len(img.shape) == 3 else img
        _, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
        contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

        results = []
        for contour in contours:
            if cv2.contourArea(contour) < min_area:
                continue
            result = self.identify(contour)
            results.append(result)

        return results

    def visualize_matches(self, img: np.ndarray, results: List[MatchResult]) -> np.ndarray:
        """可视化匹配结果"""
        canvas = img.copy() if len(img.shape) == 3 else cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

        for result in results:
            color = (0, 255, 0) if result.name != "Unknown" else (0, 0, 255)
            cv2.drawContours(canvas, [result.contour], 0, color, 2)

            x, y, w, h = cv2.boundingRect(result.contour)
            label = f"{result.name} ({result.confidence:.0f}%)"
            cv2.putText(canvas, label, (x, y - 5),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.4, color, 1)

        return canvas

    def get_hu_moments(self, contour: np.ndarray) -> np.ndarray:
        """获取轮廓的Hu矩"""
        moments = cv2.moments(contour)
        hu = cv2.HuMoments(moments)
        return hu.flatten()

    def compare_hu_moments(self, contour1: np.ndarray, contour2: np.ndarray) -> Dict:
        """比较两个轮廓的Hu矩"""
        hu1 = self.get_hu_moments(contour1)
        hu2 = self.get_hu_moments(contour2)

        # 对数变换
        hu1_log = -np.sign(hu1) * np.log10(np.abs(hu1) + 1e-10)
        hu2_log = -np.sign(hu2) * np.log10(np.abs(hu2) + 1e-10)

        return {
            'hu1': hu1_log,
            'hu2': hu2_log,
            'difference': np.abs(hu1_log - hu2_log),
            'match_score': cv2.matchShapes(contour1, contour2, self.method, 0)
        }


# 使用示例
if __name__ == "__main__":
```

```python
        self._add_template("star", img)

    def _add_template(self, name, img):
        """添加模板"""
        contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        if contours:
            self.templates[name] = contours[0]

    def recognize(self, contour):
        """识别单个轮廓"""
        best_match = "Unknown"
        best_score = float('inf')

        for name, template in self.templates.items():
            score = cv2.matchShapes(contour, template, cv2.CONTOURS_MATCH_I1, 0)
            if score < best_score:
                best_score = score
                best_match = name

        if best_score > self.threshold:
            best_match = "Unknown"

        return best_match, best_score

    def process_frame(self, frame):
        """处理单帧图像"""
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        blurred = cv2.GaussianBlur(gray, (5, 5), 0)
        _, binary = cv2.threshold(blurred, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)

        contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

        results = []
        for contour in contours:
            area = cv2.contourArea(contour)
            if area < self.min_area:
                continue

            name, score = self.recognize(contour)
            results.append((contour, name, score))

        return results, binary

    def draw_results(self, frame, results):
        """绘制识别结果"""
        output = frame.copy()

        for contour, name, score in results:
            color = self.colors.get(name, (128, 128, 128))
            cv2.drawContours(output, [contour], 0, color, 2)

            x, y, w, h = cv2.boundingRect(contour)
            confidence = max(0, (1 - score / self.threshold) * 100)
            label = f"{name} ({confidence:.0f}%)"
            cv2.putText(output, label, (x, y - 5),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

        return output


def create_random_scene():
    """创建随机测试场景"""
    frame = np.ones((480, 640, 3), np.uint8) * 240

    # 随机添加形状
    shapes = ['circle', 'square', 'triangle', 'star']
    num_shapes = np.random.randint(3, 7)

    for _ in range(num_shapes):
        shape = np.random.choice(shapes)
        x = np.random.randint(80, 560)
        y = np.random.randint(80, 400)
        size = np.random.randint(30, 60)
        angle = np.random.randint(0, 360)

        if shape == 'circle':
            cv2.circle(frame, (x, y), size, (100, 100, 100), -1)
        elif shape == 'square':
            pts = np.array([[-size, -size], [size, -size], [size, size], [-size, size]], dtype=float)
            rad = np.radians(angle)
            rot = np.array([[np.cos(rad), -np.sin(rad)], [np.sin(rad), np.cos(rad)]])
            pts = (pts @ rot.T + [x, y]).astype(int)
            cv2.fillPoly(frame, [pts], (100, 100, 100))
        elif shape == 'triangle':
            pts = []
            for i in range(3):
                a = np.radians(angle + i * 120)
                pts.append([int(x + size * np.cos(a)), int(y + size * np.sin(a))])
            cv2.fillPoly(frame, [np.array(pts)], (100, 100, 100))
        elif shape == 'star':
            pts = []
            for i in range(5):
                a_out = np.radians(angle + i * 72)
                a_in = np.radians(angle + (i + 0.5) * 72)
                pts.append([int(x + size * np.cos(a_out)), int(y + size * np.sin(a_out))])
                pts.append([int(x + size * 0.4 * np.cos(a_in)), int(y + size * 0.4 * np.sin(a_in))])
            cv2.fillPoly(frame, [np.array(pts)], (100, 100, 100))

    return frame
```

## 4. 匹配方法比较

### 4.1 不同匹配方法的特点

| 方法 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| matchShapes | 旋转缩放不变 | 对噪声敏感 | 标准形状识别 |
| matchTemplate | 精确匹配 | 需要相同尺寸 | 精确定位 |
| Hu矩 | 数学描述完整 | 需要二值化 | 形状分类 |

### 4.2 阈值选择指南

| 场景 | 建议阈值 |
|------|----------|
| 严格匹配 | < 0.05 |
| 一般匹配 | 0.05 - 0.15 |
| 宽松匹配 | 0.15 - 0.3 |
