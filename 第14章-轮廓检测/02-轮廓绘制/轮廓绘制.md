# 轮廓绘制

## 1. 轮廓绘制概述

轮廓绘制是将检测到的轮廓可视化的过程。OpenCV提供了强大的`cv2.drawContours()`函数，可以灵活地绘制轮廓及其填充效果。

### 1.1 cv2.drawContours() 函数

```python
cv2.drawContours(image, contours, contourIdx, color, thickness, lineType, hierarchy, maxLevel, offset)
```

**参数说明**：

| 参数 | 类型 | 说明 |
|------|------|------|
| image | ndarray | 目标图像（会被修改） |
| contours | list | 轮廓列表 |
| contourIdx | int | 轮廓索引，-1表示全部 |
| color | tuple | 轮廓颜色 (B, G, R) |
| thickness | int | 线条粗细，-1表示填充 |
| lineType | int | 线条类型（可选） |
| hierarchy | ndarray | 轮廓层次结构（可选） |
| maxLevel | int | 绘制的最大层级（可选） |
| offset | tuple | 轮廓偏移量（可选） |

### 1.2 常用参数值

**thickness参数**：
- `thickness > 0`：绘制轮廓线
- `thickness = -1`：填充轮廓
- `thickness = cv2.FILLED`：同-1

**lineType参数**：
- `cv2.LINE_4`：4连通线
- `cv2.LINE_8`：8连通线（默认）
- `cv2.LINE_AA`：抗锯齿线

### 示例1：基本轮廓绘制

```python
"""
示例1：基本轮廓绘制
演示drawContours()的基本用法
"""
import cv2
import numpy as np

# 创建测试图像
img = np.zeros((400, 600), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (200, 200), 255, -1)
cv2.circle(img, (400, 125), 75, 255, -1)
cv2.ellipse(img, (150, 320), (80, 40), 0, 0, 360, 255, -1)
cv2.fillPoly(img, [np.array([[400, 250], [500, 280], [480, 380], [350, 350]])], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
print(f"检测到 {len(contours)} 个轮廓")

# 创建彩色画布
canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

# 绘制所有轮廓
cv2.drawContours(canvas, contours, -1, (0, 255, 0), 2)

cv2.imshow("Original", img)
cv2.imshow("Contours", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例2：绘制单个轮廓
使用contourIdx参数指定绘制的轮廓
"""
import cv2
import numpy as np

# 创建测试图像
img = np.zeros((400, 600), dtype=np.uint8)
cv2.circle(img, (150, 200), 80, 255, -1)
cv2.rectangle(img, (300, 100), (450, 300), 255, -1)
cv2.ellipse(img, (550, 200), (40, 80), 0, 0, 360, 255, -1)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 分别绘制每个轮廓
colors = [(0, 255, 0), (255, 0, 0), (0, 0, 255), (255, 255, 0)]

# 方法1：使用索引
canvas1 = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
for i in range(len(contours)):
    color = colors[i % len(colors)]
    cv2.drawContours(canvas1, contours, i, color, 2)

# 方法2：传入单个轮廓列表
canvas2 = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
for i, contour in enumerate(contours):
    color = colors[i % len(colors)]
    cv2.drawContours(canvas2, [contour], 0, color, 2)

# 两种方法效果相同
cv2.imshow("Method 1 - Index", canvas1)
cv2.imshow("Method 2 - Single Contour", canvas2)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例4：不同线条类型
比较LINE_4、LINE_8和LINE_AA的效果

import cv2
import numpy as np

# 创建圆形轮廓
img = np.zeros((300, 300), dtype=np.uint8)
cv2.circle(img, (150, 150), 100, 255, -1)
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 不同线条类型
line_types = {
    'LINE_4': cv2.LINE_4,
    'LINE_8': cv2.LINE_8,
    'LINE_AA': cv2.LINE_AA
}

results = []
for name, line_type in line_types.items():
    canvas = np.zeros((300, 300, 3), dtype=np.uint8)
    cv2.drawContours(canvas, contours, -1, (0, 255, 0), 2, line_type)
    cv2.putText(canvas, name, (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 1)
    results.append(canvas)

# 放大显示差异
zoomed_results = []
for canvas in results:
    # 截取边缘区域
    crop = canvas[40:100, 200:260]
    # 放大
    zoomed = cv2.resize(crop, (180, 180), interpolation=cv2.INTER_NEAREST)
    zoomed_results.append(zoomed)

# 显示对比
combined = np.hstack(results)
zoomed_combined = np.hstack(zoomed_results)

cv2.imshow("Line Types Comparison", combined)
cv2.imshow("Zoomed Edge", zoomed_combined)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例6：彩色填充每个轮廓

```python
"""
示例6：彩色填充每个轮廓
为每个轮廓使用不同的颜色
"""
import cv2
import numpy as np
import random

def generate_colors(n, seed=42):
    """生成n个不同的颜色"""
    random.seed(seed)
    colors = []
    for _ in range(n):
        color = (
            random.randint(50, 255),
            random.randint(50, 255),
            random.randint(50, 255)
        )
        colors.append(color)
    return colors

# 创建测试图像
img = np.zeros((500, 700), dtype=np.uint8)

# 添加多个形状
shapes = [
    ("rectangle", (50, 50, 150, 150)),
    ("circle", (300, 100, 60)),
    ("ellipse", (550, 100, 70, 40, 30)),
    ("rectangle", (50, 250, 200, 180)),
    ("circle", (350, 330, 80)),
    ("polygon", [(500, 220), (620, 280), (600, 400), (480, 380)]),
    ("circle", (150, 420, 50)),
    ("ellipse", (400, 450, 60, 30, -20)),
]

for shape in shapes:
    if shape[0] == "rectangle":
        x, y, w, h = shape[1]
        cv2.rectangle(img, (x, y), (x+w, y+h), 255, -1)
    elif shape[0] == "circle":
        x, y, r = shape[1]
        cv2.circle(img, (x, y), r, 255, -1)
    elif shape[0] == "ellipse":
        x, y, a, b, angle = shape[1]
        cv2.ellipse(img, (x, y), (a, b), angle, 0, 360, 255, -1)
    elif shape[0] == "polygon":
        pts = np.array(shape[1])
        cv2.fillPoly(img, [pts], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 生成颜色
colors = generate_colors(len(contours))

# 创建彩色结果
canvas = np.zeros((500, 700, 3), dtype=np.uint8)

# 为每个轮廓填充不同颜色
for i, contour in enumerate(contours):
    cv2.drawContours(canvas, [contour], 0, colors[i], -1)
    cv2.drawContours(canvas, [contour], 0, (255, 255, 255), 1)  # 白色边框

    # 标注编号
    M = cv2.moments(contour)
    if M["m00"] != 0:
        cx = int(M["m10"] / M["m00"])
        cy = int(M["m01"] / M["m00"])
        cv2.putText(canvas, str(i), (cx-10, cy+5),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 0), 2)

cv2.imshow("Original", img)
cv2.imshow("Colorful Contours", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例8：绘制轮廓的外接形状
绘制外接矩形、最小外接矩形、最小外接圆等
"""
import cv2
import numpy as np

# 创建不规则形状
img = np.zeros((400, 600), dtype=np.uint8)
points = np.array([[100, 50], [200, 80], [250, 180], [200, 280],
                   [100, 300], [50, 200], [80, 120]])
cv2.fillPoly(img, [points], 255)

# 添加另一个形状
cv2.ellipse(img, (450, 200), (100, 60), 30, 0, 360, 255, -1)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 创建结果画布
canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

for contour in contours:
    # 绘制原始轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

    # 1. 直立外接矩形
    x, y, w, h = cv2.boundingRect(contour)
    cv2.rectangle(canvas, (x, y), (x+w, y+h), (255, 0, 0), 2)

    # 2. 最小外接矩形
    rect = cv2.minAreaRect(contour)
    box = cv2.boxPoints(rect)
    box = np.int32(box)
    cv2.drawContours(canvas, [box], 0, (0, 255, 255), 2)

    # 3. 最小外接圆
    (cx, cy), radius = cv2.minEnclosingCircle(contour)
    cv2.circle(canvas, (int(cx), int(cy)), int(radius), (255, 0, 255), 2)

    # 4. 拟合椭圆（需要至少5个点）
    if len(contour) >= 5:
        ellipse = cv2.fitEllipse(contour)
        cv2.ellipse(canvas, ellipse, (0, 128, 255), 2)

# 添加图例
legend_y = 20
cv2.putText(canvas, "Green: Contour", (400, legend_y),
            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
cv2.putText(canvas, "Blue: Bounding Rect", (400, legend_y + 20),
            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 0, 0), 1)
cv2.putText(canvas, "Yellow: Min Area Rect", (400, legend_y + 40),
            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 255), 1)
cv2.putText(canvas, "Magenta: Min Circle", (400, legend_y + 60),
            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 0, 255), 1)
cv2.putText(canvas, "Orange: Fit Ellipse", (400, legend_y + 80),
            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 128, 255), 1)

cv2.imshow("Enclosing Shapes", canvas)
cv2.waitKey(0)
```

### 示例9：绘制轮廓中心和方向

```python
"""
示例9：绘制轮廓中心和方向
使用矩计算质心和主方向
"""
import cv2
import numpy as np
import math

def get_orientation(contour):
    """计算轮廓的主方向角度"""
    moments = cv2.moments(contour)
    if moments["m00"] == 0:
        return 0

    # 计算中心矩
    mu20 = moments["mu20"] / moments["m00"]
    mu02 = moments["mu02"] / moments["m00"]
    mu11 = moments["mu11"] / moments["m00"]

    # 计算方向角
    theta = 0.5 * math.atan2(2 * mu11, mu20 - mu02)
    return theta

# 创建不同方向的形状
img = np.zeros((500, 700), dtype=np.uint8)

# 水平椭圆
cv2.ellipse(img, (150, 150), (80, 40), 0, 0, 360, 255, -1)

# 倾斜椭圆
cv2.ellipse(img, (400, 150), (80, 40), 45, 0, 360, 255, -1)

# 不规则形状
pts1 = np.array([[80, 300], [200, 280], [220, 400], [100, 420]])
cv2.fillPoly(img, [pts1], 255)

# 另一个不规则形状
pts2 = np.array([[350, 280], [500, 300], [550, 350], [520, 450], [380, 430], [330, 350]])
cv2.fillPoly(img, [pts2], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 绘制结果
canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

for contour in contours:
    # 绘制轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

    # 计算质心
    M = cv2.moments(contour)
    if M["m00"] == 0:
        continue

    cx = int(M["m10"] / M["m00"])
    cy = int(M["m01"] / M["m00"])

    # 绘制质心
    cv2.circle(canvas, (cx, cy), 5, (0, 0, 255), -1)

    # 计算方向
    theta = get_orientation(contour)

    # 绘制主轴
    length = 80
    x2 = int(cx + length * math.cos(theta))
    y2 = int(cy + length * math.sin(theta))
    cv2.arrowedLine(canvas, (cx, cy), (x2, y2), (255, 0, 0), 2, tipLength=0.3)

    # 绘制次轴
    x3 = int(cx + length * 0.5 * math.cos(theta + math.pi/2))
    y3 = int(cy + length * 0.5 * math.sin(theta + math.pi/2))
    cv2.arrowedLine(canvas, (cx, cy), (x3, y3), (255, 255, 0), 2, tipLength=0.3)

    # 显示角度
    angle_deg = math.degrees(theta)
    cv2.putText(canvas, f"{angle_deg:.1f} deg", (cx + 10, cy - 10),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

cv2.putText(canvas, "Red dot: Centroid", (10, 30),
            cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 255), 2)
cv2.putText(canvas, "Blue arrow: Major axis", (10, 55),
            cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 0, 0), 2)
cv2.putText(canvas, "Yellow arrow: Minor axis", (10, 80),
            cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 0), 2)

cv2.imshow("Orientation", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例10：轮廓可视化工具类

```python
"""
示例10：轮廓可视化工具类
封装各种轮廓绘制功能
"""
import cv2
import numpy as np
import math
from typing import List, Tuple, Optional
import random

class ContourVisualizer:
    """轮廓可视化工具类"""

    def __init__(self):
        self.colors = self._generate_colors(20)

    def _generate_colors(self, n: int) -> List[Tuple[int, int, int]]:
        """生成颜色列表"""
        random.seed(42)
        return [(random.randint(50, 255),
                 random.randint(50, 255),
                 random.randint(50, 255)) for _ in range(n)]

    def draw_contours(self,
                      img: np.ndarray,
                      contours: List[np.ndarray],
                      color: Tuple[int, int, int] = (0, 255, 0),
                      thickness: int = 2,
                      filled: bool = False) -> np.ndarray:
        """绘制轮廓"""
        result = self._ensure_color(img)
        t = -1 if filled else thickness
        cv2.drawContours(result, contours, -1, color, t)
        return result

    def draw_colorful(self,
                      img: np.ndarray,
                      contours: List[np.ndarray],
                      with_border: bool = True) -> np.ndarray:
        """为每个轮廓使用不同颜色"""
        result = self._ensure_color(img)
        for i, contour in enumerate(contours):
            color = self.colors[i % len(self.colors)]
            cv2.drawContours(result, [contour], 0, color, -1)
            if with_border:
                cv2.drawContours(result, [contour], 0, (255, 255, 255), 1)
        return result

    def draw_with_indices(self,
                          img: np.ndarray,
                          contours: List[np.ndarray],
                          color: Tuple[int, int, int] = (0, 255, 0),
                          font_scale: float = 0.5) -> np.ndarray:
        """绘制轮廓并标注索引"""
        result = self._ensure_color(img)
        cv2.drawContours(result, contours, -1, color, 2)

        for i, contour in enumerate(contours):
            cx, cy = self._get_centroid(contour)
            if cx and cy:
                cv2.putText(result, str(i), (cx-10, cy+5),
                           cv2.FONT_HERSHEY_SIMPLEX, font_scale, (0, 0, 255), 2)
        return result

    def draw_bounding_boxes(self,
                            img: np.ndarray,
                            contours: List[np.ndarray],
                            mode: str = 'upright') -> np.ndarray:
        """绘制外接矩形
        mode: 'upright' 直立矩形, 'rotated' 最小旋转矩形
        """
        result = self._ensure_color(img)

        for contour in contours:
            if mode == 'upright':
                x, y, w, h = cv2.boundingRect(contour)
                cv2.rectangle(result, (x, y), (x+w, y+h), (0, 255, 0), 2)
            elif mode == 'rotated':
                rect = cv2.minAreaRect(contour)
                box = cv2.boxPoints(rect)
                box = np.int32(box)
                cv2.drawContours(result, [box], 0, (0, 255, 0), 2)

        return result

    def draw_enclosing_circles(self,
                               img: np.ndarray,
                               contours: List[np.ndarray]) -> np.ndarray:
        """绘制最小外接圆"""
        result = self._ensure_color(img)

        for contour in contours:
            (x, y), radius = cv2.minEnclosingCircle(contour)
            cv2.circle(result, (int(x), int(y)), int(radius), (0, 255, 0), 2)
            cv2.circle(result, (int(x), int(y)), 3, (0, 0, 255), -1)

        return result

    def draw_centroids(self,
                       img: np.ndarray,
                       contours: List[np.ndarray],
                       radius: int = 5) -> np.ndarray:
        """绘制质心"""
        result = self._ensure_color(img)
        cv2.drawContours(result, contours, -1, (0, 255, 0), 2)

        for contour in contours:
            cx, cy = self._get_centroid(contour)
            if cx and cy:
                cv2.circle(result, (cx, cy), radius, (0, 0, 255), -1)
                cv2.circle(result, (cx, cy), radius + 2, (255, 255, 255), 1)

        return result

    def draw_orientation(self,
                         img: np.ndarray,
                         contours: List[np.ndarray],
                         arrow_length: int = 50) -> np.ndarray:
        """绘制主方向"""
        result = self._ensure_color(img)
        cv2.drawContours(result, contours, -1, (0, 255, 0), 2)

        for contour in contours:
            cx, cy = self._get_centroid(contour)
            if cx is None:
                continue

            theta = self._get_orientation(contour)
            x2 = int(cx + arrow_length * math.cos(theta))
            y2 = int(cy + arrow_length * math.sin(theta))

            cv2.circle(result, (cx, cy), 4, (0, 0, 255), -1)
            cv2.arrowedLine(result, (cx, cy), (x2, y2), (255, 0, 0), 2, tipLength=0.3)

        return result

    def draw_hierarchy(self,
                       img: np.ndarray,
                       contours: List[np.ndarray],
                       hierarchy: np.ndarray) -> np.ndarray:
        """按层级绘制轮廓"""
        result = self._ensure_color(img)
        level_colors = [
            (0, 255, 0),    # Level 0
            (255, 0, 0),    # Level 1
            (0, 0, 255),    # Level 2
            (255, 255, 0),  # Level 3
            (255, 0, 255),  # Level 4
        ]

        for i, contour in enumerate(contours):
            level = self._get_level(i, hierarchy)
            color = level_colors[level % len(level_colors)]
            cv2.drawContours(result, [contour], 0, color, 2)

        return result

    def draw_convex_hulls(self,
                          img: np.ndarray,
                          contours: List[np.ndarray]) -> np.ndarray:
        """绘制凸包"""
        result = self._ensure_color(img)

        for contour in contours:
            # 绘制原轮廓
            cv2.drawContours(result, [contour], 0, (0, 255, 0), 2)
            # 计算并绘制凸包
            hull = cv2.convexHull(contour)
            cv2.drawContours(result, [hull], 0, (0, 0, 255), 2)

        return result

    def _ensure_color(self, img: np.ndarray) -> np.ndarray:
        """确保图像是彩色的"""
        if len(img.shape) == 2:
            return cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
        return img.copy()

    def _get_centroid(self, contour: np.ndarray) -> Tuple[Optional[int], Optional[int]]:
        """获取质心"""
        M = cv2.moments(contour)
        if M["m00"] == 0:
            return None, None
        cx = int(M["m10"] / M["m00"])
        cy = int(M["m01"] / M["m00"])
        return cx, cy

    def _get_orientation(self, contour: np.ndarray) -> float:
        """获取主方向"""
        M = cv2.moments(contour)
        if M["m00"] == 0:
            return 0
        mu20 = M["mu20"] / M["m00"]
        mu02 = M["mu02"] / M["m00"]
        mu11 = M["mu11"] / M["m00"]
        return 0.5 * math.atan2(2 * mu11, mu20 - mu02)

    def _get_level(self, idx: int, hierarchy: np.ndarray) -> int:
        """获取轮廓层级"""
        level = 0
        while hierarchy[0][idx][3] != -1:
            idx = hierarchy[0][idx][3]
            level += 1
        return level
```

```python
"""
示例11：交互式轮廓绘制
鼠标点击选择轮廓并高亮显示
"""
import cv2
import numpy as np

class InteractiveContourViewer:
    """交互式轮廓查看器"""

    def __init__(self, img, contours, hierarchy=None):
        self.img = img
        self.contours = contours
        self.hierarchy = hierarchy
        self.selected = -1
        self.canvas = None
        self.window_name = "Interactive Contour Viewer"

    def find_contour_at_point(self, x, y):
        """找到包含指定点的轮廓"""
        for i, contour in enumerate(self.contours):
            # 检查点是否在轮廓内
            result = cv2.pointPolygonTest(contour, (x, y), False)
            if result >= 0:
                return i
        return -1

    def draw(self):
        """绘制当前状态"""
        self.canvas = cv2.cvtColor(self.img, cv2.COLOR_GRAY2BGR)

        for i, contour in enumerate(self.contours):
            if i == self.selected:
                # 选中的轮廓：填充高亮
                cv2.drawContours(self.canvas, [contour], 0, (0, 200, 0), -1)
                cv2.drawContours(self.canvas, [contour], 0, (0, 255, 0), 3)
            else:
                # 未选中：普通绘制
                cv2.drawContours(self.canvas, [contour], 0, (100, 100, 100), 2)

        # 显示选中轮廓的信息
        if self.selected >= 0:
            contour = self.contours[self.selected]
            area = cv2.contourArea(contour)
            perimeter = cv2.arcLength(contour, True)
            x, y, w, h = cv2.boundingRect(contour)

            info = [
                f"Contour #{self.selected}",
                f"Area: {area:.1f}",
                f"Perimeter: {perimeter:.1f}",
                f"Bounding: {w}x{h}",
            ]

            for j, text in enumerate(info):
                cv2.putText(self.canvas, text, (10, 25 + j * 25),
                           cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2)

        return self.canvas

    def mouse_callback(self, event, x, y, flags, param):
        """鼠标回调函数"""
        if event == cv2.EVENT_LBUTTONDOWN:
            self.selected = self.find_contour_at_point(x, y)
            cv2.imshow(self.window_name, self.draw())

    def run(self):
        """运行交互查看器"""
        cv2.namedWindow(self.window_name)
        cv2.setMouseCallback(self.window_name, self.mouse_callback)

        self.draw()
        cv2.imshow(self.window_name, self.canvas)

        print("点击轮廓查看信息，按任意键退出")
        cv2.waitKey(0)
        cv2.destroyAllWindows()
```

### 3.1 优化绘制性能

```python
"""绘制性能优化技巧"""
import cv2
import numpy as np

# 创建测试数据
img = np.zeros((1000, 1000), dtype=np.uint8)
for i in range(100):
    x, y = np.random.randint(50, 950, 2)
    r = np.random.randint(10, 50)
    cv2.circle(img, (x, y), r, 255, -1)

contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 方法1：一次绘制所有轮廓（最快）
canvas1 = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
cv2.drawContours(canvas1, contours, -1, (0, 255, 0), 2)

# 方法2：使用索引绘制（较快）
canvas2 = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
for i in range(len(contours)):
    cv2.drawContours(canvas2, contours, i, (0, 255, 0), 2)

# 方法3：传入单个轮廓（最慢但最灵活）
canvas3 = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
for contour in contours:
    cv2.drawContours(canvas3, [contour], 0, (0, 255, 0), 2)

print("方法1（-1索引）最适合绘制统一样式的所有轮廓")
print("方法3（单个轮廓）最适合需要不同样式的情况")
```

### 3.2 常见绘制模式

| 需求 | 推荐方法 |
|------|----------|
| 绘制所有轮廓 | `cv2.drawContours(img, contours, -1, color, thickness)` |
| 绘制单个轮廓 | `cv2.drawContours(img, [contour], 0, color, thickness)` |
| 填充轮廓 | `thickness = -1` 或 `cv2.FILLED` |
| 抗锯齿 | `lineType = cv2.LINE_AA` |
| 按层级绘制 | 配合`hierarchy`和`maxLevel`参数 |
