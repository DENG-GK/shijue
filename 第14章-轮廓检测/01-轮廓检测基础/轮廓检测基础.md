# 轮廓检测基础

## 1. 轮廓概述

轮廓是图像中连接具有相同颜色或强度的所有连续点的曲线。轮廓检测是计算机视觉中最重要的基础技术之一，广泛应用于形状分析、目标检测和图像分割等领域。

### 1.1 什么是轮廓

**轮廓的定义**：
- 轮廓是物体边界的表示
- 由一系列连续的点构成
- 描述了物体的外形特征

**轮廓与边缘的区别**：

| 特征 | 边缘 | 轮廓 |
|------|------|------|
| 表示形式 | 像素集合 | 连续曲线 |
| 连通性 | 不一定连通 | 必须连通 |
| 层次关系 | 无 | 有父子关系 |
| 信息量 | 仅位置 | 位置+拓扑 |

### 1.2 轮廓检测的应用

1. **形状分析**：识别几何形状
2. **目标检测**：定位图像中的物体
3. **物体测量**：计算面积、周长
4. **图像分割**：分离前景和背景
5. **手势识别**：识别手势轮廓
6. **工业检测**：产品缺陷检测

## 2. OpenCV轮廓检测函数

### 2.1 cv2.findContours() 函数

```python
contours, hierarchy = cv2.findContours(image, mode, method)
```

**参数说明**：

| 参数 | 说明 |
|------|------|
| image | 输入图像（二值图像） |
| mode | 轮廓检索模式 |
| method | 轮廓近似方法 |
| contours | 检测到的轮廓列表 |
| hierarchy | 轮廓层次结构 |

### 2.2 轮廓检索模式

| 模式 | 说明 |
|------|------|
| cv2.RETR_EXTERNAL | 只检索最外层轮廓 |
| cv2.RETR_LIST | 检索所有轮廓，不建立层次关系 |
| cv2.RETR_CCOMP | 检索所有轮廓，建立两级层次结构 |
| cv2.RETR_TREE | 检索所有轮廓，建立完整层次结构 |

### 2.3 轮廓近似方法

| 方法 | 说明 |
|------|------|
| cv2.CHAIN_APPROX_NONE | 存储所有轮廓点 |
| cv2.CHAIN_APPROX_SIMPLE | 压缩水平、垂直和对角方向的冗余点 |
| cv2.CHAIN_APPROX_TC89_L1 | 应用Teh-Chin链式近似算法 |
| cv2.CHAIN_APPROX_TC89_KCOS | 应用Teh-Chin链式近似算法变体 |

### 示例1：基本轮廓检测

```python
"""
示例1：基本轮廓检测
演示如何使用cv2.findContours()检测图像轮廓
"""
import cv2
import numpy as np

# 创建测试图像
img = np.zeros((400, 400), dtype=np.uint8)

# 绘制一些形状
cv2.rectangle(img, (50, 50), (150, 150), 255, -1)
cv2.circle(img, (250, 100), 50, 255, -1)
cv2.ellipse(img, (100, 300), (60, 30), 0, 0, 360, 255, -1)
cv2.fillPoly(img, [np.array([[250, 250], [350, 250], [300, 350]])], 255)

# 检测轮廓
contours, hierarchy = cv2.findContours(
    img,
    cv2.RETR_EXTERNAL,
    cv2.CHAIN_APPROX_SIMPLE
)

print(f"检测到 {len(contours)} 个轮廓")

# 在彩色图像上绘制轮廓
result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
cv2.drawContours(result, contours, -1, (0, 255, 0), 2)

# 显示结果
cv2.imshow("Original", img)
cv2.imshow("Contours", result)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例2：从真实图像检测轮廓

import cv2
import numpy as np

# 读取图像
img = cv2.imread("image.jpg")
if img is None:
    # 创建示例图像
    img = np.ones((400, 600, 3), dtype=np.uint8) * 240
    cv2.rectangle(img, (50, 50), (200, 200), (0, 100, 200), -1)

    binary,
    cv2.RETR_EXTERNAL,
    cv2.CHAIN_APPROX_SIMPLE
)

print(f"检测到 {len(contours)} 个轮廓")

# 绘制轮廓
```

```python
"""
示例4：轮廓近似方法对比
比较CHAIN_APPROX_NONE和CHAIN_APPROX_SIMPLE的区别

import cv2
import numpy as np

# 创建测试图像
img = np.zeros((400, 600), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (250, 350), 255, -1)
cv2.circle(img, (450, 200), 100, 255, -1)

# 不同近似方法
methods = {
    'CHAIN_APPROX_NONE': cv2.CHAIN_APPROX_NONE,
    'CHAIN_APPROX_SIMPLE': cv2.CHAIN_APPROX_SIMPLE
}

print("不同近似方法的轮廓点数：")
print("-" * 50)

results = []
for name, method in methods.items():
    contours, _ = cv2.findContours(
        img.copy(),
        cv2.RETR_EXTERNAL,
        method
    )

    result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

    for i, contour in enumerate(contours):
        print(f"{name} - 轮廓{i}: {len(contour)} 个点")

            cv2.circle(result, (x, y), 2, (0, 0, 255), -1)

        # 绘制轮廓线
        cv2.drawContours(result, [contour], 0, (0, 255, 0), 1)

    results.append(result)
```

### 示例5：理解轮廓层次结构

```python
"""
示例5：理解轮廓层次结构
详细解析hierarchy数组的含义
"""
import cv2
import numpy as np

# 创建复杂嵌套图像
img = np.zeros((500, 500), dtype=np.uint8)

# 第一组：嵌套的矩形
cv2.rectangle(img, (20, 20), (230, 230), 255, -1)
cv2.rectangle(img, (40, 40), (210, 210), 0, -1)
cv2.rectangle(img, (60, 60), (190, 190), 255, -1)
cv2.rectangle(img, (80, 80), (170, 170), 0, -1)

# 第二组：嵌套的圆
cv2.circle(img, (370, 125), 100, 255, -1)
cv2.circle(img, (370, 125), 70, 0, -1)
cv2.circle(img, (370, 125), 40, 255, -1)

# 第三组：独立形状
cv2.rectangle(img, (50, 300), (150, 450), 255, -1)
cv2.circle(img, (300, 375), 60, 255, -1)
cv2.ellipse(img, (430, 375), (50, 70), 0, 0, 360, 255, -1)

# 使用RETR_TREE检测轮廓
contours, hierarchy = cv2.findContours(
    img,
    cv2.RETR_TREE,
    cv2.CHAIN_APPROX_SIMPLE
)

print("轮廓层次结构分析")
print("=" * 60)
print("\nhierarchy格式: [Next, Previous, First_Child, Parent]")
print("-" * 60)

# 解析层次结构
for i, h in enumerate(hierarchy[0]):
    next_idx, prev_idx, child_idx, parent_idx = h
    print(f"轮廓{i}: Next={next_idx:2d}, Prev={prev_idx:2d}, "
          f"Child={child_idx:2d}, Parent={parent_idx:2d}")

# 可视化不同层级
result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
colors = [
    (0, 255, 0),    # 绿色 - 最外层
    (255, 0, 0),    # 蓝色 - 第二层
    (0, 0, 255),    # 红色 - 第三层
    (255, 255, 0),  # 青色 - 第四层
    (255, 0, 255),  # 洋红 - 第五层
]

def get_contour_level(idx, hierarchy):
    """获取轮廓的层级深度"""
    level = 0
    while hierarchy[0][idx][3] != -1:
        idx = hierarchy[0][idx][3]
        level += 1
    return level

for i, contour in enumerate(contours):
    level = get_contour_level(i, hierarchy)
    color = colors[level % len(colors)]
    cv2.drawContours(result, [contour], 0, color, 2)

    # 标注轮廓编号
    M = cv2.moments(contour)
    if M["m00"] != 0:
        cx = int(M["m10"] / M["m00"])
        cy = int(M["m01"] / M["m00"])
        cv2.putText(result, str(i), (cx-10, cy+5),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 2)

print("\n颜色表示层级：绿色=0级, 蓝色=1级, 红色=2级, 青色=3级")

cv2.imshow("Hierarchy Visualization", result)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例6：遍历轮廓树

```python
"""
示例6：遍历轮廓树
使用递归方式遍历轮廓层次结构
"""
import cv2
import numpy as np

def print_contour_tree(hierarchy, index=0, level=0):
    """递归打印轮廓树结构"""
    if index == -1:
        return

    indent = "  " * level
    next_idx, prev_idx, child_idx, parent_idx = hierarchy[0][index]
    print(f"{indent}├─ 轮廓 {index}")

    # 遍历子轮廓
    if child_idx != -1:
        print_contour_tree(hierarchy, child_idx, level + 1)

    # 遍历同级下一个轮廓
    if next_idx != -1:
        print_contour_tree(hierarchy, next_idx, level)

def get_children(hierarchy, idx):
    """获取指定轮廓的所有子轮廓索引"""
    children = []
    child_idx = hierarchy[0][idx][2]
    while child_idx != -1:
        children.append(child_idx)
        child_idx = hierarchy[0][child_idx][0]
    return children

def get_parent(hierarchy, idx):
    """获取指定轮廓的父轮廓索引"""
    return hierarchy[0][idx][3]

# 创建测试图像
img = np.zeros((400, 400), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (350, 350), 255, -1)
cv2.rectangle(img, (80, 80), (180, 180), 0, -1)
cv2.rectangle(img, (100, 100), (160, 160), 255, -1)
cv2.rectangle(img, (220, 80), (320, 180), 0, -1)
cv2.circle(img, (270, 130), 30, 255, -1)
cv2.rectangle(img, (100, 250), (300, 330), 0, -1)

# 检测轮廓
contours, hierarchy = cv2.findContours(
    img,
    cv2.RETR_TREE,
    cv2.CHAIN_APPROX_SIMPLE
)

print("轮廓树结构:")
print("=" * 40)

# 找到根轮廓（没有父轮廓的）
root_indices = [i for i in range(len(contours)) if hierarchy[0][i][3] == -1]

for root_idx in root_indices:
    print_contour_tree(hierarchy, root_idx)
    print()

# 演示获取特定轮廓的父子关系
print("\n轮廓关系查询:")
print("-" * 40)
for i in range(len(contours)):
    children = get_children(hierarchy, i)
    parent = get_parent(hierarchy, i)
    print(f"轮廓 {i}: 父={parent}, 子={children if children else '无'}")

# 可视化
result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
cv2.drawContours(result, contours, -1, (0, 255, 0), 2)
for i, contour in enumerate(contours):
    M = cv2.moments(contour)
    if M["m00"] != 0:
        cx = int(M["m10"] / M["m00"])
        cy = int(M["m01"] / M["m00"])
        cv2.putText(result, str(i), (cx-5, cy+5),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 2)

cv2.imshow("Contours", result)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例7：筛选轮廓

```python
"""
示例7：筛选轮廓
按面积、周长等条件筛选轮廓
"""
import cv2
import numpy as np

# 创建带多个形状的图像
np.random.seed(42)
img = np.zeros((500, 700), dtype=np.uint8)

# 添加不同大小的形状
shapes = []
# 大形状
cv2.rectangle(img, (50, 50), (200, 200), 255, -1)
cv2.circle(img, (350, 125), 80, 255, -1)
# 中等形状
cv2.rectangle(img, (500, 50), (600, 150), 255, -1)
cv2.circle(img, (100, 350), 50, 255, -1)
# 小形状（噪声）
for _ in range(20):
    x, y = np.random.randint(50, 650), np.random.randint(50, 450)
    size = np.random.randint(5, 20)
    cv2.circle(img, (x, y), size, 255, -1)

# 检测所有轮廓
contours, _ = cv2.findContours(
    img,
    cv2.RETR_EXTERNAL,
    cv2.CHAIN_APPROX_SIMPLE
)

print(f"检测到 {len(contours)} 个轮廓")

# 筛选条件
min_area = 1000
max_area = 50000
min_perimeter = 100

# 筛选轮廓
filtered_contours = []
for contour in contours:
    area = cv2.contourArea(contour)
    perimeter = cv2.arcLength(contour, True)

    if min_area <= area <= max_area and perimeter >= min_perimeter:
        filtered_contours.append(contour)

print(f"筛选后剩余 {len(filtered_contours)} 个轮廓")

# 可视化对比
original_result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
filtered_result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

cv2.drawContours(original_result, contours, -1, (0, 255, 0), 2)
cv2.drawContours(filtered_result, filtered_contours, -1, (0, 255, 0), 2)

# 在筛选结果上标注面积
for contour in filtered_contours:
    area = cv2.contourArea(contour)
    M = cv2.moments(contour)
    if M["m00"] != 0:
        cx = int(M["m10"] / M["m00"])
        cy = int(M["m01"] / M["m00"])
        cv2.putText(filtered_result, f"{int(area)}", (cx-30, cy),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 0, 255), 1)

combined = np.hstack([original_result, filtered_result])
cv2.putText(combined, "All Contours", (10, 30),
            cv2.FONT_HERSHEY_SIMPLEX, 0.8, (255, 255, 255), 2)
cv2.putText(combined, "Filtered", (710, 30),
            cv2.FONT_HERSHEY_SIMPLEX, 0.8, (255, 255, 255), 2)

cv2.imshow("Contour Filtering", combined)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例8：使用Canny边缘检测找轮廓

```python
"""
示例8：使用Canny边缘检测找轮廓
结合Canny边缘检测进行轮廓检测
"""
import cv2
import numpy as np

# 创建或读取图像
img = cv2.imread("image.jpg")
if img is None:
    # 创建示例图像
    img = np.ones((400, 600, 3), dtype=np.uint8) * 200
    cv2.rectangle(img, (50, 50), (200, 200), (50, 100, 150), -1)
    cv2.circle(img, (400, 150), 80, (150, 100, 50), -1)
    cv2.ellipse(img, (150, 300), (80, 40), 20, 0, 360, (100, 150, 50), -1)
    cv2.fillPoly(img, [np.array([[400, 250], [500, 300], [450, 380], [350, 350]])],
                 (50, 150, 100))
    # 添加渐变和噪声
    noise = np.random.normal(0, 10, img.shape).astype(np.int16)
    img = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)

# 转灰度
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 高斯模糊
blurred = cv2.GaussianBlur(gray, (5, 5), 0)

# Canny边缘检测
edges = cv2.Canny(blurred, 50, 150)

# 膨胀操作连接边缘
kernel = np.ones((3, 3), np.uint8)
dilated = cv2.dilate(edges, kernel, iterations=1)

# 检测轮廓
contours, _ = cv2.findContours(
    dilated,
    cv2.RETR_EXTERNAL,
    cv2.CHAIN_APPROX_SIMPLE
)

# 筛选有效轮廓
valid_contours = [c for c in contours if cv2.contourArea(c) > 500]

print(f"Canny检测到 {len(contours)} 个轮廓")
print(f"有效轮廓 {len(valid_contours)} 个")

# 绘制结果
result = img.copy()
cv2.drawContours(result, valid_contours, -1, (0, 255, 0), 2)

# 显示处理过程
cv2.imshow("Original", img)
cv2.imshow("Canny Edges", edges)
cv2.imshow("Dilated", dilated)
cv2.imshow("Contours", result)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例9：自适应阈值轮廓检测

```python
"""
示例9：自适应阈值轮廓检测
处理光照不均匀的图像
"""
import cv2
import numpy as np

# 创建光照不均匀的测试图像
img = np.zeros((400, 600, 3), dtype=np.uint8)

# 创建渐变背景
for i in range(600):
    brightness = int(50 + 150 * i / 600)
    img[:, i] = brightness

# 添加形状
cv2.rectangle(img, (50, 100), (150, 300), (200, 200, 200), -1)
cv2.circle(img, (300, 200), 70, (200, 200, 200), -1)
cv2.rectangle(img, (450, 100), (550, 300), (200, 200, 200), -1)

# 转灰度
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 方法1：普通阈值
_, binary_global = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)
contours_global, _ = cv2.findContours(
    binary_global, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 方法2：Otsu阈值
_, binary_otsu = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
contours_otsu, _ = cv2.findContours(
    binary_otsu, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 方法3：自适应阈值
binary_adaptive = cv2.adaptiveThreshold(
    gray, 255,
    cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
    cv2.THRESH_BINARY,
    21, 5
)
contours_adaptive, _ = cv2.findContours(
    binary_adaptive, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 筛选有效轮廓
contours_adaptive = [c for c in contours_adaptive if cv2.contourArea(c) > 1000]

print(f"普通阈值: {len(contours_global)} 个轮廓")
print(f"Otsu阈值: {len(contours_otsu)} 个轮廓")
print(f"自适应阈值: {len(contours_adaptive)} 个轮廓")

result_global = img.copy()
```

### 示例10：轮廓检测工具类

```python
"""
示例10：轮廓检测工具类
封装轮廓检测的常用功能
"""
import cv2
import numpy as np
from typing import List, Tuple, Optional

class ContourDetector:
    """轮廓检测工具类"""

    def __init__(self,
                 mode: int = cv2.RETR_EXTERNAL,
                 method: int = cv2.CHAIN_APPROX_SIMPLE):
        """
        初始化检测器

        Args:
            mode: 轮廓检索模式
            method: 轮廓近似方法
        """
        self.mode = mode
        self.method = method
        self.contours = None
        self.hierarchy = None

    def preprocess(self,
                   img: np.ndarray,
                   blur_size: int = 5,
                   threshold_method: str = 'otsu',
                   invert: bool = True) -> np.ndarray:
        """
        图像预处理

        Args:
            img: 输入图像
            blur_size: 模糊核大小
            threshold_method: 阈值方法 ('otsu', 'adaptive', 'canny')
            invert: 是否反转二值图像

        Returns:
            二值图像
        """
        # 转灰度
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img.copy()

        # 模糊
        if blur_size > 0:
            gray = cv2.GaussianBlur(gray, (blur_size, blur_size), 0)

        # 二值化
        if threshold_method == 'otsu':
            thresh_type = cv2.THRESH_BINARY_INV if invert else cv2.THRESH_BINARY
            _, binary = cv2.threshold(gray, 0, 255, thresh_type + cv2.THRESH_OTSU)
        elif threshold_method == 'adaptive':
            thresh_type = cv2.THRESH_BINARY_INV if invert else cv2.THRESH_BINARY
            binary = cv2.adaptiveThreshold(
                gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
                thresh_type, 21, 5
            )
        elif threshold_method == 'canny':
            binary = cv2.Canny(gray, 50, 150)
            kernel = np.ones((3, 3), np.uint8)
            binary = cv2.dilate(binary, kernel, iterations=1)
        else:
            raise ValueError(f"未知的阈值方法: {threshold_method}")

        return binary

    def detect(self,
               img: np.ndarray,
               preprocess: bool = True,
               **kwargs) -> Tuple[List[np.ndarray], np.ndarray]:
        """
        检测轮廓

        Args:
            img: 输入图像
            preprocess: 是否进行预处理
            **kwargs: 预处理参数

        Returns:
            轮廓列表和层次结构
        """
        if preprocess:
            binary = self.preprocess(img, **kwargs)
        else:
            binary = img

        self.contours, self.hierarchy = cv2.findContours(
            binary, self.mode, self.method
        )

        return self.contours, self.hierarchy

    def filter_by_area(self,
                       min_area: float = 0,
                       max_area: float = float('inf')) -> List[np.ndarray]:
        """按面积筛选轮廓"""
        if self.contours is None:
            return []

        return [c for c in self.contours
                if min_area <= cv2.contourArea(c) <= max_area]

    def filter_by_perimeter(self,
                           min_perimeter: float = 0,
                           max_perimeter: float = float('inf')) -> List[np.ndarray]:
        """按周长筛选轮廓"""
        if self.contours is None:
            return []

        return [c for c in self.contours
                if min_perimeter <= cv2.arcLength(c, True) <= max_perimeter]

    def filter_by_aspect_ratio(self,
                               min_ratio: float = 0,
                               max_ratio: float = float('inf')) -> List[np.ndarray]:
        """按长宽比筛选轮廓"""
        if self.contours is None:
            return []

        filtered = []
        for c in self.contours:
            x, y, w, h = cv2.boundingRect(c)
            if h == 0:
                continue
            ratio = w / h
            if min_ratio <= ratio <= max_ratio:
                filtered.append(c)

        return filtered

    def sort_contours(self,
                      contours: List[np.ndarray],
                      method: str = 'area',
                      reverse: bool = True) -> List[np.ndarray]:
        """
        排序轮廓

        Args:
            contours: 轮廓列表
            method: 排序方法 ('area', 'perimeter', 'left-right', 'top-bottom')
            reverse: 是否降序
        """
        if method == 'area':
            return sorted(contours, key=cv2.contourArea, reverse=reverse)
        elif method == 'perimeter':
            return sorted(contours, key=lambda c: cv2.arcLength(c, True), reverse=reverse)
        elif method == 'left-right':
            return sorted(contours, key=lambda c: cv2.boundingRect(c)[0], reverse=reverse)
        elif method == 'top-bottom':
            return sorted(contours, key=lambda c: cv2.boundingRect(c)[1], reverse=reverse)
        else:
            return contours

    def draw(self,
             img: np.ndarray,
             contours: Optional[List[np.ndarray]] = None,
             color: Tuple[int, int, int] = (0, 255, 0),
             thickness: int = 2,
             show_index: bool = False) -> np.ndarray:
        """
        绘制轮廓

        Args:
            img: 输入图像
            contours: 轮廓列表（默认使用检测到的所有轮廓）
            color: 轮廓颜色
            thickness: 线条粗细
            show_index: 是否显示轮廓索引

        Returns:
            绘制结果
        """
        if contours is None:
            contours = self.contours

        if contours is None:
            return img.copy()

        result = img.copy()
        if len(result.shape) == 2:
            result = cv2.cvtColor(result, cv2.COLOR_GRAY2BGR)

        cv2.drawContours(result, contours, -1, color, thickness)

        if show_index:
            for i, c in enumerate(contours):
                M = cv2.moments(c)
                if M["m00"] != 0:
                    cx = int(M["m10"] / M["m00"])
                    cy = int(M["m01"] / M["m00"])
                    cv2.putText(result, str(i), (cx-10, cy+5),
                               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 2)

        return result


# 使用示例
if __name__ == "__main__":
    # 创建测试图像
    img = np.ones((400, 600, 3), dtype=np.uint8) * 240
    cv2.rectangle(img, (50, 50), (150, 150), (100, 100, 100), -1)
    cv2.circle(img, (300, 100), 60, (100, 100, 100), -1)
    cv2.rectangle(img, (450, 50), (550, 200), (100, 100, 100), -1)
    cv2.circle(img, (100, 300), 20, (100, 100, 100), -1)  # 小圆
    cv2.circle(img, (300, 300), 70, (100, 100, 100), -1)

    # 创建检测器
    detector = ContourDetector(mode=cv2.RETR_EXTERNAL)

    # 检测轮廓
    contours, hierarchy = detector.detect(img)
    print(f"检测到 {len(contours)} 个轮廓")

    # 筛选大面积轮廓
    large_contours = detector.filter_by_area(min_area=2000)
    print(f"大面积轮廓: {len(large_contours)} 个")

    # 按面积排序
    sorted_contours = detector.sort_contours(large_contours, method='area')

    # 绘制结果
    result = detector.draw(img, sorted_contours, show_index=True)

    cv2.imshow("Original", img)
    cv2.imshow("Detected Contours", result)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
```

## 4. 注意事项

### 4.1 输入图像要求

1. **必须是二值图像**：findContours()需要二值图像作为输入
2. **前景为白色**：轮廓检测查找白色（255）区域
3. **单通道**：输入必须是灰度图像

### 4.2 常见问题

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| 检测不到轮廓 | 图像不是二值的 | 先进行二值化处理 |
| 轮廓过多 | 噪声干扰 | 先进行滤波去噪 |
| 轮廓不完整 | 边缘断裂 | 使用形态学闭运算 |
| 嵌套轮廓丢失 | 使用了RETR_EXTERNAL | 改用RETR_TREE |

### 4.3 性能优化

1. **选择合适的检索模式**：只需外轮廓时使用RETR_EXTERNAL
2. **使用SIMPLE近似**：减少轮廓点数，节省内存
3. **预先筛选**：在处理前过滤小面积轮廓
4. **ROI处理**：只在感兴趣区域进行检测

| 近似方法 | NONE、SIMPLE |
