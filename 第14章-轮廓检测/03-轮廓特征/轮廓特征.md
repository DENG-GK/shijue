# 轮廓特征

## 1. 轮廓特征概述

轮廓特征是描述轮廓几何属性的数值，用于形状分析、目标识别和图像分类。OpenCV提供了丰富的函数来计算各种轮廓特征。

### 1.1 特征分类

| 类别 | 特征 | 说明 |
|------|------|------|
| 基本特征 | 面积、周长 | 最基础的几何属性 |
| 矩特征 | 矩、质心 | 描述形状分布 |
| 形状特征 | 圆度、长宽比 | 描述形状特性 |
| 拓扑特征 | 凸性、缺陷 | 描述形状结构 |

### 1.2 常用特征函数

| 函数 | 功能 |
|------|------|
| cv2.contourArea() | 计算轮廓面积 |
| cv2.arcLength() | 计算轮廓周长 |
| cv2.moments() | 计算轮廓矩 |
| cv2.boundingRect() | 计算外接矩形 |
| cv2.minAreaRect() | 计算最小外接矩形 |
| cv2.minEnclosingCircle() | 计算最小外接圆 |
| cv2.fitEllipse() | 拟合椭圆 |
| cv2.convexHull() | 计算凸包 |
| cv2.isContourConvex() | 判断是否凸轮廓 |

### 示例1：面积和周长

```python
"""
示例1：面积和周长
计算轮廓的面积和周长
"""
import cv2
import numpy as np

# 创建测试图像
img = np.zeros((400, 600), dtype=np.uint8)

# 绘制不同形状
# 正方形 100x100
cv2.rectangle(img, (50, 50), (150, 150), 255, -1)

# 圆形 半径50
cv2.circle(img, (300, 100), 50, 255, -1)

# 矩形 150x80
cv2.rectangle(img, (420, 60), (570, 140), 255, -1)

# 椭圆
cv2.ellipse(img, (100, 280), (60, 40), 0, 0, 360, 255, -1)

# 不规则多边形
pts = np.array([[280, 220], [380, 250], [360, 350], [260, 340], [240, 280]])
cv2.fillPoly(img, [pts], 255)

# 三角形
cv2.fillPoly(img, [np.array([[500, 200], [580, 350], [420, 350]])], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 计算并显示特征
print("轮廓面积和周长:")
print("-" * 50)
print(f"{'编号':>4} {'面积':>12} {'周长':>12} {'面积/周长':>12}")
print("-" * 50)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

for i, contour in enumerate(contours):
    # 计算面积
    area = cv2.contourArea(contour)

    # 计算周长（True表示闭合曲线）
    perimeter = cv2.arcLength(contour, True)

    print(f"{i:>4} {area:>12.1f} {perimeter:>12.1f} {area/perimeter:>12.2f}")

    # 绘制轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

    # 标注信息
    M = cv2.moments(contour)
    if M["m00"] != 0:
        cx = int(M["m10"] / M["m00"])
        cy = int(M["m01"] / M["m00"])
        cv2.putText(canvas, f"A:{int(area)}", (cx-30, cy-5),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 0, 255), 1)
        cv2.putText(canvas, f"P:{int(perimeter)}", (cx-30, cy+10),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 0, 0), 1)

# 理论值对比
print("\n理论值对比:")
print(f"正方形 100x100: 面积=10000, 周长=400")
print(f"圆形 半径50: 面积≈{np.pi*50**2:.1f}, 周长≈{2*np.pi*50:.1f}")
print(f"矩形 150x80: 面积=12000, 周长=460")

cv2.imshow("Contours with Features", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例2：图像矩

```python
"""
示例2：图像矩
计算轮廓的各阶矩
"""
import cv2
import numpy as np

def print_moments(moments, name=""):
    """打印矩信息"""
    print(f"\n{name}的矩:")
    print("-" * 40)

    # 空间矩
    print("空间矩 (Spatial Moments):")
    for i in range(4):
        for j in range(4-i):
            key = f"m{i}{j}"
            if key in moments:
                print(f"  {key} = {moments[key]:.2f}")

    # 中心矩
    print("\n中心矩 (Central Moments):")
    for i in range(4):
        for j in range(4-i):
            key = f"mu{i}{j}"
            if key in moments:
                print(f"  {key} = {moments[key]:.2f}")

    # 归一化中心矩
    print("\nHu矩 (Normalized Central Moments):")
    for i in range(4):
        for j in range(4-i):
            key = f"nu{i}{j}"
            if key in moments:
                print(f"  {key} = {moments[key]:.6f}")

# 创建测试形状
shapes = {}

# 正方形
img_square = np.zeros((200, 200), dtype=np.uint8)
cv2.rectangle(img_square, (50, 50), (150, 150), 255, -1)
shapes["正方形"] = img_square

# 圆形
img_circle = np.zeros((200, 200), dtype=np.uint8)
cv2.circle(img_circle, (100, 100), 50, 255, -1)
shapes["圆形"] = img_circle

# 三角形
img_triangle = np.zeros((200, 200), dtype=np.uint8)
pts = np.array([[100, 30], [170, 170], [30, 170]])
cv2.fillPoly(img_triangle, [pts], 255)
shapes["三角形"] = img_triangle

# 计算并比较矩
results = []
for name, img in shapes.items():
    contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    if contours:
        moments = cv2.moments(contours[0])

        # 计算质心
        cx = int(moments["m10"] / moments["m00"]) if moments["m00"] != 0 else 0
        cy = int(moments["m01"] / moments["m00"]) if moments["m00"] != 0 else 0

        # 计算Hu矩
        hu = cv2.HuMoments(moments)

        results.append({
            "name": name,
            "moments": moments,
            "centroid": (cx, cy),
            "hu": hu.flatten()
        })

        print_moments(moments, name)

# 显示Hu矩对比
print("\n\nHu矩对比 (用于形状匹配):")
print("-" * 60)
print(f"{'形状':>10}", end="")
for i in range(7):
    print(f"  Hu{i+1:>8}", end="")
print()

for result in results:
    print(f"{result['name']:>10}", end="")
    for h in result["hu"]:
        print(f"  {h:>8.2e}", end="")
    print()

# 可视化
canvas = np.zeros((200, 600, 3), dtype=np.uint8)
for i, (name, img) in enumerate(shapes.items()):
    rgb = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
    canvas[:, i*200:(i+1)*200] = rgb

    # 标注质心
    cx, cy = results[i]["centroid"]
    cv2.circle(canvas, (i*200 + cx, cy), 5, (0, 0, 255), -1)
    cv2.putText(canvas, name, (i*200 + 10, 20),
               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

cv2.imshow("Shapes with Centroids", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例3：外接矩形
计算直立外接矩形和最小外接矩形
"""
import cv2
import numpy as np

# 创建旋转的形状
img = np.zeros((400, 600), dtype=np.uint8)

# 旋转的矩形
center = (150, 200)
size = (150, 80)
angle = 30
rect_points = cv2.boxPoints((center, size, angle))
rect_points = np.int32(rect_points)
cv2.fillPoly(img, [rect_points], 255)

# 旋转的椭圆
cv2.ellipse(img, (400, 200), (100, 50), -25, 0, 360, 255, -1)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

print("外接矩形比较:")
print("-" * 70)
print(f"{'轮廓':>6} {'直立矩形面积':>15} {'最小矩形面积':>15} {'面积比':>10} {'旋转角度':>12}")
print("-" * 70)

for i, contour in enumerate(contours):
    # 直立外接矩形（Bounding Rectangle）
    x, y, w, h = cv2.boundingRect(contour)
    upright_area = w * h
    cv2.rectangle(canvas, (x, y), (x+w, y+h), (255, 0, 0), 2)

    # 最小外接矩形（Rotated Rectangle）
    rect = cv2.minAreaRect(contour)
    min_area = rect[1][0] * rect[1][1]
    box = cv2.boxPoints(rect)
    box = np.int32(box)
    cv2.drawContours(canvas, [box], 0, (0, 255, 0), 2)

    # 轮廓面积
    contour_area = cv2.contourArea(contour)

    ratio = upright_area / min_area if min_area > 0 else 0
    print(f"{i:>6} {upright_area:>15.1f} {min_area:>15.1f} {ratio:>10.3f} {rect[2]:>12.1f}°")

    # 标注
    cv2.putText(canvas, f"Upright: {w}x{h}", (x, y-10),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 0, 0), 1)
    cv2.putText(canvas, f"Min: {int(rect[1][0])}x{int(rect[1][1])}", (x, y-25),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 255, 0), 1)

# 图例
cv2.putText(canvas, "Blue: Upright Rect", (10, 30),
           cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 0, 0), 2)
cv2.putText(canvas, "Green: Min Area Rect", (10, 55),
           cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

cv2.imshow("Bounding Rectangles", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
        print(f"{i:>6} {contour_area:>12.1f} ({int(cx):>4},{int(cy):>4}) "
              f"{radius:>8.1f} {'N/A':>15} {'N/A':>15}")

    # 绘制原轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 0, 255), 1)
```

### 示例5：形状因子

```python
"""
示例5：形状因子
计算各种形状因子：圆度、矩形度、长宽比等
"""
import cv2
import numpy as np

def calculate_shape_features(contour):
    """计算形状特征"""
    features = {}

    # 基本特征
    area = cv2.contourArea(contour)
    perimeter = cv2.arcLength(contour, True)
    features["area"] = area
    features["perimeter"] = perimeter

    # 外接矩形
    x, y, w, h = cv2.boundingRect(contour)
    features["bbox"] = (x, y, w, h)
    features["bbox_area"] = w * h

    # 长宽比 (Aspect Ratio)
    features["aspect_ratio"] = w / h if h > 0 else 0

    # 矩形度 (Extent) - 轮廓面积与外接矩形面积之比
    features["extent"] = area / (w * h) if w * h > 0 else 0

    # 圆度 (Circularity) - 4π×面积/周长²，圆为1
    features["circularity"] = 4 * np.pi * area / (perimeter ** 2) if perimeter > 0 else 0

    # 等效直径 (Equivalent Diameter)
    features["equiv_diameter"] = np.sqrt(4 * area / np.pi)

    # 凸包
    hull = cv2.convexHull(contour)
    hull_area = cv2.contourArea(hull)

    # 固实度 (Solidity) - 轮廓面积与凸包面积之比
    features["solidity"] = area / hull_area if hull_area > 0 else 0

    # 凸度 (Convexity) - 凸包周长与轮廓周长之比
    hull_perimeter = cv2.arcLength(hull, True)
    features["convexity"] = hull_perimeter / perimeter if perimeter > 0 else 0

    # 最小外接矩形
    if len(contour) >= 5:
        rect = cv2.minAreaRect(contour)
        min_rect_area = rect[1][0] * rect[1][1]
        features["min_rect_area"] = min_rect_area

        # 矩形度（最小外接矩形）
        features["rectangularity"] = area / min_rect_area if min_rect_area > 0 else 0
    else:
        features["rectangularity"] = 0

    return features

# 创建测试形状
img = np.zeros((500, 800), dtype=np.uint8)

# 圆形 - 圆度应接近1
cv2.circle(img, (100, 100), 60, 255, -1)

# 正方形 - 长宽比=1，圆度≈0.785
cv2.rectangle(img, (200, 40), (320, 160), 255, -1)

# 矩形 - 不同长宽比
cv2.rectangle(img, (380, 40), (550, 160), 255, -1)

# 三角形
cv2.fillPoly(img, [np.array([[650, 40], [750, 160], [550, 160]])], 255)

# 五角星（低固实度）
def draw_star(img, center, outer_r, inner_r):
    pts = []
    for i in range(5):
        angle_out = np.pi/2 + i * 2*np.pi/5
        angle_in = np.pi/2 + (i + 0.5) * 2*np.pi/5
        pts.append([int(center[0] + outer_r * np.cos(angle_out)),
                   int(center[1] - outer_r * np.sin(angle_out))])
        pts.append([int(center[0] + inner_r * np.cos(angle_in)),
                   int(center[1] - inner_r * np.sin(angle_in))])
    cv2.fillPoly(img, [np.array(pts)], 255)

draw_star(img, (100, 300), 70, 30)

# L形（低矩形度）
cv2.rectangle(img, (200, 220), (260, 380), 255, -1)
cv2.rectangle(img, (260, 320), (350, 380), 255, -1)

# 椭圆
cv2.ellipse(img, (480, 300), (100, 50), 0, 0, 360, 255, -1)

# 不规则形状
pts = np.array([[600, 220], [750, 250], [720, 350], [640, 380], [580, 300]])
cv2.fillPoly(img, [pts], 255)

# 检测轮廓并计算特征
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 形状名称（按从左到右、从上到下的顺序）
shape_names = ["圆形", "正方形", "矩形", "三角形", "星形", "L形", "椭圆", "不规则"]

# 按位置排序轮廓
sorted_contours = sorted(contours, key=lambda c: (cv2.boundingRect(c)[1] // 200,
                                                    cv2.boundingRect(c)[0]))

# 打印特征表
print("形状特征分析:")
print("=" * 100)
print(f"{'形状':>8} {'面积':>10} {'圆度':>8} {'矩形度':>8} {'长宽比':>8} "
      f"{'固实度':>8} {'凸度':>8}")
print("-" * 100)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

for i, contour in enumerate(sorted_contours):
    if i >= len(shape_names):
        break

    features = calculate_shape_features(contour)

    print(f"{shape_names[i]:>8} {features['area']:>10.0f} "
          f"{features['circularity']:>8.3f} {features['extent']:>8.3f} "
          f"{features['aspect_ratio']:>8.3f} {features['solidity']:>8.3f} "
          f"{features['convexity']:>8.3f}")

    # 绘制轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

    # 标注形状名称
    x, y, w, h = features["bbox"]
    cv2.putText(canvas, shape_names[i], (x, y-5),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 255, 255), 1)

print("-" * 100)
print("\n说明: 圆度=1表示完美圆, 固实度<1表示有凹陷")

cv2.imshow("Shape Features", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例6：凸包和凸缺陷

```python
"""
示例6：凸包和凸缺陷
计算轮廓的凸包和凸缺陷
"""
import cv2
import numpy as np

# 创建手形图像
img = np.zeros((500, 500), dtype=np.uint8)

# 绘制简化的手掌形状
palm = np.array([[200, 350], [300, 350], [320, 300], [330, 250],
                 [300, 150], [280, 80], [260, 80], [260, 150],  # 中指
                 [240, 80], [220, 80], [220, 150],  # 食指
                 [200, 100], [180, 100], [180, 170],  # 拇指
                 [180, 250], [170, 300]])
cv2.fillPoly(img, [palm], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
contour = contours[0]

# 计算凸包
hull = cv2.convexHull(contour)
hull_indices = cv2.convexHull(contour, returnPoints=False)

# 计算凸缺陷
defects = cv2.convexityDefects(contour, hull_indices)

# 绘制结果
canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

# 绘制原轮廓
cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

# 绘制凸包
cv2.drawContours(canvas, [hull], 0, (255, 0, 0), 2)

# 绘制凸缺陷
print("凸缺陷信息:")
print("-" * 60)
print(f"{'编号':>4} {'起点':>12} {'终点':>12} {'最远点':>12} {'深度':>10}")
print("-" * 60)

if defects is not None:
    for i, defect in enumerate(defects):
        s, e, f, d = defect[0]

        start = tuple(contour[s][0])
        end = tuple(contour[e][0])
        far = tuple(contour[f][0])
        depth = d / 256.0

        print(f"{i:>4} {str(start):>12} {str(end):>12} {str(far):>12} {depth:>10.1f}")

        # 只显示深度较大的缺陷
        if depth > 10:
            # 绘制缺陷点
            cv2.circle(canvas, far, 5, (0, 0, 255), -1)
            # 绘制从凸包到缺陷点的线
            cv2.line(canvas, start, far, (0, 255, 255), 1)
            cv2.line(canvas, end, far, (0, 255, 255), 1)

# 判断凸性
is_convex = cv2.isContourConvex(contour)
print(f"\n轮廓是否为凸轮廓: {is_convex}")
print(f"凸包点数: {len(hull)}")
print(f"轮廓点数: {len(contour)}")
print(f"凸缺陷数: {len(defects) if defects is not None else 0}")

# 图例
cv2.putText(canvas, "Green: Contour", (10, 25),
```

### 示例9：轮廓特征分析器

```python
"""
示例9：轮廓特征分析器
综合计算和展示所有轮廓特征
"""
import cv2
import numpy as np
from dataclasses import dataclass
from typing import Tuple, List, Optional
import math

@dataclass
class ContourFeatures:
    """轮廓特征数据类"""
    # 基本特征
    area: float
    perimeter: float

    # 外接形状
    bounding_rect: Tuple[int, int, int, int]
    min_area_rect: Tuple
    min_enclosing_circle: Tuple[Tuple[float, float], float]

    # 质心
    centroid: Tuple[float, float]

    # 形状因子
    aspect_ratio: float
    extent: float
    circularity: float
    solidity: float
    convexity: float

    # 凸包信息
    convex_hull_area: float
    is_convex: bool

    # 方向
    orientation: float

class ContourAnalyzer:
    """轮廓特征分析器"""

    def analyze(self, contour: np.ndarray) -> ContourFeatures:
        """分析轮廓特征"""
        # 基本特征
        area = cv2.contourArea(contour)
        perimeter = cv2.arcLength(contour, True)

        # 外接形状
        bounding_rect = cv2.boundingRect(contour)
        x, y, w, h = bounding_rect

        min_area_rect = cv2.minAreaRect(contour) if len(contour) >= 5 else None
        min_enclosing_circle = cv2.minEnclosingCircle(contour)

        # 矩和质心
        M = cv2.moments(contour)
        if M["m00"] != 0:
            cx = M["m10"] / M["m00"]
            cy = M["m01"] / M["m00"]
        else:
            cx, cy = 0, 0

        # 形状因子
        aspect_ratio = w / h if h > 0 else 0
        extent = area / (w * h) if w * h > 0 else 0
        circularity = 4 * np.pi * area / (perimeter ** 2) if perimeter > 0 else 0

        # 凸包
        hull = cv2.convexHull(contour)
        hull_area = cv2.contourArea(hull)
        hull_perimeter = cv2.arcLength(hull, True)

        solidity = area / hull_area if hull_area > 0 else 0
        convexity = hull_perimeter / perimeter if perimeter > 0 else 0
        is_convex = cv2.isContourConvex(contour)

        # 方向
        if M["m00"] != 0:
            mu20 = M["mu20"] / M["m00"]
            mu02 = M["mu02"] / M["m00"]
            mu11 = M["mu11"] / M["m00"]
            orientation = 0.5 * math.atan2(2 * mu11, mu20 - mu02)
        else:
            orientation = 0

        return ContourFeatures(
            area=area,
            perimeter=perimeter,
            bounding_rect=bounding_rect,
            min_area_rect=min_area_rect,
            min_enclosing_circle=min_enclosing_circle,
            centroid=(cx, cy),
            aspect_ratio=aspect_ratio,
            extent=extent,
            circularity=circularity,
            solidity=solidity,
            convexity=convexity,
            convex_hull_area=hull_area,
            is_convex=is_convex,
            orientation=orientation
        )

    def classify_shape(self, features: ContourFeatures) -> str:
        """基于特征分类形状"""
        # 使用简单规则分类
        if features.circularity > 0.85:
            return "圆形"
        elif features.circularity > 0.75 and 0.9 < features.aspect_ratio < 1.1:
            return "正方形"
        elif features.extent > 0.9 and features.solidity > 0.95:
            return "矩形"
        elif features.solidity < 0.7:
            return "星形/凹形"
        elif 0.45 < features.extent < 0.55:
            return "三角形"
        else:
            return "不规则"

    def visualize(self,
                  img: np.ndarray,
                  contour: np.ndarray,
                  features: ContourFeatures) -> np.ndarray:
        """可视化轮廓特征"""
        if len(img.shape) == 2:
            canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
        else:
            canvas = img.copy()

        # 绘制轮廓
        cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

        # 绘制外接矩形
        x, y, w, h = features.bounding_rect
        cv2.rectangle(canvas, (x, y), (x+w, y+h), (255, 0, 0), 1)

        # 绘制最小外接圆
        center, radius = features.min_enclosing_circle
        cv2.circle(canvas, (int(center[0]), int(center[1])), int(radius), (255, 0, 255), 1)

        # 绘制质心
        cx, cy = int(features.centroid[0]), int(features.centroid[1])
        cv2.circle(canvas, (cx, cy), 4, (0, 0, 255), -1)

        # 绘制方向
        length = 50
        angle = features.orientation
        x2 = int(cx + length * math.cos(angle))
        y2 = int(cy + length * math.sin(angle))
        cv2.arrowedLine(canvas, (cx, cy), (x2, y2), (0, 255, 255), 2, tipLength=0.3)

        return canvas

    def print_features(self, features: ContourFeatures, name: str = ""):
        """打印特征信息"""
        print(f"\n{name}的特征分析:")
        print("=" * 50)
        print(f"基本特征:")
        print(f"  面积: {features.area:.1f}")
        print(f"  周长: {features.perimeter:.1f}")
        print(f"\n外接形状:")
        print(f"  外接矩形: {features.bounding_rect}")
        print(f"  最小外接圆: 圆心{features.min_enclosing_circle[0]}, 半径{features.min_enclosing_circle[1]:.1f}")
        print(f"\n形状因子:")
        print(f"  长宽比: {features.aspect_ratio:.3f}")
        print(f"  矩形度: {features.extent:.3f}")
        print(f"  圆度: {features.circularity:.3f}")
        print(f"  固实度: {features.solidity:.3f}")
        print(f"  凸度: {features.convexity:.3f}")
        print(f"\n其他:")
        print(f"  质心: ({features.centroid[0]:.1f}, {features.centroid[1]:.1f})")
        print(f"  方向: {math.degrees(features.orientation):.1f}°")
        print(f"  是否凸轮廓: {features.is_convex}")
        print(f"\n形状分类: {self.classify_shape(features)}")


# 使用示例
if __name__ == "__main__":
```

```python
"""
示例10：基于特征的形状检测
使用轮廓特征识别不同的几何形状
"""
import cv2
import numpy as np

class ShapeDetector:
    """几何形状检测器"""

    def detect(self, contour):
        """
        检测轮廓的形状类型

        Returns:
            (shape_name, confidence)
        """
        # 计算特征
        area = cv2.contourArea(contour)
        perimeter = cv2.arcLength(contour, True)
        circularity = 4 * np.pi * area / (perimeter ** 2) if perimeter > 0 else 0

        x, y, w, h = cv2.boundingRect(contour)
        aspect_ratio = w / h if h > 0 else 0
        extent = area / (w * h) if w * h > 0 else 0

        hull = cv2.convexHull(contour)
        hull_area = cv2.contourArea(hull)
        solidity = area / hull_area if hull_area > 0 else 0

        # 多边形近似
        epsilon = 0.02 * perimeter
        approx = cv2.approxPolyDP(contour, epsilon, True)
        vertices = len(approx)

        # 形状判断规则
        if circularity > 0.85:
            return "circle", circularity

        if vertices == 3:
            return "triangle", extent

        if vertices == 4:
            if 0.9 < aspect_ratio < 1.1 and extent > 0.9:
                return "square", extent
            else:
                return "rectangle", extent

        if vertices == 5:
            if solidity < 0.75:
                return "star", 1 - solidity
            else:
                return "pentagon", extent

        if vertices == 6:
            return "hexagon", extent

        if vertices > 6 and circularity > 0.7:
            return "ellipse", circularity

        return "polygon", solidity

    def analyze_image(self, img):
        """分析图像中的所有形状"""
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img

        # 预处理
        blurred = cv2.GaussianBlur(gray, (5, 5), 0)
        _, binary = cv2.threshold(blurred, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

        # 检测轮廓
        contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

        results = []
        for contour in contours:
            area = cv2.contourArea(contour)
            if area < 100:  # 过滤小轮廓
                continue

            shape, confidence = self.detect(contour)
            x, y, w, h = cv2.boundingRect(contour)

            results.append({
                'contour': contour,
                'shape': shape,
                'confidence': confidence,
                'bbox': (x, y, w, h),
                'area': area
            })

        return results

        # 绘制结果
        color = color_map.get(shape, (200, 200, 200))
        cv2.drawContours(canvas, [result['contour']], 0, color, 2)

        # 标注
        x, y, w, h = bbox
        cv2.putText(canvas, shape, (x, y - 5),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

    # 添加图例
    legend_x = 10
    legend_y = 420
    for i, (name, color) in enumerate(color_map.items()):
        row = i // 5
        col = i % 5
        x = legend_x + col * 150
        y = legend_y + row * 25
        cv2.circle(canvas, (x, y), 5, color, -1)
        cv2.putText(canvas, name, (x + 10, y + 5),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.4, color, 1)

    cv2.imshow("Shape Detection", canvas)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
```
