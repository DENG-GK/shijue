# 轮廓近似

## 1. 轮廓近似概述

轮廓近似是将复杂轮廓简化为更少点数的过程。这在形状识别、数据压缩和图形简化等场景中非常有用。

### 1.1 为什么需要轮廓近似

1. **减少数据量**：简化存储和传输
2. **降低计算复杂度**：加速后续处理
3. **形状识别**：提取关键几何特征
4. **噪声过滤**：去除微小的轮廓变化

### 1.2 常用近似方法

| 方法 | 函数 | 特点 |
|------|------|------|
| 多边形近似 | cv2.approxPolyDP() | Douglas-Peucker算法 |
| 凸包 | cv2.convexHull() | 最小凸多边形 |
| 外接矩形 | cv2.boundingRect() | 直立矩形 |
| 最小矩形 | cv2.minAreaRect() | 任意角度矩形 |
| 最小圆 | cv2.minEnclosingCircle() | 包围圆 |
| 拟合椭圆 | cv2.fitEllipse() | 最佳拟合椭圆 |
| 拟合直线 | cv2.fitLine() | 最佳拟合直线 |

## 2. Douglas-Peucker算法

### 2.1 算法原理

Douglas-Peucker算法（也称Ramer-Douglas-Peucker算法）是一种递归的曲线简化算法：

1. 连接曲线的起点和终点
2. 找到距离这条线最远的点
3. 如果最大距离小于阈值ε，则用直线替代
4. 否则，以该点为分界，递归处理两段

### 2.2 cv2.approxPolyDP() 函数

```python
approx = cv2.approxPolyDP(curve, epsilon, closed)
```

**参数说明**：

| 参数 | 说明 |
|------|------|
| curve | 输入轮廓 |
| epsilon | 近似精度（最大距离阈值） |
| closed | 是否闭合曲线 |
| approx | 近似后的轮廓 |

## 3. 代码示例

### 示例1：基本多边形近似

```python
"""
示例1：基本多边形近似
使用approxPolyDP()简化轮廓
"""
import cv2
import numpy as np

# 创建复杂轮廓
img = np.zeros((400, 500), dtype=np.uint8)

# 绘制一个有很多点的圆形
cv2.circle(img, (150, 200), 100, 255, -1)

# 绘制一个不规则形状
pts = np.array([[300, 80], [420, 100], [450, 180], [430, 280],
                [350, 320], [280, 280], [260, 180]])
cv2.fillPoly(img, [pts], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)

print("多边形近似演示:")
print("-" * 50)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

for i, contour in enumerate(contours):
    original_points = len(contour)

    # 计算周长
    perimeter = cv2.arcLength(contour, True)

    # 使用不同的epsilon值
    epsilon = 0.02 * perimeter  # 2%的周长
    approx = cv2.approxPolyDP(contour, epsilon, True)

    print(f"轮廓{i}: 原始点数={original_points}, 近似后={len(approx)}")

    # 绘制原始轮廓（细线）
    cv2.drawContours(canvas, [contour], 0, (100, 100, 100), 1)

    # 绘制近似轮廓（粗线）
    cv2.drawContours(canvas, [approx], 0, (0, 255, 0), 2)

    # 标注顶点
    for point in approx:
        x, y = point[0]
        cv2.circle(canvas, (x, y), 4, (0, 0, 255), -1)

cv2.putText(canvas, "Gray: Original, Green: Approximated", (10, 30),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

cv2.imshow("Polygon Approximation", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例2：不同epsilon值的效果

```python
"""
示例2：不同epsilon值的效果
比较不同近似精度的结果
"""
import cv2
import numpy as np

# 创建复杂形状
img = np.zeros((400, 400), dtype=np.uint8)

# 绘制一个复杂的不规则形状
pts = np.array([[100, 50], [200, 30], [280, 80], [320, 150],
                [300, 250], [250, 320], [150, 350], [80, 300],
                [50, 200], [70, 120]])
cv2.fillPoly(img, [pts], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
contour = contours[0]

perimeter = cv2.arcLength(contour, True)
original_points = len(contour)

print("不同epsilon值的效果:")
print("-" * 50)
print(f"原始轮廓点数: {original_points}")
print("-" * 50)

# 不同的epsilon比例
epsilon_ratios = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]

results = []
for ratio in epsilon_ratios:
    epsilon = ratio * perimeter
    approx = cv2.approxPolyDP(contour, epsilon, True)
    results.append((ratio, len(approx), approx))
    print(f"epsilon = {ratio:.3f} * 周长: {len(approx)} 个点")

# 可视化
fig_rows = 2
fig_cols = 3
cell_size = 200
canvas = np.zeros((fig_rows * cell_size, fig_cols * cell_size, 3), dtype=np.uint8)

for idx, (ratio, points, approx) in enumerate(results):
    row = idx // fig_cols
    col = idx % fig_cols
    x_offset = col * cell_size
    y_offset = row * cell_size

    # 缩放轮廓
    scaled_approx = approx.copy()
    scaled_approx[:, :, 0] = (approx[:, :, 0] * 0.4 + x_offset + 20).astype(int)
    scaled_approx[:, :, 1] = (approx[:, :, 1] * 0.4 + y_offset + 20).astype(int)

    scaled_contour = contour.copy()
    scaled_contour[:, :, 0] = (contour[:, :, 0] * 0.4 + x_offset + 20).astype(int)
    scaled_contour[:, :, 1] = (contour[:, :, 1] * 0.4 + y_offset + 20).astype(int)

    # 绘制原始轮廓
    cv2.drawContours(canvas, [scaled_contour], 0, (50, 50, 50), 1)

    # 绘制近似轮廓
    cv2.drawContours(canvas, [scaled_approx], 0, (0, 255, 0), 2)

    # 绘制顶点
    for point in scaled_approx:
        x, y = point[0]
        cv2.circle(canvas, (x, y), 3, (0, 0, 255), -1)

    # 标注
    cv2.putText(canvas, f"eps={ratio:.3f}", (x_offset + 5, y_offset + 15),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 255, 255), 1)
    cv2.putText(canvas, f"pts={points}", (x_offset + 5, y_offset + 30),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 255, 0), 1)

cv2.imshow("Epsilon Comparison", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例3：使用近似识别几何形状

```python
"""
示例3：使用近似识别几何形状
基于顶点数识别基本几何形状
"""
import cv2
import numpy as np

def detect_shape(contour):
    """基于多边形近似识别形状"""
    perimeter = cv2.arcLength(contour, True)
    approx = cv2.approxPolyDP(contour, 0.04 * perimeter, True)
    vertices = len(approx)

    # 计算圆度
    area = cv2.contourArea(contour)
    circularity = 4 * np.pi * area / (perimeter ** 2) if perimeter > 0 else 0

    if vertices == 3:
        return "Triangle", approx
    elif vertices == 4:
        # 区分正方形和矩形
        x, y, w, h = cv2.boundingRect(approx)
        aspect_ratio = w / float(h)
        if 0.95 <= aspect_ratio <= 1.05:
            return "Square", approx
        else:
            return "Rectangle", approx
    elif vertices == 5:
        return "Pentagon", approx
    elif vertices == 6:
        return "Hexagon", approx
    elif vertices > 6:
        if circularity > 0.8:
            return "Circle", approx
        else:
            return "Ellipse", approx
    else:
        return "Unknown", approx

# 创建测试图像
img = np.ones((500, 800, 3), dtype=np.uint8) * 240

# 绘制不同形状
shapes_info = []

# 三角形
pts = np.array([[100, 50], [180, 170], [20, 170]])
cv2.fillPoly(img, [pts], (100, 100, 100))
shapes_info.append("Triangle")

# 正方形
cv2.rectangle(img, (220, 50), (340, 170), (100, 100, 100), -1)
shapes_info.append("Square")

# 矩形
cv2.rectangle(img, (380, 60), (520, 160), (100, 100, 100), -1)
shapes_info.append("Rectangle")

# 五边形
pts = []
for i in range(5):
    angle = -np.pi/2 + i * 2*np.pi/5
    pts.append([int(620 + 60 * np.cos(angle)), int(110 + 60 * np.sin(angle))])
cv2.fillPoly(img, [np.array(pts)], (100, 100, 100))
shapes_info.append("Pentagon")

# 六边形
pts = []
for i in range(6):
    angle = i * np.pi/3
    pts.append([int(750 + 50 * np.cos(angle)), int(110 + 50 * np.sin(angle))])
cv2.fillPoly(img, [np.array(pts)], (100, 100, 100))
shapes_info.append("Hexagon")

# 圆形
cv2.circle(img, (80, 320), 60, (100, 100, 100), -1)
shapes_info.append("Circle")

# 椭圆
cv2.ellipse(img, (250, 320), (80, 50), 0, 0, 360, (100, 100, 100), -1)
shapes_info.append("Ellipse")

# 检测形状
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
_, binary = cv2.threshold(gray, 150, 255, cv2.THRESH_BINARY_INV)
contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 结果画布
canvas = img.copy()

print("形状识别结果:")
print("-" * 50)
print(f"{'序号':>4} {'识别结果':>12} {'顶点数':>8} {'面积':>10}")
print("-" * 50)

# 按位置排序
sorted_contours = sorted(contours,
                         key=lambda c: (cv2.boundingRect(c)[1] // 200, cv2.boundingRect(c)[0]))

colors = {
    'Triangle': (0, 255, 0),
    'Square': (255, 0, 0),
    'Rectangle': (0, 0, 255),
    'Pentagon': (255, 255, 0),
    'Hexagon': (255, 0, 255),
    'Circle': (0, 255, 255),
    'Ellipse': (128, 255, 128),
    'Unknown': (128, 128, 128)
}

for i, contour in enumerate(sorted_contours):
    area = cv2.contourArea(contour)
    if area < 500:  # 过滤小轮廓
        continue

    shape_name, approx = detect_shape(contour)
    color = colors.get(shape_name, (200, 200, 200))

    print(f"{i:>4} {shape_name:>12} {len(approx):>8} {area:>10.0f}")

    # 绘制轮廓
    cv2.drawContours(canvas, [contour], 0, color, 2)

    # 绘制近似多边形顶点
    for point in approx:
        x, y = point[0]
        cv2.circle(canvas, (x, y), 4, (0, 0, 0), -1)

    # 标注
    x, y, w, h = cv2.boundingRect(contour)
    cv2.putText(canvas, shape_name, (x, y - 5),
               cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

cv2.imshow("Shape Detection", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例4：凸包近似

```python
"""
示例4：凸包近似
使用凸包简化轮廓
"""
import cv2
import numpy as np

# 创建复杂的凹形状
img = np.zeros((400, 500), dtype=np.uint8)

# 星形（有凹陷）
pts = []
for i in range(5):
    angle_out = -np.pi/2 + i * 2*np.pi/5
    angle_in = -np.pi/2 + (i + 0.5) * 2*np.pi/5
    pts.append([int(150 + 100 * np.cos(angle_out)), int(200 + 100 * np.sin(angle_out))])
    pts.append([int(150 + 40 * np.cos(angle_in)), int(200 + 40 * np.sin(angle_in))])
cv2.fillPoly(img, [np.array(pts)], 255)

# L形（凹形）
cv2.rectangle(img, (300, 80), (360, 320), 255, -1)
cv2.rectangle(img, (360, 260), (460, 320), 255, -1)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

print("凸包近似:")
print("-" * 60)
print(f"{'轮廓':>4} {'原始点数':>10} {'凸包点数':>10} {'原始面积':>12} {'凸包面积':>12}")
print("-" * 60)

for i, contour in enumerate(contours):
    # 计算凸包
    hull = cv2.convexHull(contour)

    # 计算面积
    original_area = cv2.contourArea(contour)
    hull_area = cv2.contourArea(hull)

    print(f"{i:>4} {len(contour):>10} {len(hull):>10} "
          f"{original_area:>12.0f} {hull_area:>12.0f}")

    # 绘制原始轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

    # 绘制凸包
    cv2.drawContours(canvas, [hull], 0, (255, 0, 0), 2)

    # 标注凸包顶点
    for point in hull:
        x, y = point[0]
        cv2.circle(canvas, (x, y), 4, (0, 0, 255), -1)

# 图例
cv2.putText(canvas, "Green: Original", (10, 25),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
cv2.putText(canvas, "Blue: Convex Hull", (10, 45),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 0, 0), 1)
cv2.putText(canvas, "Red: Hull vertices", (10, 65),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 1)

cv2.imshow("Convex Hull Approximation", canvas)

cv2.destroyAllWindows()
```

### 示例5：直线拟合

```python
"""
示例5：直线拟合
使用cv2.fitLine()拟合轮廓的最佳直线
"""
import cv2
import numpy as np

# 创建倾斜的形状
img = np.zeros((400, 500), dtype=np.uint8)

# 细长的倾斜矩形1
center1, size1, angle1 = (150, 150), (200, 30), 30
rect1 = cv2.boxPoints((center1, size1, angle1))
cv2.fillPoly(img, [np.int32(rect1)], 255)

# 细长的倾斜矩形2
center2, size2, angle2 = (350, 250), (180, 25), -45
rect2 = cv2.boxPoints((center2, size2, angle2))
cv2.fillPoly(img, [np.int32(rect2)], 255)

# 散点集合（近似直线）
for i in range(50):
    x = 80 + i * 3 + np.random.randint(-5, 5)
    y = 320 + int(0.5 * i) + np.random.randint(-5, 5)
    cv2.circle(img, (x, y), 3, 255, -1)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

print("直线拟合结果:")
print("-" * 60)
print(f"{'轮廓':>4} {'方向向量':>20} {'通过点':>20} {'角度':>10}")
print("-" * 60)

for i, contour in enumerate(contours):
    if len(contour) < 5:
        continue

    # 拟合直线
    # fitLine返回 [vx, vy, x0, y0]
    # vx, vy: 方向向量
    # x0, y0: 直线上的点
    line = cv2.fitLine(contour, cv2.DIST_L2, 0, 0.01, 0.01)
    vx, vy, x0, y0 = line.flatten()

    # 计算角度
    angle = np.degrees(np.arctan2(vy, vx))

    print(f"{i:>4} ({vx:>7.3f}, {vy:>7.3f}) ({x0:>7.1f}, {y0:>7.1f}) {angle:>10.1f}°")

    # 绘制轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

    # 绘制拟合直线（延长到图像边界）
    # 计算直线与图像边界的交点
    lefty = int((-x0 * vy / vx) + y0) if vx != 0 else 0
    righty = int(((500 - x0) * vy / vx) + y0) if vx != 0 else 0

    cv2.line(canvas, (0, lefty), (500, righty), (255, 0, 0), 2)

    # 标记通过点
    cv2.circle(canvas, (int(x0), int(y0)), 5, (0, 0, 255), -1)

cv2.putText(canvas, "Green: Contour, Blue: Fitted Line", (10, 25),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

cv2.imshow("Line Fitting", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例6：椭圆拟合

```python
"""
示例6：椭圆拟合
使用cv2.fitEllipse()拟合轮廓的最佳椭圆
"""
import cv2
import numpy as np

# 创建不同形状的图像
img = np.zeros((400, 600), dtype=np.uint8)

# 椭圆形状
cv2.ellipse(img, (100, 150), (60, 40), 20, 0, 360, 255, -1)

# 圆形
cv2.circle(img, (280, 150), 50, 255, -1)

# 不规则形状（近似椭圆）
pts = np.array([[400, 80], [500, 100], [530, 180], [480, 240],
                [400, 250], [330, 200], [340, 120]])
cv2.fillPoly(img, [pts], 255)

# 长条形
center = (150, 320)
size = (150, 40)
angle = 25
rect = cv2.boxPoints((center, size, angle))
cv2.fillPoly(img, [np.int32(rect)], 255)

# 近似圆形的多边形
pts = []
for i in range(8):
    a = i * 2 * np.pi / 8
    r = 45 + np.random.randint(-5, 5)
    pts.append([int(420 + r * np.cos(a)), int(320 + r * np.sin(a))])
cv2.fillPoly(img, [np.array(pts)], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

print("椭圆拟合结果:")
print("-" * 70)
print(f"{'轮廓':>4} {'中心':>15} {'长轴':>8} {'短轴':>8} {'角度':>8} {'拟合误差':>12}")
print("-" * 70)

for i, contour in enumerate(contours):
    if len(contour) < 5:  # 椭圆拟合需要至少5个点
        continue

    # 拟合椭圆
    ellipse = cv2.fitEllipse(contour)
    center, axes, angle = ellipse

    # 计算拟合误差
    # 使用轮廓面积与椭圆面积的比值
    contour_area = cv2.contourArea(contour)
    ellipse_area = np.pi * axes[0] * axes[1] / 4
    fit_error = abs(contour_area - ellipse_area) / contour_area * 100

    print(f"{i:>4} ({center[0]:>5.1f},{center[1]:>5.1f}) {axes[0]:>8.1f} "
          f"{axes[1]:>8.1f} {angle:>8.1f}° {fit_error:>11.1f}%")

    # 绘制原始轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 255, 0), 2)

    # 绘制拟合椭圆
    cv2.ellipse(canvas, ellipse, (255, 0, 0), 2)

    # 绘制椭圆中心
    cv2.circle(canvas, (int(center[0]), int(center[1])), 4, (0, 0, 255), -1)

    # 绘制主轴
    length = axes[0] / 2
    rad = np.radians(angle)
    x2 = int(center[0] + length * np.cos(rad))
    y2 = int(center[1] + length * np.sin(rad))
    cv2.line(canvas, (int(center[0]), int(center[1])), (x2, y2), (255, 255, 0), 1)

cv2.putText(canvas, "Green: Contour, Blue: Fitted Ellipse", (10, 25),
           cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

cv2.imshow("Ellipse Fitting", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例7：外接矩形近似
比较直立外接矩形和最小外接矩形
"""
import cv2
import numpy as np

# 创建旋转的形状
img = np.zeros((400, 600), dtype=np.uint8)

# 旋转的矩形
for i, (center, size, angle) in enumerate([
    ((100, 150), (120, 60), 30),
    ((280, 150), (100, 50), 60),
    ((450, 150), (150, 40), -20),
]):
    rect = cv2.boxPoints((center, size, angle))
    cv2.fillPoly(img, [np.int32(rect)], 255)

# 旋转的椭圆
cv2.ellipse(img, (100, 320), (80, 40), 45, 0, 360, 255, -1)

# 不规则多边形
pts = np.array([[250, 250], [350, 280], [380, 350], [300, 400], [220, 350]])
cv2.fillPoly(img, [pts], 255)

# 旋转的三角形
pts = np.array([[430, 280], [550, 320], [480, 400]])
cv2.fillPoly(img, [pts], 255)

# 检测轮廓
contours, _ = cv2.findContours(img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

canvas = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

print("外接矩形比较:")
print("-" * 80)
print(f"{'轮廓':>4} {'直立矩形':>25} {'直立面积':>10} {'最小矩形角度':>12} {'最小面积':>10} {'效率':>8}")
print("-" * 80)

for i, contour in enumerate(contours):
    # 直立外接矩形
    x, y, w, h = cv2.boundingRect(contour)
    upright_area = w * h
    cv2.rectangle(canvas, (x, y), (x+w, y+h), (255, 0, 0), 2)

    # 最小外接矩形
    rect = cv2.minAreaRect(contour)
    min_area = rect[1][0] * rect[1][1]
    box = cv2.boxPoints(rect)
    box = np.int32(box)
    cv2.drawContours(canvas, [box], 0, (0, 255, 0), 2)

    # 轮廓面积
    contour_area = cv2.contourArea(contour)

    # 效率 = 轮廓面积 / 最小矩形面积
    efficiency = contour_area / min_area * 100 if min_area > 0 else 0

    print(f"{i:>4} ({x:>3},{y:>3}) {w:>3}x{h:>3} {upright_area:>10.0f} "
          f"{rect[2]:>12.1f}° {min_area:>10.0f} {efficiency:>7.1f}%")

    # 绘制轮廓
    cv2.drawContours(canvas, [contour], 0, (0, 0, 255), 1)

cv2.putText(canvas, "Blue: Upright Rect", (10, 25),
```

### 示例8：轮廓简化工具类

```python
"""
示例8：轮廓简化工具类
封装各种轮廓近似方法
"""
import cv2
import numpy as np
from typing import List, Tuple, Optional
from dataclasses import dataclass

@dataclass
class ApproximationResult:
    """近似结果数据类"""
    method: str
    points: np.ndarray
    point_count: int
    area: float
    compression_ratio: float

class ContourSimplifier:
    """轮廓简化工具类"""

    def __init__(self, contour: np.ndarray):
        self.contour = contour
        self.original_points = len(contour)
        self.original_area = cv2.contourArea(contour)
        self.perimeter = cv2.arcLength(contour, True)

    def polygon_approx(self, epsilon_ratio: float = 0.02) -> ApproximationResult:
        """多边形近似"""
        epsilon = epsilon_ratio * self.perimeter
        approx = cv2.approxPolyDP(self.contour, epsilon, True)

        return ApproximationResult(
            method=f"Polygon (ε={epsilon_ratio})",
            points=approx,
            point_count=len(approx),
            area=cv2.contourArea(approx),
            compression_ratio=len(approx) / self.original_points
        )

    def convex_hull(self) -> ApproximationResult:
        """凸包近似"""
        hull = cv2.convexHull(self.contour)

        return ApproximationResult(
            method="Convex Hull",
            points=hull,
            point_count=len(hull),
            area=cv2.contourArea(hull),
            compression_ratio=len(hull) / self.original_points
        )

    def bounding_rect(self) -> ApproximationResult:
        """直立外接矩形"""
        x, y, w, h = cv2.boundingRect(self.contour)
        points = np.array([[[x, y]], [[x+w, y]], [[x+w, y+h]], [[x, y+h]]])

        return ApproximationResult(
            method="Bounding Rect",
            points=points,
            point_count=4,
            area=w * h,
            compression_ratio=4 / self.original_points
        )

    def min_area_rect(self) -> ApproximationResult:
        """最小外接矩形"""
        rect = cv2.minAreaRect(self.contour)
        box = cv2.boxPoints(rect)
        points = np.int32(box).reshape(-1, 1, 2)

        return ApproximationResult(
            method="Min Area Rect",
            points=points,
            point_count=4,
            area=rect[1][0] * rect[1][1],
            compression_ratio=4 / self.original_points
        )

    def min_enclosing_circle(self) -> Tuple[Tuple[float, float], float]:
        """最小外接圆"""
        return cv2.minEnclosingCircle(self.contour)

    def fit_ellipse(self) -> Optional[Tuple]:
        """拟合椭圆"""
        if len(self.contour) >= 5:
            return cv2.fitEllipse(self.contour)
        return None

    def adaptive_simplify(self, target_points: int) -> ApproximationResult:
        """自适应简化到目标点数"""
        # 二分搜索找到合适的epsilon
        low, high = 0.001, 0.5
        best_approx = None
        best_diff = float('inf')

        while high - low > 0.0001:
            mid = (low + high) / 2
            epsilon = mid * self.perimeter
            approx = cv2.approxPolyDP(self.contour, epsilon, True)

            diff = abs(len(approx) - target_points)
            if diff < best_diff:
                best_diff = diff
                best_approx = approx

            if len(approx) > target_points:
                low = mid
            else:
                high = mid

        return ApproximationResult(
            method=f"Adaptive ({target_points} pts)",
            points=best_approx,
            point_count=len(best_approx),
            area=cv2.contourArea(best_approx),
            compression_ratio=len(best_approx) / self.original_points
        )

    def compare_all(self) -> List[ApproximationResult]:
        """比较所有近似方法"""
        results = [
            self.polygon_approx(0.01),
            self.polygon_approx(0.02),
            self.polygon_approx(0.05),
            self.convex_hull(),
            self.bounding_rect(),
            self.min_area_rect(),
        ]
        return results

    def visualize(self, img: np.ndarray) -> np.ndarray:
        """可视化所有近似方法"""
        results = self.compare_all()

        # 创建大画布
        rows, cols = 2, 3
        cell_h, cell_w = img.shape[:2]
        canvas = np.zeros((rows * cell_h, cols * cell_w, 3), dtype=np.uint8)

        for idx, result in enumerate(results):
            row = idx // cols
            col = idx % cols
            x_off = col * cell_w
            y_off = row * cell_h

            # 复制原图
            cell = cv2.cvtColor(img.copy(), cv2.COLOR_GRAY2BGR)

            # 绘制原轮廓
            cv2.drawContours(cell, [self.contour], 0, (100, 100, 100), 1)

            # 绘制近似结果
            cv2.drawContours(cell, [result.points], 0, (0, 255, 0), 2)

            # 标注顶点
            for point in result.points:
                x, y = point[0]
                cv2.circle(cell, (x, y), 3, (0, 0, 255), -1)

            # 添加标签
            cv2.putText(cell, result.method, (5, 20),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 255, 255), 1)
            cv2.putText(cell, f"Points: {result.point_count}", (5, 40),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 255, 0), 1)
            cv2.putText(cell, f"Ratio: {result.compression_ratio:.2%}", (5, 60),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 255, 0), 1)

            canvas[y_off:y_off+cell_h, x_off:x_off+cell_w] = cell

        return canvas


# 使用示例
if __name__ == "__main__":
    # 创建复杂形状
    img = np.zeros((300, 300), dtype=np.uint8)
    pts = np.array([[50, 100], [150, 30], [250, 80], [280, 180],
```

```python
"""

使用滑块调整epsilon值

import cv2
import numpy as np

class InteractiveApproximation:
    """交互式轮廓近似工具"""

    def __init__(self, img, contour):
        self.img = img
        self.contour = contour
        self.perimeter = cv2.arcLength(contour, True)
        self.window_name = "Interactive Approximation"
        self.epsilon_percent = 2  # 默认2%

    def update(self, val):
        """滑块回调函数"""
        self.epsilon_percent = max(1, val)  # 至少1%

        # 计算近似
        epsilon = self.epsilon_percent / 100.0 * self.perimeter
        approx = cv2.approxPolyDP(self.contour, epsilon, True)

        # 绘制结果
        canvas = cv2.cvtColor(self.img, cv2.COLOR_GRAY2BGR)

        # 原始轮廓
        cv2.drawContours(canvas, [self.contour], 0, (100, 100, 100), 1)

        # 近似轮廓
        cv2.drawContours(canvas, [approx], 0, (0, 255, 0), 2)

        # 顶点
        for point in approx:
            x, y = point[0]
            cv2.circle(canvas, (x, y), 4, (0, 0, 255), -1)

        # 信息显示
        info = [
            f"Epsilon: {self.epsilon_percent}% of perimeter",
            f"Original points: {len(self.contour)}",
            f"Approximated: {len(approx)} points",
            f"Compression: {len(approx)/len(self.contour)*100:.1f}%"
        ]

        for i, text in enumerate(info):
            cv2.putText(canvas, text, (10, 25 + i * 20),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

        """运行交互界面"""
        cv2.namedWindow(self.window_name)
        cv2.createTrackbar("Epsilon %", self.window_name, self.epsilon_percent, 20, self.update)

        self.update(self.epsilon_percent)

        print("拖动滑块调整近似精度，按任意键退出")
        cv2.waitKey(0)
        cv2.destroyAllWindows()
```

```python
"""
示例10：轮廓平滑
使用不同方法平滑轮廓
"""
import cv2
import numpy as np
from scipy import ndimage

def smooth_contour_gaussian(contour, sigma=3):
    """使用高斯滤波平滑轮廓"""
    # 将轮廓点展平
    x = contour[:, 0, 0].astype(float)
    y = contour[:, 0, 1].astype(float)

    # 高斯平滑
    x_smooth = ndimage.gaussian_filter1d(x, sigma, mode='wrap')
    y_smooth = ndimage.gaussian_filter1d(y, sigma, mode='wrap')

    # 重建轮廓
    smoothed = np.zeros_like(contour)
    smoothed[:, 0, 0] = x_smooth.astype(int)
    smoothed[:, 0, 1] = y_smooth.astype(int)

    return smoothed

def smooth_contour_moving_average(contour, window=5):
    """使用移动平均平滑轮廓"""
    x = contour[:, 0, 0].astype(float)
    y = contour[:, 0, 1].astype(float)

    # 移动平均（循环边界）
    kernel = np.ones(window) / window
    x_smooth = np.convolve(np.tile(x, 3), kernel, mode='same')[len(x):2*len(x)]
    y_smooth = np.convolve(np.tile(y, 3), kernel, mode='same')[len(y):2*len(y)]

    smoothed = np.zeros_like(contour)
    smoothed[:, 0, 0] = x_smooth.astype(int)
    smoothed[:, 0, 1] = y_smooth.astype(int)

    return smoothed

def smooth_contour_approx(contour, epsilon_ratio=0.005):
    """使用多边形近似平滑轮廓"""
    perimeter = cv2.arcLength(contour, True)
    epsilon = epsilon_ratio * perimeter
    return cv2.approxPolyDP(contour, epsilon, True)
```

## 4. 近似方法选择

### 4.1 选择指南

| 场景 | 推荐方法 | 说明 |
|------|----------|------|
| 形状识别 | approxPolyDP | 保留几何特征 |
| 边界简化 | approxPolyDP | 控制精度 |
| 凸包围 | convexHull | 忽略凹陷 |
| 定位框 | minAreaRect | 最紧密包围 |
| 圆形物体 | minEnclosingCircle | 快速定位 |
| 椭圆物体 | fitEllipse | 形状拟合 |

### 4.2 epsilon选择技巧

| epsilon值 | 效果 |
|-----------|------|
| 0.001 | 几乎保留原始形状 |
| 0.01 | 轻微简化 |
| 0.02 | 适中简化（推荐起始值） |
| 0.05 | 明显简化 |
| 0.1+ | 激进简化 |

|------|------|

| epsilon参数 | 控制近似精度 |

| 应用场景 | 形状识别、数据简化 |
