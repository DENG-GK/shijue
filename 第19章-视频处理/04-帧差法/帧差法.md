## 一、帧差法原理

### 1.1 基本概念

帧差法（Frame Difference）是一种基于时间差分的运动检测方法：

- **核心思想**：比较相邻帧之间的像素差异
- **基本假设**：静态背景下，运动物体会导致像素值变化
- **检测原理**：差值大于阈值的像素被认为是运动区域

### 1.2 数学表达

**两帧差分：**
$$D(x,y) = |I_t(x,y) - I_{t-1}(x,y)|$$

**二值化：**
$$M(x,y) = \begin{cases} 1 & \text{if } D(x,y) > T \\ 0 & \text{otherwise} \end{cases}$$

其中：
- $I_t$ 是当前帧
- $I_{t-1}$ 是前一帧
- $T$ 是阈值
- $M$ 是运动掩码

### 1.3 帧差法变体

1. **两帧差分**：相邻两帧做差
2. **三帧差分**：使用三帧消除"鬼影"
3. **自适应帧差**：动态调整阈值

## 二、代码示例

### 示例1：基础两帧差分

```python
"""
示例1：基础两帧差分
这是最简单的帧差法，笨蛋先从这里学起！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 读取第一帧
ret, prev_frame = cap.read()
if not ret:
    print("无法读取帧！")
    exit()

# 转换为灰度图
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

# 阈值
threshold = 30

print("两帧差分运动检测")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 转换为灰度图
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算帧差
    frame_diff = cv2.absdiff(gray, prev_gray)

    # 二值化
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    # 显示结果
    cv2.imshow('Original', frame)
    cv2.imshow('Frame Difference', frame_diff)
    cv2.imshow('Motion Mask', motion_mask)

    # 更新前一帧
    prev_gray = gray.copy()

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例2：三帧差分法

```python
"""
示例2：三帧差分法
三帧差分可以消除"鬼影"效果，本小姐来演示一下！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 读取前两帧
ret, frame1 = cap.read()
ret, frame2 = cap.read()

gray1 = cv2.cvtColor(frame1, cv2.COLOR_BGR2GRAY)
gray2 = cv2.cvtColor(frame2, cv2.COLOR_BGR2GRAY)

threshold = 25

print("三帧差分运动检测")
print("按 'q' 退出")

while True:
    ret, frame3 = cap.read()
    if not ret:
        break

    gray3 = cv2.cvtColor(frame3, cv2.COLOR_BGR2GRAY)

    # 计算两次帧差
    diff1 = cv2.absdiff(gray2, gray1)
    diff2 = cv2.absdiff(gray3, gray2)

    # 二值化
    _, thresh1 = cv2.threshold(diff1, threshold, 255, cv2.THRESH_BINARY)
    _, thresh2 = cv2.threshold(diff2, threshold, 255, cv2.THRESH_BINARY)

    # 与操作：只保留两次都检测到运动的区域
    motion_mask = cv2.bitwise_and(thresh1, thresh2)

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    motion_mask = cv2.morphologyEx(motion_mask, cv2.MORPH_CLOSE, kernel)
    motion_mask = cv2.morphologyEx(motion_mask, cv2.MORPH_OPEN, kernel)

    # 在原图上标记运动区域
    result = frame3.copy()
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    for contour in contours:
        if cv2.contourArea(contour) > 500:  # 过滤小区域
            x, y, w, h = cv2.boundingRect(contour)
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    # 显示
    cv2.imshow('Original', frame3)
    cv2.imshow('Motion Mask', motion_mask)
    cv2.imshow('Detection Result', result)

    # 更新帧
    gray1 = gray2.copy()
    gray2 = gray3.copy()

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例3：自适应阈值帧差法

```python
"""
示例3：自适应阈值帧差法
根据场景自动调整阈值，更加智能！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

# 初始阈值和自适应参数
base_threshold = 25
adaptive_factor = 1.5

print("自适应阈值帧差法")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 计算帧差
    frame_diff = cv2.absdiff(gray, prev_gray)

    # 计算自适应阈值（基于帧差的标准差）
    mean_diff = np.mean(frame_diff)
    std_diff = np.std(frame_diff)
    adaptive_threshold = max(base_threshold, mean_diff + adaptive_factor * std_diff)

    # 二值化
    _, motion_mask = cv2.threshold(frame_diff, adaptive_threshold, 255, cv2.THRESH_BINARY)

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    motion_mask = cv2.morphologyEx(motion_mask, cv2.MORPH_CLOSE, kernel)

    # 显示信息
    result = frame.copy()
    cv2.putText(result, f'Threshold: {adaptive_threshold:.1f}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(result, f'Mean Diff: {mean_diff:.1f}', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    # 标记运动区域
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)
    for contour in contours:
        if cv2.contourArea(contour) > 500:
            x, y, w, h = cv2.boundingRect(contour)
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    cv2.imshow('Result', result)
    cv2.imshow('Motion Mask', motion_mask)

    prev_gray = gray.copy()

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例4：高斯模糊预处理

```python
"""
示例4：高斯模糊预处理
模糊处理可以减少噪声干扰，提高检测准确性！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)
# 高斯模糊
prev_gray = cv2.GaussianBlur(prev_gray, (21, 21), 0)

threshold = 25

print("高斯模糊预处理帧差法")
print("按 'b' 切换模糊开关，按 'q' 退出")

use_blur = True

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # 可选高斯模糊
    if use_blur:
        gray_processed = cv2.GaussianBlur(gray, (21, 21), 0)
    else:
        gray_processed = gray

    # 计算帧差
    frame_diff = cv2.absdiff(gray_processed, prev_gray)

    # 二值化
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    # 膨胀填充空洞
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9))
    motion_mask = cv2.dilate(motion_mask, kernel, iterations=2)

    # 标记运动区域
    result = frame.copy()
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    motion_detected = False
    for contour in contours:
        if cv2.contourArea(contour) > 1000:
            motion_detected = True
            x, y, w, h = cv2.boundingRect(contour)
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    # 显示状态
    blur_status = "ON" if use_blur else "OFF"
    cv2.putText(result, f'Blur: {blur_status}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    if motion_detected:
        cv2.putText(result, 'MOTION DETECTED!', (10, 60),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

    cv2.imshow('Result', result)
    cv2.imshow('Motion Mask', motion_mask)

    prev_gray = gray_processed.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('b'):
        use_blur = not use_blur
        print(f"模糊: {blur_status}")

cap.release()
cv2.destroyAllWindows()
```

### 示例5：运动区域追踪

```python
"""
示例5：运动区域追踪
追踪运动物体的轨迹，这可是高级技能呢！
"""
import cv2
import numpy as np
from collections import deque

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)
prev_gray = cv2.GaussianBlur(prev_gray, (21, 21), 0)

# 轨迹点队列
trajectory = deque(maxlen=50)
threshold = 25

print("运动轨迹追踪")
print("按 'c' 清除轨迹，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    gray = cv2.GaussianBlur(gray, (21, 21), 0)

    # 帧差
    frame_diff = cv2.absdiff(gray, prev_gray)
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9))
    motion_mask = cv2.dilate(motion_mask, kernel, iterations=2)

    # 查找轮廓
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    result = frame.copy()

    # 找最大运动区域的中心
    if contours:
        largest_contour = max(contours, key=cv2.contourArea)
        if cv2.contourArea(largest_contour) > 1000:
            # 计算中心点
            M = cv2.moments(largest_contour)
            if M['m00'] > 0:
                cx = int(M['m10'] / M['m00'])
                cy = int(M['m01'] / M['m00'])

                # 添加到轨迹
                trajectory.append((cx, cy))

                # 绘制边界框
                x, y, w, h = cv2.boundingRect(largest_contour)
                cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

                # 绘制中心点
                cv2.circle(result, (cx, cy), 5, (0, 0, 255), -1)

    # 绘制轨迹
    for i in range(1, len(trajectory)):
        thickness = int(np.sqrt(50 / float(i + 1)) * 2.5)
        cv2.line(result, trajectory[i - 1], trajectory[i], (255, 0, 0), thickness)

    cv2.putText(result, f'Trajectory points: {len(trajectory)}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Motion Tracking', result)
    cv2.imshow('Motion Mask', motion_mask)

    prev_gray = gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('c'):
        trajectory.clear()
        print("轨迹已清除")

cap.release()
cv2.destroyAllWindows()
```

### 示例6：运动热力图

```python
"""
示例6：运动热力图
累积运动区域生成热力图，超级炫酷！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)
prev_gray = cv2.GaussianBlur(prev_gray, (21, 21), 0)

# 热力图累积器
frame_height, frame_width = prev_gray.shape
heatmap = np.zeros((frame_height, frame_width), dtype=np.float32)

threshold = 25
decay_factor = 0.95  # 衰减因子

print("运动热力图")
print("按 'r' 重置热力图，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    gray = cv2.GaussianBlur(gray, (21, 21), 0)

    # 帧差
    frame_diff = cv2.absdiff(gray, prev_gray)
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    # 累积到热力图（衰减旧数据，添加新数据）
    heatmap = heatmap * decay_factor + motion_mask.astype(np.float32) * 0.1

    # 归一化热力图
    heatmap_normalized = np.clip(heatmap, 0, 255).astype(np.uint8)

    # 应用颜色映射
    heatmap_color = cv2.applyColorMap(heatmap_normalized, cv2.COLORMAP_JET)

    # 叠加到原图
    alpha = 0.6
    overlay = cv2.addWeighted(frame, 1 - alpha, heatmap_color, alpha, 0)

    cv2.imshow('Original', frame)
    cv2.imshow('Heatmap', heatmap_color)
    cv2.imshow('Overlay', overlay)

    prev_gray = gray.copy()

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        heatmap = np.zeros((frame_height, frame_width), dtype=np.float32)
        print("热力图已重置")

cap.release()
cv2.destroyAllWindows()
```

### 示例7：运动方向分析

```python
"""
示例7：运动方向分析
分析运动物体的移动方向，这可是高级分析技术！
"""
import cv2
import numpy as np
from collections import deque

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)
prev_gray = cv2.GaussianBlur(prev_gray, (21, 21), 0)

# 位置历史
position_history = deque(maxlen=10)
threshold = 25

def get_direction(positions):
    """根据位置历史计算移动方向"""
    if len(positions) < 2:
        return "静止", 0, 0

    dx = positions[-1][0] - positions[0][0]
    dy = positions[-1][1] - positions[0][1]

    # 计算距离
    distance = np.sqrt(dx**2 + dy**2)

    if distance < 10:
        return "静止", 0, 0

    # 确定主方向
    if abs(dx) > abs(dy):
        if dx > 0:
            direction = "向右"
        else:
            direction = "向左"
    else:
        if dy > 0:
            direction = "向下"
        else:
            direction = "向上"

    return direction, dx, dy

print("运动方向分析")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    gray = cv2.GaussianBlur(gray, (21, 21), 0)

    # 帧差
    frame_diff = cv2.absdiff(gray, prev_gray)
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9))
    motion_mask = cv2.dilate(motion_mask, kernel, iterations=2)

    result = frame.copy()

    # 查找运动区域
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    if contours:
        largest_contour = max(contours, key=cv2.contourArea)
        if cv2.contourArea(largest_contour) > 1000:
            M = cv2.moments(largest_contour)
            if M['m00'] > 0:
                cx = int(M['m10'] / M['m00'])
                cy = int(M['m01'] / M['m00'])

                position_history.append((cx, cy))

                # 绘制边界框
                x, y, w, h = cv2.boundingRect(largest_contour)
                cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

                # 分析方向
                direction, dx, dy = get_direction(position_history)

                # 绘制方向箭头
                if direction != "静止":
                    end_x = cx + int(dx * 2)
                    end_y = cy + int(dy * 2)
                    cv2.arrowedLine(result, (cx, cy), (end_x, end_y),
                                    (0, 0, 255), 3, tipLength=0.3)

                cv2.putText(result, f'Direction: {direction}', (10, 30),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
                cv2.putText(result, f'dx: {dx:.0f}, dy: {dy:.0f}', (10, 60),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    else:
        position_history.clear()
        cv2.putText(result, 'No motion detected', (10, 30),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Direction Analysis', result)

    prev_gray = gray.copy()

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例8：运动触发录制

```python
"""
示例8：运动触发录制
检测到运动就开始录制，安防监控必备！
"""
import cv2
import numpy as np
from datetime import datetime
import os

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
fps = 30

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)
prev_gray = cv2.GaussianBlur(prev_gray, (21, 21), 0)

threshold = 25
min_area = 2000

# 录制状态
recording = False
out = None
no_motion_frames = 0
max_no_motion_frames = 90  # 3秒无运动停止录制

# 创建输出目录
output_dir = 'motion_recordings'
os.makedirs(output_dir, exist_ok=True)

print("运动触发录制系统")
print("检测到运动自动开始录制")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    gray = cv2.GaussianBlur(gray, (21, 21), 0)

    # 帧差
    frame_diff = cv2.absdiff(gray, prev_gray)
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9))
    motion_mask = cv2.dilate(motion_mask, kernel, iterations=2)

    # 检测运动
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    motion_detected = False
    for contour in contours:
        if cv2.contourArea(contour) > min_area:
            motion_detected = True
            break

    result = frame.copy()

    if motion_detected:
        no_motion_frames = 0

        if not recording:
            # 开始录制
            timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
            filename = os.path.join(output_dir, f'motion_{timestamp}.avi')
            fourcc = cv2.VideoWriter_fourcc(*'XVID')
            out = cv2.VideoWriter(filename, fourcc, fps, (frame_width, frame_height))
            recording = True
            print(f"运动检测！开始录制：{filename}")

        # 绘制运动区域
        for contour in contours:
            if cv2.contourArea(contour) > min_area:
                x, y, w, h = cv2.boundingRect(contour)
                cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

        cv2.putText(result, 'MOTION DETECTED', (10, 30),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
    else:
        no_motion_frames += 1

    # 录制中
    if recording:
        out.write(frame)
        cv2.circle(result, (frame_width - 30, 30), 10, (0, 0, 255), -1)
        cv2.putText(result, 'REC', (frame_width - 70, 35),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 2)

        # 无运动超时停止录制
        if no_motion_frames > max_no_motion_frames:
            recording = False
            out.release()
            out = None
            print("运动停止，录制结束")

    cv2.imshow('Motion Triggered Recording', result)

    prev_gray = gray.copy()

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

if out:
    out.release()
cap.release()
cv2.destroyAllWindows()
```

### 示例9：多区域运动检测

```python
"""
示例9：多区域运动检测
设置多个检测区域，分别监控！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

ret, prev_frame = cap.read()
frame_height, frame_width = prev_frame.shape[:2]
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)
prev_gray = cv2.GaussianBlur(prev_gray, (21, 21), 0)

# 定义检测区域 (x, y, w, h, name, color)
detection_zones = [
    (0, 0, frame_width // 2, frame_height // 2, "Zone A", (255, 0, 0)),
    (frame_width // 2, 0, frame_width // 2, frame_height // 2, "Zone B", (0, 255, 0)),
    (0, frame_height // 2, frame_width // 2, frame_height // 2, "Zone C", (0, 0, 255)),
    (frame_width // 2, frame_height // 2, frame_width // 2, frame_height // 2, "Zone D", (255, 255, 0)),
]

threshold = 25
min_area = 500

print("多区域运动检测")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    gray = cv2.GaussianBlur(gray, (21, 21), 0)

    # 帧差
    frame_diff = cv2.absdiff(gray, prev_gray)
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    motion_mask = cv2.dilate(motion_mask, kernel, iterations=2)

    result = frame.copy()

    # 检查每个区域
    for x, y, w, h, name, color in detection_zones:
        # 提取区域
        zone_mask = motion_mask[y:y+h, x:x+w]

        # 计算运动像素占比
        motion_pixels = cv2.countNonZero(zone_mask)
        total_pixels = w * h
        motion_ratio = motion_pixels / total_pixels * 100

        # 绘制区域边框
        thickness = 3 if motion_ratio > 1 else 1
        cv2.rectangle(result, (x, y), (x + w, y + h), color, thickness)

        # 显示区域名称和运动比例
        status = "MOTION!" if motion_ratio > 1 else "Clear"
        cv2.putText(result, f'{name}: {motion_ratio:.1f}%', (x + 5, y + 25),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)
        cv2.putText(result, status, (x + 5, y + 45),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

    cv2.imshow('Multi-Zone Detection', result)
    cv2.imshow('Motion Mask', motion_mask)

    prev_gray = gray.copy()

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例10：帧差法参数调优工具

```python
"""
示例10：帧差法参数调优工具
可视化调节参数，找到最佳配置！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

ret, prev_frame = cap.read()
prev_gray = cv2.cvtColor(prev_frame, cv2.COLOR_BGR2GRAY)

# 创建窗口和滑动条
cv2.namedWindow('Settings')

def nothing(x):
    pass

cv2.createTrackbar('Threshold', 'Settings', 25, 100, nothing)
cv2.createTrackbar('Blur Size', 'Settings', 10, 30, nothing)
cv2.createTrackbar('Dilate', 'Settings', 2, 10, nothing)
cv2.createTrackbar('Min Area', 'Settings', 5, 50, nothing)  # *100

print("帧差法参数调优工具")
print("调节滑动条找到最佳参数")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 获取参数
    threshold = cv2.getTrackbarPos('Threshold', 'Settings')
    blur_size = cv2.getTrackbarPos('Blur Size', 'Settings') * 2 + 1
    dilate_iter = cv2.getTrackbarPos('Dilate', 'Settings')
    min_area = cv2.getTrackbarPos('Min Area', 'Settings') * 100

    # 预处理
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    if blur_size > 1:
        gray = cv2.GaussianBlur(gray, (blur_size, blur_size), 0)
        prev_gray_blur = cv2.GaussianBlur(prev_gray, (blur_size, blur_size), 0)
    else:
        prev_gray_blur = prev_gray

    # 帧差
    frame_diff = cv2.absdiff(gray if blur_size > 1 else gray, prev_gray_blur)
    _, motion_mask = cv2.threshold(frame_diff, threshold, 255, cv2.THRESH_BINARY)

    # 形态学处理
    if dilate_iter > 0:
        kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
        motion_mask = cv2.dilate(motion_mask, kernel, iterations=dilate_iter)

    # 检测轮廓
    result = frame.copy()
    contours, _ = cv2.findContours(motion_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    motion_count = 0
    for contour in contours:
        area = cv2.contourArea(contour)
        if area > min_area:
            motion_count += 1
            x, y, w, h = cv2.boundingRect(contour)
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)
            cv2.putText(result, f'{area:.0f}', (x, y - 5),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 255, 0), 1)

    # 显示参数信息
    info = [
        f'Threshold: {threshold}',
        f'Blur: {blur_size}',
        f'Dilate: {dilate_iter}',
        f'Min Area: {min_area}',
        f'Detections: {motion_count}'
    ]

    for i, text in enumerate(info):
        cv2.putText(result, text, (10, 25 + i * 25),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

    cv2.imshow('Settings', result)
    cv2.imshow('Motion Mask', motion_mask)
    cv2.imshow('Frame Difference', frame_diff)

    prev_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

## 三、帧差法优缺点

### 优点
- 算法简单，计算速度快
- 对光照变化不敏感（相对于背景建模）
- 内存占用小
- 实时性好

### 缺点
- 无法检测静止目标
- 快速运动物体可能产生"鬼影"
- 容易受噪声影响
- 无法提取完整的运动目标
