## 一、背景分割原理

### 1.1 基本概念

背景分割（Background Subtraction）是一种前景检测技术：

- **核心思想**：建立背景模型，将当前帧与背景模型比较
- **前景**：与背景模型差异显著的区域
- **背景**：场景中相对静止的部分
- **自适应**：背景模型随时间动态更新

### 1.2 与帧差法的区别

| 特性 | 帧差法 | 背景分割 |
|------|--------|----------|
| 背景模型 | 无 | 有 |
| 静止目标检测 | 不能 | 能 |
| 完整目标提取 | 不完整 | 较完整 |
| 光照适应性 | 较差 | 较好 |
| 计算复杂度 | 低 | 中高 |

### 1.3 OpenCV背景分割器

OpenCV提供多种背景分割算法：

- **MOG2**：基于高斯混合模型，自适应背景建模
- **KNN**：基于K近邻的背景分割
- **GMG**：结合统计方法（需要contrib模块）

## 二、代码示例

### 示例1：MOG2背景分割器

```python
"""
示例1：MOG2背景分割器
这是OpenCV中最常用的背景分割器，本小姐先教你这个！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 创建MOG2背景分割器
# history: 用于建模的历史帧数
# varThreshold: 前景判定阈值
# detectShadows: 是否检测阴影
bg_subtractor = cv2.createBackgroundSubtractorMOG2(
    history=500,
    varThreshold=16,
    detectShadows=True
)

print("MOG2背景分割")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 应用背景分割
    fg_mask = bg_subtractor.apply(frame)

    # 获取背景图像
    background = bg_subtractor.getBackgroundImage()

    # 形态学处理去噪
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    fg_mask_clean = cv2.morphologyEx(fg_mask, cv2.MORPH_OPEN, kernel)
    fg_mask_clean = cv2.morphologyEx(fg_mask_clean, cv2.MORPH_CLOSE, kernel)

    # 提取前景
    foreground = cv2.bitwise_and(frame, frame, mask=fg_mask_clean)

    # 显示结果
    cv2.imshow('Original', frame)
    cv2.imshow('FG Mask', fg_mask)
    cv2.imshow('FG Mask Clean', fg_mask_clean)
    cv2.imshow('Foreground', foreground)
    if background is not None:
        cv2.imshow('Background', background)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例2：KNN背景分割器

```python
"""
示例2：KNN背景分割器
基于K近邻的方法，对复杂背景效果更好！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 创建KNN背景分割器
bg_subtractor = cv2.createBackgroundSubtractorKNN(
    history=500,
    dist2Threshold=400.0,
    detectShadows=True
)

print("KNN背景分割")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 应用背景分割
    fg_mask = bg_subtractor.apply(frame)

    # 获取背景
    background = bg_subtractor.getBackgroundImage()

    # 去除阴影（阴影像素值为127）
    # 255为前景，127为阴影，0为背景
    fg_mask_no_shadow = cv2.threshold(fg_mask, 200, 255, cv2.THRESH_BINARY)[1]

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    fg_mask_clean = cv2.morphologyEx(fg_mask_no_shadow, cv2.MORPH_OPEN, kernel)
    fg_mask_clean = cv2.morphologyEx(fg_mask_clean, cv2.MORPH_CLOSE, kernel)

    # 在原图上标记前景
    result = frame.copy()
    contours, _ = cv2.findContours(fg_mask_clean, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    for contour in contours:
        if cv2.contourArea(contour) > 500:
            x, y, w, h = cv2.boundingRect(contour)
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    cv2.imshow('Original', frame)
    cv2.imshow('FG Mask (with shadow)', fg_mask)
    cv2.imshow('FG Mask (no shadow)', fg_mask_clean)
    cv2.imshow('Detection Result', result)
    if background is not None:
        cv2.imshow('Background Model', background)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

```python
"""
示例4：学习率控制
学习率决定背景更新速度，很重要的参数！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

bg_subtractor = cv2.createBackgroundSubtractorMOG2(detectShadows=False)

# 学习率模式
learning_modes = [
    ('Auto', -1),          # 自动学习率
    ('Fast', 0.1),         # 快速学习
    ('Normal', 0.01),      # 正常学习
    ('Slow', 0.001),       # 慢速学习
    ('Frozen', 0),         # 冻结（不更新）
]
current_mode = 0

print("学习率控制演示")
print("按 'l' 切换学习率模式")
print("按 'r' 重置背景模型")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    mode_name, learning_rate = learning_modes[current_mode]

    # 应用背景分割
    fg_mask = bg_subtractor.apply(frame, learningRate=learning_rate)

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    fg_mask_clean = cv2.morphologyEx(fg_mask, cv2.MORPH_OPEN, kernel)

    # 获取背景
    background = bg_subtractor.getBackgroundImage()

    # 显示信息
    result = frame.copy()
    cv2.putText(result, f'Learning Mode: {mode_name}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(result, f'Learning Rate: {learning_rate}', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Original', result)
    cv2.imshow('FG Mask', fg_mask_clean)
    if background is not None:
        cv2.imshow('Background Model', background)

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('l'):
        current_mode = (current_mode + 1) % len(learning_modes)
        print(f"切换到：{learning_modes[current_mode][0]}")
    elif key == ord('r'):
        bg_subtractor = cv2.createBackgroundSubtractorMOG2(detectShadows=False)
        print("背景模型已重置")

cap.release()
cv2.destroyAllWindows()
```

### 示例5：阴影检测与去除

```python
"""
示例5：阴影检测与去除
阴影处理是背景分割的重要环节！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 创建带阴影检测的MOG2
bg_subtractor = cv2.createBackgroundSubtractorMOG2(
    history=500,
    varThreshold=16,
    detectShadows=True
)

# 设置阴影相关参数
bg_subtractor.setShadowThreshold(0.5)  # 阴影阈值
bg_subtractor.setShadowValue(127)       # 阴影像素值

print("阴影检测与去除")
print("按 's' 切换阴影显示模式")
print("按 'q' 退出")

show_shadow = True

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 应用背景分割
    fg_mask = bg_subtractor.apply(frame)

    # 分离前景和阴影
    # 255 = 前景, 127 = 阴影, 0 = 背景
    foreground_only = np.where(fg_mask == 255, 255, 0).astype(np.uint8)
    shadow_only = np.where(fg_mask == 127, 255, 0).astype(np.uint8)

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    foreground_clean = cv2.morphologyEx(foreground_only, cv2.MORPH_OPEN, kernel)
    foreground_clean = cv2.morphologyEx(foreground_clean, cv2.MORPH_CLOSE, kernel)

    # 创建可视化
    result = frame.copy()

    # 标记前景区域
    contours, _ = cv2.findContours(foreground_clean, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)
    for contour in contours:
        if cv2.contourArea(contour) > 500:
            x, y, w, h = cv2.boundingRect(contour)
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    # 可选显示阴影
    if show_shadow:
        # 用蓝色标记阴影
        shadow_overlay = np.zeros_like(frame)
        shadow_overlay[:, :, 0] = shadow_only  # 蓝色通道
        result = cv2.addWeighted(result, 1, shadow_overlay, 0.3, 0)

    # 创建彩色掩码显示
    color_mask = np.zeros_like(frame)
    color_mask[:, :, 1] = foreground_only  # 绿色 = 前景
    color_mask[:, :, 0] = shadow_only       # 蓝色 = 阴影

    cv2.putText(result, f'Shadow Display: {"ON" if show_shadow else "OFF"}',
                (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Result', result)
    cv2.imshow('Original Mask', fg_mask)
    cv2.imshow('Color Mask (G=FG, B=Shadow)', color_mask)
    cv2.imshow('Foreground Only', foreground_clean)

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('s'):
        show_shadow = not show_shadow

cap.release()
cv2.destroyAllWindows()
```

### 示例6：前景目标追踪

```python
"""
示例6：前景目标追踪
检测并追踪前景中的运动目标！
"""
import cv2
import numpy as np
from collections import OrderedDict

class CentroidTracker:
    """质心追踪器"""
    def __init__(self, max_disappeared=50):
        self.next_object_id = 0
        self.objects = OrderedDict()
        self.disappeared = OrderedDict()
        self.max_disappeared = max_disappeared

    def register(self, centroid):
        self.objects[self.next_object_id] = centroid
        self.disappeared[self.next_object_id] = 0
        self.next_object_id += 1

    def deregister(self, object_id):
        del self.objects[object_id]
        del self.disappeared[object_id]

    def update(self, rects):
        if len(rects) == 0:
            for object_id in list(self.disappeared.keys()):
                self.disappeared[object_id] += 1
                if self.disappeared[object_id] > self.max_disappeared:
                    self.deregister(object_id)
            return self.objects

        input_centroids = np.zeros((len(rects), 2), dtype="int")
        for i, (x, y, w, h) in enumerate(rects):
            cx = x + w // 2
            cy = y + h // 2
            input_centroids[i] = (cx, cy)

        if len(self.objects) == 0:
            for centroid in input_centroids:
                self.register(centroid)
        else:
            object_ids = list(self.objects.keys())
            object_centroids = list(self.objects.values())

            # 计算距离矩阵
            D = np.zeros((len(object_centroids), len(input_centroids)))
            for i, oc in enumerate(object_centroids):
                for j, ic in enumerate(input_centroids):
                    D[i, j] = np.linalg.norm(np.array(oc) - np.array(ic))

            rows = D.min(axis=1).argsort()
            cols = D.argmin(axis=1)[rows]

            used_rows = set()
            used_cols = set()

            for row, col in zip(rows, cols):
                if row in used_rows or col in used_cols:
                    continue
                if D[row, col] > 50:  # 最大距离阈值
                    continue

                object_id = object_ids[row]
                self.objects[object_id] = input_centroids[col]
                self.disappeared[object_id] = 0

                used_rows.add(row)
                used_cols.add(col)

            unused_rows = set(range(len(object_centroids))) - used_rows
            unused_cols = set(range(len(input_centroids))) - used_cols

            for row in unused_rows:
                object_id = object_ids[row]
                self.disappeared[object_id] += 1
                if self.disappeared[object_id] > self.max_disappeared:
                    self.deregister(object_id)

            for col in unused_cols:
                self.register(input_centroids[col])

        return self.objects

# 初始化
cap = cv2.VideoCapture(0)
bg_subtractor = cv2.createBackgroundSubtractorMOG2(detectShadows=True)
tracker = CentroidTracker(max_disappeared=30)

print("前景目标追踪")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 背景分割
    fg_mask = bg_subtractor.apply(frame)
    fg_mask = cv2.threshold(fg_mask, 200, 255, cv2.THRESH_BINARY)[1]

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    fg_mask = cv2.morphologyEx(fg_mask, cv2.MORPH_CLOSE, kernel)
    fg_mask = cv2.morphologyEx(fg_mask, cv2.MORPH_OPEN, kernel)

    # 查找轮廓
    contours, _ = cv2.findContours(fg_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    rects = []
    for contour in contours:
        if cv2.contourArea(contour) > 1000:
            rects.append(cv2.boundingRect(contour))

    # 更新追踪
    objects = tracker.update(rects)

    # 绘制结果
    result = frame.copy()
    for object_id, centroid in objects.items():
        cv2.circle(result, tuple(centroid), 5, (0, 255, 0), -1)
        cv2.putText(result, f'ID {object_id}', (centroid[0] - 20, centroid[1] - 20),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

    for x, y, w, h in rects:
        cv2.rectangle(result, (x, y), (x + w, y + h), (255, 0, 0), 2)

    cv2.putText(result, f'Objects: {len(objects)}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Tracking', result)
    cv2.imshow('FG Mask', fg_mask)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例7：人数统计

```python
"""
示例7：人数统计
基于背景分割的人数统计功能！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))

bg_subtractor = cv2.createBackgroundSubtractorMOG2(
    history=500,
    varThreshold=50,
    detectShadows=True
)

# 计数线位置
count_line_y = frame_height // 2

# 计数器
enter_count = 0
exit_count = 0

# 物体追踪
tracked_objects = {}
next_id = 0

min_area = 2000

print("人数统计系统")
print("按 'r' 重置计数，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 背景分割
    fg_mask = bg_subtractor.apply(frame)
    fg_mask = cv2.threshold(fg_mask, 200, 255, cv2.THRESH_BINARY)[1]

    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (7, 7))
    fg_mask = cv2.morphologyEx(fg_mask, cv2.MORPH_CLOSE, kernel)
    fg_mask = cv2.morphologyEx(fg_mask, cv2.MORPH_OPEN, kernel)

    # 查找轮廓
    contours, _ = cv2.findContours(fg_mask, cv2.RETR_EXTERNAL,
                                    cv2.CHAIN_APPROX_SIMPLE)

    current_objects = []

    for contour in contours:
        if cv2.contourArea(contour) > min_area:
            x, y, w, h = cv2.boundingRect(contour)
            cx, cy = x + w // 2, y + h // 2
            current_objects.append((cx, cy, x, y, w, h))

    # 更新追踪
    new_tracked = {}

    for cx, cy, x, y, w, h in current_objects:
        matched = False

        for obj_id, (prev_cx, prev_cy, prev_crossed) in tracked_objects.items():
            if abs(cx - prev_cx) < 50 and abs(cy - prev_cy) < 50:
                # 检查是否越过计数线
                if not prev_crossed:
                    if prev_cy < count_line_y <= cy:
                        enter_count += 1
                        print(f"进入！总计：{enter_count}")
                        new_tracked[obj_id] = (cx, cy, True)
                    elif prev_cy > count_line_y >= cy:
                        exit_count += 1
                        print(f"离开！总计：{exit_count}")
                        new_tracked[obj_id] = (cx, cy, True)
                    else:
                        new_tracked[obj_id] = (cx, cy, False)
                else:
                    new_tracked[obj_id] = (cx, cy, True)
                matched = True
                break

        if not matched:
            new_tracked[next_id] = (cx, cy, False)
            next_id += 1

    tracked_objects = new_tracked

    # 绘制结果
    result = frame.copy()

    # 绘制计数线
    cv2.line(result, (0, count_line_y), (frame_width, count_line_y),
             (0, 255, 255), 2)

    # 绘制检测到的对象
    for cx, cy, x, y, w, h in current_objects:
        cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)
        cv2.circle(result, (cx, cy), 5, (0, 0, 255), -1)

    # 显示计数
    cv2.putText(result, f'Enter: {enter_count}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(result, f'Exit: {exit_count}', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
    cv2.putText(result, f'Current: {len(tracked_objects)}', (10, 90),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 0), 2)

    cv2.imshow('People Counting', result)
    cv2.imshow('FG Mask', fg_mask)

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        enter_count = 0
        exit_count = 0
```

### 示例8：背景替换

```python
"""
示例8：背景替换
提取前景并替换背景，超酷的特效！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))

bg_subtractor = cv2.createBackgroundSubtractorMOG2(
    history=500,
    varThreshold=50,
    detectShadows=False
)

# 创建不同的背景
backgrounds = []

# 1. 纯色背景
blue_bg = np.full((frame_height, frame_width, 3), (255, 100, 0), dtype=np.uint8)
backgrounds.append(('Blue', blue_bg))

# 2. 渐变背景
gradient_bg = np.zeros((frame_height, frame_width, 3), dtype=np.uint8)
for i in range(frame_height):
    gradient_bg[i, :] = [int(255 * i / frame_height), 100, int(255 * (1 - i / frame_height))]
backgrounds.append(('Gradient', gradient_bg))

# 3. 图案背景
pattern_bg = np.zeros((frame_height, frame_width, 3), dtype=np.uint8)
for i in range(0, frame_height, 20):
    for j in range(0, frame_width, 20):
        color = (0, 255, 0) if (i + j) % 40 == 0 else (0, 100, 0)
        cv2.rectangle(pattern_bg, (j, i), (j + 20, i + 20), color, -1)
backgrounds.append(('Pattern', pattern_bg))

current_bg = 0

print("背景替换效果")
print("按 'b' 切换背景")
print("按 'q' 退出")

# 预热背景模型
print("正在学习背景，请保持静止...")
for _ in range(60):
    ret, frame = cap.read()
    if ret:
        bg_subtractor.apply(frame, learningRate=0.5)
print("学习完成！")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 获取前景掩码
    fg_mask = bg_subtractor.apply(frame, learningRate=0)

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (7, 7))
    fg_mask = cv2.morphologyEx(fg_mask, cv2.MORPH_CLOSE, kernel)
    fg_mask = cv2.morphologyEx(fg_mask, cv2.MORPH_OPEN, kernel)

    # 高斯模糊边缘
    fg_mask = cv2.GaussianBlur(fg_mask, (5, 5), 0)

    # 归一化掩码
    fg_mask_norm = fg_mask.astype(np.float32) / 255.0
    fg_mask_3ch = cv2.merge([fg_mask_norm, fg_mask_norm, fg_mask_norm])

    # 获取当前背景
    bg_name, bg_image = backgrounds[current_bg]

    # 合成图像
    foreground = frame.astype(np.float32) * fg_mask_3ch
    background = bg_image.astype(np.float32) * (1 - fg_mask_3ch)
    composite = (foreground + background).astype(np.uint8)

    # 显示信息
    cv2.putText(composite, f'Background: {bg_name}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)

    cv2.imshow('Original', frame)
    cv2.imshow('FG Mask', fg_mask)
    cv2.imshow('Background Replacement', composite)

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('b'):
        current_bg = (current_bg + 1) % len(backgrounds)
        print(f"切换到背景：{backgrounds[current_bg][0]}")

cap.release()
cv2.destroyAllWindows()
```

### 示例9：运动检测报警

```python
"""
示例9：运动检测报警
检测到运动就报警，安防监控必备！
"""
import cv2
import numpy as np
from datetime import datetime
import winsound  # Windows音效

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

bg_subtractor = cv2.createBackgroundSubtractorMOG2(
    history=500,
    varThreshold=50,
    detectShadows=True
)

# 报警设置
alarm_enabled = True
alarm_threshold = 5000  # 前景像素阈值
alarm_cooldown = 0
cooldown_frames = 60  # 报警冷却帧数

print("运动检测报警系统")
print("按 'a' 启用/禁用报警")
print("按 '+'/'-' 调整灵敏度")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 背景分割
    fg_mask = bg_subtractor.apply(frame)
    fg_mask = cv2.threshold(fg_mask, 200, 255, cv2.THRESH_BINARY)[1]

    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    fg_mask = cv2.morphologyEx(fg_mask, cv2.MORPH_CLOSE, kernel)

    # 计算前景像素数量
    fg_pixels = cv2.countNonZero(fg_mask)

    # 检测运动
    motion_detected = fg_pixels > alarm_threshold

    result = frame.copy()

    # 报警逻辑
    if alarm_cooldown > 0:
        alarm_cooldown -= 1

    if motion_detected and alarm_enabled and alarm_cooldown == 0:
        # 触发报警
        alarm_cooldown = cooldown_frames

        # 播放报警声（Windows）
        try:
            winsound.Beep(1000, 200)
        except:
            pass

        # 记录时间
        timestamp = datetime.now().strftime('%H:%M:%S')
        print(f"[{timestamp}] 检测到运动！前景像素：{fg_pixels}")

        # 红色边框
        cv2.rectangle(result, (0, 0), (result.shape[1]-1, result.shape[0]-1),
                      (0, 0, 255), 5)

    # 绘制状态信息
    status_color = (0, 255, 0) if alarm_enabled else (0, 0, 255)
    cv2.putText(result, f'Alarm: {"ON" if alarm_enabled else "OFF"}',
                (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, status_color, 2)
    cv2.putText(result, f'Threshold: {alarm_threshold}',
                (10, 60), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(result, f'FG Pixels: {fg_pixels}',
                (10, 90), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    if motion_detected:
        cv2.putText(result, 'MOTION!', (10, 120),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

    # 绘制灵敏度条
    bar_width = int((fg_pixels / alarm_threshold) * 200)
    bar_width = min(bar_width, 200)
    cv2.rectangle(result, (10, 130), (10 + bar_width, 145),
                  (0, 0, 255) if motion_detected else (0, 255, 0), -1)
    cv2.rectangle(result, (10, 130), (210, 145), (255, 255, 255), 1)

    cv2.imshow('Motion Alarm', result)
    cv2.imshow('FG Mask', fg_mask)

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('a'):
        alarm_enabled = not alarm_enabled
        print(f"报警：{'启用' if alarm_enabled else '禁用'}")
    elif key in [ord('+'), ord('=')]:
        alarm_threshold += 1000
        print(f"阈值：{alarm_threshold}")
    elif key == ord('-'):
        alarm_threshold = max(1000, alarm_threshold - 1000)
        print(f"阈值：{alarm_threshold}")

cap.release()
cv2.destroyAllWindows()
```

### 示例10：MOG2与KNN对比

```python
"""
示例10：MOG2与KNN对比
同时运行两种算法进行对比！
"""
import cv2
import numpy as np
import time

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 创建两种背景分割器
mog2 = cv2.createBackgroundSubtractorMOG2(
    history=500,
    varThreshold=16,
    detectShadows=True
)

knn = cv2.createBackgroundSubtractorKNN(
    history=500,
    dist2Threshold=400.0,
    detectShadows=True
)

# 性能统计
mog2_times = []
knn_times = []

print("MOG2 vs KNN 背景分割对比")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # MOG2
    start_time = time.time()
    mog2_mask = mog2.apply(frame)
    mog2_time = (time.time() - start_time) * 1000
    mog2_times.append(mog2_time)

    # KNN
    start_time = time.time()
    knn_mask = knn.apply(frame)
    knn_time = (time.time() - start_time) * 1000
    knn_times.append(knn_time)

    # 保持最近100帧的统计
    if len(mog2_times) > 100:
        mog2_times.pop(0)
        knn_times.pop(0)

    avg_mog2 = np.mean(mog2_times)
    avg_knn = np.mean(knn_times)

    # 二值化处理
    mog2_binary = cv2.threshold(mog2_mask, 200, 255, cv2.THRESH_BINARY)[1]
    knn_binary = cv2.threshold(knn_mask, 200, 255, cv2.THRESH_BINARY)[1]

    # 形态学处理
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    mog2_clean = cv2.morphologyEx(mog2_binary, cv2.MORPH_CLOSE, kernel)
    knn_clean = cv2.morphologyEx(knn_binary, cv2.MORPH_CLOSE, kernel)

    # 计算前景像素
    mog2_pixels = cv2.countNonZero(mog2_clean)
    knn_pixels = cv2.countNonZero(knn_clean)

    # 创建对比显示
    mog2_display = cv2.cvtColor(mog2_clean, cv2.COLOR_GRAY2BGR)
    knn_display = cv2.cvtColor(knn_clean, cv2.COLOR_GRAY2BGR)

    # 添加信息
    cv2.putText(mog2_display, 'MOG2', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)
    cv2.putText(mog2_display, f'Time: {mog2_time:.1f}ms (avg: {avg_mog2:.1f}ms)',
                (10, 60), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
    cv2.putText(mog2_display, f'FG Pixels: {mog2_pixels}',
                (10, 80), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

    cv2.putText(knn_display, 'KNN', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)
    cv2.putText(knn_display, f'Time: {knn_time:.1f}ms (avg: {avg_knn:.1f}ms)',
                (10, 60), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
    cv2.putText(knn_display, f'FG Pixels: {knn_pixels}',
                (10, 80), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)

    # 并排显示
    comparison = np.hstack([mog2_display, knn_display])

    cv2.imshow('Original', frame)
    cv2.imshow('MOG2 vs KNN', comparison)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

print("\n性能统计：")
print(f"MOG2 平均耗时：{np.mean(mog2_times):.2f}ms")
print(f"KNN 平均耗时：{np.mean(knn_times):.2f}ms")

cap.release()
cv2.destroyAllWindows()
```
