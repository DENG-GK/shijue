## 一、视频基础知识

### 1.1 什么是视频

视频本质上是由一系列连续的静态图像（帧）按照一定的速率播放形成的动态画面：

- **帧（Frame）**：视频的基本单位，每一帧都是一幅完整的图像
- **帧率（FPS）**：每秒播放的帧数，常见有24、30、60fps
- **分辨率**：每帧图像的像素尺寸，如1920×1080
- **编码格式**：视频压缩方式，如H.264、H.265、VP9等
- **容器格式**：视频文件封装格式，如MP4、AVI、MKV等

### 1.2 OpenCV视频处理

OpenCV通过`cv2.VideoCapture`类来处理视频，它可以：

- 读取本地视频文件
- 捕获摄像头实时画面
- 读取网络视频流

### 示例1：基础视频读取

```python
"""
示例1：基础视频读取
哼，连视频都不会读取的话，本小姐会笑话你的！
"""
import cv2

# 创建VideoCapture对象
cap = cv2.VideoCapture('test_video.mp4')

# 检查视频是否成功打开
if not cap.isOpened():
    print("错误：无法打开视频文件！")
    exit()

# 读取一帧
ret, frame = cap.read()

if ret:
    print(f"成功读取一帧，帧的形状：{frame.shape}")
    cv2.imshow('Frame', frame)
    cv2.waitKey(0)
else:
    print("读取帧失败！")

# 释放资源
cap.release()
cv2.destroyAllWindows()
```

### 示例2：获取视频属性

```python
"""
示例2：获取视频属性
本小姐来教你如何查看视频的各种参数，记好了！
"""
import cv2

cap = cv2.VideoCapture('test_video.mp4')

if cap.isOpened():
    # 获取视频属性
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = cap.get(cv2.CAP_PROP_FPS)
    frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    fourcc = int(cap.get(cv2.CAP_PROP_FOURCC))

    # 解码fourcc
    fourcc_str = "".join([chr((fourcc >> 8 * i) & 0xFF) for i in range(4)])

    print("=" * 40)
    print("视频属性信息")
    print("=" * 40)
    print(f"分辨率：{width} x {height}")
    print(f"帧率：{fps:.2f} FPS")
    print(f"总帧数：{frame_count}")
    print(f"时长：{frame_count / fps:.2f} 秒")
    print(f"编码格式：{fourcc_str}")
    print("=" * 40)

cap.release()
```

### 示例3：循环播放视频

```python
"""
示例3：循环播放视频
这才是视频播放的正确方式呢！别搞错了，笨蛋！
"""
import cv2

cap = cv2.VideoCapture('test_video.mp4')

if not cap.isOpened():
    print("无法打开视频！")
    exit()

# 获取帧率，用于控制播放速度
fps = cap.get(cv2.CAP_PROP_FPS)
delay = int(1000 / fps)  # 每帧延迟时间（毫秒）

while True:
    ret, frame = cap.read()

    # 视频结束
    if not ret:
        print("视频播放完毕！")
        break

    cv2.imshow('Video Player', frame)

    # 按'q'键退出
    if cv2.waitKey(delay) & 0xFF == ord('q'):
        print("用户退出播放")
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例4：视频循环播放（重复播放）

```python
"""
示例4：视频循环播放
哼，想让视频重复播放？本小姐教你正确的做法！
"""
import cv2

cap = cv2.VideoCapture('test_video.mp4')

if not cap.isOpened():
    print("无法打开视频！")
    exit()

fps = cap.get(cv2.CAP_PROP_FPS)
delay = int(1000 / fps)
loop_count = 0

while True:
    ret, frame = cap.read()

    if not ret:
        # 视频结束，重新定位到开头
        cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
        loop_count += 1
        print(f"循环播放第 {loop_count} 次")
        continue

    # 在帧上显示循环次数
    cv2.putText(frame, f'Loop: {loop_count}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

    cv2.imshow('Loop Video', frame)

    if cv2.waitKey(delay) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例6：视频进度条控制

```python
"""
示例6：视频进度条控制
这个进度条功能可是本小姐精心设计的呢！
"""
import cv2

def on_trackbar(val):
    """进度条回调函数"""
    global cap
    cap.set(cv2.CAP_PROP_POS_FRAMES, val)

cap = cv2.VideoCapture('test_video.mp4')

if not cap.isOpened():
    print("无法打开视频！")
    exit()

total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
fps = cap.get(cv2.CAP_PROP_FPS)
delay = int(1000 / fps)

# 创建窗口和进度条
cv2.namedWindow('Video with Trackbar')
cv2.createTrackbar('Progress', 'Video with Trackbar', 0, total_frames - 1, on_trackbar)

paused = False

while True:
    if not paused:
        ret, frame = cap.read()

        if not ret:
            break

        # 更新进度条位置
        current_frame = int(cap.get(cv2.CAP_PROP_POS_FRAMES))
        cv2.setTrackbarPos('Progress', 'Video with Trackbar', current_frame)

        # 显示当前时间
        current_time = current_frame / fps
        cv2.putText(frame, f'Time: {current_time:.2f}s', (10, 30),
                    cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

        cv2.imshow('Video with Trackbar', frame)

    key = cv2.waitKey(delay) & 0xFF
    if key == ord('q'):
        break
    elif key == ord(' '):  # 空格键暂停/继续
        paused = not paused
        print("暂停" if paused else "继续播放")

cap.release()
cv2.destroyAllWindows()
```

### 示例7：多视频同时显示

```python
"""
示例7：多视频同时显示
同时处理多个视频对本小姐来说就是小菜一碟！
"""
import cv2
import numpy as np

# 打开多个视频
cap1 = cv2.VideoCapture('video1.mp4')
cap2 = cv2.VideoCapture('video2.mp4')

if not cap1.isOpened() or not cap2.isOpened():
    print("无法打开视频文件！")
    exit()

# 获取两个视频的fps，取较小值
fps1 = cap1.get(cv2.CAP_PROP_FPS)
fps2 = cap2.get(cv2.CAP_PROP_FPS)
fps = min(fps1, fps2)
delay = int(1000 / fps)

# 目标显示尺寸
target_size = (640, 360)

while True:
    ret1, frame1 = cap1.read()
    ret2, frame2 = cap2.read()

    # 如果任一视频结束，则停止
    if not ret1 or not ret2:
        break

    # 调整到相同尺寸
    frame1 = cv2.resize(frame1, target_size)
    frame2 = cv2.resize(frame2, target_size)

    # 水平拼接
    combined = np.hstack([frame1, frame2])

    cv2.imshow('Multi Video', combined)

    if cv2.waitKey(delay) & 0xFF == ord('q'):
        break

cap1.release()
cap2.release()
cv2.destroyAllWindows()
```

### 示例8：视频帧处理（实时滤镜效果）

```python
"""
示例8：视频帧处理（实时滤镜效果）
哼，给视频加滤镜可是本小姐的拿手好戏呢！
"""
import cv2
import numpy as np

cap = cv2.VideoCapture('test_video.mp4')

if not cap.isOpened():
    print("无法打开视频！")
    exit()

fps = cap.get(cv2.CAP_PROP_FPS)
delay = int(1000 / fps)

# 滤镜模式
filter_mode = 0
filter_names = ['原始', '灰度', '边缘检测', '模糊', '反色', '浮雕']

def apply_filter(frame, mode):
    """应用滤镜效果"""
    if mode == 0:  # 原始
        return frame
    elif mode == 1:  # 灰度
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        return cv2.cvtColor(gray, cv2.COLOR_GRAY2BGR)
    elif mode == 2:  # 边缘检测
        edges = cv2.Canny(frame, 100, 200)
        return cv2.cvtColor(edges, cv2.COLOR_GRAY2BGR)
    elif mode == 3:  # 模糊
        return cv2.GaussianBlur(frame, (15, 15), 0)
    elif mode == 4:  # 反色
        return cv2.bitwise_not(frame)
    elif mode == 5:  # 浮雕效果
        kernel = np.array([[-2, -1, 0],
                          [-1, 1, 1],
                          [0, 1, 2]])
        return cv2.filter2D(frame, -1, kernel) + 128
    return frame

print("按数字键 0-5 切换滤镜效果，按 q 退出")

while True:
    ret, frame = cap.read()

    if not ret:
        cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
        continue

    # 应用滤镜
    processed = apply_filter(frame, filter_mode)

    # 显示当前滤镜名称
    cv2.putText(processed, f'Filter: {filter_names[filter_mode]}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

    cv2.imshow('Video Filters', processed)

    key = cv2.waitKey(delay) & 0xFF
    if key == ord('q'):
        break
    elif ord('0') <= key <= ord('5'):
        filter_mode = key - ord('0')
        print(f"切换到滤镜：{filter_names[filter_mode]}")

cap.release()
cv2.destroyAllWindows()
```

### 示例9：提取视频关键帧

```python
"""
示例9：提取视频关键帧
自动提取关键帧可是很有用的技能呢！
"""
import cv2
import os
import numpy as np

def extract_keyframes(video_path, output_dir, threshold=30.0):
    """
    基于帧差法提取关键帧

    参数:
        video_path: 视频路径
        output_dir: 输出目录
        threshold: 帧差阈值
    """
    cap = cv2.VideoCapture(video_path)

    if not cap.isOpened():
        print("无法打开视频！")
        return

    os.makedirs(output_dir, exist_ok=True)

    prev_frame = None
    frame_idx = 0
    keyframe_count = 0

    while True:
        ret, frame = cap.read()

        if not ret:
            break

        # 转换为灰度图
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        if prev_frame is not None:
            # 计算帧差
            diff = cv2.absdiff(prev_frame, gray)
            mean_diff = np.mean(diff)

            # 如果差异超过阈值，保存为关键帧
            if mean_diff > threshold:
                keyframe_path = os.path.join(output_dir, f'keyframe_{keyframe_count:04d}.jpg')
                cv2.imwrite(keyframe_path, frame)
                print(f"保存关键帧 {keyframe_count}，帧号：{frame_idx}，差异：{mean_diff:.2f}")
                keyframe_count += 1
        else:
            # 第一帧作为关键帧
            keyframe_path = os.path.join(output_dir, f'keyframe_{keyframe_count:04d}.jpg')
            cv2.imwrite(keyframe_path, frame)
            keyframe_count += 1

        prev_frame = gray
        frame_idx += 1

    cap.release()
    print(f"\n提取完成！共 {keyframe_count} 个关键帧，保存到 {output_dir}")

# 使用示例
extract_keyframes('test_video.mp4', 'keyframes', threshold=25.0)
```

### 示例10：视频帧率控制与变速播放

```python
"""
示例10：视频帧率控制与变速播放
想要快进或慢放？本小姐这就教你！
"""
import cv2

cap = cv2.VideoCapture('test_video.mp4')

if not cap.isOpened():
    print("无法打开视频！")
    exit()

original_fps = cap.get(cv2.CAP_PROP_FPS)
speed_multiplier = 1.0

print("控制说明：")
print("  '+' 或 '=' : 加速")
print("  '-' : 减速")
print("  'r' : 恢复正常速度")
print("  'q' : 退出")

while True:
    ret, frame = cap.read()

    if not ret:
        cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
        continue

    # 根据速度倍率计算延迟
    delay = int(1000 / (original_fps * speed_multiplier))
    delay = max(1, delay)  # 确保延迟至少1毫秒

    # 显示速度信息
    cv2.putText(frame, f'Speed: {speed_multiplier:.1f}x', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)
    cv2.putText(frame, f'FPS: {original_fps * speed_multiplier:.1f}', (10, 70),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

    cv2.imshow('Variable Speed Player', frame)

    key = cv2.waitKey(delay) & 0xFF
    if key == ord('q'):
        break
    elif key in [ord('+'), ord('=')]:
        speed_multiplier = min(4.0, speed_multiplier + 0.25)
        print(f"当前速度：{speed_multiplier:.1f}x")
    elif key == ord('-'):
        speed_multiplier = max(0.25, speed_multiplier - 0.25)
        print(f"当前速度：{speed_multiplier:.1f}x")
    elif key == ord('r'):
        speed_multiplier = 1.0
        print("恢复正常速度")

cap.release()
cv2.destroyAllWindows()
```

## 三、VideoCapture常用属性

| 属性 | 说明 |
|------|------|
| `CAP_PROP_FRAME_WIDTH` | 帧宽度 |
| `CAP_PROP_FRAME_HEIGHT` | 帧高度 |
| `CAP_PROP_FPS` | 帧率 |
| `CAP_PROP_FRAME_COUNT` | 总帧数 |
| `CAP_PROP_POS_FRAMES` | 当前帧位置 |
| `CAP_PROP_POS_MSEC` | 当前时间位置（毫秒） |
| `CAP_PROP_POS_AVI_RATIO` | 相对位置（0-1） |
| `CAP_PROP_FOURCC` | 编码格式 |
| `CAP_PROP_BRIGHTNESS` | 亮度 |
| `CAP_PROP_CONTRAST` | 对比度 |
