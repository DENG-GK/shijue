## 一、摄像头基础

### 1.1 摄像头类型

OpenCV支持多种摄像头类型：

- **USB摄像头**：通过USB接口连接的摄像头，最常见
- **内置摄像头**：笔记本电脑内置的摄像头
- **IP摄像头**：通过网络连接的摄像头，使用RTSP/HTTP协议
- **虚拟摄像头**：软件模拟的摄像头设备

### 1.2 摄像头索引

- 索引0通常是默认摄像头（笔记本内置或第一个USB摄像头）
- 多个摄像头按照连接顺序编号：0, 1, 2, ...
- 在Windows上，可以使用`cv2.CAP_DSHOW`后端提高兼容性

## 二、代码示例

### 示例1：基础摄像头读取

```python
"""
示例1：基础摄像头读取
哼，连摄像头都不会开的话，你还学什么计算机视觉！
"""
import cv2

# 打开默认摄像头（索引0）
cap = cv2.VideoCapture(0)

# 检查是否成功打开
if not cap.isOpened():
    print("错误：无法打开摄像头！")
    exit()

print("摄像头已打开，按 'q' 键退出")

while True:
    # 读取一帧
    ret, frame = cap.read()

    if not ret:
        print("读取帧失败！")
        break

    # 显示帧
    cv2.imshow('Camera', frame)

    # 按'q'退出
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

# 释放资源
cap.release()
cv2.destroyAllWindows()
```

```python
"""
示例3：设置摄像头分辨率
想要高清画质？本小姐教你怎么设置！
"""
import cv2

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 获取原始分辨率
original_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
original_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
print(f"原始分辨率：{original_width}x{original_height}")

# 尝试设置更高分辨率（1920x1080）
cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1920)
cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 1080)

# 验证设置结果
new_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
new_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
print(f"设置后分辨率：{new_width}x{new_height}")

# 常见分辨率列表
resolutions = [
    ("480p", 640, 480),
    ("720p", 1280, 720),
    ("1080p", 1920, 1080),
    ("4K", 3840, 2160)
]

print("\n测试摄像头支持的分辨率：")
for name, w, h in resolutions:
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, w)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, h)
    actual_w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    actual_h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    supported = "✓" if (actual_w == w and actual_h == h) else "✗"
    print(f"  {name} ({w}x{h}): {supported} -> 实际 {actual_w}x{actual_h}")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 显示当前分辨率
    h, w = frame.shape[:2]
    cv2.putText(frame, f'Resolution: {w}x{h}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

    cv2.imshow('High Resolution Camera', frame)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例4：设置摄像头帧率

```python
"""
示例4：设置摄像头帧率
帧率控制也是很重要的，笨蛋要记好了！
"""
import cv2
import time

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 获取原始帧率
original_fps = cap.get(cv2.CAP_PROP_FPS)
print(f"原始帧率：{original_fps}")

# 设置帧率
target_fps = 60
cap.set(cv2.CAP_PROP_FPS, target_fps)

# 验证设置
actual_fps = cap.get(cv2.CAP_PROP_FPS)
print(f"设置后帧率：{actual_fps}")

# 实际测量帧率
frame_count = 0
start_time = time.time()

while True:
    ret, frame = cap.read()
    if not ret:
        break

    frame_count += 1
    elapsed_time = time.time() - start_time

    # 每秒更新一次实际帧率
    if elapsed_time >= 1.0:
        measured_fps = frame_count / elapsed_time
        frame_count = 0
        start_time = time.time()

        # 显示帧率信息
        info = f"Measured FPS: {measured_fps:.1f}"
    else:
        info = "Calculating FPS..."

    cv2.putText(frame, info, (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

    cv2.imshow('FPS Test', frame)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例5：摄像头参数调节

```python
"""
示例5：摄像头参数调节
调节亮度、对比度、曝光等参数，这可是专业技能呢！
"""
import cv2

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 创建调节窗口
cv2.namedWindow('Camera Settings')

# 创建滑动条
def nothing(x):
    pass

# 获取当前参数值作为初始值
brightness = int(cap.get(cv2.CAP_PROP_BRIGHTNESS)) if cap.get(cv2.CAP_PROP_BRIGHTNESS) > 0 else 50
contrast = int(cap.get(cv2.CAP_PROP_CONTRAST)) if cap.get(cv2.CAP_PROP_CONTRAST) > 0 else 50
saturation = int(cap.get(cv2.CAP_PROP_SATURATION)) if cap.get(cv2.CAP_PROP_SATURATION) > 0 else 50

cv2.createTrackbar('Brightness', 'Camera Settings', brightness, 100, nothing)
cv2.createTrackbar('Contrast', 'Camera Settings', contrast, 100, nothing)
cv2.createTrackbar('Saturation', 'Camera Settings', saturation, 100, nothing)

print("使用滑动条调节摄像头参数")
print("按 'r' 重置参数，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 获取滑动条值
    brightness = cv2.getTrackbarPos('Brightness', 'Camera Settings')
    contrast = cv2.getTrackbarPos('Contrast', 'Camera Settings')
    saturation = cv2.getTrackbarPos('Saturation', 'Camera Settings')

    # 设置摄像头参数
    cap.set(cv2.CAP_PROP_BRIGHTNESS, brightness)
    cap.set(cv2.CAP_PROP_CONTRAST, contrast)
    cap.set(cv2.CAP_PROP_SATURATION, saturation)

    # 显示参数信息
    cv2.putText(frame, f'Brightness: {brightness}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(frame, f'Contrast: {contrast}', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(frame, f'Saturation: {saturation}', (10, 90),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Camera Settings', frame)

    key = cv2.waitKey(1) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('r'):
        # 重置参数
        cv2.setTrackbarPos('Brightness', 'Camera Settings', 50)
        cv2.setTrackbarPos('Contrast', 'Camera Settings', 50)
        cv2.setTrackbarPos('Saturation', 'Camera Settings', 50)

cap.release()
cv2.destroyAllWindows()
```

### 示例6：多摄像头同时操作

```python
"""
示例6：多摄像头同时操作
同时控制多个摄像头？这对本小姐来说轻而易举！
"""
import cv2
import numpy as np

# 尝试打开多个摄像头
cameras = []
max_cameras = 4  # 最多尝试4个摄像头

for i in range(max_cameras):
    cap = cv2.VideoCapture(i)
    if cap.isOpened():
        cameras.append((i, cap))
        print(f"摄像头 {i} 已连接")
    else:
        cap.release()

if not cameras:
    print("没有找到可用的摄像头！")
    exit()

print(f"\n共找到 {len(cameras)} 个摄像头")

# 统一分辨率
target_size = (320, 240)

while True:
    frames = []

    for idx, cap in cameras:
        ret, frame = cap.read()
        if ret:
            # 调整大小
            frame = cv2.resize(frame, target_size)
            # 添加摄像头标签
            cv2.putText(frame, f'Camera {idx}', (10, 25),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
            frames.append(frame)
        else:
            # 如果读取失败，用黑色图像替代
            black = np.zeros((target_size[1], target_size[0], 3), dtype=np.uint8)
            cv2.putText(black, f'Camera {idx} - No Signal', (10, 120),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 1)
            frames.append(black)

    # 拼接所有摄像头画面
    if len(frames) == 1:
        combined = frames[0]
    elif len(frames) == 2:
        combined = np.hstack(frames)
    elif len(frames) <= 4:
        # 2x2 网格
        while len(frames) < 4:
            black = np.zeros((target_size[1], target_size[0], 3), dtype=np.uint8)
            frames.append(black)
        row1 = np.hstack(frames[:2])
        row2 = np.hstack(frames[2:4])
        combined = np.vstack([row1, row2])
    else:
        combined = np.hstack(frames[:4])

    cv2.imshow('Multi Camera View', combined)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

# 释放所有摄像头
for idx, cap in cameras:
    cap.release()
cv2.destroyAllWindows()
```

### 示例7：IP摄像头连接

```python
"""
示例7：IP摄像头连接
网络摄像头也难不倒本小姐呢！
"""
import cv2

# IP摄像头URL格式示例
# RTSP: rtsp://username:password@ip:port/stream
# HTTP: http://ip:port/video

# 常见IP摄像头URL格式
ip_camera_urls = {
    'hikvision': 'rtsp://admin:password@192.168.1.100:554/Streaming/Channels/101',
    'dahua': 'rtsp://admin:password@192.168.1.100:554/cam/realmonitor?channel=1&subtype=0',
    'generic': 'rtsp://192.168.1.100:554/stream1',
    'mjpeg': 'http://192.168.1.100:8080/video',
}

def connect_ip_camera(url, timeout=10):
    """连接IP摄像头"""
    print(f"正在连接：{url}")

    cap = cv2.VideoCapture(url)

    # 设置超时（某些后端支持）
    cap.set(cv2.CAP_PROP_OPEN_TIMEOUT_MSEC, timeout * 1000)

    if not cap.isOpened():
        print("无法连接到IP摄像头！")
        return None

    print("连接成功！")
    return cap

# 示例：连接本地RTSP流
# 替换为实际的IP摄像头地址
rtsp_url = 'rtsp://192.168.1.100:554/stream'

# 如果没有实际的IP摄像头，使用本地摄像头演示
cap = cv2.VideoCapture(0)  # 替换为 connect_ip_camera(rtsp_url)

if cap is None or not cap.isOpened():
    print("使用本地摄像头作为演示")
    cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开任何摄像头！")
    exit()

# 设置缓冲区大小（减少延迟）
cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)

while True:
    ret, frame = cap.read()
    if not ret:
        print("读取帧失败，尝试重连...")
        cap.release()
        cap = cv2.VideoCapture(0)
        continue

    cv2.imshow('IP Camera', frame)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

### 示例8：摄像头截图功能

```python
"""
示例8：摄像头截图功能
按一下就能截图，多方便啊！
"""
import cv2
import os
from datetime import datetime

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 创建截图保存目录
screenshot_dir = 'screenshots'
os.makedirs(screenshot_dir, exist_ok=True)

screenshot_count = 0

print("按 's' 截图，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 显示提示
    display_frame = frame.copy()
    cv2.putText(display_frame, "Press 's' to screenshot", (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(display_frame, f"Screenshots: {screenshot_count}", (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Screenshot Camera', display_frame)

    key = cv2.waitKey(1) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('s'):
        # 生成文件名（时间戳）
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = os.path.join(screenshot_dir, f'screenshot_{timestamp}.jpg')

        # 保存截图（保存原始帧，不含提示文字）
        cv2.imwrite(filename, frame)
        screenshot_count += 1
        print(f"截图已保存：{filename}")

        # 显示截图成功提示
        cv2.putText(display_frame, "Screenshot saved!", (10, 90),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
        cv2.imshow('Screenshot Camera', display_frame)
        cv2.waitKey(500)  # 显示0.5秒

cap.release()
cv2.destroyAllWindows()
print(f"\n共保存 {screenshot_count} 张截图到 {screenshot_dir}")
```

### 示例9：摄像头画面镜像与翻转

```python
"""
示例9：摄像头画面镜像与翻转
自拍镜像效果，爱美的你肯定需要这个！
"""
import cv2

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 翻转模式
flip_mode = 0
mode_names = ['原始', '水平镜像', '垂直翻转', '180°旋转']

print("按 'f' 切换翻转模式，按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 应用翻转
    if flip_mode == 1:
        # 水平镜像（自拍模式）
        frame = cv2.flip(frame, 1)
    elif flip_mode == 2:
        # 垂直翻转
        frame = cv2.flip(frame, 0)
    elif flip_mode == 3:
        # 180度旋转（同时水平和垂直翻转）
        frame = cv2.flip(frame, -1)

    # 显示当前模式
    cv2.putText(frame, f'Mode: {mode_names[flip_mode]}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)

    cv2.imshow('Flip Camera', frame)

    key = cv2.waitKey(1) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('f'):
        flip_mode = (flip_mode + 1) % 4
        print(f"切换到：{mode_names[flip_mode]}")

cap.release()
cv2.destroyAllWindows()
```

### 示例10：摄像头自动曝光控制

```python
"""
示例10：摄像头自动曝光控制
曝光控制对于拍出好照片很重要呢！
"""
import cv2

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

# 创建窗口
cv2.namedWindow('Exposure Control')

def nothing(x):
    pass

# 检查是否支持自动曝光
auto_exposure = cap.get(cv2.CAP_PROP_AUTO_EXPOSURE)
print(f"自动曝光状态：{auto_exposure}")

# 创建滑动条
cv2.createTrackbar('Auto Exposure', 'Exposure Control', 1, 1, nothing)
cv2.createTrackbar('Exposure', 'Exposure Control', 0, 10, nothing)

print("自动曝光：0=关闭，1=开启")
print("曝光值范围取决于摄像头")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 获取滑动条值
    auto_exp = cv2.getTrackbarPos('Auto Exposure', 'Exposure Control')
    exposure = cv2.getTrackbarPos('Exposure', 'Exposure Control')

    # 设置自动曝光
    if auto_exp:
        cap.set(cv2.CAP_PROP_AUTO_EXPOSURE, 3)  # 自动模式
    else:
        cap.set(cv2.CAP_PROP_AUTO_EXPOSURE, 1)  # 手动模式
        # 设置曝光值（负值，值越小曝光越短）
        cap.set(cv2.CAP_PROP_EXPOSURE, -exposure)

    # 获取当前曝光值
    current_exposure = cap.get(cv2.CAP_PROP_EXPOSURE)

    # 显示信息
    mode_text = "Auto" if auto_exp else "Manual"
    cv2.putText(frame, f'Mode: {mode_text}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(frame, f'Exposure: {current_exposure}', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Exposure Control', frame)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```
