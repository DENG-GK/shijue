## 一、视频保存基础

### 1.1 VideoWriter类

OpenCV使用`cv2.VideoWriter`类来保存视频：

```python
cv2.VideoWriter(filename, fourcc, fps, frameSize, isColor)
```

参数说明：
- **filename**：输出视频文件名
- **fourcc**：视频编码格式（四字符编码）
- **fps**：帧率
- **frameSize**：帧尺寸 (宽度, 高度)
- **isColor**：是否为彩色视频

### 1.2 常用编码格式（FourCC）

| FourCC | 格式 | 文件扩展名 | 说明 |
|--------|------|------------|------|
| XVID | MPEG-4 | .avi | 兼容性好 |
| MJPG | Motion JPEG | .avi | 质量好，文件较大 |
| mp4v | MPEG-4 | .mp4 | 通用MP4格式 |
| avc1/H264 | H.264 | .mp4 | 高压缩率 |
| X264 | H.264 | .mp4 | 需要额外编解码器 |
| DIVX | DivX | .avi | 经典格式 |

### 示例1：基础视频保存

```python
"""
示例1：基础视频保存
哼，这是最基本的视频保存方式，笨蛋可别学不会！
"""
import cv2
import numpy as np

# 视频参数
output_filename = 'output_basic.avi'
fps = 30
frame_width = 640
frame_height = 480
duration = 5  # 秒

# 创建VideoWriter
fourcc = cv2.VideoWriter_fourcc(*'XVID')
out = cv2.VideoWriter(output_filename, fourcc, fps, (frame_width, frame_height))

# 检查是否成功创建
if not out.isOpened():
    print("无法创建视频文件！")
    exit()

total_frames = fps * duration
print(f"正在生成 {duration} 秒视频...")

for i in range(total_frames):
    # 创建渐变颜色帧
    frame = np.zeros((frame_height, frame_width, 3), dtype=np.uint8)

    # 随时间变化的颜色
    b = int((i / total_frames) * 255)
    g = int(((total_frames - i) / total_frames) * 255)
    r = 128

    frame[:] = (b, g, r)

    # 添加帧号
    cv2.putText(frame, f'Frame: {i + 1}/{total_frames}', (50, 50),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (255, 255, 255), 2)

    # 写入帧
    out.write(frame)

    # 显示进度
    if (i + 1) % fps == 0:
        print(f"已处理 {i + 1}/{total_frames} 帧")

# 释放资源
out.release()
print(f"视频已保存为：{output_filename}")
```

### 示例3：不同编码格式比较

```python
"""
示例3：不同编码格式比较
本小姐来给你演示各种编码格式的区别！
"""
import cv2
import numpy as np
import os

# 测试用的帧参数
frame_width = 640
frame_height = 480
fps = 30
duration = 3
total_frames = fps * duration

# 要测试的编码格式
codecs = [
    ('XVID', '.avi'),
    ('MJPG', '.avi'),
    ('mp4v', '.mp4'),
    ('DIVX', '.avi'),
]

def generate_test_frame(frame_num):
    """生成测试帧"""
    frame = np.zeros((frame_height, frame_width, 3), dtype=np.uint8)

    # 创建移动的圆
    center_x = int((frame_num / total_frames) * frame_width)
    center_y = frame_height // 2
    cv2.circle(frame, (center_x, center_y), 50, (0, 255, 0), -1)

    # 添加一些细节
    cv2.rectangle(frame, (10, 10), (200, 60), (255, 255, 255), 2)
    cv2.putText(frame, f'Frame {frame_num}', (20, 45),
                cv2.FONT_HERSHEY_SIMPLEX, 1, (255, 255, 255), 2)

    return frame

print("测试不同视频编码格式...")
print("=" * 50)

results = []

for codec, ext in codecs:
    output_file = f'test_{codec}{ext}'

    try:
        fourcc = cv2.VideoWriter_fourcc(*codec)
        out = cv2.VideoWriter(output_file, fourcc, fps, (frame_width, frame_height))

        if not out.isOpened():
            print(f"[{codec}] 编码器不可用")
            continue

        # 写入测试帧
        for i in range(total_frames):
            frame = generate_test_frame(i)
            out.write(frame)

        out.release()

        # 获取文件大小
        file_size = os.path.getsize(output_file)
        file_size_kb = file_size / 1024

        results.append({
            'codec': codec,
            'ext': ext,
            'size': file_size_kb,
            'file': output_file
        })

        print(f"[{codec}] 成功 - 文件大小: {file_size_kb:.2f} KB")

    except Exception as e:
        print(f"[{codec}] 失败 - {e}")

print("=" * 50)
print("\n编码格式比较结果：")
print("-" * 40)
print(f"{'编码':<10} {'扩展名':<8} {'大小(KB)':<15}")
print("-" * 40)
for r in results:
    print(f"{r['codec']:<10} {r['ext']:<8} {r['size']:<15.2f}")
print("-" * 40)

if results:
    min_size = min(results, key=lambda x: x['size'])
    print(f"\n最小文件：{min_size['codec']} ({min_size['size']:.2f} KB)")
```

```python
    """

    参数:
        input_video: 输入视频路径
        output_video: 输出视频路径

        frame_num += 1

    print(f"视频处理完成！保存到：{temp_output}")
```

### 示例5：实时效果录制

```python
"""
示例5：实时效果录制
边处理边录制，本小姐的专业技能！
"""
import cv2
import numpy as np
from datetime import datetime

cap = cv2.VideoCapture(0)

if not cap.isOpened():
    print("无法打开摄像头！")
    exit()

frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
fps = 30

# 效果列表
effects = ['normal', 'gray', 'blur', 'edge', 'cartoon']
current_effect = 0

# 录制状态
recording = False
out = None
frame_count = 0

def apply_effect(frame, effect_name):
    """应用视觉效果"""
    if effect_name == 'normal':
        return frame
    elif effect_name == 'gray':
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        return cv2.cvtColor(gray, cv2.COLOR_GRAY2BGR)
    elif effect_name == 'blur':
        return cv2.GaussianBlur(frame, (21, 21), 0)
    elif effect_name == 'edge':
        edges = cv2.Canny(frame, 100, 200)
        return cv2.cvtColor(edges, cv2.COLOR_GRAY2BGR)
    elif effect_name == 'cartoon':
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        gray = cv2.medianBlur(gray, 5)
        edges = cv2.adaptiveThreshold(gray, 255,
                                      cv2.ADAPTIVE_THRESH_MEAN_C,
                                      cv2.THRESH_BINARY, 9, 9)
        color = cv2.bilateralFilter(frame, 9, 300, 300)
        cartoon = cv2.bitwise_and(color, color, mask=edges)
        return cartoon
    return frame

print("控制说明：")
print("  'e' : 切换效果")
print("  'r' : 开始/停止录制")
print("  'q' : 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # 应用当前效果
    processed = apply_effect(frame, effects[current_effect])

    # 录制处理后的帧
    if recording and out is not None:
        out.write(processed)
        frame_count += 1

    # 显示信息
    info_frame = processed.copy()
    cv2.putText(info_frame, f'Effect: {effects[current_effect]}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    if recording:
        cv2.circle(info_frame, (frame_width - 30, 30), 10, (0, 0, 255), -1)
        cv2.putText(info_frame, f'REC: {frame_count}', (frame_width - 150, 35),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

    cv2.imshow('Effect Recorder', info_frame)

    key = cv2.waitKey(1) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('e'):
        current_effect = (current_effect + 1) % len(effects)
        print(f"切换效果：{effects[current_effect]}")
    elif key == ord('r'):
        if not recording:
            # 开始录制
            timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
            filename = f'effect_recording_{timestamp}.avi'
            fourcc = cv2.VideoWriter_fourcc(*'XVID')
            out = cv2.VideoWriter(filename, fourcc, fps, (frame_width, frame_height))
            recording = True
            frame_count = 0
            print(f"开始录制：{filename}")
        else:
            # 停止录制
            recording = False
            if out:
                out.release()
                out = None
            print(f"停止录制，共 {frame_count} 帧")

cap.release()
if out:
    out.release()
cv2.destroyAllWindows()
```

### 示例6：定时录制

```python
"""
示例6：定时录制
设定时长自动录制，适合监控场景！
"""
import cv2
import time
from datetime import datetime

def timed_recording(duration_seconds, output_prefix='timed_rec'):
    """
    定时录制指定时长的视频

    参数:
        duration_seconds: 录制时长（秒）
        output_prefix: 输出文件名前缀
    """
    cap = cv2.VideoCapture(0)

    if not cap.isOpened():
        print("无法打开摄像头！")
        return

    frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = 30

    # 生成文件名
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    output_file = f'{output_prefix}_{timestamp}.avi'

    fourcc = cv2.VideoWriter_fourcc(*'XVID')
    out = cv2.VideoWriter(output_file, fourcc, fps, (frame_width, frame_height))

    start_time = time.time()
    frame_count = 0

    print(f"开始录制 {duration_seconds} 秒...")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        elapsed_time = time.time() - start_time
        remaining_time = duration_seconds - elapsed_time

        if remaining_time <= 0:
            break

        # 写入帧
        out.write(frame)
        frame_count += 1

        # 显示倒计时
        display_frame = frame.copy()
        cv2.putText(display_frame, f'Recording: {remaining_time:.1f}s remaining',
                    (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
        cv2.putText(display_frame, f'Frames: {frame_count}',
                    (10, 60), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

        # 进度条
        progress = int((elapsed_time / duration_seconds) * frame_width)
        cv2.rectangle(display_frame, (0, frame_height - 20),
                      (progress, frame_height), (0, 255, 0), -1)

        cv2.imshow('Timed Recording', display_frame)

        if cv2.waitKey(1) & 0xFF == ord('q'):
            print("用户中断录制")
            break

    cap.release()
    out.release()
    cv2.destroyAllWindows()

    actual_duration = frame_count / fps
    print(f"\n录制完成！")
    print(f"  文件：{output_file}")
    print(f"  帧数：{frame_count}")
    print(f"  时长：{actual_duration:.2f} 秒")

# 使用示例：录制10秒
timed_recording(10)
```

### 示例7：分段录制

```python
"""
示例7：分段录制
自动分割成多个视频文件，适合长时间录制！
"""
import cv2
import time
from datetime import datetime

def segmented_recording(segment_duration=60, max_segments=10):
    """
    分段录制视频

    参数:
        segment_duration: 每段视频时长（秒）
        max_segments: 最大段数
    """
    cap = cv2.VideoCapture(0)

    if not cap.isOpened():
        print("无法打开摄像头！")
        return

    frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = 30
    fourcc = cv2.VideoWriter_fourcc(*'XVID')

    segment_count = 0
    frame_count = 0
    out = None
    segment_start_time = None

    def start_new_segment():
        nonlocal out, segment_count, frame_count, segment_start_time

        # 关闭上一段
        if out is not None:
            out.release()

        segment_count += 1
        frame_count = 0

        # 创建新文件
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = f'segment_{segment_count:03d}_{timestamp}.avi'
        out = cv2.VideoWriter(filename, fourcc, fps, (frame_width, frame_height))
        segment_start_time = time.time()

        print(f"\n开始录制第 {segment_count} 段：{filename}")
        return filename

    print(f"分段录制模式：每段 {segment_duration} 秒，最多 {max_segments} 段")
    print("按 'q' 停止录制\n")

    current_file = start_new_segment()

    while segment_count <= max_segments:
        ret, frame = cap.read()
        if not ret:
            break

        # 检查是否需要开始新段
        elapsed = time.time() - segment_start_time
        if elapsed >= segment_duration:
            current_file = start_new_segment()

        # 写入帧
        out.write(frame)
        frame_count += 1

        # 显示信息
        display_frame = frame.copy()
        remaining = segment_duration - elapsed
        cv2.putText(display_frame, f'Segment: {segment_count}/{max_segments}',
                    (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.putText(display_frame, f'Remaining: {remaining:.1f}s',
                    (10, 60), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.putText(display_frame, f'Frames: {frame_count}',
                    (10, 90), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

        # 录制指示
        cv2.circle(display_frame, (frame_width - 30, 30), 10, (0, 0, 255), -1)

        cv2.imshow('Segmented Recording', display_frame)

        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

    # 清理
    if out:
        out.release()
    cap.release()
    cv2.destroyAllWindows()

    print(f"\n录制结束，共 {segment_count} 段")

# 使用示例：每段30秒，最多5段
segmented_recording(segment_duration=30, max_segments=5)
```

### 示例8：GIF动图保存

```python
"""
示例8：GIF动图保存
保存成GIF动图，发表情包超方便！
"""
import cv2
import numpy as np

# 注意：OpenCV不直接支持GIF，我们需要使用imageio或PIL
try:
    import imageio
    HAS_IMAGEIO = True
except ImportError:
    HAS_IMAGEIO = False
    print("提示：安装 imageio 库以支持GIF保存")
    print("pip install imageio")

def create_gif_from_camera(output_file='output.gif', duration=3, fps=10):
    """
    从摄像头录制GIF

    参数:
        output_file: 输出GIF文件名
        duration: 录制时长（秒）
        fps: GIF帧率
    """
    if not HAS_IMAGEIO:
        print("需要安装 imageio 库！")
        return

    cap = cv2.VideoCapture(0)

    if not cap.isOpened():
        print("无法打开摄像头！")
        return

    frames = []
    total_frames = duration * fps
    frame_interval = int(30 / fps)  # 假设摄像头30fps

    print(f"录制 {duration} 秒 GIF，帧率 {fps}fps...")
    print("按 'q' 提前结束")

    frame_count = 0
    capture_count = 0

    while capture_count < total_frames:
        ret, frame = cap.read()
        if not ret:
            break

        frame_count += 1

        # 按间隔采样
        if frame_count % frame_interval == 0:
            # 调整大小以减小GIF体积
            small_frame = cv2.resize(frame, (320, 240))
            # BGR转RGB
            rgb_frame = cv2.cvtColor(small_frame, cv2.COLOR_BGR2RGB)
            frames.append(rgb_frame)
            capture_count += 1

        # 显示
        display = frame.copy()
        cv2.putText(display, f'GIF Recording: {capture_count}/{total_frames}',
                    (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.imshow('GIF Recorder', display)

        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

    cap.release()
    cv2.destroyAllWindows()

    if frames:
        print(f"保存GIF：{len(frames)} 帧...")
        imageio.mimsave(output_file, frames, fps=fps, loop=0)
        print(f"GIF已保存：{output_file}")
    else:
        print("没有录制到帧！")

# 不使用imageio的替代方案

    保存帧序列（可用其他工具转换为GIF）
    """

    os.makedirs(output_dir, exist_ok=True)

    cap = cv2.VideoCapture(0)

    if not cap.isOpened():
        print("无法打开摄像头！")
        return

    total_frames = duration * fps
    frame_interval = int(30 / fps)

    frame_count = 0
    save_count = 0

    print(f"保存帧序列到 {output_dir}/...")

    while save_count < total_frames:
        ret, frame = cap.read()
        if not ret:
            break

        frame_count += 1

        if frame_count % frame_interval == 0:
            small_frame = cv2.resize(frame, (320, 240))
            filename = os.path.join(output_dir, f'frame_{save_count:04d}.png')
            cv2.imwrite(filename, small_frame)
            save_count += 1

        display = frame.copy()
        cv2.putText(display, f'Saving: {save_count}/{total_frames}',
                    (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.imshow('Frame Saver', display)

        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

    cap.release()
    cv2.destroyAllWindows()

    print(f"已保存 {save_count} 帧到 {output_dir}/")
    print("可使用以下命令转换为GIF：")
    print(f"ffmpeg -framerate {fps} -i {output_dir}/frame_%04d.png output.gif")
```

```python
"""
示例9：视频水印添加
给视频加水印，保护你的作品！
"""
import cv2
import numpy as np

def add_watermark_to_video(input_video, output_video, watermark_text="哈雷酱",
                           position='bottom-right', opacity=0.7):
    """
    给视频添加文字水印

    参数:
        input_video: 输入视频
        output_video: 输出视频
        watermark_text: 水印文字
        position: 位置 (top-left, top-right, bottom-left, bottom-right, center)
        opacity: 透明度 (0-1)
    """
    cap = cv2.VideoCapture(input_video)

    if not cap.isOpened():
        print("无法打开视频！")
        return

    frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = cap.get(cv2.CAP_PROP_FPS)
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

    fourcc = cv2.VideoWriter_fourcc(*'XVID')
    out = cv2.VideoWriter(output_video, fourcc, fps, (frame_width, frame_height))

    # 计算水印位置
    font = cv2.FONT_HERSHEY_SIMPLEX
    font_scale = 1.0
    thickness = 2
    (text_width, text_height), baseline = cv2.getTextSize(
        watermark_text, font, font_scale, thickness)

    margin = 10
    positions = {
        'top-left': (margin, text_height + margin),
        'top-right': (frame_width - text_width - margin, text_height + margin),
        'bottom-left': (margin, frame_height - margin),
        'bottom-right': (frame_width - text_width - margin, frame_height - margin),
        'center': ((frame_width - text_width) // 2, (frame_height + text_height) // 2)
    }
    text_pos = positions.get(position, positions['bottom-right'])

    frame_count = 0
    print(f"添加水印中...")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        # 创建水印层
        overlay = frame.copy()
        cv2.putText(overlay, watermark_text, text_pos, font, font_scale,
                    (255, 255, 255), thickness + 2)  # 描边
        cv2.putText(overlay, watermark_text, text_pos, font, font_scale,
                    (0, 0, 0), thickness)  # 文字

        # 混合
        frame = cv2.addWeighted(overlay, opacity, frame, 1 - opacity, 0)

        out.write(frame)
        frame_count += 1

        if frame_count % 100 == 0:
            print(f"进度：{frame_count}/{total_frames}")

    cap.release()
    out.release()
    print(f"水印视频已保存：{output_video}")

# 图片水印版本
def add_image_watermark(input_video, output_video, watermark_image,
                        position='bottom-right', scale=0.2, opacity=0.7):
    """
    给视频添加图片水印
    """
    cap = cv2.VideoCapture(input_video)
    watermark = cv2.imread(watermark_image, cv2.IMREAD_UNCHANGED)

    if not cap.isOpened():
        print("无法打开视频！")
        return

    if watermark is None:
        print("无法加载水印图片！")
        return

    frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = cap.get(cv2.CAP_PROP_FPS)

    fourcc = cv2.VideoWriter_fourcc(*'XVID')
    out = cv2.VideoWriter(output_video, fourcc, fps, (frame_width, frame_height))

    # 缩放水印
    wm_height = int(frame_height * scale)
    wm_width = int(watermark.shape[1] * (wm_height / watermark.shape[0]))
    watermark = cv2.resize(watermark, (wm_width, wm_height))

    # 计算位置
    margin = 10
    if position == 'bottom-right':
        x = frame_width - wm_width - margin
        y = frame_height - wm_height - margin
    elif position == 'bottom-left':
        x = margin
        y = frame_height - wm_height - margin
    else:
        x = margin
        y = margin

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        # 叠加水印
        if watermark.shape[2] == 4:  # 有Alpha通道
            alpha = watermark[:, :, 3] / 255.0 * opacity
            for c in range(3):
                frame[y:y+wm_height, x:x+wm_width, c] = \
                    frame[y:y+wm_height, x:x+wm_width, c] * (1 - alpha) + \
                    watermark[:, :, c] * alpha
        else:
            roi = frame[y:y+wm_height, x:x+wm_width]
            frame[y:y+wm_height, x:x+wm_width] = cv2.addWeighted(
                watermark[:,:,:3], opacity, roi, 1 - opacity, 0)

        out.write(frame)

    cap.release()
    out.release()
    print(f"图片水印视频已保存：{output_video}")
```

```python
"""
示例10：视频质量设置
控制输出视频的质量和大小，专业人士必备技能！
"""
import cv2
import os

def record_with_quality(output_prefix='quality_test', duration=5):
    """
    以不同质量设置录制视频进行比较
    """
    cap = cv2.VideoCapture(0)

    if not cap.isOpened():
        print("无法打开摄像头！")
        return

    frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = 30

    # 不同质量配置
    quality_configs = [
        {'name': 'low', 'resolution': (320, 240), 'codec': 'XVID'},
        {'name': 'medium', 'resolution': (640, 480), 'codec': 'XVID'},
        {'name': 'high', 'resolution': (1280, 720), 'codec': 'MJPG'},
    ]

    writers = []
    for config in quality_configs:
        filename = f"{output_prefix}_{config['name']}.avi"
        fourcc = cv2.VideoWriter_fourcc(*config['codec'])
        out = cv2.VideoWriter(filename, fourcc, fps, config['resolution'])
        writers.append({
            'writer': out,
            'config': config,
            'filename': filename
        })

    frame_count = 0
    total_frames = fps * duration

    print(f"同时录制 {len(quality_configs)} 种质量设置...")

    while frame_count < total_frames:
        ret, frame = cap.read()
        if not ret:
            break

        # 写入各种质量版本
        for w in writers:
            resized = cv2.resize(frame, w['config']['resolution'])
            w['writer'].write(resized)

        frame_count += 1

        # 显示进度
        display = frame.copy()
        cv2.putText(display, f'Recording: {frame_count}/{total_frames}',
                    (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.imshow('Quality Test', display)

        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

    # 释放资源
    cap.release()
    for w in writers:
        w['writer'].release()
    cv2.destroyAllWindows()

    # 显示结果
    print("\n质量比较结果：")
    print("-" * 60)
    print(f"{'质量':<10} {'分辨率':<15} {'编码':<10} {'文件大小':<15}")
    print("-" * 60)

    for w in writers:
        if os.path.exists(w['filename']):
            size_kb = os.path.getsize(w['filename']) / 1024

            print(f"{w['config']['name']:<10} {res:<15} {w['config']['codec']:<10} {size_kb:.2f} KB")

    print("-" * 60)
```

## 三、常用FourCC编码

| FourCC | 说明 | 优点 | 缺点 |
|--------|------|------|------|
| XVID | MPEG-4编码 | 兼容性好，压缩率高 | 需要解码器 |
| MJPG | Motion JPEG | 质量好，编辑方便 | 文件较大 |
| mp4v | MPEG-4 Part 2 | MP4格式通用 | 压缩效率一般 |
| avc1 | H.264/AVC | 高压缩率，质量好 | 编码慢 |
| DIVX | DivX MPEG-4 | 兼容老设备 | 需要解码器 |
