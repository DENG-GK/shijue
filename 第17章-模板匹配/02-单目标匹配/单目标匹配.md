# 单目标匹配

## 1. 单目标匹配概述

哼，笨蛋，单目标匹配是模板匹配最基本的应用场景！就是在图像中找到唯一一个最佳匹配位置！本小姐来教你怎么做！(￣▽￣)／

### 1.1 什么是单目标匹配

单目标匹配假设：
- 图像中只有一个目标
- 或者只需要找到最佳匹配的那一个

### 1.2 基本流程

```
1. 读取源图像和模板
       ↓
2. 执行 cv2.matchTemplate()
       ↓
3. 使用 cv2.minMaxLoc() 找最佳位置
       ↓
4. 根据方法选择最大/最小值位置
       ↓
5. 绘制检测框
```

## 2. 标准实现

### 2.1 代码示例1：完整单目标检测

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def single_object_matching(source, template, method=cv2.TM_CCOEFF_NORMED):
    """
    单目标模板匹配

    参数:
        source: 源图像（灰度）
        template: 模板图像（灰度）
        method: 匹配方法

    返回:
        location: 匹配位置 (x, y)
        confidence: 匹配置信度
    """
    # 执行模板匹配
    result = cv2.matchTemplate(source, template, method)

    # 根据方法确定最佳位置
    min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)

    # SQDIFF方法使用最小值，其他使用最大值
    if method in [cv2.TM_SQDIFF, cv2.TM_SQDIFF_NORMED]:
        location = min_loc
        # 对于SQDIFF_NORMED，值越小越好，转换为置信度
        if method == cv2.TM_SQDIFF_NORMED:
            confidence = 1 - min_val
        else:
            confidence = min_val  # 原始值
    else:
        location = max_loc
        confidence = max_val

    return location, confidence, result

# 创建测试图像
source = np.zeros((400, 500), dtype=np.uint8)
source[:] = 100  # 灰色背景

# 添加目标
cv2.rectangle(source, (200, 150), (280, 230), 220, -1)
cv2.circle(source, (240, 190), 25, 255, -1)

# 添加干扰物
cv2.rectangle(source, (50, 80), (100, 130), 180, -1)
cv2.circle(source, (400, 300), 40, 200, -1)

# 创建模板
template = np.zeros((80, 80), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (80, 80), 220, -1)
cv2.circle(template, (40, 40), 25, 255, -1)

# 执行匹配
location, confidence, result = single_object_matching(source, template)

print(f"检测结果:")
print(f"  匹配位置: {location}")
print(f"  置信度: {confidence:.4f}")

# 绘制结果
th, tw = template.shape
source_color = cv2.cvtColor(source, cv2.COLOR_GRAY2BGR)
cv2.rectangle(source_color, location,
              (location[0] + tw, location[1] + th),
              (0, 255, 0), 2)
cv2.putText(source_color, f'{confidence:.3f}',
            (location[0], location[1] - 10),
            cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

# 可视化
fig, axes = plt.subplots(2, 2, figsize=(12, 10))

axes[0, 0].imshow(source, cmap='gray')
axes[0, 0].set_title('源图像')
axes[0, 0].axis('off')

axes[0, 1].imshow(template, cmap='gray')
axes[0, 1].set_title('模板')
axes[0, 1].axis('off')

im = axes[1, 0].imshow(result, cmap='hot')
axes[1, 0].plot(location[0], location[1], 'g*', markersize=15)
axes[1, 0].set_title('匹配热力图')
plt.colorbar(im, ax=axes[1, 0], fraction=0.046)

axes[1, 1].imshow(cv2.cvtColor(source_color, cv2.COLOR_BGR2RGB))
axes[1, 1].set_title(f'检测结果 (置信度: {confidence:.3f})')
axes[1, 1].axis('off')

plt.tight_layout()
plt.savefig('single_object_matching.png', dpi=150, bbox_inches='tight')
plt.show()
```

### 2.2 代码示例2：带阈值的检测

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def detect_with_threshold(source, template, method=cv2.TM_CCOEFF_NORMED, threshold=0.8):
    """
    带阈值的单目标检测

    参数:
        threshold: 置信度阈值，低于此值认为未检测到

    返回:
        detected: 是否检测到目标
        location: 检测位置
        confidence: 置信度
    """
    result = cv2.matchTemplate(source, template, method)
    min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)

    if method in [cv2.TM_SQDIFF, cv2.TM_SQDIFF_NORMED]:
        location = min_loc
        confidence = 1 - min_val if method == cv2.TM_SQDIFF_NORMED else -min_val
    else:
        location = max_loc
        confidence = max_val

    detected = confidence >= threshold

    return detected, location, confidence

# 测试不同场景
def create_test_scene(has_target=True):
    scene = np.zeros((300, 400), dtype=np.uint8)
    scene[:] = 100

    if has_target:
        # 添加目标
        cv2.rectangle(scene, (150, 100), (220, 170), 220, -1)
        cv2.circle(scene, (185, 135), 20, 255, -1)

    # 添加干扰物
    cv2.rectangle(scene, (50, 50), (100, 100), 180, -1)
    cv2.ellipse(scene, (320, 200), (40, 30), 0, 0, 360, 200, -1)

    return scene

# 模板
template = np.zeros((70, 70), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (70, 70), 220, -1)
cv2.circle(template, (35, 35), 20, 255, -1)

# 测试有目标和无目标的情况
scenes = [
    (create_test_scene(True), '有目标'),
    (create_test_scene(False), '无目标'),
]

fig, axes = plt.subplots(2, 2, figsize=(12, 10))

threshold = 0.7

for i, (scene, title) in enumerate(scenes):
    detected, location, confidence = detect_with_threshold(scene, template, threshold=threshold)

    scene_color = cv2.cvtColor(scene, cv2.COLOR_GRAY2BGR)
    th, tw = template.shape

    if detected:
        cv2.rectangle(scene_color, location,
                     (location[0] + tw, location[1] + th),
                     (0, 255, 0), 2)
        status = "检测到目标"
        color = 'green'
    else:
        status = "未检测到目标"
        color = 'red'

    axes[i, 0].imshow(scene, cmap='gray')
    axes[i, 0].set_title(f'{title}')
    axes[i, 0].axis('off')

    axes[i, 1].imshow(cv2.cvtColor(scene_color, cv2.COLOR_BGR2RGB))
    axes[i, 1].set_title(f'{status}\n置信度: {confidence:.3f} (阈值: {threshold})', color=color)
    axes[i, 1].axis('off')

plt.suptitle('带阈值的目标检测', fontsize=14)
plt.tight_layout()
plt.savefig('threshold_detection.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 3. 彩色图像匹配

### 3.1 代码示例3：彩色图像处理

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def color_template_matching(source_color, template_color, method=cv2.TM_CCOEFF_NORMED):
    """
    彩色图像模板匹配

    策略1: 转换为灰度后匹配
    策略2: 分通道匹配后融合
    """
    # 策略1: 灰度匹配
    source_gray = cv2.cvtColor(source_color, cv2.COLOR_BGR2GRAY)
    template_gray = cv2.cvtColor(template_color, cv2.COLOR_BGR2GRAY)

    result_gray = cv2.matchTemplate(source_gray, template_gray, method)
    _, max_val_gray, _, max_loc_gray = cv2.minMaxLoc(result_gray)

    # 策略2: 分通道匹配
    results_channels = []
    for i in range(3):
        result_ch = cv2.matchTemplate(source_color[:,:,i],
                                      template_color[:,:,i], method)
        results_channels.append(result_ch)

    # 融合三通道结果（取平均）
    result_combined = np.mean(results_channels, axis=0)
    _, max_val_combined, _, max_loc_combined = cv2.minMaxLoc(result_combined)

    return {
        'gray': {'result': result_gray, 'location': max_loc_gray, 'confidence': max_val_gray},
        'combined': {'result': result_combined, 'location': max_loc_combined, 'confidence': max_val_combined}
    }

# 创建彩色测试图像
source = np.zeros((300, 400, 3), dtype=np.uint8)
source[:] = [100, 120, 100]  # 绿色调背景

# 红色目标
cv2.rectangle(source, (150, 100), (230, 180), (0, 0, 220), -1)
cv2.circle(source, (190, 140), 25, (50, 50, 255), -1)

# 蓝色干扰
cv2.rectangle(source, (50, 50), (100, 100), (200, 100, 50), -1)

# 创建模板
template = np.zeros((80, 80, 3), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (80, 80), (0, 0, 220), -1)
cv2.circle(template, (40, 40), 25, (50, 50, 255), -1)

# 执行匹配
results = color_template_matching(source, template)

# 可视化
fig, axes = plt.subplots(2, 3, figsize=(15, 10))

axes[0, 0].imshow(cv2.cvtColor(source, cv2.COLOR_BGR2RGB))
axes[0, 0].set_title('源图像（彩色）')
axes[0, 0].axis('off')

axes[0, 1].imshow(cv2.cvtColor(template, cv2.COLOR_BGR2RGB))
axes[0, 1].set_title('模板（彩色）')
axes[0, 1].axis('off')

axes[0, 2].imshow(cv2.cvtColor(source, cv2.COLOR_BGR2GRAY), cmap='gray')
axes[0, 2].set_title('灰度转换')
axes[0, 2].axis('off')

# 灰度匹配结果
th, tw = template.shape[:2]
gray_res = results['gray']
source_result1 = source.copy()
cv2.rectangle(source_result1, gray_res['location'],
              (gray_res['location'][0]+tw, gray_res['location'][1]+th),
              (0, 255, 0), 2)

axes[1, 0].imshow(cv2.cvtColor(source_result1, cv2.COLOR_BGR2RGB))
axes[1, 0].set_title(f"灰度匹配\n置信度: {gray_res['confidence']:.3f}")
axes[1, 0].axis('off')

# 热力图
im1 = axes[1, 1].imshow(gray_res['result'], cmap='hot')
axes[1, 1].set_title('灰度匹配热力图')
plt.colorbar(im1, ax=axes[1, 1], fraction=0.046)

# 分通道融合结果
combined_res = results['combined']
source_result2 = source.copy()
cv2.rectangle(source_result2, combined_res['location'],
              (combined_res['location'][0]+tw, combined_res['location'][1]+th),
              (255, 0, 0), 2)

im2 = axes[1, 2].imshow(combined_res['result'], cmap='hot')
axes[1, 2].set_title(f"分通道融合\n置信度: {combined_res['confidence']:.3f}")
plt.colorbar(im2, ax=axes[1, 2], fraction=0.046)

plt.tight_layout()
plt.savefig('color_matching.png', dpi=150, bbox_inches='tight')
plt.show()

print("匹配结果对比:")
print(f"灰度方法 - 位置: {gray_res['location']}, 置信度: {gray_res['confidence']:.4f}")
print(f"分通道方法 - 位置: {combined_res['location']}, 置信度: {combined_res['confidence']:.4f}")
```

## 4. ROI区域匹配

### 4.1 代码示例4：指定区域内匹配

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def roi_template_matching(source, template, roi=None, method=cv2.TM_CCOEFF_NORMED):
    """
    在指定ROI区域内进行模板匹配

    参数:
        roi: (x, y, w, h) 感兴趣区域，None表示全图搜索
    """
    if roi is not None:
        x, y, w, h = roi
        source_roi = source[y:y+h, x:x+w]
        offset = (x, y)
    else:
        source_roi = source
        offset = (0, 0)

    # 检查ROI是否足够大
    th, tw = template.shape[:2]
    if source_roi.shape[0] < th or source_roi.shape[1] < tw:
        raise ValueError("ROI区域小于模板尺寸")

    # 执行匹配
    result = cv2.matchTemplate(source_roi, template, method)

    if method in [cv2.TM_SQDIFF, cv2.TM_SQDIFF_NORMED]:
        _, _, loc, _ = cv2.minMaxLoc(result)
        confidence = 1 - cv2.minMaxLoc(result)[0] if method == cv2.TM_SQDIFF_NORMED else None
    else:
        _, confidence, _, loc = cv2.minMaxLoc(result)

    # 转换回原图坐标
    global_loc = (loc[0] + offset[0], loc[1] + offset[1])

    return global_loc, confidence, result

# 创建大图像
source = np.zeros((500, 700), dtype=np.uint8)
source[:] = 80

# 在不同位置添加相似目标
positions = [(100, 100), (400, 150), (300, 350)]
for pos in positions:
    cv2.rectangle(source, pos, (pos[0]+60, pos[1]+60), 200, -1)
    cv2.circle(source, (pos[0]+30, pos[1]+30), 15, 255, -1)

# 模板
template = np.zeros((60, 60), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (60, 60), 200, -1)
cv2.circle(template, (30, 30), 15, 255, -1)

# 定义不同的ROI
rois = [
    None,                    # 全图搜索
    (50, 50, 200, 200),      # 左上区域
    (250, 100, 250, 200),    # 中间区域
    (200, 280, 250, 200),    # 下方区域
]

fig, axes = plt.subplots(2, 2, figsize=(14, 12))

for idx, roi in enumerate(rois):
    ax = axes[idx // 2, idx % 2]

    # 执行ROI匹配
    location, confidence, _ = roi_template_matching(source, template, roi)

    # 绘制结果
    source_color = cv2.cvtColor(source, cv2.COLOR_GRAY2BGR)
    th, tw = template.shape

    # 绘制ROI区域
    if roi is not None:
        x, y, w, h = roi
        cv2.rectangle(source_color, (x, y), (x+w, y+h), (255, 255, 0), 2)

    # 绘制检测结果
    cv2.rectangle(source_color, location,
                  (location[0]+tw, location[1]+th),
                  (0, 255, 0), 2)

    roi_text = f'ROI: {roi}' if roi else '全图搜索'
    ax.imshow(cv2.cvtColor(source_color, cv2.COLOR_BGR2RGB))
    ax.set_title(f'{roi_text}\n检测位置: {location}, 置信度: {confidence:.3f}')
    ax.axis('off')

plt.suptitle('ROI区域模板匹配', fontsize=14)
plt.tight_layout()
plt.savefig('roi_matching.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 6. 实际应用案例

### 6.1 代码示例6：图标检测

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def detect_icon(screenshot, icon_template, threshold=0.8):
    """
    在截图中检测图标

    应用场景：自动化测试、RPA
    """
    # 转灰度
    if len(screenshot.shape) == 3:
        screenshot_gray = cv2.cvtColor(screenshot, cv2.COLOR_BGR2GRAY)
    else:
        screenshot_gray = screenshot

    if len(icon_template.shape) == 3:
        template_gray = cv2.cvtColor(icon_template, cv2.COLOR_BGR2GRAY)
    else:
        template_gray = icon_template

    # 模板匹配
    result = cv2.matchTemplate(screenshot_gray, template_gray, cv2.TM_CCOEFF_NORMED)
    _, max_val, _, max_loc = cv2.minMaxLoc(result)

    # 判断是否检测到
    if max_val >= threshold:
        th, tw = template_gray.shape
        center = (max_loc[0] + tw // 2, max_loc[1] + th // 2)
        return {
            'found': True,
            'location': max_loc,
            'center': center,
            'confidence': max_val,
            'bbox': (max_loc[0], max_loc[1], tw, th)
        }
    else:
        return {
            'found': False,
            'confidence': max_val
        }

# 模拟屏幕截图
screenshot = np.zeros((400, 600, 3), dtype=np.uint8)
screenshot[:] = [240, 240, 240]  # 灰色背景

# 添加模拟按钮
button_positions = [(50, 50), (200, 50), (350, 50)]
for i, pos in enumerate(button_positions):
    color = [(100, 100, 200), (100, 200, 100), (200, 100, 100)][i]
    cv2.rectangle(screenshot, pos, (pos[0]+80, pos[1]+40), color, -1)
    cv2.rectangle(screenshot, pos, (pos[0]+80, pos[1]+40), (50, 50, 50), 2)

# 添加目标图标
target_pos = (250, 200)
cv2.rectangle(screenshot, target_pos, (target_pos[0]+60, target_pos[1]+60),
              (50, 150, 50), -1)
cv2.circle(screenshot, (target_pos[0]+30, target_pos[1]+30), 20, (100, 255, 100), -1)

# 创建要查找的图标模板
icon = np.zeros((60, 60, 3), dtype=np.uint8)
cv2.rectangle(icon, (0, 0), (60, 60), (50, 150, 50), -1)
cv2.circle(icon, (30, 30), 20, (100, 255, 100), -1)

# 检测图标
result = detect_icon(screenshot, icon)

print("图标检测结果:")
if result['found']:
    print(f"  状态: 找到图标")
    print(f"  位置: {result['location']}")
    print(f"  中心点: {result['center']}")
    print(f"  置信度: {result['confidence']:.4f}")
else:
    print(f"  状态: 未找到图标")
    print(f"  最高置信度: {result['confidence']:.4f}")

# 可视化
fig, axes = plt.subplots(1, 3, figsize=(15, 5))

axes[0].imshow(cv2.cvtColor(screenshot, cv2.COLOR_BGR2RGB))
axes[0].set_title('屏幕截图')
axes[0].axis('off')

axes[1].imshow(cv2.cvtColor(icon, cv2.COLOR_BGR2RGB))
axes[1].set_title('目标图标')
axes[1].axis('off')

# 绘制检测结果
screenshot_result = screenshot.copy()
if result['found']:
    x, y, w, h = result['bbox']
    cv2.rectangle(screenshot_result, (x, y), (x+w, y+h), (0, 255, 0), 3)
```

```python
import cv2
import numpy as np
import time
import matplotlib.pyplot as plt

class RealtimeTemplateTracker:
    """实时模板追踪器"""

    def __init__(self, template, method=cv2.TM_CCOEFF_NORMED, threshold=0.7):
        self.template = template
        self.method = method
        self.threshold = threshold
        self.th, self.tw = template.shape[:2]

        # 转灰度
        if len(template.shape) == 3:
            self.template_gray = cv2.cvtColor(template, cv2.COLOR_BGR2GRAY)
        else:
            self.template_gray = template

    def detect(self, frame):
        """检测单帧"""
        # 转灰度
        if len(frame.shape) == 3:
            frame_gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        else:
            frame_gray = frame

        # 模板匹配
        result = cv2.matchTemplate(frame_gray, self.template_gray, self.method)
        _, max_val, _, max_loc = cv2.minMaxLoc(result)

        if max_val >= self.threshold:
            return max_loc, max_val
        return None, max_val

    def draw_result(self, frame, location, confidence):
        """绘制检测结果"""
        frame_draw = frame.copy()
        if location is not None:
            cv2.rectangle(frame_draw, location,
                         (location[0] + self.tw, location[1] + self.th),
                         (0, 255, 0), 2)
            cv2.putText(frame_draw, f'{confidence:.2f}',
                       (location[0], location[1] - 10),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
        return frame_draw

# 模拟视频帧序列
def generate_video_frames(num_frames=50):
    """生成模拟视频帧"""
    frames = []

    # 创建移动的目标
    template = np.zeros((50, 50), dtype=np.uint8)
    cv2.rectangle(template, (5, 5), (45, 45), 200, -1)
    cv2.circle(template, (25, 25), 12, 255, -1)

    for i in range(num_frames):
        frame = np.zeros((300, 400), dtype=np.uint8)
        frame[:] = 80

        # 目标位置（正弦移动）
        x = int(100 + 150 * (0.5 + 0.5 * np.sin(i * 0.15)))
        y = int(100 + 50 * np.sin(i * 0.1))

        # 添加目标
        frame[y:y+50, x:x+50] = template

        # 添加噪声
        noise = np.random.randint(0, 20, frame.shape, dtype=np.uint8)
        frame = cv2.add(frame, noise)

        frames.append(frame)

    return frames, template

# 生成视频
frames, template = generate_video_frames(50)

# 创建追踪器
tracker = RealtimeTemplateTracker(template, threshold=0.6)

# 处理所有帧
processing_times = []
detections = []

for frame in frames:
    start = time.time()
    location, confidence = tracker.detect(frame)
    processing_times.append((time.time() - start) * 1000)
    detections.append((location, confidence))

avg_time = np.mean(processing_times)
fps = 1000 / avg_time
detection_rate = sum(1 for loc, _ in detections if loc is not None) / len(detections)

print(f"性能统计:")
print(f"  平均处理时间: {avg_time:.2f} ms")
print(f"  理论帧率: {fps:.1f} FPS")
print(f"  检测成功率: {detection_rate:.1%}")

# 可视化几帧结果
fig, axes = plt.subplots(2, 4, figsize=(16, 8))

sample_indices = [0, 10, 20, 30, 5, 15, 25, 35]

for idx, frame_idx in enumerate(sample_indices):
    ax = axes[idx // 4, idx % 4]
    frame = frames[frame_idx]
    location, confidence = detections[frame_idx]
    frame_result = tracker.draw_result(cv2.cvtColor(frame, cv2.COLOR_GRAY2BGR), location, confidence)

    ax.imshow(cv2.cvtColor(frame_result, cv2.COLOR_BGR2RGB))
    ax.set_title(f'帧 {frame_idx}\n置信度: {confidence:.2f}')
    ax.axis('off')

plt.suptitle(f'实时模板追踪 (平均 {fps:.1f} FPS)', fontsize=14)
plt.tight_layout()
plt.savefig('realtime_tracking.png', dpi=150, bbox_inches='tight')
```

```python
import cv2

def detect_document_marker(document, marker_template, threshold=0.75):
    """
    在文档图像中检测标记（如二维码角点、校准标记等）
    """
    # 预处理
    if len(document.shape) == 3:
        doc_gray = cv2.cvtColor(document, cv2.COLOR_BGR2GRAY)
    else:
        doc_gray = document

    if len(marker_template.shape) == 3:
        marker_gray = cv2.cvtColor(marker_template, cv2.COLOR_BGR2GRAY)
    else:
        marker_gray = marker_template

    # 二值化增强对比度
    _, doc_binary = cv2.threshold(doc_gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    _, marker_binary = cv2.threshold(marker_gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    # 模板匹配（使用二值化图像）
    result = cv2.matchTemplate(doc_binary, marker_binary, cv2.TM_CCOEFF_NORMED)
    _, max_val, _, max_loc = cv2.minMaxLoc(result)

    return {
        'found': max_val >= threshold,
        'location': max_loc,
        'confidence': max_val,
        'result_map': result
    }
```

```python
import cv2
import numpy as np
import time
import matplotlib.pyplot as plt

def pyramid_template_matching(source, template, scales=[1.0, 0.5, 0.25],
                              method=cv2.TM_CCOEFF_NORMED):
    """
    使用图像金字塔加速模板匹配

    策略：先在低分辨率图像上粗定位，再在高分辨率上精定位
    """
    results = []

    for scale in scales:
        # 缩放图像和模板
        scaled_source = cv2.resize(source, None, fx=scale, fy=scale)
        scaled_template = cv2.resize(template, None, fx=scale, fy=scale)

        # 确保模板不会太小
        if scaled_template.shape[0] < 10 or scaled_template.shape[1] < 10:
            continue

        start = time.time()
        result = cv2.matchTemplate(scaled_source, scaled_template, method)
        elapsed = (time.time() - start) * 1000

        _, max_val, _, max_loc = cv2.minMaxLoc(result)

        # 转换回原始坐标
        original_loc = (int(max_loc[0] / scale), int(max_loc[1] / scale))

        results.append({
            'scale': scale,
            'location': original_loc,
            'confidence': max_val,
            'time_ms': elapsed
        })

    return results
```
