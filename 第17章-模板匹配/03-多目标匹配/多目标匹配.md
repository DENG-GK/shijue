# 多目标匹配

## 1. 多目标匹配概述

哼，笨蛋，之前讲的单目标匹配只能找一个目标，但实际应用中经常需要找到图像中的所有匹配目标！本小姐今天就来教你多目标匹配！(￣▽￣)／

### 1.1 多目标 vs 单目标

| 特性 | 单目标匹配 | 多目标匹配 |
|------|------------|------------|
| 目标数量 | 1个 | 多个 |
| 使用函数 | minMaxLoc() | 阈值过滤 |
| 处理方式 | 取最优值 | 遍历所有高于阈值的位置 |
| 后处理 | 无需 | 需要NMS去重 |

### 1.2 基本思路

```
1. 执行模板匹配得到结果矩阵
       ↓
2. 设置匹配阈值
       ↓
3. 找出所有超过阈值的位置
       ↓
4. 应用非极大值抑制(NMS)去除重叠
       ↓
5. 返回所有检测结果
```

## 2. 基于阈值的多目标检测

### 2.1 代码示例1：基本多目标检测

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def multi_object_matching(source, template, threshold=0.8, method=cv2.TM_CCOEFF_NORMED):
    """
    多目标模板匹配

    参数:
        source: 源图像
        template: 模板
        threshold: 匹配阈值
        method: 匹配方法

    返回:
        locations: 所有匹配位置列表
    """
    # 执行模板匹配
    result = cv2.matchTemplate(source, template, method)

    # 根据方法确定阈值条件
    if method in [cv2.TM_SQDIFF, cv2.TM_SQDIFF_NORMED]:
        # SQDIFF越小越好
        locations = np.where(result <= (1 - threshold) if method == cv2.TM_SQDIFF_NORMED else threshold)
    else:
        # 其他方法越大越好
        locations = np.where(result >= threshold)

    # 转换为坐标列表
    matches = list(zip(locations[1], locations[0]))  # (x, y)格式

    return matches, result

# 创建测试图像
source = np.zeros((400, 600), dtype=np.uint8)
source[:] = 80

# 添加多个相同的目标
target_positions = [(50, 50), (200, 80), (400, 60), (100, 250), (350, 280)]
for pos in target_positions:
    cv2.rectangle(source, pos, (pos[0]+50, pos[1]+50), 200, -1)
    cv2.circle(source, (pos[0]+25, pos[1]+25), 15, 255, -1)

# 创建模板
template = np.zeros((50, 50), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (50, 50), 200, -1)
cv2.circle(template, (25, 25), 15, 255, -1)

# 执行多目标匹配
matches, result = multi_object_matching(source, template, threshold=0.9)

print(f"检测到 {len(matches)} 个匹配位置")
print(f"实际目标数: {len(target_positions)}")

# 绘制结果
th, tw = template.shape
source_color = cv2.cvtColor(source, cv2.COLOR_GRAY2BGR)

for loc in matches:
    cv2.rectangle(source_color, loc, (loc[0]+tw, loc[1]+th), (0, 255, 0), 2)

# 可视化
fig, axes = plt.subplots(1, 3, figsize=(15, 5))

axes[0].imshow(source, cmap='gray')
axes[0].set_title(f'源图像 ({len(target_positions)}个目标)')
axes[0].axis('off')

plt.tight_layout()
plt.savefig('multi_object_basic.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 3. 非极大值抑制(NMS)

### 3.1 NMS原理

NMS通过抑制重叠的检测框来消除冗余检测：
1. 按置信度排序所有检测框
2. 选择置信度最高的框
3. 删除与该框IoU大于阈值的其他框
4. 重复直到处理完所有框

### 3.2 代码示例3：实现NMS

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def non_max_suppression(boxes, scores, iou_threshold=0.5):
    """
    非极大值抑制

    参数:
        boxes: 检测框列表 [(x, y, w, h), ...]
        scores: 对应的置信度列表
        iou_threshold: IoU阈值

    返回:
        keep_indices: 保留的框索引
    """
    if len(boxes) == 0:
        return []

    boxes = np.array(boxes, dtype=np.float32)
    scores = np.array(scores)

    # 计算坐标
    x1 = boxes[:, 0]
    y1 = boxes[:, 1]
    x2 = boxes[:, 0] + boxes[:, 2]
    y2 = boxes[:, 1] + boxes[:, 3]

    # 计算面积
    areas = boxes[:, 2] * boxes[:, 3]

    # 按分数排序（降序）
    order = scores.argsort()[::-1]

    keep = []

    while order.size > 0:
        # 取当前最高分的索引
        i = order[0]
        keep.append(i)

        if order.size == 1:
            break

        # 计算与其他框的IoU
        xx1 = np.maximum(x1[i], x1[order[1:]])
        yy1 = np.maximum(y1[i], y1[order[1:]])
        xx2 = np.minimum(x2[i], x2[order[1:]])
        yy2 = np.minimum(y2[i], y2[order[1:]])

        w = np.maximum(0, xx2 - xx1)
        h = np.maximum(0, yy2 - yy1)

        intersection = w * h
        union = areas[i] + areas[order[1:]] - intersection
        iou = intersection / (union + 1e-6)

        # 保留IoU小于阈值的框
        mask = iou <= iou_threshold
        order = order[1:][mask]

    return keep

def multi_object_with_nms(source, template, threshold=0.8, nms_threshold=0.3,
                          method=cv2.TM_CCOEFF_NORMED):
    """
    带NMS的多目标匹配
    """
    result = cv2.matchTemplate(source, template, method)
    th, tw = template.shape[:2]

    # 找出所有超过阈值的位置
    if method in [cv2.TM_SQDIFF, cv2.TM_SQDIFF_NORMED]:
        locations = np.where(result <= (1 - threshold))
        scores = 1 - result[locations]
    else:
        locations = np.where(result >= threshold)
        scores = result[locations]

    # 转换为boxes格式
    boxes = [(locations[1][i], locations[0][i], tw, th)
             for i in range(len(locations[0]))]

    if len(boxes) == 0:
        return [], [], result

    # 应用NMS
    keep = non_max_suppression(boxes, scores, nms_threshold)

    final_boxes = [boxes[i] for i in keep]
    final_scores = [scores[i] for i in keep]

    return final_boxes, final_scores, result
```

## 4. 局部最大值方法

### 4.1 代码示例5：使用膨胀操作找局部最大值

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def find_local_maxima(result, threshold=0.8, min_distance=10):
    """
    使用形态学膨胀找局部最大值

    参数:
        result: 匹配结果矩阵
        threshold: 匹配阈值
        min_distance: 最小距离（对应膨胀核大小）
    """
    # 阈值过滤
    mask = result >= threshold

    # 膨胀操作找局部最大值
    kernel_size = min_distance * 2 + 1
    kernel = np.ones((kernel_size, kernel_size), dtype=np.uint8)

    # 膨胀后的结果
    dilated = cv2.dilate(result, kernel)

    # 局部最大值：膨胀后值等于原值的点
    local_max = (result == dilated) & mask

    # 获取局部最大值位置
    locations = np.where(local_max)
    matches = list(zip(locations[1], locations[0]))

    # 获取对应的分数
    scores = [result[y, x] for x, y in matches]

    return matches, scores
```

## 5. 完整多目标检测器

### 5.1 代码示例6：封装完整的检测器类

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

class MultiTemplateDetector:
    """多目标模板检测器"""

    def __init__(self, template, method=cv2.TM_CCOEFF_NORMED):
        self.template = template
        self.method = method

        if len(template.shape) == 3:
            self.template_gray = cv2.cvtColor(template, cv2.COLOR_BGR2GRAY)
        else:
            self.template_gray = template

        self.th, self.tw = self.template_gray.shape

    def detect(self, image, threshold=0.8, nms_threshold=0.3, max_detections=100):
        """
        检测所有匹配目标

        参数:
            image: 输入图像
            threshold: 匹配阈值
            nms_threshold: NMS的IoU阈值
            max_detections: 最大检测数量

        返回:
            detections: 检测结果列表
        """
        # 转灰度
        if len(image.shape) == 3:
            image_gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            image_gray = image

        # 模板匹配
        result = cv2.matchTemplate(image_gray, self.template_gray, self.method)

        # 找出超过阈值的位置
        if self.method in [cv2.TM_SQDIFF, cv2.TM_SQDIFF_NORMED]:
            locations = np.where(result <= (1 - threshold))
            scores = 1 - result[locations]
        else:
            locations = np.where(result >= threshold)
            scores = result[locations]

        if len(locations[0]) == 0:
            return []

        # 构建检测框
        boxes = np.array([[locations[1][i], locations[0][i], self.tw, self.th]
                          for i in range(len(locations[0]))], dtype=np.float32)
        scores = np.array(scores)

        # OpenCV的NMS
        indices = cv2.dnn.NMSBoxes(
            boxes.tolist(),
            scores.tolist(),
            threshold,
            nms_threshold
        )

        if len(indices) == 0:
            return []

        # 构建检测结果
        detections = []
        for i in indices.flatten()[:max_detections]:
            x, y, w, h = boxes[i]
            detections.append({
                'bbox': (int(x), int(y), int(w), int(h)),
                'center': (int(x + w/2), int(y + h/2)),
                'confidence': float(scores[i])
            })

        # 按置信度排序
        detections.sort(key=lambda x: x['confidence'], reverse=True)

        return detections

    def draw_detections(self, image, detections, color=(0, 255, 0), thickness=2):
        """绘制检测结果"""
        result = image.copy()
        if len(result.shape) == 2:
            result = cv2.cvtColor(result, cv2.COLOR_GRAY2BGR)

        for det in detections:
            x, y, w, h = det['bbox']
            conf = det['confidence']

            cv2.rectangle(result, (x, y), (x+w, y+h), color, thickness)
            cv2.putText(result, f'{conf:.2f}', (x, y-5),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 1)

        return result

# 测试检测器
source = np.zeros((450, 600), dtype=np.uint8)
source[:] = 80

# 添加多个目标
positions = [
    (50, 50), (180, 70), (320, 40), (480, 80),
    (80, 200), (250, 220), (420, 180),
    (150, 350), (350, 330)
]

for pos in positions:
    cv2.rectangle(source, pos, (pos[0]+55, pos[1]+55), 200, -1)
    cv2.circle(source, (pos[0]+27, pos[1]+27), 16, 255, -1)

# 模板

cv2.rectangle(template, (0, 0), (55, 55), 200, -1)
cv2.circle(template, (27, 27), 16, 255, -1)

# 检测

    print(f"  目标{i+1}: 位置{det['bbox'][:2]}, 置信度{det['confidence']:.3f}")

# 可视化

fig, axes = plt.subplots(1, 2, figsize=(14, 6))

axes[0].imshow(source, cmap='gray')
axes[0].set_title(f'源图像 ({len(positions)}个目标)')
axes[0].axis('off')

axes[1].imshow(cv2.cvtColor(result_image, cv2.COLOR_BGR2RGB))
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

class MultiScaleMultiObjectDetector:
    """多尺度多目标检测器"""

    def __init__(self, template, scales=[0.8, 1.0, 1.2]):
        self.original_template = template
        self.scales = scales

        if len(template.shape) == 3:
            self.template_gray = cv2.cvtColor(template, cv2.COLOR_BGR2GRAY)
        else:
            self.template_gray = template

    def detect(self, image, threshold=0.8, nms_threshold=0.3):
        """多尺度检测"""
        if len(image.shape) == 3:
            image_gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            image_gray = image

        all_boxes = []
        all_scores = []

        for scale in self.scales:
            # 缩放模板
            scaled_template = cv2.resize(
                self.template_gray, None,
                fx=scale, fy=scale,
                interpolation=cv2.INTER_LINEAR
            )

            th, tw = scaled_template.shape

            # 检查模板是否太大
            if th > image_gray.shape[0] or tw > image_gray.shape[1]:
                continue

            # 匹配
            result = cv2.matchTemplate(image_gray, scaled_template, cv2.TM_CCOEFF_NORMED)

            # 找超过阈值的位置
            locations = np.where(result >= threshold)

            for i in range(len(locations[0])):
                x = locations[1][i]
                y = locations[0][i]
                score = result[y, x]

                all_boxes.append([x, y, tw, th])
                all_scores.append(score)

        if len(all_boxes) == 0:
            return []

        # NMS
        boxes = np.array(all_boxes, dtype=np.float32)
        scores = np.array(all_scores)

        indices = cv2.dnn.NMSBoxes(
            boxes.tolist(), scores.tolist(),
            threshold, nms_threshold
        )

        if len(indices) == 0:
            return []

        detections = []
        for i in indices.flatten():
            x, y, w, h = boxes[i]
            detections.append({
                'bbox': (int(x), int(y), int(w), int(h)),
                'confidence': float(scores[i]),
                'scale': w / self.template_gray.shape[1]
            })

        return detections

# 创建包含不同尺度目标的图像
source = np.zeros((400, 600), dtype=np.uint8)
source[:] = 80

# 添加不同尺度的目标
def add_target(img, x, y, size):
    cv2.rectangle(img, (x, y), (x+size, y+size), 200, -1)
    cv2.circle(img, (x+size//2, y+size//2), size//3, 255, -1)

# 不同尺度
add_target(source, 50, 50, 50)      # 尺度 1.0

add_target(source, 400, 250, 45)    # 尺度 0.9

# 标准尺度模板
template = np.zeros((50, 50), dtype=np.uint8)
add_target(template, 0, 0, 50)

# 创建多尺度检测器
detector = MultiScaleMultiObjectDetector(template, scales=[0.8, 0.9, 1.0, 1.1, 1.2])

# 检测

print(f"多尺度检测结果: {len(detections)} 个目标")
```

### 8.1 代码示例10：批量多模板匹配

```python
import cv2
import numpy as np
import time
import matplotlib.pyplot as plt

def batch_multi_template_matching(source, templates, threshold=0.8, nms_threshold=0.3):
    """
    批量多模板匹配

    参数:
        templates: 模板列表，每个元素为(template, label)
    """
    all_detections = []

    if len(source.shape) == 3:
        source_gray = cv2.cvtColor(source, cv2.COLOR_BGR2GRAY)
    else:
        source_gray = source

    for template, label in templates:
        if len(template.shape) == 3:
            template_gray = cv2.cvtColor(template, cv2.COLOR_BGR2GRAY)
        else:
            template_gray = template

        th, tw = template_gray.shape

        # 匹配
        result = cv2.matchTemplate(source_gray, template_gray, cv2.TM_CCOEFF_NORMED)

        locations = np.where(result >= threshold)

        for i in range(len(locations[0])):
            x, y = locations[1][i], locations[0][i]
            score = result[y, x]
            all_detections.append({
                'bbox': (x, y, tw, th),
                'confidence': score,
                'label': label
            })

    if len(all_detections) == 0:
        return []

    # 全局NMS
    boxes = np.array([d['bbox'] for d in all_detections], dtype=np.float32)
    scores = np.array([d['confidence'] for d in all_detections])

    indices = cv2.dnn.NMSBoxes(
        boxes.tolist(), scores.tolist(),
        threshold, nms_threshold
    )

    if len(indices) == 0:
        return []

    return [all_detections[i] for i in indices.flatten()]

# 创建测试场景
source = np.zeros((400, 600), dtype=np.uint8)
source[:] = 100

# 添加不同类型的目标
# 类型1: 矩形
for pos in [(50, 50), (200, 100), (450, 80)]:
    cv2.rectangle(source, pos, (pos[0]+50, pos[1]+50), 200, -1)

# 类型2: 圆形
for pos in [(100, 250), (300, 280), (480, 250)]:
    cv2.circle(source, pos, 25, 220, -1)

# 类型3: 三角形
for pos in [(200, 200), (380, 180)]:
    pts = np.array([
        [pos[0], pos[1]+40],
        [pos[0]+20, pos[1]],
        [pos[0]+40, pos[1]+40]
    ])
    cv2.fillPoly(source, [pts], 180)

# 创建模板
template_rect = np.zeros((50, 50), dtype=np.uint8)
cv2.rectangle(template_rect, (0, 0), (50, 50), 200, -1)

template_circle = np.zeros((50, 50), dtype=np.uint8)
cv2.circle(template_circle, (25, 25), 25, 220, -1)

template_tri = np.zeros((40, 40), dtype=np.uint8)
pts = np.array([[0, 40], [20, 0], [40, 40]])
cv2.fillPoly(template_tri, [pts], 180)

templates = [
    (template_rect, 'Rectangle'),
    (template_circle, 'Circle'),
    (template_tri, 'Triangle')
]

# 检测
start = time.time()
detections = batch_multi_template_matching(source, templates, threshold=0.7)
elapsed = (time.time() - start) * 1000

print(f"检测耗时: {elapsed:.2f}ms")
print(f"检测结果: {len(detections)} 个目标")

for det in detections:
    print(f"  {det['label']}: 位置{det['bbox'][:2]}, 置信度{det['confidence']:.3f}")

# 可视化
source_color = cv2.cvtColor(source, cv2.COLOR_GRAY2BGR)
colors = {'Rectangle': (0, 255, 0), 'Circle': (255, 0, 0), 'Triangle': (0, 0, 255)}

for det in detections:
    x, y, w, h = det['bbox']
    color = colors.get(det['label'], (255, 255, 255))
    cv2.rectangle(source_color, (x, y), (x+w, y+h), color, 2)
    cv2.putText(source_color, det['label'][:3], (x, y-5),
               cv2.FONT_HERSHEY_SIMPLEX, 0.4, color, 1)

fig, axes = plt.subplots(1, 2, figsize=(14, 6))

axes[0].imshow(source, cmap='gray')
axes[0].set_title('源图像（多类型目标）')
axes[0].axis('off')

axes[1].imshow(cv2.cvtColor(source_color, cv2.COLOR_BGR2RGB))
axes[1].set_title(f'批量多模板检测结果\n绿=矩形 蓝=圆形 红=三角形')
axes[1].axis('off')

plt.tight_layout()
plt.savefig('batch_matching.png', dpi=150, bbox_inches='tight')
plt.show()
```

### 核心要点

1. **基本方法**
   - 阈值过滤找所有超过阈值的位置
   - 需要NMS去除重叠检测

2. **NMS非极大值抑制**
   - 按置信度排序
   - 迭代删除高IoU的框
   - IoU阈值影响检测密度

3. **进阶技术**
   - 局部最大值方法
   - 多尺度检测
   - 批量多模板检测

4. **实际应用**
   - 重复图案检测
   - 角点检测
   - 多类型目标检测
