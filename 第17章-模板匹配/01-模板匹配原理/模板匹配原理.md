# 模板匹配原理

## 1. 什么是模板匹配

哼，笨蛋，模板匹配可是图像处理中最直观的目标定位方法之一！本小姐今天就来给你好好讲讲！(￣▽￣)／

### 1.1 基本概念

模板匹配（Template Matching）是一种在大图像中搜索并找到模板图像位置的技术：

```
大图像（源图像）          模板图像
┌─────────────────┐      ┌─────┐
│                 │      │ 🎯  │
│    ┌─────┐      │  ←   └─────┘
│    │ 🎯  │      │
│    └─────┘      │      在源图像中
│                 │      找到模板位置
└─────────────────┘
```

### 1.2 工作原理

模板匹配的核心思想是**滑动窗口**：
1. 将模板在源图像上从左到右、从上到下滑动
2. 在每个位置计算模板与对应区域的相似度
3. 找到相似度最高（或差异最小）的位置

### 1.3 代码示例1：理解滑动窗口

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建源图像和模板
source = np.zeros((200, 300), dtype=np.uint8)
cv2.rectangle(source, (100, 60), (160, 120), 255, -1)  # 目标位置
cv2.rectangle(source, (50, 130), (90, 170), 150, -1)   # 干扰物

# 创建模板
template = np.zeros((60, 60), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (60, 60), 255, -1)

# 可视化滑动过程
fig, axes = plt.subplots(2, 3, figsize=(15, 10))

# 模拟几个滑动位置
positions = [(20, 20), (60, 40), (100, 60), (150, 80), (200, 100)]
th, tw = template.shape

for idx, (x, y) in enumerate(positions[:5]):
    if idx < 3:
        ax = axes[0, idx]
    else:
        ax = axes[1, idx-3]

    # 复制源图像
    display = cv2.cvtColor(source.copy(), cv2.COLOR_GRAY2BGR)

    # 绘制当前滑动窗口位置
    cv2.rectangle(display, (x, y), (x+tw, y+th), (0, 255, 0), 2)

    # 计算该位置的匹配度（简单的SSD）
    if y+th <= source.shape[0] and x+tw <= source.shape[1]:
        region = source[y:y+th, x:x+tw]
        diff = np.sum((region.astype(float) - template.astype(float))**2)
        similarity = 1 / (1 + diff/1e6)
    else:
        similarity = 0

    ax.imshow(cv2.cvtColor(display, cv2.COLOR_BGR2RGB))
    ax.set_title(f'位置({x},{y})\n相似度: {similarity:.4f}')
    ax.axis('off')

# 显示模板
axes[1, 2].imshow(template, cmap='gray')
axes[1, 2].set_title('模板图像')
axes[1, 2].axis('off')

plt.suptitle('模板匹配滑动窗口示意', fontsize=14)
plt.tight_layout()
plt.savefig('sliding_window.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 2. OpenCV中的模板匹配

### 2.1 cv2.matchTemplate函数

```python
result = cv2.matchTemplate(image, template, method)
```

**参数说明：**
- `image`：源图像（灰度图）
- `template`：模板图像（必须小于等于源图像）
- `method`：匹配方法

**返回值：**
- `result`：匹配结果矩阵，大小为 `(H-h+1, W-w+1)`
  - H, W：源图像尺寸
  - h, w：模板尺寸

### 2.2 六种匹配方法

| 方法 | 说明 | 最佳值 |
|------|------|--------|
| `cv2.TM_SQDIFF` | 平方差匹配 | 最小值 |
| `cv2.TM_SQDIFF_NORMED` | 归一化平方差 | 最小值 |
| `cv2.TM_CCORR` | 相关匹配 | 最大值 |
| `cv2.TM_CCORR_NORMED` | 归一化相关 | 最大值 |
| `cv2.TM_CCOEFF` | 相关系数匹配 | 最大值 |
| `cv2.TM_CCOEFF_NORMED` | 归一化相关系数 | 最大值 |

### 2.3 代码示例2：基本模板匹配

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建测试图像
source = np.zeros((300, 400), dtype=np.uint8)
source[:] = 100  # 灰色背景

# 添加目标和干扰物
cv2.rectangle(source, (150, 100), (220, 170), 255, -1)  # 目标
cv2.circle(source, (80, 200), 30, 180, -1)               # 干扰
cv2.rectangle(source, (300, 50), (350, 100), 200, -1)    # 干扰

# 创建模板（与目标相同）
template = np.zeros((70, 70), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (70, 70), 255, -1)

# 执行模板匹配
result = cv2.matchTemplate(source, template, cv2.TM_CCOEFF_NORMED)

# 找到最佳匹配位置
min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)

print(f"匹配结果矩阵大小: {result.shape}")
print(f"源图像大小: {source.shape}")
print(f"模板大小: {template.shape}")
print(f"最大相关系数: {max_val:.4f}")
print(f"最佳匹配位置: {max_loc}")

# 绘制结果
th, tw = template.shape
top_left = max_loc
bottom_right = (top_left[0] + tw, top_left[1] + th)

source_color = cv2.cvtColor(source, cv2.COLOR_GRAY2BGR)
cv2.rectangle(source_color, top_left, bottom_right, (0, 255, 0), 2)

# 可视化
fig, axes = plt.subplots(1, 3, figsize=(15, 5))

axes[0].imshow(source, cmap='gray')
axes[0].set_title('源图像')
axes[0].axis('off')

axes[1].imshow(template, cmap='gray')
axes[1].set_title('模板')
axes[1].axis('off')

im = axes[2].imshow(result, cmap='hot')
axes[2].set_title('匹配结果热力图')
axes[2].plot(max_loc[0], max_loc[1], 'g*', markersize=15)
plt.colorbar(im, ax=axes[2], fraction=0.046)

plt.tight_layout()
plt.savefig('basic_matching.png', dpi=150, bbox_inches='tight')
plt.show()

# 显示检测结果
plt.figure(figsize=(8, 6))
plt.imshow(cv2.cvtColor(source_color, cv2.COLOR_BGR2RGB))
plt.title(f'检测结果 (相关系数: {max_val:.4f})')
plt.axis('off')
plt.savefig('detection_result.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 3. 匹配方法详解

### 3.1 平方差匹配 (TM_SQDIFF)

计算模板与图像块的平方差之和：

$$R(x,y) = \sum_{x',y'} [T(x',y') - I(x+x', y+y')]^2$$

- 值越小，匹配越好
- 完美匹配时值为0

### 3.2 代码示例3：平方差匹配

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建测试图像
source = np.zeros((250, 350), dtype=np.uint8)
source[:] = 80

# 目标：一个十字形
target_x, target_y = 150, 100
cv2.line(source, (target_x-30, target_y), (target_x+30, target_y), 255, 5)
cv2.line(source, (target_x, target_y-30), (target_x, target_y+30), 255, 5)

# 创建十字形模板
template = np.zeros((70, 70), dtype=np.uint8)
cv2.line(template, (5, 35), (65, 35), 255, 5)
cv2.line(template, (35, 5), (35, 65), 255, 5)

# 平方差匹配
result_sqdiff = cv2.matchTemplate(source, template, cv2.TM_SQDIFF)
result_sqdiff_normed = cv2.matchTemplate(source, template, cv2.TM_SQDIFF_NORMED)

# 找最小值位置（平方差越小越好）
min_val1, _, min_loc1, _ = cv2.minMaxLoc(result_sqdiff)
min_val2, _, min_loc2, _ = cv2.minMaxLoc(result_sqdiff_normed)

print(f"TM_SQDIFF - 最小值: {min_val1:.2f}, 位置: {min_loc1}")
print(f"TM_SQDIFF_NORMED - 最小值: {min_val2:.4f}, 位置: {min_loc2}")

# 可视化
fig, axes = plt.subplots(2, 2, figsize=(12, 10))

axes[0, 0].imshow(source, cmap='gray')
axes[0, 0].set_title('源图像')
axes[0, 0].axis('off')

axes[0, 1].imshow(template, cmap='gray')
axes[0, 1].set_title('模板')
axes[0, 1].axis('off')

im1 = axes[1, 0].imshow(result_sqdiff, cmap='hot')
axes[1, 0].plot(min_loc1[0], min_loc1[1], 'b*', markersize=15)
axes[1, 0].set_title(f'TM_SQDIFF\n最小值: {min_val1:.0f}')
plt.colorbar(im1, ax=axes[1, 0], fraction=0.046)

im2 = axes[1, 1].imshow(result_sqdiff_normed, cmap='hot')
axes[1, 1].plot(min_loc2[0], min_loc2[1], 'b*', markersize=15)
axes[1, 1].set_title(f'TM_SQDIFF_NORMED\n最小值: {min_val2:.4f}')
plt.colorbar(im2, ax=axes[1, 1], fraction=0.046)

plt.tight_layout()
plt.savefig('sqdiff_matching.png', dpi=150, bbox_inches='tight')
plt.show()
```

### 3.3 相关匹配 (TM_CCORR)

计算模板与图像块的相关性：

$$R(x,y) = \sum_{x',y'} [T(x',y') \cdot I(x+x', y+y')]$$

- 值越大，匹配越好

### 3.5 相关系数匹配 (TM_CCOEFF)

计算去均值后的相关系数：

$$R(x,y) = \sum_{x',y'} [T'(x',y') \cdot I'(x+x', y+y')]$$

其中 $T'$ 和 $I'$ 是去均值后的模板和图像块。

- 值越大，匹配越好
- 对亮度变化有一定鲁棒性

### 3.6 代码示例5：相关系数匹配

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建测试图像
source = np.zeros((250, 350), dtype=np.uint8)
source[:] = 50  # 暗背景

cv2.rectangle(source, (100, 70), (180, 150), 200, -1)
cv2.rectangle(source, (220, 120), (300, 200), 150, -1)  # 较暗的类似形状

# 创建模板（中等亮度）
template = np.zeros((80, 80), dtype=np.uint8)
cv2.rectangle(template, (0, 0), (80, 80), 180, -1)

# 三种方法对比
methods = [
    ('TM_CCORR_NORMED', cv2.TM_CCORR_NORMED),
    ('TM_CCOEFF', cv2.TM_CCOEFF),
    ('TM_CCOEFF_NORMED', cv2.TM_CCOEFF_NORMED),
]

fig, axes = plt.subplots(2, 2, figsize=(12, 10))

axes[0, 0].imshow(source, cmap='gray')
axes[0, 0].set_title('源图像（两个不同亮度的矩形）')
axes[0, 0].axis('off')

axes[0, 1].imshow(template, cmap='gray')
axes[0, 1].set_title('模板')
axes[0, 1].axis('off')

for idx, (name, method) in enumerate(methods[:2]):

    ax = axes[1, idx]
    im = ax.imshow(result, cmap='hot')

    plt.colorbar(im, ax=ax, fraction=0.046)

plt.tight_layout()
plt.savefig('ccoeff_matching.png', dpi=150, bbox_inches='tight')
plt.show()

# 详细对比归一化方法
print("\n归一化方法对比:")
for name, method in methods:
    result = cv2.matchTemplate(source, template, method)
    min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)
    print(f"{name}:")
    print(f"  最大值: {max_val:.4f}, 位置: {max_loc}")
```

## 4. 归一化的重要性

### 4.1 代码示例6：归一化 vs 非归一化

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建具有不同亮度区域的图像
source = np.zeros((200, 400), dtype=np.uint8)
# 左半部分较暗
source[:, :200] = 60
# 右半部分较亮
source[:, 200:] = 180

# 在两个区域各放一个相同的目标
target = np.zeros((50, 50), dtype=np.uint8)
cv2.rectangle(target, (10, 10), (40, 40), 255, -1)

# 放置目标（相对于背景的对比度相同）
source[75:125, 75:125] = source[75:125, 75:125] + 80  # 左侧
source[75:125, 275:325] = source[75:125, 275:325] + 80  # 右侧

# 模板
template = np.zeros((50, 50), dtype=np.uint8)
template[:] = 140
cv2.rectangle(template, (10, 10), (40, 40), 220, -1)

# 不同方法对比
methods = [
    ('TM_SQDIFF', cv2.TM_SQDIFF, 'min'),
    ('TM_SQDIFF_NORMED', cv2.TM_SQDIFF_NORMED, 'min'),
    ('TM_CCORR', cv2.TM_CCORR, 'max'),
    ('TM_CCORR_NORMED', cv2.TM_CCORR_NORMED, 'max'),
]

fig, axes = plt.subplots(2, 3, figsize=(15, 10))

axes[0, 0].imshow(source, cmap='gray')
axes[0, 0].set_title('源图像\n(左暗右亮，各有一个目标)')
axes[0, 0].axis('off')

axes[0, 1].imshow(template, cmap='gray')
axes[0, 1].set_title('模板')
axes[0, 1].axis('off')

axes[0, 2].axis('off')

for idx, (name, method, best) in enumerate(methods):
    result = cv2.matchTemplate(source, template, method)

    if best == 'min':
        opt_val, _, opt_loc, _ = cv2.minMaxLoc(result)
    else:
        _, opt_val, _, opt_loc = cv2.minMaxLoc(result)

    ax = axes[1, idx] if idx < 3 else axes[1, idx-1]
    if idx < 2:
        ax = axes[1, idx]
    else:
        continue

    im = ax.imshow(result, cmap='hot')
    ax.plot(opt_loc[0], opt_loc[1], 'g*' if best == 'max' else 'b*', markersize=12)
    ax.set_title(f'{name}\n最佳位置: {opt_loc}')
    plt.colorbar(im, ax=ax, fraction=0.046)

# 显示归一化方法
for idx, (name, method, best) in enumerate(methods[2:4]):
    result = cv2.matchTemplate(source, template, method)

    if best == 'min':
        opt_val, _, opt_loc, _ = cv2.minMaxLoc(result)
    else:
        _, opt_val, _, opt_loc = cv2.minMaxLoc(result)

    ax = axes[1, idx]
    im = ax.imshow(result, cmap='hot')
    ax.plot(opt_loc[0], opt_loc[1], 'g*', markersize=12)
    ax.set_title(f'{name}\n最佳位置: {opt_loc}')
    plt.colorbar(im, ax=ax, fraction=0.046)

axes[1, 2].axis('off')

plt.tight_layout()
plt.savefig('normalization_comparison.png', dpi=150, bbox_inches='tight')
plt.show()

print("分析：归一化方法对光照变化更鲁棒")
```

### 5.2 代码示例8：匹配置信度评估

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def evaluate_match_confidence(result, method):
    """评估匹配置信度"""
    if method in [cv2.TM_SQDIFF, cv2.TM_SQDIFF_NORMED]:
        # 对于SQDIFF，最小值是最佳匹配
        best_val, _, best_loc, _ = cv2.minMaxLoc(result)
        is_min_best = True
    else:
        # 对于其他方法，最大值是最佳匹配
        _, best_val, _, best_loc = cv2.minMaxLoc(result)
        is_min_best = False

    # 计算统计信息
    mean_val = np.mean(result)
    std_val = np.std(result)

    # 计算峰值显著性
    if is_min_best:
        # 最佳值与平均值的差距
        significance = (mean_val - best_val) / std_val if std_val > 0 else 0
    else:
        significance = (best_val - mean_val) / std_val if std_val > 0 else 0

    return {
        'best_val': best_val,
        'best_loc': best_loc,
        'mean': mean_val,
        'std': std_val,
        'significance': significance
    }

# 创建不同匹配质量的测试
def create_test_cases():
    cases = []

    # Case 1: 完美匹配
    source1 = np.zeros((150, 200), dtype=np.uint8)
    cv2.rectangle(source1, (70, 50), (130, 110), 200, -1)
    template1 = np.zeros((60, 60), dtype=np.uint8)
    cv2.rectangle(template1, (0, 0), (60, 60), 200, -1)
    cases.append(('完美匹配', source1, template1))

    # Case 2: 部分匹配
    source2 = np.zeros((150, 200), dtype=np.uint8)
    cv2.rectangle(source2, (70, 50), (130, 110), 180, -1)  # 略有不同
    cases.append(('部分匹配', source2, template1))

    # Case 3: 无匹配
    source3 = np.zeros((150, 200), dtype=np.uint8)
    cv2.circle(source3, (100, 75), 40, 200, -1)  # 圆形，不是矩形
    cases.append(('无匹配', source3, template1))

    return cases

cases = create_test_cases()

fig, axes = plt.subplots(len(cases), 3, figsize=(15, 12))

for i, (name, source, template) in enumerate(cases):
    result = cv2.matchTemplate(source, template, cv2.TM_CCOEFF_NORMED)
    stats = evaluate_match_confidence(result, cv2.TM_CCOEFF_NORMED)

    # 显示源图像
    axes[i, 0].imshow(source, cmap='gray')
    axes[i, 0].set_title(f'{name}\n源图像')
    axes[i, 0].axis('off')

    # 显示结果热力图
    im = axes[i, 1].imshow(result, cmap='hot', vmin=-1, vmax=1)
    axes[i, 1].plot(stats['best_loc'][0], stats['best_loc'][1], 'g*', markersize=12)
    axes[i, 1].set_title(f"匹配结果\n最佳值: {stats['best_val']:.3f}")
    plt.colorbar(im, ax=axes[i, 1], fraction=0.046)

    # 显示直方图
    axes[i, 2].hist(result.flatten(), bins=50, edgecolor='black', alpha=0.7)
    axes[i, 2].axvline(stats['best_val'], color='r', linestyle='--',
                       label=f"最佳: {stats['best_val']:.3f}")
    axes[i, 2].axvline(stats['mean'], color='g', linestyle='--',
                       label=f"均值: {stats['mean']:.3f}")
    axes[i, 2].set_title(f"值分布\n显著性: {stats['significance']:.2f}")
    axes[i, 2].legend(fontsize=8)
    axes[i, 2].set_xlabel('匹配值')

plt.tight_layout()
plt.savefig('confidence_evaluation.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 6. 模板匹配的局限性

### 6.1 主要局限

1. **尺度敏感**：模板大小必须与目标一致
2. **旋转敏感**：模板方向必须与目标一致
3. **光照敏感**：亮度变化会影响匹配
4. **计算量大**：大图像和大模板匹配较慢

### 6.2 代码示例9：展示局限性

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建原始目标
original = np.zeros((80, 80), dtype=np.uint8)
cv2.rectangle(original, (15, 15), (65, 65), 200, -1)
cv2.circle(original, (40, 40), 15, 255, -1)

# 模板
template = original.copy()

# 创建不同变换的场景
def create_scene(transform_type):
    scene = np.zeros((200, 250), dtype=np.uint8)
    scene[:] = 80

    if transform_type == 'original':
        scene[60:140, 85:165] = original
        return scene, '原始（应该匹配）'

    elif transform_type == 'scaled':
        scaled = cv2.resize(original, None, fx=1.3, fy=1.3)
        h, w = scaled.shape
        y, x = 60, 80
        scene[y:y+h, x:x+w] = scaled
        return scene, '缩放1.3x（匹配困难）'

    elif transform_type == 'rotated':
        M = cv2.getRotationMatrix2D((40, 40), 30, 1.0)
        rotated = cv2.warpAffine(original, M, (80, 80))
        scene[60:140, 85:165] = rotated
        return scene, '旋转30°（匹配困难）'

    elif transform_type == 'bright':
        bright = cv2.add(original, 50)
        scene[60:140, 85:165] = bright
        return scene, '亮度变化（部分影响）'

transforms = ['original', 'scaled', 'rotated', 'bright']

fig, axes = plt.subplots(2, 4, figsize=(16, 8))

for idx, t in enumerate(transforms):
    scene, title = create_scene(t)
    result = cv2.matchTemplate(scene, template, cv2.TM_CCOEFF_NORMED)
    _, max_val, _, max_loc = cv2.minMaxLoc(result)

    # 绘制检测框
    scene_color = cv2.cvtColor(scene, cv2.COLOR_GRAY2BGR)
    th, tw = template.shape
    cv2.rectangle(scene_color, max_loc, (max_loc[0]+tw, max_loc[1]+th), (0, 255, 0), 2)

    axes[0, idx].imshow(cv2.cvtColor(scene_color, cv2.COLOR_BGR2RGB))
    axes[0, idx].set_title(f'{title}\n匹配值: {max_val:.3f}')
    axes[0, idx].axis('off')

    im = axes[1, idx].imshow(result, cmap='hot', vmin=0, vmax=1)
    axes[1, idx].plot(max_loc[0], max_loc[1], 'g*', markersize=10)
    axes[1, idx].set_title('匹配热力图')
    plt.colorbar(im, ax=axes[1, idx], fraction=0.046)

plt.suptitle('模板匹配的局限性演示', fontsize=14)
plt.tight_layout()
plt.savefig('matching_limitations.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 7. 本章小结

### 核心要点

1. **基本原理**
   - 滑动窗口逐像素计算相似度
   - 找到相似度最高/差异最小的位置

2. **六种匹配方法**
   - SQDIFF：平方差，越小越好
   - CCORR：相关性，越大越好
   - CCOEFF：相关系数，越大越好
   - 带NORMED：归一化版本，更鲁棒

3. **结果矩阵**
   - 大小为 (H-h+1, W-w+1)
   - 每个像素表示对应位置的匹配度

4. **局限性**
   - 对尺度、旋转、光照敏感
   - 计算量较大
