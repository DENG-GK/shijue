# 特征提取概述

## 1. 特征提取简介

特征提取是计算机视觉的核心技术之一，目的是从图像中提取具有区分性的信息，用于图像匹配、目标识别、场景理解等任务。

### 1.1 什么是图像特征

**图像特征**是指图像中具有以下特性的信息：
- **可区分性**：能够区分不同的图像内容
- **可重复性**：在不同条件下能被重复检测
- **可匹配性**：能够在不同图像间进行匹配

### 1.2 特征的分类

| 类别 | 说明 | 示例 |
|------|------|------|
| 点特征 | 图像中的特殊点 | 角点、斑点 |
| 边缘特征 | 灰度变化明显的边界 | Canny边缘 |
| 区域特征 | 具有相似属性的区域 | 超像素 |
| 全局特征 | 描述整个图像 | 颜色直方图 |
| 局部特征 | 描述图像局部区域 | SIFT、ORB |

### 1.3 特征提取的流程

```
图像 → 特征检测 → 特征描述 → 特征匹配 → 应用
        (定位)      (编码)       (比较)
```

## 2. 特征检测器与描述子

### 2.1 特征检测器

特征检测器负责在图像中定位特征点的位置。

**常见检测器**：

| 检测器 | 特点 | 速度 |
|--------|------|------|
| Harris | 旋转不变 | 快 |
| Shi-Tomasi | 更稳定 | 快 |
| FAST | 速度快 | 很快 |
| DoG (SIFT) | 尺度不变 | 慢 |
| BRISK | 多尺度 | 中等 |

### 2.2 特征描述子

特征描述子将特征点周围的局部区域编码为向量。

**常见描述子**：

| 描述子 | 维度 | 类型 | 特点 |
|--------|------|------|------|
| SIFT | 128 | 浮点 | 精确、尺度旋转不变 |
| SURF | 64/128 | 浮点 | 快速、尺度旋转不变 |
| ORB | 256 bits | 二进制 | 快速、旋转不变 |
| BRIEF | 256 bits | 二进制 | 非常快 |
| BRISK | 512 bits | 二进制 | 多尺度 |

### 2.3 不变性

| 不变性 | 说明 | 支持算法 |
|--------|------|----------|
| 平移不变 | 位置变化不影响 | 所有局部特征 |
| 旋转不变 | 旋转不影响 | SIFT, ORB, SURF |
| 尺度不变 | 缩放不影响 | SIFT, SURF |
| 光照不变 | 亮度变化不影响 | 大部分 |
| 仿射不变 | 仿射变换不影响 | ASIFT |

```python
"""
示例1：OpenCV特征检测器概览
展示OpenCV中可用的特征检测器
"""
import cv2
import numpy as np

def list_feature_detectors():
    """列出OpenCV中的特征检测器"""
    detectors = {
        'ORB': lambda: cv2.ORB_create(),
        'BRISK': lambda: cv2.BRISK_create(),
        'AKAZE': lambda: cv2.AKAZE_create(),
        'KAZE': lambda: cv2.KAZE_create(),
        'SIFT': lambda: cv2.SIFT_create(),
        'FAST': lambda: cv2.FastFeatureDetector_create(),
        'AGAST': lambda: cv2.AgastFeatureDetector_create(),
        'GFTT': lambda: cv2.GFTTDetector_create(),  # Good Features To Track
        'MSER': lambda: cv2.MSER_create(),
        'SimpleBlobDetector': lambda: cv2.SimpleBlobDetector_create(),
    }

    print("OpenCV可用的特征检测器:")
    print("=" * 60)

    available = []
    for name, create_func in detectors.items():
        try:
            detector = create_func()
            available.append(name)
            print(f"✓ {name}")
        except Exception as e:
            print(f"✗ {name}: {e}")

    return available

# 创建测试图像
img = np.zeros((400, 500), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (200, 200), 200, -1)
cv2.circle(img, (350, 150), 80, 200, -1)
cv2.rectangle(img, (100, 280), (400, 380), 150, -1)

# 添加纹理
noise = np.random.normal(0, 10, img.shape).astype(np.int16)
img = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)

# 列出可用检测器
available_detectors = list_feature_detectors()

print(f"\n共 {len(available_detectors)} 个可用检测器")

# 使用几个常见检测器进行对比
print("\n特征检测对比:")
print("-" * 50)
print(f"{'检测器':>15} {'特征点数':>15}")
print("-" * 50)

test_detectors = ['ORB', 'BRISK', 'AKAZE', 'SIFT', 'FAST']
results = []

for name in test_detectors:
    try:
        if name == 'ORB':
            detector = cv2.ORB_create()
        elif name == 'BRISK':
            detector = cv2.BRISK_create()
        elif name == 'AKAZE':
            detector = cv2.AKAZE_create()
        elif name == 'SIFT':
            detector = cv2.SIFT_create()
        elif name == 'FAST':
            detector = cv2.FastFeatureDetector_create()

        keypoints = detector.detect(img, None)
        print(f"{name:>15} {len(keypoints):>15}")
        results.append((name, keypoints))
    except Exception as e:
        print(f"{name:>15} 错误: {e}")

# 可视化
canvas = np.zeros((400, len(results) * 500, 3), dtype=np.uint8)

for i, (name, kps) in enumerate(results):
    result = cv2.cvtColor(img.copy(), cv2.COLOR_GRAY2BGR)
    result = cv2.drawKeypoints(result, kps, None,
                               color=(0, 255, 0),
                               flags=cv2.DRAW_MATCHES_FLAGS_DRAW_RICH_KEYPOINTS)
    cv2.putText(result, f"{name}: {len(kps)}", (10, 30),
               cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    canvas[:, i*500:(i+1)*500] = result

cv2.imshow("Feature Detectors Comparison", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""

import cv2
import numpy as np

# 创建测试图像
img = np.zeros((400, 500), dtype=np.uint8)
cv2.rectangle(img, (80, 80), (220, 220), 200, -1)
cv2.circle(img, (380, 150), 70, 200, -1)

# 添加纹理
for i in range(0, 500, 20):
    cv2.line(img, (i, 0), (i, 400), 128, 1)
for j in range(0, 400, 20):
    cv2.line(img, (0, j), (500, j), 128, 1)

# 使用ORB检测特征点
orb = cv2.ORB_create(nfeatures=50)
keypoints = orb.detect(img, None)

print("KeyPoint对象属性分析:")
print("=" * 70)

print("\n前10个特征点的属性:")
print("-" * 70)
print(f"{'#':>3} {'位置':>15} {'大小':>8} {'角度':>8} {'响应':>10} {'octave':>8}")
print("-" * 70)

for i, kp in enumerate(keypoints[:10]):
    print(f"{i:>3} ({kp.pt[0]:>6.2f}, {kp.pt[1]:>5.2f}) "
          f"{kp.size:>8.2f} {kp.angle:>8.2f} {kp.response:>10.4f} {kp.octave:>8}")

print("\n属性说明:")
print("-" * 70)
print("pt      : 特征点的(x, y)坐标")
print("size    : 特征点的直径（邻域大小）")
print("angle   : 特征点的方向（度，-1表示未计算）")
print("response: 特征点的响应强度")
print("octave  : 特征点所在的金字塔层级")
print("class_id: 特征点的类别ID（用于分类）")

# 统计分析
sizes = [kp.size for kp in keypoints]
responses = [kp.response for kp in keypoints]
angles = [kp.angle for kp in keypoints if kp.angle >= 0]

print("\n统计信息:")
print("-" * 70)
print(f"特征点总数: {len(keypoints)}")
print(f"大小范围: [{min(sizes):.2f}, {max(sizes):.2f}]")
print(f"响应范围: [{min(responses):.4f}, {max(responses):.4f}]")
if angles:
    print(f"角度范围: [{min(angles):.2f}, {max(angles):.2f}]")

# 可视化
result = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)

# 按响应强度着色
max_response = max(responses)
for kp in keypoints:
    x, y = int(kp.pt[0]), int(kp.pt[1])
    r = int(kp.size / 2)

    # 颜色根据响应强度
    intensity = int(255 * kp.response / max_response)
    color = (0, intensity, 255 - intensity)

    cv2.circle(result, (x, y), r, color, 2)

    # 绘制方向
    if kp.angle >= 0:
        angle_rad = np.radians(kp.angle)
        x2 = int(x + r * np.cos(angle_rad))
        y2 = int(y + r * np.sin(angle_rad))
        cv2.line(result, (x, y), (x2, y2), color, 1)

cv2.imshow("Keypoints with Properties", result)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 示例3：检测器性能比较

```python
"""
示例3：检测器性能比较
比较不同检测器的速度和特征数量
"""
import cv2
import numpy as np
import time

def benchmark_detector(detector, img, iterations=10):
    """测试检测器性能"""
    times = []
    keypoints = None

    for _ in range(iterations):
        start = time.time()
        keypoints = detector.detect(img, None)
        elapsed = time.time() - start
        times.append(elapsed)

    return {
        'keypoints': keypoints,
        'count': len(keypoints) if keypoints else 0,
        'avg_time': np.mean(times) * 1000,  # ms
        'std_time': np.std(times) * 1000
    }

# 创建测试图像
img = cv2.imread("image.jpg", cv2.IMREAD_GRAYSCALE)
if img is None:
    # 创建合成图像
    img = np.random.randint(100, 200, (480, 640), dtype=np.uint8)
    # 添加一些结构
    cv2.rectangle(img, (100, 100), (300, 300), 50, -1)
    cv2.circle(img, (500, 200), 100, 50, -1)
    # 添加棋盘格纹理
    for i in range(0, 640, 40):
        for j in range(0, 480, 40):
            if (i // 40 + j // 40) % 2 == 0:
                cv2.rectangle(img, (i, j), (i+20, j+20), 200, -1)

# 创建检测器
detectors = {
    'FAST': cv2.FastFeatureDetector_create(),
    'ORB': cv2.ORB_create(nfeatures=500),
    'BRISK': cv2.BRISK_create(),
    'AKAZE': cv2.AKAZE_create(),
    'SIFT': cv2.SIFT_create(nfeatures=500),
}

print("特征检测器性能比较:")
print("=" * 70)
print(f"图像大小: {img.shape[1]}x{img.shape[0]}")
print("-" * 70)
print(f"{'检测器':>10} {'特征数':>10} {'平均时间(ms)':>15} {'标准差(ms)':>12}")
print("-" * 70)

results = {}
for name, detector in detectors.items():
    result = benchmark_detector(detector, img)
    results[name] = result
    print(f"{name:>10} {result['count']:>10} {result['avg_time']:>15.2f} {result['std_time']:>12.2f}")

print("-" * 70)

# 速度排名
sorted_by_speed = sorted(results.items(), key=lambda x: x[1]['avg_time'])
print("\n按速度排名（从快到慢）:")
for i, (name, result) in enumerate(sorted_by_speed, 1):
    print(f"{i}. {name}: {result['avg_time']:.2f}ms")

# 可视化
n_detectors = len(results)
canvas = np.zeros((480, 640 * min(n_detectors, 3), 3), dtype=np.uint8)

for i, (name, result) in enumerate(list(results.items())[:3]):
    vis = cv2.cvtColor(img.copy(), cv2.COLOR_GRAY2BGR)
    vis = cv2.drawKeypoints(vis, result['keypoints'], None,
                            color=(0, 255, 0),
                            flags=cv2.DRAW_MATCHES_FLAGS_DRAW_RICH_KEYPOINTS)
    cv2.putText(vis, f"{name}: {result['count']} pts", (10, 30),
               cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(vis, f"{result['avg_time']:.1f}ms", (10, 60),
               cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 0), 2)

    canvas[:, i*640:(i+1)*640] = vis

cv2.imshow("Detector Comparison", canvas)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

```python
"""
示例4：描述子提取与比较
比较不同描述子的维度和类型
"""
import cv2
import numpy as np

# 创建测试图像
img = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img, (50, 50), (150, 150), 200, -1)
cv2.circle(img, (280, 100), 60, 200, -1)

# 添加纹理
noise = np.random.normal(0, 15, img.shape).astype(np.int16)
img = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)

# 不同的特征提取器（检测器+描述子）
extractors = {
    'ORB': cv2.ORB_create(nfeatures=100),
    'BRISK': cv2.BRISK_create(),
    'AKAZE': cv2.AKAZE_create(),
    'SIFT': cv2.SIFT_create(nfeatures=100),
}

print("特征描述子比较:")
print("=" * 70)
print(f"{'算法':>10} {'特征数':>10} {'描述子维度':>15} {'数据类型':>15} {'大小(KB)':>12}")
print("-" * 70)

for name, extractor in extractors.items():
    # 检测并计算描述子
    keypoints, descriptors = extractor.detectAndCompute(img, None)

    if descriptors is not None:
        dim = descriptors.shape[1] if len(descriptors.shape) > 1 else 1
        dtype = descriptors.dtype
        size_kb = descriptors.nbytes / 1024

        # 判断是二进制还是浮点描述子
        if dtype == np.uint8:
            desc_type = f"Binary ({dim * 8} bits)"
        else:
            desc_type = f"Float ({dim}D)"

        print(f"{name:>10} {len(keypoints):>10} {desc_type:>15} {str(dtype):>15} {size_kb:>12.2f}")
    else:
        print(f"{name:>10} {len(keypoints):>10} {'N/A':>15} {'N/A':>15} {'N/A':>12}")

print("-" * 70)

# 分析单个描述子
print("\n单个描述子示例（ORB）:")
orb = cv2.ORB_create(nfeatures=5)
kps, descs = orb.detectAndCompute(img, None)

if descs is not None and len(descs) > 0:
    print(f"描述子形状: {descs.shape}")
    print(f"第一个描述子（前32字节）:")
    print(descs[0][:32])

    # 计算描述子之间的汉明距离
    if len(descs) >= 2:
        dist = cv2.norm(descs[0], descs[1], cv2.NORM_HAMMING)
        print(f"\n描述子0和1的汉明距离: {dist}")

print("\n单个描述子示例（SIFT）:")
sift = cv2.SIFT_create(nfeatures=5)
kps, descs = sift.detectAndCompute(img, None)

if descs is not None and len(descs) > 0:
    print(f"描述子形状: {descs.shape}")
    print(f"第一个描述子（前16维）:")
    print(descs[0][:16])

    # 计算描述子之间的欧氏距离
    if len(descs) >= 2:
        dist = cv2.norm(descs[0], descs[1], cv2.NORM_L2)
        print(f"\n描述子0和1的欧氏距离: {dist:.2f}")
```

```python
sift = cv2.SIFT_create()
keypoints, descriptors = sift.detectAndCompute(img, None)

print("特征尺度和方向分析:")

print(f"检测到 {len(keypoints)} 个特征点")

    count = sum(1 for o in octaves if o == oct)

        cv2.line(result, (x, y), (x2, y2), (0, 0, 255), 2)
```

### 示例6：特征提取工具类

```python
"""
示例6：特征提取工具类
封装常用的特征提取功能
"""
import cv2
import numpy as np
from typing import List, Tuple, Optional, Dict
from dataclasses import dataclass
import time

@dataclass
class FeatureResult:
    """特征提取结果"""
    keypoints: List
    descriptors: np.ndarray
    extraction_time: float
    detector_name: str

class FeatureExtractor:
    """特征提取器"""

    SUPPORTED_METHODS = ['ORB', 'BRISK', 'AKAZE', 'SIFT', 'KAZE']

    def __init__(self, method: str = 'ORB', **kwargs):
        """
        初始化特征提取器

        Args:
            method: 算法名称
            **kwargs: 算法参数
        """
        self.method = method.upper()
        self.extractor = self._create_extractor(**kwargs)

    def _create_extractor(self, **kwargs):
        """创建特征提取器"""
        if self.method == 'ORB':
            return cv2.ORB_create(**kwargs)
        elif self.method == 'BRISK':
            return cv2.BRISK_create(**kwargs)
        elif self.method == 'AKAZE':
            return cv2.AKAZE_create(**kwargs)
        elif self.method == 'SIFT':
            return cv2.SIFT_create(**kwargs)
        elif self.method == 'KAZE':
            return cv2.KAZE_create(**kwargs)
        else:
            raise ValueError(f"不支持的方法: {self.method}")

    def extract(self,
                img: np.ndarray,
                mask: Optional[np.ndarray] = None) -> FeatureResult:
        """
        提取特征

        Args:
            img: 输入图像
            mask: 掩码

        Returns:
            FeatureResult对象
        """
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img

        start = time.time()
        keypoints, descriptors = self.extractor.detectAndCompute(gray, mask)
        elapsed = time.time() - start

        return FeatureResult(
            keypoints=keypoints,
            descriptors=descriptors,
            extraction_time=elapsed,
            detector_name=self.method
        )

    def detect_only(self, img: np.ndarray, mask: Optional[np.ndarray] = None):
        """只检测特征点"""
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img
        return self.extractor.detect(gray, mask)

    def compute_only(self, img: np.ndarray, keypoints: List):
        """只计算描述子"""
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        else:
            gray = img
        return self.extractor.compute(gray, keypoints)

    def visualize(self,
                  img: np.ndarray,
                  result: FeatureResult,
                  show_rich: bool = True) -> np.ndarray:
        """可视化特征"""
        if len(img.shape) == 2:
            vis = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
        else:
            vis = img.copy()

        flags = cv2.DRAW_MATCHES_FLAGS_DRAW_RICH_KEYPOINTS if show_rich else 0
        vis = cv2.drawKeypoints(vis, result.keypoints, None,
                               color=(0, 255, 0), flags=flags)

        # 添加信息
        cv2.putText(vis, f"{result.detector_name}: {len(result.keypoints)} features",
                   (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.putText(vis, f"Time: {result.extraction_time*1000:.1f}ms",
                   (10, 60), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 0), 2)

        return vis

    def get_stats(self, result: FeatureResult) -> Dict:
        """获取统计信息"""
        if not result.keypoints:
            return {}

        sizes = [kp.size for kp in result.keypoints]
        responses = [kp.response for kp in result.keypoints]

        stats = {
            'count': len(result.keypoints),
            'size_min': min(sizes),
            'size_max': max(sizes),
            'size_mean': np.mean(sizes),
            'response_min': min(responses),
            'response_max': max(responses),
            'extraction_time_ms': result.extraction_time * 1000
        }

        if result.descriptors is not None:
            stats['descriptor_dim'] = result.descriptors.shape[1]
            stats['descriptor_type'] = str(result.descriptors.dtype)

        return stats
```

```python
"""
示例7：特征匹配预览
简单展示特征匹配的基本流程
"""
import cv2
import numpy as np

# 创建两个相似的图像（模拟不同视角）
img1 = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img1, (100, 80), (250, 200), 200, -1)
cv2.circle(img1, (320, 140), 50, 200, -1)

# 第二个图像：旋转和平移
img2 = np.zeros((300, 400), dtype=np.uint8)
center = (200, 150)
pts = np.array([[100, 80], [250, 80], [250, 200], [100, 200]], dtype=float)
pts = pts - [175, 140]  # 中心化
angle = 15
rad = np.radians(angle)
rot = np.array([[np.cos(rad), -np.sin(rad)], [np.sin(rad), np.cos(rad)]])
pts = (pts @ rot.T + [200, 160]).astype(int)
cv2.fillPoly(img2, [pts], 200)
cv2.circle(img2, (330, 160), 50, 200, -1)

# 添加纹理
for img in [img1, img2]:
    noise = np.random.normal(0, 10, img.shape).astype(np.int16)
    img[:] = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)

# 特征提取
orb = cv2.ORB_create(nfeatures=100)
kp1, desc1 = orb.detectAndCompute(img1, None)
kp2, desc2 = orb.detectAndCompute(img2, None)

print(f"图像1: {len(kp1)} 个特征点")
print(f"图像2: {len(kp2)} 个特征点")

# 特征匹配（BFMatcher + 汉明距离）
bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
matches = bf.match(desc1, desc2)

# 按距离排序
matches = sorted(matches, key=lambda x: x.distance)

print(f"匹配点对: {len(matches)}")
print(f"\n前5个匹配的距离:")
for i, m in enumerate(matches[:5]):
    print(f"  匹配 {i}: 距离={m.distance}")

# 可视化
result = cv2.drawMatches(img1, kp1, img2, kp2, matches[:20], None,
                         matchColor=(0, 255, 0),
                         singlePointColor=(255, 0, 0),
                         flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)

cv2.putText(result, f"Matches: {len(matches)}", (10, 30),
           cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)

cv2.imshow("Feature Matching Preview", result)
cv2.waitKey(0)
cv2.destroyAllWindows()
```
