# 特征匹配

## 1. 特征匹配概述

哼，笨蛋，特征匹配可是计算机视觉中最重要的技术之一！本小姐今天就来给你好好讲讲各种匹配方法！(￣▽￣)／

### 1.1 什么是特征匹配

特征匹配是指在两幅或多幅图像中找到对应的特征点，建立特征点之间的对应关系。

```
图像1的特征点  ←→  图像2的特征点
     ↓                 ↓
   描述符            描述符
     ↓                 ↓
   ← 距离度量/相似度计算 →
             ↓
        匹配结果
```

### 1.2 匹配的应用场景

| 应用 | 描述 |
|------|------|
| 图像拼接 | 全景图生成 |
| 物体识别 | 特定物体定位 |
| 3D重建 | 立体视觉、SfM |
| 视觉SLAM | 机器人定位 |
| 图像配准 | 医学图像对齐 |
| 增强现实 | AR标记追踪 |

### 1.3 距离度量方法

| 描述符类型 | 推荐距离 | 说明 |
|------------|----------|------|
| SIFT/SURF (浮点) | L2 (欧氏距离) | 默认选择 |
| ORB/BRIEF (二进制) | Hamming距离 | 位运算，速度快 |
| 归一化描述符 | 余弦距离 | 适合高维特征 |

## 2. 暴力匹配(Brute-Force)

### 2.1 原理

暴力匹配将一幅图像中的每个描述符与另一幅图像中的所有描述符进行比较，找出距离最小的作为匹配。

### 2.2 代码示例1：基本暴力匹配

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建测试图像对
def create_image_pair():
    # 原始图像
    img1 = np.zeros((300, 400), dtype=np.uint8)
    cv2.rectangle(img1, (50, 50), (150, 150), 255, -1)
    cv2.circle(img1, (280, 100), 50, 200, -1)
    cv2.putText(img1, 'A', (180, 250), cv2.FONT_HERSHEY_SIMPLEX, 2, 180, 3)

    # 旋转变换的图像
    center = (200, 150)
    M = cv2.getRotationMatrix2D(center, 15, 0.9)
    img2 = cv2.warpAffine(img1, M, (400, 300))

    return img1, img2

img1, img2 = create_image_pair()

# 使用ORB检测和描述
orb = cv2.ORB_create(nfeatures=500)
kp1, des1 = orb.detectAndCompute(img1, None)
kp2, des2 = orb.detectAndCompute(img2, None)

print(f"图像1关键点: {len(kp1)}")
print(f"图像2关键点: {len(kp2)}")

# 创建暴力匹配器
# cv2.NORM_HAMMING用于二进制描述符（ORB, BRIEF）
# cv2.NORM_L2用于浮点描述符（SIFT, SURF）
bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)

# 进行匹配
matches = bf.match(des1, des2)

# 按距离排序
matches = sorted(matches, key=lambda x: x.distance)

print(f"匹配数量: {len(matches)}")
print(f"最小距离: {matches[0].distance}")
print(f"最大距离: {matches[-1].distance}")

# 绘制匹配结果
img_matches = cv2.drawMatches(img1, kp1, img2, kp2, matches[:50], None,
                              matchColor=(0, 255, 0),
                              flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)

plt.figure(figsize=(14, 6))
plt.imshow(img_matches)
plt.title(f'暴力匹配结果 (显示前50个匹配)')
plt.axis('off')
plt.savefig('bf_basic_matching.png', dpi=150, bbox_inches='tight')
plt.show()
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建测试图像
img1 = np.zeros((250, 350), dtype=np.uint8)
cv2.rectangle(img1, (40, 40), (140, 140), 255, -1)
cv2.circle(img1, (250, 80), 40, 200, -1)

M = np.float32([[1, 0.1, 10], [0.05, 1, 5]])

kp1, des1 = orb.detectAndCompute(img1, None)
kp2, des2 = orb.detectAndCompute(img2, None)

# 不使用crossCheck
bf_no_cross = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=False)
matches_no_cross = bf_no_cross.match(des1, des2)

# 使用crossCheck
bf_cross = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
matches_cross = bf_cross.match(des1, des2)

print("crossCheck原理:")
print("  crossCheck=False: A->B最近匹配")
print("  crossCheck=True:  A->B最近且B->A最近 (互为最佳匹配)")
print(f"\n不使用crossCheck: {len(matches_no_cross)} 个匹配")
print(f"使用crossCheck: {len(matches_cross)} 个匹配")

# 可视化对比
fig, axes = plt.subplots(2, 1, figsize=(14, 10))

img_no_cross = cv2.drawMatches(img1, kp1, img2, kp2,
                                sorted(matches_no_cross, key=lambda x: x.distance)[:30],
                                None, flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
axes[0].imshow(img_no_cross)
axes[0].set_title(f'crossCheck=False ({len(matches_no_cross)}个匹配，显示前30)')
axes[0].axis('off')

img_cross = cv2.drawMatches(img1, kp1, img2, kp2,
                            sorted(matches_cross, key=lambda x: x.distance)[:30],
                            None, flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
axes[1].imshow(img_cross)
axes[1].set_title(f'crossCheck=True ({len(matches_cross)}个匹配，显示前30)')
axes[1].axis('off')

plt.tight_layout()
plt.savefig('bf_crosscheck.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 3. KNN匹配与比率测试

### 3.1 Lowe's比率测试

David Lowe提出的比率测试是过滤错误匹配的有效方法：
- 对每个特征点找到k个最近邻（通常k=2）
- 如果最近距离/次近距离 < 阈值（通常0.7-0.8），则认为是好匹配
- 原理：正确匹配的最近邻应该明显优于其他候选

### 3.2 代码示例3：KNN匹配与比率测试

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建测试图像对
img1 = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img1, (50, 50), (180, 180), 255, -1)
cv2.circle(img1, (300, 120), 60, 200, -1)
cv2.rectangle(img1, (80, 220), (200, 280), 180, -1)

# 添加一些噪声和变换
noise = np.random.randint(0, 30, img1.shape, dtype=np.uint8)
img1_noisy = cv2.add(img1, noise)

M = cv2.getRotationMatrix2D((200, 150), 10, 0.95)
img2 = cv2.warpAffine(img1_noisy, M, (400, 300))
noise2 = np.random.randint(0, 30, img2.shape, dtype=np.uint8)
img2 = cv2.add(img2, noise2)

# SIFT检测（浮点描述符）
sift = cv2.SIFT_create(nfeatures=500)
kp1, des1 = sift.detectAndCompute(img1, None)
kp2, des2 = sift.detectAndCompute(img2, None)

# 创建BF匹配器（使用L2范数）
bf = cv2.BFMatcher(cv2.NORM_L2)

# KNN匹配，k=2
matches = bf.knnMatch(des1, des2, k=2)

print(f"总匹配数: {len(matches)}")

# 比率测试
ratio_thresholds = [0.5, 0.7, 0.8, 0.9]
results = {}

for ratio in ratio_thresholds:
    good = []
    for m, n in matches:
        if m.distance < ratio * n.distance:
            good.append(m)
    results[ratio] = good
    print(f"比率阈值 {ratio}: {len(good)} 个好匹配")

# 可视化不同比率阈值的结果
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

for idx, ratio in enumerate(ratio_thresholds):
    ax = axes[idx // 2, idx % 2]
    good = results[ratio]

    img_match = cv2.drawMatches(img1, kp1, img2, kp2, good[:40], None,
                                flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
    ax.imshow(img_match)
    ax.set_title(f'比率阈值={ratio}, 好匹配数={len(good)}')
    ax.axis('off')

plt.suptitle('Lowe比率测试对比', fontsize=14)
plt.tight_layout()
plt.savefig('knn_ratio_test.png', dpi=150, bbox_inches='tight')
plt.show()
```

### 3.3 代码示例4：距离分布分析

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建图像对
img1 = np.zeros((300, 400), dtype=np.uint8)
cv2.rectangle(img1, (50, 50), (180, 180), 255, -1)
cv2.circle(img1, (300, 120), 60, 200, -1)

M = cv2.getRotationMatrix2D((200, 150), 20, 0.85)
img2 = cv2.warpAffine(img1, M, (400, 300))

# SIFT检测
sift = cv2.SIFT_create()
kp1, des1 = sift.detectAndCompute(img1, None)
kp2, des2 = sift.detectAndCompute(img2, None)

# KNN匹配
bf = cv2.BFMatcher(cv2.NORM_L2)
matches = bf.knnMatch(des1, des2, k=2)

# 分析距离分布
first_distances = [m.distance for m, n in matches]
second_distances = [n.distance for m, n in matches]
ratios = [m.distance / n.distance for m, n in matches]

fig, axes = plt.subplots(2, 2, figsize=(12, 10))

# 最近邻距离分布
axes[0, 0].hist(first_distances, bins=30, edgecolor='black', alpha=0.7)
axes[0, 0].set_xlabel('距离')
axes[0, 0].set_ylabel('频数')
axes[0, 0].set_title('最近邻距离分布')
axes[0, 0].axvline(np.median(first_distances), color='r', linestyle='--',
                   label=f'中位数: {np.median(first_distances):.1f}')
axes[0, 0].legend()

# 次近邻距离分布
axes[0, 1].hist(second_distances, bins=30, edgecolor='black', alpha=0.7, color='orange')
axes[0, 1].set_xlabel('距离')
axes[0, 1].set_ylabel('频数')
axes[0, 1].set_title('次近邻距离分布')
axes[0, 1].axvline(np.median(second_distances), color='r', linestyle='--',
                   label=f'中位数: {np.median(second_distances):.1f}')
axes[0, 1].legend()

# 比率分布
axes[1, 0].hist(ratios, bins=30, edgecolor='black', alpha=0.7, color='green')
axes[1, 0].set_xlabel('距离比率')
axes[1, 0].set_ylabel('频数')
axes[1, 0].set_title('距离比率分布')
axes[1, 0].axvline(0.75, color='r', linestyle='--', label='阈值 0.75')
axes[1, 0].legend()

# 散点图：最近邻 vs 次近邻
good_mask = np.array(ratios) < 0.75
axes[1, 1].scatter(np.array(first_distances)[~good_mask],
                   np.array(second_distances)[~good_mask],
                   alpha=0.5, label='过滤掉', c='red', s=20)
axes[1, 1].scatter(np.array(first_distances)[good_mask],
                   np.array(second_distances)[good_mask],
                   alpha=0.5, label='保留', c='green', s=20)
axes[1, 1].set_xlabel('最近邻距离')
axes[1, 1].set_ylabel('次近邻距离')
axes[1, 1].set_title('匹配点分布')
axes[1, 1].legend()

# 绘制分界线（ratio=0.75）
x = np.linspace(0, max(first_distances), 100)
axes[1, 1].plot(x, x / 0.75, 'k--', alpha=0.5, label='ratio=0.75')

plt.tight_layout()
plt.savefig('distance_analysis.png', dpi=150, bbox_inches='tight')
plt.show()

# 统计信息
print(f"匹配统计:")
print(f"  总匹配数: {len(matches)}")
print(f"  好匹配数 (ratio<0.75): {sum(good_mask)}")
print(f"  平均比率: {np.mean(ratios):.3f}")
print(f"  比率中位数: {np.median(ratios):.3f}")
```

## 4. FLANN匹配器

### 4.1 FLANN简介

FLANN（Fast Library for Approximate Nearest Neighbors）是一种快速近似最近邻搜索算法，对于大量特征点的匹配比暴力匹配更高效。

### 4.2 代码示例5：FLANN匹配（浮点描述符）

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt
import time

# 创建较大的测试图像
np.random.seed(42)
img1 = np.random.randint(50, 200, (600, 800), dtype=np.uint8)
cv2.rectangle(img1, (100, 100), (300, 300), 255, -1)
cv2.circle(img1, (500, 200), 100, 200, -1)
cv2.rectangle(img1, (200, 350), (500, 550), 180, -1)

M = cv2.getRotationMatrix2D((400, 300), 12, 0.92)
img2 = cv2.warpAffine(img1, M, (800, 600))

# SIFT检测
sift = cv2.SIFT_create(nfeatures=2000)
kp1, des1 = sift.detectAndCompute(img1, None)
kp2, des2 = sift.detectAndCompute(img2, None)

print(f"关键点数量: {len(kp1)}, {len(kp2)}")

# FLANN参数（用于SIFT/SURF等浮点描述符）
FLANN_INDEX_KDTREE = 1
index_params = dict(algorithm=FLANN_INDEX_KDTREE, trees=5)
search_params = dict(checks=50)  # 检查次数，越高越精确但更慢

# 创建FLANN匹配器
flann = cv2.FlannBasedMatcher(index_params, search_params)

# 计时对比
# FLANN匹配
start = time.time()
matches_flann = flann.knnMatch(des1, des2, k=2)
flann_time = (time.time() - start) * 1000

# BF匹配
bf = cv2.BFMatcher(cv2.NORM_L2)
start = time.time()
matches_bf = bf.knnMatch(des1, des2, k=2)
bf_time = (time.time() - start) * 1000

print(f"\nFLANN匹配时间: {flann_time:.2f}ms")
print(f"BF匹配时间: {bf_time:.2f}ms")
print(f"加速比: {bf_time/flann_time:.2f}x")

# 比率测试
good_flann = []
for m, n in matches_flann:
    if m.distance < 0.75 * n.distance:
        good_flann.append(m)

good_bf = []
for m, n in matches_bf:
    if m.distance < 0.75 * n.distance:
        good_bf.append(m)

print(f"\nFLANN好匹配: {len(good_flann)}")
print(f"BF好匹配: {len(good_bf)}")

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(14, 6))

img_flann = cv2.drawMatches(img1, kp1, img2, kp2, good_flann[:50], None,
                            flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
axes[0].imshow(img_flann)
axes[0].set_title(f'FLANN匹配 ({flann_time:.1f}ms, {len(good_flann)}个好匹配)')
axes[0].axis('off')

img_bf = cv2.drawMatches(img1, kp1, img2, kp2, good_bf[:50], None,
                         flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
axes[1].imshow(img_bf)
axes[1].set_title(f'BF匹配 ({bf_time:.1f}ms, {len(good_bf)}个好匹配)')
axes[1].axis('off')

plt.tight_layout()
plt.savefig('flann_vs_bf.png', dpi=150, bbox_inches='tight')
plt.show()
```

## 5. 匹配过滤与验证

### 5.1 代码示例7：使用RANSAC过滤误匹配

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# 创建测试图像对
img1 = np.zeros((400, 500), dtype=np.uint8)
cv2.rectangle(img1, (50, 50), (200, 200), 255, -1)
cv2.circle(img1, (350, 150), 80, 200, -1)
cv2.rectangle(img1, (100, 250), (300, 380), 180, -1)

# 透视变换
pts1 = np.float32([[0, 0], [500, 0], [500, 400], [0, 400]])
pts2 = np.float32([[20, 30], [480, 10], [490, 380], [10, 390]])
M = cv2.getPerspectiveTransform(pts1, pts2)
img2 = cv2.warpPerspective(img1, M, (500, 400))

# SIFT检测
sift = cv2.SIFT_create()
kp1, des1 = sift.detectAndCompute(img1, None)
kp2, des2 = sift.detectAndCompute(img2, None)

# FLANN匹配
FLANN_INDEX_KDTREE = 1
index_params = dict(algorithm=FLANN_INDEX_KDTREE, trees=5)
search_params = dict(checks=50)
flann = cv2.FlannBasedMatcher(index_params, search_params)

matches = flann.knnMatch(des1, des2, k=2)

# 比率测试
good = []
for m, n in matches:
    if m.distance < 0.75 * n.distance:
        good.append(m)

print(f"比率测试后: {len(good)} 个匹配")

# 提取匹配点
src_pts = np.float32([kp1[m.queryIdx].pt for m in good]).reshape(-1, 1, 2)
dst_pts = np.float32([kp2[m.trainIdx].pt for m in good]).reshape(-1, 1, 2)

# 使用RANSAC计算单应性矩阵
H, mask = cv2.findHomography(src_pts, dst_pts, cv2.RANSAC, 5.0)
matchesMask = mask.ravel().tolist()

inliers = sum(matchesMask)
outliers = len(matchesMask) - inliers

print(f"RANSAC结果:")
print(f"  内点 (inliers): {inliers}")
print(f"  外点 (outliers): {outliers}")
print(f"  内点率: {inliers/len(matchesMask)*100:.1f}%")

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(14, 6))

# RANSAC前（所有匹配）
draw_params_before = dict(
    matchColor=(0, 255, 0),
    singlePointColor=None,
    flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS
)
img_before = cv2.drawMatches(img1, kp1, img2, kp2, good, None, **draw_params_before)
axes[0].imshow(img_before)
axes[0].set_title(f'RANSAC前 ({len(good)}个匹配)')
axes[0].axis('off')

# RANSAC后（只显示内点）
draw_params_after = dict(
    matchColor=(0, 255, 0),
    singlePointColor=None,
    matchesMask=matchesMask,
    flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS
)
img_after = cv2.drawMatches(img1, kp1, img2, kp2, good, None, **draw_params_after)
axes[1].imshow(img_after)
axes[1].set_title(f'RANSAC后 ({inliers}个内点)')
axes[1].axis('off')

plt.tight_layout()
plt.savefig('ransac_filtering.png', dpi=150, bbox_inches='tight')
plt.show()

# 验证单应性矩阵
if H is not None:
    print(f"\n单应性矩阵H:")
    print(H)

    # 使用H变换图像边界
    h, w = img1.shape
    pts = np.float32([[0, 0], [0, h-1], [w-1, h-1], [w-1, 0]]).reshape(-1, 1, 2)
    dst = cv2.perspectiveTransform(pts, H)
    print(f"\n变换后的边界点:")
    print(dst.reshape(-1, 2))
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

def stitch_images(img1, img2):
    """拼接两张图像"""
    # SIFT检测
    sift = cv2.SIFT_create()
    kp1, des1 = sift.detectAndCompute(img1, None)
    kp2, des2 = sift.detectAndCompute(img2, None)

    # FLANN匹配
    FLANN_INDEX_KDTREE = 1
    index_params = dict(algorithm=FLANN_INDEX_KDTREE, trees=5)
    search_params = dict(checks=50)
    flann = cv2.FlannBasedMatcher(index_params, search_params)

    matches = flann.knnMatch(des1, des2, k=2)

    # 比率测试
    good = []
    for m, n in matches:
        if m.distance < 0.7 * n.distance:
            good.append(m)

    print(f"好匹配数: {len(good)}")

    if len(good) < 4:
        print("匹配点不足，无法拼接")
        return None

    # 提取匹配点
    src_pts = np.float32([kp1[m.queryIdx].pt for m in good]).reshape(-1, 1, 2)
    dst_pts = np.float32([kp2[m.trainIdx].pt for m in good]).reshape(-1, 1, 2)

    # 计算单应性矩阵
    H, mask = cv2.findHomography(src_pts, dst_pts, cv2.RANSAC, 5.0)

    if H is None:
        print("无法计算单应性矩阵")
        return None

    # 计算输出图像尺寸
    h1, w1 = img1.shape[:2]
    h2, w2 = img2.shape[:2]

    # 变换img1的四个角点
    corners1 = np.float32([[0, 0], [0, h1], [w1, h1], [w1, 0]]).reshape(-1, 1, 2)
    corners1_transformed = cv2.perspectiveTransform(corners1, H)

    # 合并所有角点
    corners2 = np.float32([[0, 0], [0, h2], [w2, h2], [w2, 0]]).reshape(-1, 1, 2)
    all_corners = np.concatenate([corners1_transformed, corners2], axis=0)

    # 计算边界
    x_min = int(np.floor(all_corners[:, 0, 0].min()))
    x_max = int(np.ceil(all_corners[:, 0, 0].max()))
    y_min = int(np.floor(all_corners[:, 0, 1].min()))
    y_max = int(np.ceil(all_corners[:, 0, 1].max()))

    # 平移矩阵
    translation = np.array([
        [1, 0, -x_min],
        [0, 1, -y_min],
        [0, 0, 1]
    ])

    # 输出尺寸
    output_size = (x_max - x_min, y_max - y_min)

    # 变换并拼接
    H_translated = translation @ H
    result = cv2.warpPerspective(img1, H_translated, output_size)

    # 放置img2
    result[-y_min:-y_min+h2, -x_min:-x_min+w2] = img2

    return result, H

# 创建测试图像
img_full = np.zeros((300, 600), dtype=np.uint8)
```

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt

class FeatureTracker:
    """基于特征匹配的物体追踪器"""

    def __init__(self, template):
        self.template = template
        self.orb = cv2.ORB_create(nfeatures=500)
        self.bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=False)

        # 提取模板特征
        self.kp_template, self.des_template = self.orb.detectAndCompute(template, None)
        print(f"模板特征点: {len(self.kp_template)}")

    def track(self, frame):
        """在帧中追踪物体"""
        # 检测帧中的特征
        kp_frame, des_frame = self.orb.detectAndCompute(frame, None)

        if des_frame is None or len(des_frame) < 2:
            return None, 0

        # KNN匹配
        matches = self.bf.knnMatch(self.des_template, des_frame, k=2)

        # 比率测试
        good = []
        for match in matches:
            if len(match) == 2:
                m, n = match
                if m.distance < 0.75 * n.distance:
                    good.append(m)

        if len(good) < 4:
            return None, len(good)

        # 提取匹配点
        src_pts = np.float32([self.kp_template[m.queryIdx].pt for m in good]).reshape(-1, 1, 2)
        dst_pts = np.float32([kp_frame[m.trainIdx].pt for m in good]).reshape(-1, 1, 2)

        # 计算单应性矩阵
        H, mask = cv2.findHomography(src_pts, dst_pts, cv2.RANSAC, 5.0)

        if H is None:
            return None, len(good)

        # 变换模板边界
        h, w = self.template.shape[:2]
        pts = np.float32([[0, 0], [0, h-1], [w-1, h-1], [w-1, 0]]).reshape(-1, 1, 2)
        dst = cv2.perspectiveTransform(pts, H)

        return dst, len(good)
```

```python
import cv2

def benchmark_matchers():

    sift = cv2.SIFT_create(nfeatures=5000)
    kp1, des1 = sift.detectAndCompute(img1, None)
    kp2, des2 = sift.detectAndCompute(img2, None)

    orb = cv2.ORB_create(nfeatures=5000)
    kp1_orb, des1_orb = orb.detectAndCompute(img1, None)

    FLANN_INDEX_KDTREE = 1
    index_params = dict(algorithm=FLANN_INDEX_KDTREE, trees=5)
```

### 核心要点回顾

1. **匹配器类型**
   - 暴力匹配（BFMatcher）：精确但较慢
   - FLANN匹配：近似但快速

2. **距离度量**
   - L2范数：用于SIFT/SURF等浮点描述符
   - Hamming距离：用于ORB/BRIEF等二进制描述符

3. **匹配过滤**
   - crossCheck：互为最佳匹配
   - 比率测试：Lowe's ratio test
   - RANSAC：几何验证

4. **应用技巧**
   - 根据描述符类型选择合适的匹配器和距离度量
   - 使用比率测试过滤误匹配
   - 使用RANSAC进一步验证几何一致性

### 匹配器选择指南

| 场景 | 推荐配置 |
|------|----------|
| 高精度要求 | BF + 比率测试 + RANSAC |
| 实时应用 | FLANN + ORB |
| 大规模特征 | FLANN (KD-Tree/LSH) |
| 小规模特征 | BF + crossCheck |
