## 一、OpenCV跟踪器总览

### 1.1 跟踪器对比

| 跟踪器 | 算法原理 | 速度 | 精度 | 尺度自适应 |
|--------|----------|------|------|------------|
| BOOSTING | AdaBoost | 慢 | 低 | 否 |
| MIL | 多实例学习 | 中 | 中 | 否 |
| KCF | 相关滤波 | 快 | 高 | 否 |
| TLD | 跟踪-学习-检测 | 中 | 高 | 是 |
| MEDIANFLOW | 中值流 | 快 | 中 | 是 |
| GOTURN | 深度学习 | 慢 | 高 | 是 |
| MOSSE | 相关滤波 | 很快 | 中 | 否 |
| CSRT | 判别相关滤波 | 中 | 很高 | 是 |

### 1.2 选择指南

- **需要速度**：MOSSE > KCF > MEDIANFLOW
- **需要精度**：CSRT > TLD > KCF
- **需要尺度自适应**：CSRT > TLD > MEDIANFLOW
- **长期跟踪**：TLD（有重检测能力）

### 示例1：MOSSE快速跟踪器

```python
"""
示例1：MOSSE快速跟踪器
最快的跟踪器，适合对速度要求极高的场景！
"""
import cv2
import time

cap = cv2.VideoCapture(0)

ret, frame = cap.read()
if not ret:
    exit()

print("选择跟踪目标...")
bbox = cv2.selectROI('Select', frame, False, True)
cv2.destroyWindow('Select')

# 创建MOSSE跟踪器
tracker = cv2.legacy.TrackerMOSSE_create()
tracker.init(frame, bbox)

fps_history = []

print("MOSSE跟踪器 - 最快的跟踪器")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    start_time = time.time()
    success, bbox = tracker.update(frame)
    elapsed = (time.time() - start_time) * 1000

    fps = 1000 / elapsed if elapsed > 0 else 0
    fps_history.append(fps)
    if len(fps_history) > 30:
        fps_history.pop(0)

    result = frame.copy()

    if success:
        x, y, w, h = [int(v) for v in bbox]
        cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    avg_fps = sum(fps_history) / len(fps_history)
    cv2.putText(result, f'MOSSE - FPS: {fps:.0f} (avg: {avg_fps:.0f})', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('MOSSE Tracker', result)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

print(f"平均FPS: {avg_fps:.0f}")
cap.release()
cv2.destroyAllWindows()
```

```python
"""
```

```python
import time
import numpy as np

def create_tracker(tracker_type):
    """创建跟踪器"""
    creators = {
        'MOSSE': cv2.legacy.TrackerMOSSE_create,
        'KCF': cv2.legacy.TrackerKCF_create,
        'CSRT': cv2.legacy.TrackerCSRT_create,

if not ret:
    exit()

trackers = {}
stats = {}

for tt in tracker_types:
    tracker = create_tracker(tt)
    tracker.init(frame, bbox)
    trackers[tt] = tracker
    stats[tt] = {
        'times': [],
        'success': 0,
```

### 示例6：跟踪器切换系统

```python
"""
示例6：跟踪器切换系统
运行时动态切换跟踪器！
"""
import cv2
import time

class SwitchableTracker:
    def __init__(self):
        self.tracker_types = ['MOSSE', 'KCF', 'CSRT', 'MEDIANFLOW']
        self.current_index = 0
        self.tracker = None
        self.bbox = None

    def init(self, frame, bbox):
        self.bbox = bbox
        self._create_tracker()
        self.tracker.init(frame, bbox)

    def _create_tracker(self):
        tt = self.tracker_types[self.current_index]
        if tt == 'MOSSE':
            self.tracker = cv2.legacy.TrackerMOSSE_create()
        elif tt == 'KCF':
            self.tracker = cv2.legacy.TrackerKCF_create()
        elif tt == 'CSRT':
            self.tracker = cv2.legacy.TrackerCSRT_create()
        elif tt == 'MEDIANFLOW':
            self.tracker = cv2.legacy.TrackerMedianFlow_create()

    def switch_tracker(self, frame):
        """切换到下一个跟踪器"""
        self.current_index = (self.current_index + 1) % len(self.tracker_types)
        self._create_tracker()
        if self.bbox is not None:
            x, y, w, h = [int(v) for v in self.bbox]
            self.tracker.init(frame, (x, y, w, h))
        return self.tracker_types[self.current_index]

    def update(self, frame):
        if self.tracker is None:
            return False, None

        success, bbox = self.tracker.update(frame)
        if success:
            self.bbox = bbox
        return success, bbox

    def current_type(self):
        return self.tracker_types[self.current_index]

# 使用
cap = cv2.VideoCapture(0)

ret, frame = cap.read()
bbox = cv2.selectROI('Select', frame, False, True)
cv2.destroyWindow('Select')

tracker = SwitchableTracker()
tracker.init(frame, bbox)

print("跟踪器切换系统")
print("按 's' 切换跟踪器")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    start_time = time.time()
    success, bbox = tracker.update(frame)
    elapsed = (time.time() - start_time) * 1000

    result = frame.copy()

    if success:
        x, y, w, h = [int(v) for v in bbox]
        cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    fps = 1000 / elapsed if elapsed > 0 else 0
    cv2.putText(result, f'{tracker.current_type()}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
    cv2.putText(result, f'FPS: {fps:.1f}', (10, 60),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
    cv2.putText(result, "Press 's' to switch", (10, 90),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)

    cv2.imshow('Switchable Tracker', result)

    key = cv2.waitKey(30) & 0xFF
    if key == ord('q'):
        break
    elif key == ord('s'):
        new_type = tracker.switch_tracker(frame)
        print(f"切换到: {new_type}")

cap.release()
cv2.destroyAllWindows()
```

### 示例7：自动最佳跟踪器选择

```python
"""
示例7：自动最佳跟踪器选择
根据场景自动选择最佳跟踪器！
"""
import cv2
import numpy as np
import time

class AutoTracker:
    def __init__(self):
        self.trackers = {}
        self.scores = {}
        self.current_best = 'KCF'
        self.eval_interval = 30
        self.frame_count = 0

    def init(self, frame, bbox):
        tracker_types = ['MOSSE', 'KCF', 'CSRT']

        for tt in tracker_types:
            if tt == 'MOSSE':
                tracker = cv2.legacy.TrackerMOSSE_create()
            elif tt == 'KCF':
                tracker = cv2.legacy.TrackerKCF_create()
            elif tt == 'CSRT':
                tracker = cv2.legacy.TrackerCSRT_create()

            tracker.init(frame, bbox)
            self.trackers[tt] = {
                'tracker': tracker,
                'success_count': 0,
                'time_sum': 0
            }
            self.scores[tt] = 0

    def update(self, frame):
        self.frame_count += 1
        results = {}

        for tt, data in self.trackers.items():
            start = time.time()
            success, bbox = data['tracker'].update(frame)
            elapsed = (time.time() - start) * 1000

            data['time_sum'] += elapsed
            if success:
                data['success_count'] += 1

            results[tt] = {
                'success': success,
                'bbox': bbox,
                'time': elapsed
            }

        # 定期评估并选择最佳
        if self.frame_count % self.eval_interval == 0:
            self._evaluate()

        return results[self.current_best]['success'], \
               results[self.current_best]['bbox'], \
               self.current_best

    def _evaluate(self):
        best_score = -1
        best_tracker = self.current_best

        for tt, data in self.trackers.items():
            if self.frame_count > 0:
                success_rate = data['success_count'] / self.frame_count
                avg_time = data['time_sum'] / self.frame_count
                speed_score = 1 / (1 + avg_time / 20)

                score = success_rate * 0.6 + speed_score * 0.4
                self.scores[tt] = score

                if score > best_score:
                    best_score = score
                    best_tracker = tt

        if best_tracker != self.current_best:
            print(f"切换最佳跟踪器: {self.current_best} -> {best_tracker}")
            self.current_best = best_tracker

# 使用
cap = cv2.VideoCapture(0)

ret, frame = cap.read()
bbox = cv2.selectROI('Select', frame, False, True)
cv2.destroyWindow('Select')

tracker = AutoTracker()
tracker.init(frame, bbox)

print("自动跟踪器选择")
print("系统会自动选择最佳跟踪器")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    success, bbox, best_type = tracker.update(frame)

    result = frame.copy()

    if success:
        x, y, w, h = [int(v) for v in bbox]
        cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

    cv2.putText(result, f'Best: {best_type}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    # 显示分数
    y_offset = 60
    for tt, score in tracker.scores.items():
        color = (0, 255, 0) if tt == best_type else (255, 255, 255)
        cv2.putText(result, f'{tt}: {score:.2f}', (10, y_offset),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 1)
        y_offset += 20

    cv2.imshow('Auto Tracker', result)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

```python
"""
示例8：级联跟踪器
快速跟踪器失败时切换到精确跟踪器！
"""
import cv2
import time

class CascadeTracker:
    def __init__(self):
        self.fast_tracker = None
        self.accurate_tracker = None
        self.using_accurate = False
        self.fail_count = 0
        self.max_fails = 5

    def init(self, frame, bbox):
        self.fast_tracker = cv2.legacy.TrackerMOSSE_create()
        self.accurate_tracker = cv2.legacy.TrackerCSRT_create()

        self.fast_tracker.init(frame, bbox)
        self.accurate_tracker.init(frame, bbox)

        self.using_accurate = False
        self.fail_count = 0

    def update(self, frame):
        if not self.using_accurate:
            # 使用快速跟踪器
            success, bbox = self.fast_tracker.update(frame)

            if not success:
                self.fail_count += 1
                if self.fail_count >= self.max_fails:
                    print("切换到精确跟踪器")
                    self.using_accurate = True
                    self.fail_count = 0
            else:
                self.fail_count = 0
                # 同步更新精确跟踪器
                x, y, w, h = [int(v) for v in bbox]
                self.accurate_tracker = cv2.legacy.TrackerCSRT_create()
                self.accurate_tracker.init(frame, (x, y, w, h))

            return success, bbox, 'MOSSE'
        else:
            # 使用精确跟踪器
            success, bbox = self.accurate_tracker.update(frame)

            if success:
                self.fail_count = 0
                # 尝试恢复快速跟踪
                if self.fail_count == 0:
                    x, y, w, h = [int(v) for v in bbox]
                    self.fast_tracker = cv2.legacy.TrackerMOSSE_create()
                    self.fast_tracker.init(frame, (x, y, w, h))
                    self.using_accurate = False
                    print("恢复快速跟踪器")

            return success, bbox, 'CSRT'

# 使用

ret, frame = cap.read()
bbox = cv2.selectROI('Select', frame, False, True)
cv2.destroyWindow('Select')

tracker = CascadeTracker()
tracker.init(frame, bbox)

print("级联跟踪器")
print("快速模式失败时自动切换到精确模式")
print("按 'q' 退出")

while True:
    ret, frame = cap.read()
    if not ret:
        break

    start = time.time()
    success, bbox, tracker_type = tracker.update(frame)
    elapsed = (time.time() - start) * 1000

    result = frame.copy()

    if success:
        x, y, w, h = [int(v) for v in bbox]
        color = (0, 255, 0) if tracker_type == 'MOSSE' else (0, 0, 255)
        cv2.rectangle(result, (x, y), (x + w, y + h), color, 2)

    fps = 1000 / elapsed if elapsed > 0 else 0
    cv2.putText(result, f'{tracker_type} - FPS: {fps:.0f}', (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

    cv2.imshow('Cascade Tracker', result)

    if cv2.waitKey(30) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

```python
"""
示例9：跟踪器融合
融合多个跟踪器的结果提高鲁棒性！
"""
import cv2
import numpy as np

class FusionTracker:
    def __init__(self):
        self.trackers = []
        self.weights = []

    def init(self, frame, bbox):
        configs = [
            ('KCF', 0.4),
            ('CSRT', 0.4),
            ('MEDIANFLOW', 0.2),
        ]

        for tt, weight in configs:
            if tt == 'KCF':
                tracker = cv2.legacy.TrackerKCF_create()
            elif tt == 'CSRT':
                tracker = cv2.legacy.TrackerCSRT_create()
            elif tt == 'MEDIANFLOW':
                tracker = cv2.legacy.TrackerMedianFlow_create()

            tracker.init(frame, bbox)
            self.trackers.append({'type': tt, 'tracker': tracker, 'weight': weight})

    def update(self, frame):
        results = []
        weights = []

        for item in self.trackers:
            success, bbox = item['tracker'].update(frame)

            if success:
                results.append(bbox)
                weights.append(item['weight'])

        if not results:
            return False, None

        # 加权融合
        weights = np.array(weights) / sum(weights)
        bboxes = np.array(results)

        fused_bbox = np.sum(bboxes * weights[:, np.newaxis], axis=0)

        return True, tuple(fused_bbox)

# 使用
cap = cv2.VideoCapture(0)

ret, frame = cap.read()
bbox = cv2.selectROI('Select', frame, False, True)
cv2.destroyWindow('Select')

tracker = FusionTracker()
```

### 示例10：完整跟踪器工具箱

```python
"""
示例10：完整跟踪器工具箱
集成所有跟踪器的完整工具！
"""
import cv2
import numpy as np
import time
from collections import deque

class TrackerToolbox:
    TRACKERS = {
        '1': ('MOSSE', cv2.legacy.TrackerMOSSE_create),
        '2': ('KCF', cv2.legacy.TrackerKCF_create),
        '3': ('CSRT', cv2.legacy.TrackerCSRT_create),
        '4': ('MEDIANFLOW', cv2.legacy.TrackerMedianFlow_create),
        '5': ('MIL', cv2.legacy.TrackerMIL_create),
    }

    def __init__(self):
        self.tracker = None
        self.tracker_type = None
        self.trajectory = deque(maxlen=100)
        self.fps_history = deque(maxlen=30)
        self.initialized = False

    def init(self, frame, bbox, tracker_key='2'):
        if tracker_key not in self.TRACKERS:
            tracker_key = '2'

        name, creator = self.TRACKERS[tracker_key]
        self.tracker_type = name
        self.tracker = creator()
        self.tracker.init(frame, bbox)

        x, y, w, h = [int(v) for v in bbox]
        self.trajectory.clear()
        self.trajectory.append((x + w//2, y + h//2))

        self.initialized = True

    def update(self, frame):
        if not self.initialized:
            return False, None

        start = time.time()
        success, bbox = self.tracker.update(frame)
        elapsed = (time.time() - start) * 1000

        self.fps_history.append(1000 / elapsed if elapsed > 0 else 0)

        if success:
            x, y, w, h = [int(v) for v in bbox]
            self.trajectory.append((x + w//2, y + h//2))

        return success, bbox

    def get_fps(self):
        return np.mean(self.fps_history) if self.fps_history else 0

    def draw(self, frame, bbox, success):
        result = frame.copy()

        # 绘制轨迹
        for i in range(1, len(self.trajectory)):
            alpha = i / len(self.trajectory)
            color = (int(255*(1-alpha)), int(255*alpha), 0)
            cv2.line(result, self.trajectory[i-1], self.trajectory[i], color, 2)

        # 绘制边界框
        if success and bbox is not None:
            x, y, w, h = [int(v) for v in bbox]
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

        return result

def main():
    cap = cv2.VideoCapture(0)
    toolbox = TrackerToolbox()

    selecting = False
    start_point = None
    current_bbox = None

    def mouse_callback(event, x, y, flags, param):
        nonlocal selecting, start_point, current_bbox

        if event == cv2.EVENT_LBUTTONDOWN:
            selecting = True
            start_point = (x, y)
        elif event == cv2.EVENT_LBUTTONUP and selecting:
            selecting = False
            x1, y1 = start_point
            w, h = x - x1, y - y1
            if w > 10 and h > 10:
                current_bbox = (x1, y1, w, h)

    cv2.namedWindow('Tracker Toolbox')
    cv2.setMouseCallback('Tracker Toolbox', mouse_callback)

    print("="*50)
    print("跟踪器工具箱")
    print("="*50)
    print("用鼠标框选目标")
    print("按 1-5 切换跟踪器:")
    for key, (name, _) in TrackerToolbox.TRACKERS.items():
        print(f"  {key}: {name}")
    print("按 'r' 重置")
    print("按 'q' 退出")
    print("="*50)

    current_tracker_key = '2'

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        # 初始化
        if current_bbox is not None:
            toolbox.init(frame, current_bbox, current_tracker_key)
            current_bbox = None

        result = frame.copy()

        # 更新
        if toolbox.initialized:
            success, bbox = toolbox.update(frame)
            result = toolbox.draw(result, bbox, success)

            # 信息
            fps = toolbox.get_fps()
            cv2.putText(result, f'{toolbox.tracker_type}', (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
            cv2.putText(result, f'FPS: {fps:.1f}', (10, 60),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
            cv2.putText(result, f'Status: {"OK" if success else "Lost"}', (10, 90),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.6,
                        (0, 255, 0) if success else (0, 0, 255), 2)
        else:
            cv2.putText(result, 'Draw box to start', (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)

        # 选择框
        if selecting and start_point:
            cv2.rectangle(result, start_point,
                          (cv2.getWindowImageRect('Tracker Toolbox')[2],
                           cv2.getWindowImageRect('Tracker Toolbox')[3]),
                          (0, 255, 255), 2)

        cv2.imshow('Tracker Toolbox', result)

        key = cv2.waitKey(30) & 0xFF

        if key == ord('q'):
            break
        elif key == ord('r'):
            toolbox = TrackerToolbox()
        elif chr(key) in TrackerToolbox.TRACKERS:
            current_tracker_key = chr(key)
            if toolbox.initialized and toolbox.trajectory:
                # 用当前位置重新初始化
                ret, frame = cap.read()
                if ret:
                    last = toolbox.trajectory[-1]
                    w, h = 100, 100  # 默认大小
                    bbox = (last[0] - w//2, last[1] - h//2, w, h)
                    toolbox.init(frame, bbox, current_tracker_key)
                    print(f"切换到: {TrackerToolbox.TRACKERS[current_tracker_key][0]}")

    cap.release()
    cv2.destroyAllWindows()

if __name__ == '__main__':
    main()
```
