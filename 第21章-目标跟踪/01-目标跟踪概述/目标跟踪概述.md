## 一、目标跟踪基础

### 1.1 什么是目标跟踪

**目标跟踪（Object Tracking）** 是指在视频序列中持续定位特定目标的技术：

- **输入**：第一帧中目标的位置（边界框）
- **输出**：后续每帧中目标的位置
- **挑战**：遮挡、形变、光照变化、快速运动等

### 1.2 跟踪 vs 检测

| 特性 | 目标跟踪 | 目标检测 |
|------|----------|----------|
| 初始化 | 需要初始位置 | 不需要 |
| 速度 | 快 | 相对慢 |
| 类别 | 跟踪特定实例 | 检测所有实例 |
| 遮挡处理 | 可预测 | 可能丢失 |

### 1.3 跟踪算法分类

**按方法分类：**
1. **生成式方法**：建立目标外观模型（MeanShift、粒子滤波）
2. **判别式方法**：区分目标与背景（KCF、MOSSE）
3. **深度学习方法**：端到端学习（SiamFC、GOTURN）

**按特征分类：**
1. **颜色特征**：直方图（MeanShift、CamShift）
2. **纹理特征**：HOG、LBP
3. **深度特征**：CNN特征

### 1.4 OpenCV跟踪器

OpenCV提供多种内置跟踪器：

| 跟踪器 | 特点 | 速度 | 精度 |
|--------|------|------|------|
| BOOSTING | 基于AdaBoost | 慢 | 低 |
| MIL | 多实例学习 | 慢 | 中 |
| KCF | 核相关滤波 | 快 | 高 |
| TLD | 跟踪-学习-检测 | 中 | 高 |
| MEDIANFLOW | 中值流 | 快 | 中 |
| GOTURN | 深度学习 | 慢 | 高 |
| MOSSE | 最小输出平方和 | 很快 | 中 |
| CSRT | 判别相关滤波 | 中 | 很高 |

```python
"""
示例1：理解跟踪流程
本小姐来演示目标跟踪的基本流程！
"""
import cv2
import numpy as np

def simple_tracking_demo():
    """简单跟踪演示"""
    cap = cv2.VideoCapture(0)

    # 跟踪状态
    tracking = False
    bbox = None
    template = None

    def mouse_callback(event, x, y, flags, param):
        nonlocal tracking, bbox, template

        if event == cv2.EVENT_LBUTTONDOWN:
            bbox = [x, y, 0, 0]
            tracking = False

        elif event == cv2.EVENT_MOUSEMOVE and bbox is not None and not tracking:
            bbox[2] = x - bbox[0]
            bbox[3] = y - bbox[1]

        elif event == cv2.EVENT_LBUTTONUP and bbox is not None:
            if bbox[2] > 10 and bbox[3] > 10:
                tracking = True
                # 保存模板
                ret, frame = cap.read()
                if ret:
                    x, y, w, h = bbox
                    template = frame[y:y+h, x:x+w].copy()
                    print(f"开始跟踪，模板大小: {w}x{h}")
            else:
                bbox = None

    cv2.namedWindow('Tracking Demo')
    cv2.setMouseCallback('Tracking Demo', mouse_callback)

    print("简单跟踪演示")
    print("用鼠标框选目标开始跟踪")
    print("按 'r' 重置，按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        result = frame.copy()

        # 绘制选择框
        if bbox is not None and not tracking:
            x, y, w, h = bbox
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

        # 简单模板匹配跟踪
        if tracking and template is not None:
            # 使用模板匹配
            gray_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            gray_template = cv2.cvtColor(template, cv2.COLOR_BGR2GRAY)

            match = cv2.matchTemplate(gray_frame, gray_template, cv2.TM_CCOEFF_NORMED)
            _, max_val, _, max_loc = cv2.minMaxLoc(match)

            if max_val > 0.5:  # 匹配阈值
                x, y = max_loc
                w, h = template.shape[1], template.shape[0]
                bbox = [x, y, w, h]

                cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)
                cv2.putText(result, f'Confidence: {max_val:.2f}', (x, y - 10),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)
            else:
                cv2.putText(result, 'Target Lost!', (10, 30),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

        cv2.imshow('Tracking Demo', result)

        key = cv2.waitKey(30) & 0xFF
        if key == ord('q'):
            break
        elif key == ord('r'):
            tracking = False
            bbox = None
            template = None

    cap.release()
    cv2.destroyAllWindows()
```

```python
"""

本小姐来介绍OpenCV跟踪器的统一接口！

import cv2

def create_tracker(tracker_type):
    """创建指定类型的跟踪器"""
    trackers = {
        'BOOSTING': cv2.legacy.TrackerBoosting_create,
        'MIL': cv2.legacy.TrackerMIL_create,
        'KCF': cv2.legacy.TrackerKCF_create,
        'TLD': cv2.legacy.TrackerTLD_create,
        'MEDIANFLOW': cv2.legacy.TrackerMedianFlow_create,
        'MOSSE': cv2.legacy.TrackerMOSSE_create,
        'CSRT': cv2.legacy.TrackerCSRT_create,
    }

    if tracker_type in trackers:
        return trackers[tracker_type]()
    else:
        print(f"未知跟踪器类型: {tracker_type}")
        return None

def tracking_demo(tracker_type='KCF'):
    """使用OpenCV跟踪器的演示"""
    cap = cv2.VideoCapture(0)

    # 读取第一帧
    ret, frame = cap.read()
    if not ret:
        print("无法读取视频！")
        return

    # 选择ROI
    print(f"使用 {tracker_type} 跟踪器")
    print("请用鼠标选择跟踪目标，按Enter确认")
    bbox = cv2.selectROI('Select Target', frame, fromCenter=False, showCrosshair=True)
    cv2.destroyWindow('Select Target')

    if bbox[2] == 0 or bbox[3] == 0:
        print("未选择目标！")
        return

    # 创建跟踪器
    tracker = create_tracker(tracker_type)
    if tracker is None:
        return

    # 初始化跟踪器
    tracker.init(frame, bbox)

    print("开始跟踪...")
    print("按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        # 更新跟踪器
        success, bbox = tracker.update(frame)

        if success:
            # 绘制边界框
            x, y, w, h = [int(v) for v in bbox]
            cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 0), 2)
            cv2.putText(frame, f'{tracker_type} Tracking', (x, y - 10),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
        else:
            cv2.putText(frame, 'Tracking Failed!', (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

        cv2.imshow('Tracking', frame)

        if cv2.waitKey(30) & 0xFF == ord('q'):
            break

    cap.release()
    cv2.destroyAllWindows()
```

```python
"""
示例3：多跟踪器对比
同时运行多个跟踪器进行对比！

import cv2
import time

def multi_tracker_comparison():
    """多跟踪器对比演示"""
    cap = cv2.VideoCapture(0)

    ret, frame = cap.read()
    if not ret:
        return

    # 选择目标
    print("请选择跟踪目标...")
    bbox = cv2.selectROI('Select', frame, False, True)
    cv2.destroyWindow('Select')

    if bbox[2] == 0 or bbox[3] == 0:
        return

    # 创建多个跟踪器
    tracker_types = ['KCF', 'MOSSE', 'CSRT', 'MEDIANFLOW']
    trackers = []
    colors = [(255, 0, 0), (0, 255, 0), (0, 0, 255), (255, 255, 0)]

    for tt in tracker_types:
        if tt == 'KCF':
            tracker = cv2.legacy.TrackerKCF_create()
        elif tt == 'MOSSE':
            tracker = cv2.legacy.TrackerMOSSE_create()
        elif tt == 'CSRT':
            tracker = cv2.legacy.TrackerCSRT_create()
        elif tt == 'MEDIANFLOW':
            tracker = cv2.legacy.TrackerMedianFlow_create()

        tracker.init(frame, bbox)
        trackers.append((tt, tracker))

    print("开始对比跟踪...")
    print("按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        result = frame.copy()

        # 更新所有跟踪器
        for i, (name, tracker) in enumerate(trackers):
            start_time = time.time()
            success, box = tracker.update(frame)
            elapsed = (time.time() - start_time) * 1000

            if success:
                x, y, w, h = [int(v) for v in box]
                cv2.rectangle(result, (x, y), (x + w, y + h), colors[i], 2)

            # 显示信息
            status = "OK" if success else "Lost"
            cv2.putText(result, f'{name}: {status} ({elapsed:.1f}ms)',
                        (10, 30 + i * 25), cv2.FONT_HERSHEY_SIMPLEX,
                        0.6, colors[i], 2)

        cv2.imshow('Multi Tracker Comparison', result)

        if cv2.waitKey(30) & 0xFF == ord('q'):
            break

    cap.release()
    cv2.destroyAllWindows()
```

```python
"""
示例4：跟踪失败检测
检测跟踪何时失败并尝试恢复！
"""
import cv2
import numpy as np

class RobustTracker:
    def __init__(self, tracker_type='KCF'):
        self.tracker_type = tracker_type
        self.tracker = None
        self.template = None
        self.bbox = None
        self.lost_count = 0
        self.max_lost = 30

    def init(self, frame, bbox):
        """初始化跟踪器"""
        self.bbox = bbox
        self.template = self._extract_template(frame, bbox)
        self._create_tracker()
        self.tracker.init(frame, bbox)
        self.lost_count = 0

    def _create_tracker(self):
        """创建跟踪器"""
        if self.tracker_type == 'KCF':
            self.tracker = cv2.legacy.TrackerKCF_create()
        elif self.tracker_type == 'CSRT':
            self.tracker = cv2.legacy.TrackerCSRT_create()
        else:
            self.tracker = cv2.legacy.TrackerKCF_create()

    def _extract_template(self, frame, bbox):
        """提取模板"""
        x, y, w, h = [int(v) for v in bbox]
        return frame[y:y+h, x:x+w].copy()

    def update(self, frame):
        """更新跟踪"""
        if self.tracker is None:
            return False, None

        success, bbox = self.tracker.update(frame)

        if success:
            # 验证跟踪结果
            confidence = self._compute_confidence(frame, bbox)

            if confidence > 0.3:
                self.bbox = bbox
                self.lost_count = 0
                return True, bbox
            else:
                success = False

        if not success:
            self.lost_count += 1

            if self.lost_count < self.max_lost:
                # 尝试模板匹配恢复
                recovered_bbox = self._try_recover(frame)
                if recovered_bbox is not None:
                    self.bbox = recovered_bbox
                    self._create_tracker()
                    self.tracker.init(frame, tuple(recovered_bbox))
                    self.lost_count = 0
                    return True, recovered_bbox

        return False, self.bbox

    def _compute_confidence(self, frame, bbox):
        """计算置信度"""
        if self.template is None:
            return 1.0

        x, y, w, h = [int(v) for v in bbox]

        # 边界检查
        if x < 0 or y < 0 or x + w > frame.shape[1] or y + h > frame.shape[0]:
            return 0.0

        current = frame[y:y+h, x:x+w]

        # 调整大小
        current = cv2.resize(current, (self.template.shape[1], self.template.shape[0]))

        # 计算相似度
        result = cv2.matchTemplate(current, self.template, cv2.TM_CCOEFF_NORMED)
        return result[0][0]

    def _try_recover(self, frame):
        """尝试恢复跟踪"""
        if self.template is None or self.bbox is None:
            return None

        # 在上一位置附近搜索
        x, y, w, h = [int(v) for v in self.bbox]
        search_margin = 50

        # 搜索区域
        sx = max(0, x - search_margin)
        sy = max(0, y - search_margin)
        ex = min(frame.shape[1], x + w + search_margin)
        ey = min(frame.shape[0], y + h + search_margin)

        search_region = frame[sy:ey, sx:ex]

        # 模板匹配
        result = cv2.matchTemplate(search_region, self.template, cv2.TM_CCOEFF_NORMED)
        _, max_val, _, max_loc = cv2.minMaxLoc(result)

        if max_val > 0.5:
            new_x = sx + max_loc[0]
            new_y = sy + max_loc[1]
            return (new_x, new_y, self.template.shape[1], self.template.shape[0])

        return None

# 使用示例
def robust_tracking_demo():
    cap = cv2.VideoCapture(0)

    ret, frame = cap.read()
    if not ret:
        return

    print("选择跟踪目标...")
    bbox = cv2.selectROI('Select', frame, False, True)
    cv2.destroyWindow('Select')

    tracker = RobustTracker('KCF')
    tracker.init(frame, bbox)

    print("鲁棒跟踪演示")
    print("按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        success, bbox = tracker.update(frame)

        result = frame.copy()

        if success:
            x, y, w, h = [int(v) for v in bbox]
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)
            cv2.putText(result, 'Tracking', (x, y - 10),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
        else:
            cv2.putText(result, f'Lost ({tracker.lost_count})', (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)

        cv2.imshow('Robust Tracking', result)

        if cv2.waitKey(30) & 0xFF == ord('q'):
            break

    cap.release()
    cv2.destroyAllWindows()
```

```python
"""
示例5：多目标跟踪
同时跟踪多个目标！
"""
import cv2
import numpy as np

class MultiObjectTracker:
    def __init__(self, tracker_type='KCF'):
        self.tracker_type = tracker_type
        self.trackers = []
        self.colors = []
        self.next_id = 0

    def add_target(self, frame, bbox):
        """添加跟踪目标"""
        if self.tracker_type == 'KCF':
            tracker = cv2.legacy.TrackerKCF_create()
        elif self.tracker_type == 'CSRT':
            tracker = cv2.legacy.TrackerCSRT_create()
        else:
            tracker = cv2.legacy.TrackerMOSSE_create()

        tracker.init(frame, bbox)

        # 随机颜色
        color = tuple(map(int, np.random.randint(0, 255, 3)))

        self.trackers.append({
            'id': self.next_id,
            'tracker': tracker,
            'bbox': bbox,
            'active': True
        })
        self.colors.append(color)
        self.next_id += 1

        print(f"添加目标 ID: {self.next_id - 1}")

    def update(self, frame):
        """更新所有跟踪器"""
        results = []

        for i, item in enumerate(self.trackers):
            if item['active']:
                success, bbox = item['tracker'].update(frame)
                item['bbox'] = bbox
                item['active'] = success

                results.append({
                    'id': item['id'],
                    'bbox': bbox,
                    'active': success,
                    'color': self.colors[i]
                })

        return results

    def remove_inactive(self):
        """移除失效的跟踪器"""
        self.trackers = [t for t in self.trackers if t['active']]

# 使用演示
def multi_object_tracking_demo():
    cap = cv2.VideoCapture(0)
    mot = MultiObjectTracker('KCF')

    selecting = False
    start_point = None

    def mouse_callback(event, x, y, flags, param):
        nonlocal selecting, start_point

        if event == cv2.EVENT_LBUTTONDOWN:
            selecting = True
            start_point = (x, y)

        elif event == cv2.EVENT_LBUTTONUP and selecting:
            selecting = False
            x1, y1 = start_point
            w, h = x - x1, y - y1

            if w > 10 and h > 10:
                ret, frame = cap.read()
                if ret:
                    mot.add_target(frame, (x1, y1, w, h))

    cv2.namedWindow('Multi Object Tracking')
    cv2.setMouseCallback('Multi Object Tracking', mouse_callback)

    print("多目标跟踪演示")
    print("用鼠标框选添加跟踪目标")
    print("按 'c' 清除所有目标")
    print("按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        # 更新跟踪
        results = mot.update(frame)

        # 绘制结果
        for r in results:
            if r['active']:
                x, y, w, h = [int(v) for v in r['bbox']]
                cv2.rectangle(frame, (x, y), (x + w, y + h), r['color'], 2)
                cv2.putText(frame, f"ID: {r['id']}", (x, y - 10),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.6, r['color'], 2)

        # 绘制选择框
        if selecting and start_point:
            cv2.rectangle(frame, start_point,
                          (cv2.getWindowImageRect('Multi Object Tracking')[2],
                           cv2.getWindowImageRect('Multi Object Tracking')[3]),

        cv2.putText(frame, f'Tracking: {len([r for r in results if r["active"]])} objects',

        key = cv2.waitKey(30) & 0xFF

            print("清除所有目标")

    cv2.destroyAllWindows()
```

### 示例6：跟踪性能评估

```python
"""
示例6：跟踪性能评估
评估跟踪器的性能指标！
"""
import cv2
import numpy as np
import time

def compute_iou(box1, box2):
    """计算IoU（交并比）"""
    x1, y1, w1, h1 = box1
    x2, y2, w2, h2 = box2

    # 计算交集
    xi1 = max(x1, x2)
    yi1 = max(y1, y2)
    xi2 = min(x1 + w1, x2 + w2)
    yi2 = min(y1 + h1, y2 + h2)

    if xi2 <= xi1 or yi2 <= yi1:
        return 0.0

    inter_area = (xi2 - xi1) * (yi2 - yi1)

    # 计算并集
    box1_area = w1 * h1
    box2_area = w2 * h2
    union_area = box1_area + box2_area - inter_area

    return inter_area / union_area if union_area > 0 else 0.0

class TrackerEvaluator:
    def __init__(self):
        self.frame_times = []
        self.ious = []
        self.success_count = 0
        self.total_count = 0

    def update(self, elapsed_time, predicted_box, gt_box=None):
        """更新评估指标"""
        self.frame_times.append(elapsed_time)
        self.total_count += 1

        if gt_box is not None:
            iou = compute_iou(predicted_box, gt_box)
            self.ious.append(iou)
            if iou > 0.5:
                self.success_count += 1

    def get_metrics(self):
        """获取性能指标"""
        avg_fps = 1000 / np.mean(self.frame_times) if self.frame_times else 0
        avg_iou = np.mean(self.ious) if self.ious else 0
        success_rate = self.success_count / self.total_count if self.total_count > 0 else 0

        return {
            'avg_fps': avg_fps,
            'avg_time_ms': np.mean(self.frame_times) if self.frame_times else 0,
            'avg_iou': avg_iou,
            'success_rate': success_rate,
            'total_frames': self.total_count
        }

def tracking_evaluation_demo():
    """跟踪性能评估演示"""
    cap = cv2.VideoCapture(0)

    ret, frame = cap.read()
    if not ret:
        return

    print("选择跟踪目标...")
    bbox = cv2.selectROI('Select', frame, False, True)
    cv2.destroyWindow('Select')

    # 创建跟踪器
    tracker = cv2.legacy.TrackerKCF_create()
    tracker.init(frame, bbox)

    evaluator = TrackerEvaluator()

    print("跟踪性能评估")
    print("按 'q' 退出查看统计")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        start_time = time.time()
        success, current_bbox = tracker.update(frame)
        elapsed = (time.time() - start_time) * 1000

        if success:
            evaluator.update(elapsed, current_bbox)

            x, y, w, h = [int(v) for v in current_bbox]
            cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 0), 2)

        # 显示实时指标
        metrics = evaluator.get_metrics()
        cv2.putText(frame, f"FPS: {metrics['avg_fps']:.1f}", (10, 30),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
        cv2.putText(frame, f"Time: {elapsed:.1f}ms", (10, 55),
```

```python
    tracker.init(frame, bbox)

    # 轨迹历史

    velocity_history = deque(maxlen=10)

            trajectory.append(center)

            # 计算速度
            if prev_center is not None:
                velocity = np.sqrt((center[0] - prev_center[0])**2 +
                                   (center[1] - prev_center[1])**2)

            # 绘制边界框
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

            # 绘制轨迹
            for i in range(1, len(trajectory)):
                # 渐变颜色
                alpha = i / len(trajectory)
                color = (int(255 * (1 - alpha)), int(255 * alpha), 0)
                thickness = max(1, int(3 * alpha))
                cv2.line(result, trajectory[i-1], trajectory[i], color, thickness)

            # 绘制当前位置
            cv2.circle(result, center, 5, (0, 0, 255), -1)

            # 显示速度
            if velocity_history:
                avg_velocity = np.mean(velocity_history)

                            (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
```

```python
"""
示例8：跟踪区域预测
预测目标下一帧的位置！
"""
import cv2
import numpy as np
from collections import deque

class PredictiveTracker:
    def __init__(self):
        self.tracker = cv2.legacy.TrackerKCF_create()
        self.positions = deque(maxlen=10)
        self.velocities = deque(maxlen=5)
        self.initialized = False

    def init(self, frame, bbox):
        self.tracker.init(frame, bbox)
        x, y, w, h = bbox
        self.positions.append((x + w/2, y + h/2))
        self.initialized = True

    def predict_next_position(self):
        """预测下一帧位置"""
        if len(self.positions) < 2:
            return None

        # 计算平均速度
        velocities = []
        for i in range(1, len(self.positions)):
            vx = self.positions[i][0] - self.positions[i-1][0]
            vy = self.positions[i][1] - self.positions[i-1][1]
            velocities.append((vx, vy))

        avg_vx = np.mean([v[0] for v in velocities])
        avg_vy = np.mean([v[1] for v in velocities])

        # 预测
        last_pos = self.positions[-1]
        predicted = (last_pos[0] + avg_vx, last_pos[1] + avg_vy)

        return predicted

    def update(self, frame):
        if not self.initialized:
            return False, None, None

        success, bbox = self.tracker.update(frame)

        predicted_pos = self.predict_next_position()

        if success:
            x, y, w, h = bbox
            center = (x + w/2, y + h/2)
            self.positions.append(center)

        return success, bbox, predicted_pos

def predictive_tracking_demo():
    cap = cv2.VideoCapture(0)

        return

    cv2.destroyWindow('Select')

    tracker = PredictiveTracker()
    tracker.init(frame, bbox)

    print("预测跟踪演示")
    print("绿框=当前位置，蓝圈=预测位置")
    print("按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        success, bbox, predicted = tracker.update(frame)

        result = frame.copy()

        if success:
            x, y, w, h = [int(v) for v in bbox]
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

            # 绘制预测位置

                px, py = int(predicted[0]), int(predicted[1])
```

```python
"""
示例9：自适应跟踪器选择
根据场景自动选择最佳跟踪器！
"""
import cv2
import numpy as np
import time

class AdaptiveTracker:
    def __init__(self):
        self.trackers = {}
        self.current_tracker = None
        self.current_name = None
        self.performance_history = {}
        self.frame_count = 0

    def init(self, frame, bbox):
        """初始化所有跟踪器"""
        tracker_types = {
            'KCF': cv2.legacy.TrackerKCF_create,
            'MOSSE': cv2.legacy.TrackerMOSSE_create,
            'CSRT': cv2.legacy.TrackerCSRT_create,
        }

        for name, creator in tracker_types.items():
            tracker = creator()
            tracker.init(frame, bbox)
            self.trackers[name] = {
                'tracker': tracker,
                'success_count': 0,
                'total_time': 0
            }
            self.performance_history[name] = []

        # 默认使用KCF
        self.current_name = 'KCF'
        self.current_tracker = self.trackers['KCF']['tracker']

    def update(self, frame):
        """更新跟踪"""
        self.frame_count += 1

        results = {}

        # 更新所有跟踪器
        for name, data in self.trackers.items():
            start_time = time.time()
            success, bbox = data['tracker'].update(frame)
            elapsed = (time.time() - start_time) * 1000

            data['total_time'] += elapsed
            if success:
                data['success_count'] += 1

            results[name] = {
                'success': success,
                'bbox': bbox,
                'time': elapsed
            }

            # 记录性能
            self.performance_history[name].append({
                'success': success,
                'time': elapsed
            })

        # 每50帧选择最佳跟踪器
        if self.frame_count % 50 == 0:
            self._select_best_tracker()

        return results[self.current_name]['success'], results[self.current_name]['bbox'], self.current_name

    def _select_best_tracker(self):
        """选择最佳跟踪器"""
        best_score = -1
        best_name = self.current_name

        for name, data in self.trackers.items():
            if data['total_time'] > 0:
                # 综合考虑成功率和速度
                success_rate = data['success_count'] / self.frame_count
                avg_time = data['total_time'] / self.frame_count
                speed_score = 1 / (1 + avg_time / 10)  # 时间越短分数越高

                score = success_rate * 0.7 + speed_score * 0.3

                if score > best_score:
                    best_score = score
                    best_name = name

        if best_name != self.current_name:
            print(f"切换跟踪器: {self.current_name} -> {best_name}")
            self.current_name = best_name

def adaptive_tracking_demo():
    cap = cv2.VideoCapture(0)

    ret, frame = cap.read()
    if not ret:
        return

    print("选择跟踪目标...")
    bbox = cv2.selectROI('Select', frame, False, True)
    cv2.destroyWindow('Select')

    tracker = AdaptiveTracker()
    tracker.init(frame, bbox)

    print("自适应跟踪演示")
    print("系统会自动选择最佳跟踪器")
    print("按 'q' 退出")

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        success, bbox, tracker_name = tracker.update(frame)

        result = frame.copy()

        if success:
            x, y, w, h = [int(v) for v in bbox]
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

        cv2.putText(result, f'Tracker: {tracker_name}', (10, 30),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.putText(result, f'Status: {"OK" if success else "Lost"}', (10, 60),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7,

        cv2.imshow('Adaptive Tracking', result)

        if cv2.waitKey(30) & 0xFF == ord('q'):
```

```python
"""
示例10：跟踪综合系统
完整的目标跟踪系统！
"""
import cv2
import numpy as np
from collections import deque
import time

class ComprehensiveTracker:
    def __init__(self, tracker_type='CSRT'):
        self.tracker_type = tracker_type
        self.tracker = None
        self.bbox = None
        self.trajectory = deque(maxlen=100)
        self.frame_times = deque(maxlen=30)
        self.initialized = False
        self.lost_frames = 0
        self.total_frames = 0

    def init(self, frame, bbox):
        if self.tracker_type == 'KCF':
            self.tracker = cv2.legacy.TrackerKCF_create()
        elif self.tracker_type == 'CSRT':
            self.tracker = cv2.legacy.TrackerCSRT_create()
        elif self.tracker_type == 'MOSSE':
            self.tracker = cv2.legacy.TrackerMOSSE_create()

        self.tracker.init(frame, bbox)
        self.bbox = bbox
        self.initialized = True
        self.trajectory.clear()

        x, y, w, h = bbox
        self.trajectory.append((x + w//2, y + h//2))

    def update(self, frame):
        if not self.initialized:
            return False, None

        start_time = time.time()
        success, bbox = self.tracker.update(frame)
        elapsed = (time.time() - start_time) * 1000

        self.frame_times.append(elapsed)
        self.total_frames += 1

        if success:
            self.bbox = bbox
            x, y, w, h = [int(v) for v in bbox]
            self.trajectory.append((x + w//2, y + h//2))
            self.lost_frames = 0
        else:
            self.lost_frames += 1

        return success, bbox

    def get_stats(self):
        avg_fps = 1000 / np.mean(self.frame_times) if self.frame_times else 0
        success_rate = (self.total_frames - self.lost_frames) / max(1, self.total_frames)

        return {
            'fps': avg_fps,
            'success_rate': success_rate,
            'total_frames': self.total_frames,
            'trajectory_length': len(self.trajectory)
        }

    def draw(self, frame):
        result = frame.copy()

        # 绘制轨迹
        for i in range(1, len(self.trajectory)):
            alpha = i / len(self.trajectory)
            color = (int(255 * (1-alpha)), int(255 * alpha), 0)
            thickness = max(1, int(3 * alpha))
            cv2.line(result, self.trajectory[i-1], self.trajectory[i], color, thickness)

        # 绘制边界框
        if self.bbox is not None:
            x, y, w, h = [int(v) for v in self.bbox]
            cv2.rectangle(result, (x, y), (x + w, y + h), (0, 255, 0), 2)

        return result

def comprehensive_tracking_demo():

    ret, frame = cap.read()

    tracker = None
    selecting = False
    start_point = None
    current_bbox = None

    def mouse_callback(event, x, y, flags, param):
        nonlocal selecting, start_point, current_bbox, tracker

        if event == cv2.EVENT_LBUTTONDOWN:
            selecting = True
            start_point = (x, y)

        elif event == cv2.EVENT_LBUTTONUP and selecting:
            selecting = False
            x1, y1 = start_point
            w, h = x - x1, y - y1

            if w > 10 and h > 10:
                current_bbox = (x1, y1, w, h)
                ret, frame = cap.read()
                if ret:
                    tracker = ComprehensiveTracker('CSRT')
                    tracker.init(frame, current_bbox)
                    print("开始跟踪")

    cv2.namedWindow('Comprehensive Tracker')
    cv2.setMouseCallback('Comprehensive Tracker', mouse_callback)

    print("综合跟踪系统")
    print("用鼠标框选目标开始跟踪")
    print("按 'r' 重置")
    print("按 'q' 退出")

    while True:

        if not ret:
```
